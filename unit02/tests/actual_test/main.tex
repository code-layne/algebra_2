\parthead{Part B --- Solving by Factoring (20 pts)}
Factor completely: take out the GCF first. To solve, get the equation equal to zero, factor, and
set each factor equal to zero.

\begin{enumerate}[leftmargin=1.9em, itemsep=6pt, label=\arabic*., start=5]
  \item (8 pts) Factor completely.
    \begin{enumerate}[label=(\alph*), leftmargin=1.9em, itemsep=8pt]
      \item $4x^2+8x-32$
        \begin{work}
          4x^2+8x-32 &= 4(x^2+2x-8)
        \end{work}
        \hfill \slot{4.2cm}{$4(x+4)(x-2)$}
      \item $x^2-64$ \hfill \slot{4.2cm}{$(x+8)(x-8)$}
      \item $x^2+12x+36$ \hfill \slot{4.2cm}{$(x+6)^2$}
      \item $2x^2-3x-5$ \quad (by grouping)
        \begin{work}
          2x^2-3x-5 &= 2x^2+2x-5x-5 \\
                    &= 2x(x+1)-5(x+1)
        \end{work}
        \hfill \slot{4.2cm}{$(2x-5)(x+1)$}
    \end{enumerate}
\end{enumerate}
