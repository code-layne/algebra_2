\boxguard[12]
\begin{notesbox}{1. Slope Is the Rate of Change}
For \emph{any} two points $(x_1,y_1)$ and $(x_2,y_2)$ on a line,
\[
  m = \frac{y_2 - y_1}{x_2 - x_1} = \frac{\text{rise}}{\text{run}}.
\]
Because a line's rate of change is \emph{constant}, it does not matter which two points you pick.
\textbf{Try it} with $(1,4)$ and $(3,0)$:
\[
  m = \frac{\blank{0.8cm} - \blank{0.8cm}}{\blank{0.8cm} - \blank{0.8cm}}
    = \frac{\blank{0.8cm}}{\blank{0.8cm}} = \blank{1.0cm}.
\]
Warning: subtract the $y$'s and the $x$'s in the \emph{same order}, or the sign flips.
\end{notesbox}

\begin{notesbox}{2. Three Forms, One Line}
Here is the single line $y = 2x - 4$ written three different ways --- same line, different
``outfits.'' Fill in what each form makes easy to read.

\vspace{2pt}
\renewcommand{\arraystretch}{1.5}
\begin{tabularx}{\linewidth}{@{}p{0.30\linewidth} l X@{}}
\textbf{Slope-intercept} & $y = mx + b$ & Reads off the slope $m=\blank{0.9cm}$ and $y$-intercept $(0,\blank{0.9cm})$. \\
\textbf{Point-slope} & $y - y_1 = m(x - x_1)$ & Built from a slope and a point; e.g.\ $(2,0)$: $\;y - 0 = \blank{0.7cm}(x - \blank{0.7cm})$. \\
\textbf{Standard} & $Ax + By = C$ & Integers, $A\ge 0$; tidy for intercepts. Here: $\blank{0.7cm}\,x - y = \blank{0.9cm}$. \\
\end{tabularx}

\vspace{3pt}
\textbf{They are equal.} Simplify the point-slope $y - 0 = 2(x-2)$ by distributing:
\begin{work}
  y - 0 &= 2(x - 2) \\
      y &= 2x - 4
\end{work}
That is exactly the slope-intercept form above --- each form is the \emph{same} line.
\end{notesbox}

\begin{notesbox}{3. Writing the Equation of a Line}
You can always write a line once you have a \textbf{slope} and \textbf{one point}. Match the given
information to a starting form:
\begin{enumerate}[leftmargin=1.7em, itemsep=4pt]
  \item \textbf{Slope $+$ $y$-intercept} $\to$ drop straight into $y = mx + b$.\\
        Slope $3$, $y$-intercept $(0,-2)$: $\;y = \blank{2.6cm}$.
  \item \textbf{Slope $+$ any point} $\to$ point-slope, then simplify.\\
        Slope $-\tfrac12$ through $(4,1)$: $\;y - 1 = \blank{1.4cm}(x - \blank{0.8cm})$,
        so $y = \blank{2.6cm}$.
  \item \textbf{Two points} $\to$ find $m$ first, then use point-slope.\\
        Through $(1,4)$ and $(3,0)$: $\;m = \blank{1.0cm}$; using $(3,0)$,
        $y - 0 = \blank{1.4cm}(x - \blank{0.8cm})$, so $y = \blank{2.4cm}$.
\end{enumerate}
\end{notesbox}

\boxguard[20]
\begin{notesbox}{4. Graphing by Transforming the Parent $y = x$}
Every non-vertical line is the parent $y = x$ after two moves: a vertical
\textbf{stretch/reflection} (which sets the \emph{slope}) and a vertical \textbf{shift} (which sets
the \emph{$y$-intercept}). Below, the dashed line is the parent $y=x$ and the solid line is
$y = 2x - 4$.
\begin{center}
\begin{tikzpicture}[scale=0.5]
  \lgrid{-2}{6}{-6}{6}
  \draw[very thick, forest!55, dashed] (-2,-2)--(6,6) node[right, font=\scriptsize, forest!70]{$y=x$};
  \draw[very thick, forest] (-1,-6)--(5,6) node[right, font=\scriptsize]{$y=2x-4$};
  \fill[charcoal] (0,0) circle (2.5pt);
  \fill[charcoal] (0,-4) circle (3pt) node[left, font=\scriptsize]{$(0,-4)$};
\end{tikzpicture}
\end{center}
\begin{itemize}[leftmargin=1.5em, itemsep=3pt]
  \item The slope changed from $1$ to $2$: a vertical \textbf{stretch} by a factor of \blank{1.0cm}
        \; (this is $k\,f(x)$ with $k=\blank{0.8cm}$).
  \item The $y$-intercept moved from $(0,0)$ to $(0,-4)$: a \textbf{shift down} \blank{1.0cm} units
        \; (this is $f(x)+k$ with $k=\blank{1.0cm}$).
  \item If the slope were \emph{negative}, the parent would also be \textbf{reflected} over the
        $x$-axis. \textbf{Big idea:} this $k\,f(x)$ and $f(x)+k$ language will graph \emph{every}
        function family this year.
\end{itemize}
\end{notesbox}

\begin{practicebox}
Write the equation of the line through $(-2,3)$ and $(2,-5)$ in \emph{all three forms}.
\begin{enumerate}[leftmargin=1.6em, itemsep=3pt]
  \item Slope: $m = \dfrac{\blank{0.8cm}-\blank{0.8cm}}{\blank{0.8cm}-\blank{0.8cm}} = \blank{1.0cm}$.
  \item Point-slope (using $(2,-5)$): $\;y - (\blank{0.8cm}) = \blank{1.2cm}\,(x - \blank{0.8cm})$.
  \item Simplify that point-slope form into slope-intercept form:
    \begin{work}
      y + 5 &= -2(x - 2) \\
      y + 5 &= -2x + 4   \\
          y &= -2x - 1
    \end{work}
  \item Standard form (integers, $A \ge 0$): $\;\blank{3.0cm}$.
\end{enumerate}
\end{practicebox}
