\begin{tcolorbox}[colback=forestbg, colframe=forest, boxrule=0.9pt, arc=2mm,
    left=2mm, right=2mm, top=1.5mm, bottom=1.5mm]
    \textbf{Primary Objective:} Students describe any quadratic as a transformation of the
    parent $y=x^2$ --- shifts from $h$ and $k$, a stretch or compression from $|a|$, a reflection
    when $a<0$ --- and read the graph's features from whichever of the \textbf{three forms} is in
    front of them: the vertex and axis from vertex form, the $y$-intercept and axis
    $x=-\frac{b}{2a}$ from standard form, the zeros from intercept form. \textbf{No factoring and
    no solving} --- a factored form is given and read; 2.2 produces it.
    \\[2pt]
    \textbf{Standards (2023 VA SOL):} \textbf{A.F.2b, A.F.2c, A.F.2d} (the Algebra~1
    quadratic-graphing cluster: zeros and $y$-intercept, the vertex as max or min, the axis of
    symmetry), extended by \textbf{A2.F.1} (transformations of the parent, including the
    horizontal shift $f(x-h)$), \textbf{A2.F.2a} (domain, range, zeros, intercepts), and
    \textbf{A2.F.2d} (extrema).
    \\[2pt]
    \textbf{Lesson model:} \emph{Slides first.} The deck is the lesson and its printed handout is
    the student's notes, and students are \emph{solving the whole hour}: a warm-up on screen as
    they walk in; four rounds of a short \textbf{definition} and \textbf{example} ($\le$3 min),
    a \textbf{ladder} of three problems (\rung1\ \rung2\ \rung3), and a \textbf{Show me} or
    \textbf{Find the mistake}; a self-scored \textbf{Final Round}; and the homework started in
    class. One curve in three forms carries it: $(x-1)^2-4=x^2-2x-3=(x+1)(x-3)$.
    \textbf{Homework source:} DeltaMath, printed --- the exported set is dropped into the packet
    as \texttt{homework/main.pdf}.
\end{tcolorbox}

\begin{skillbox}[Priority Ideas \& Skills]{goldbox}
\begin{tabularx}{\linewidth}{@{}X|X@{}}
\textbf{Priority Ideas \& Skills} & \textbf{Key Understandings (the ``why'')} \\[2pt]
\hline
\vspace{-6pt}
\begin{itemize}[leftmargin=1.1em, nosep, after=\vspace{2pt}]
  \item Describe $y=a(x-h)^2+k$ as shifts, stretch/compression, and reflection of $y=x^2$.
  \item Read the vertex $(h,k)$ and axis $x=h$; say max or min from the sign of $a$.
  \item From $ax^2+bx+c$: the $y$-intercept $c$, the axis $x=-\frac{b}{2a}$, then the vertex.
  \item From $a(x-p)(x-q)$: the zeros $p$, $q$, and the axis halfway between.
\end{itemize}
&
\vspace{-6pt}
\begin{itemize}[leftmargin=1.1em, nosep, after=\vspace{2pt}]
  \item The three forms are one curve; each hands over one feature for free.
  \item $a$ controls shape and direction; $h$ and $k$ control position --- read them in that
        order.
  \item \textbf{Target misconception: the bracket sign flips.} $(x-1)^2$ moves the parent
        \emph{right}; $(x+1)(x-3)$ has zeros $-1$ and $3$. Students read the printed sign.
  \item Opening down makes the vertex a maximum --- and the range $y\le k$.
\end{itemize}
\end{tabularx}
\end{skillbox}

\begin{skillbox}[{Vocabulary, Concepts, \& Theorems --- one definition frame each}]{sky}
\renewcommand{\arraystretch}{1.3}
\begin{tabularx}{\linewidth}{@{}l X@{}}
\textbf{Parent function} & $y=x^2$, vertex $(0,0)$ --- every quadratic is it transformed. \\
\textbf{Vertex form} & $y=a(x-h)^2+k$: vertex $(h,k)$, \textbf{axis of symmetry} $x=h$; right
                       $h$, up $k$. \\
\textbf{Stretch / compression} & $|a|>1$ narrower (vertical stretch); $0<|a|<1$ wider (vertical
                                 compression). \\
\textbf{Reflection} & $a<0$: opens down, the vertex is a \textbf{maximum}; $a>0$: a
                      \textbf{minimum}. \\
\textbf{Standard form} & $y=ax^2+bx+c$: $y$-intercept $c$, axis $x=-\frac{b}{2a}$. \\
\textbf{Intercept form} & $y=a(x-p)(x-q)$: \textbf{zeros} $p$ and $q$, axis
                          $x=\frac{p+q}{2}$. \\
\end{tabularx}
\end{skillbox}

\begin{skillbox}[Engagement --- how this hour keeps everyone solving]{goldbox}
\textbf{Nobody waits:} every practice frame is a ladder --- \rung1\ everyone (a quick win),
\rung2\ everyone, \rung3\ early finishers. \textbf{Everyone commits, safely:} \emph{Show me} is
1--4 fingers on the count of three, all at once. \textbf{Critique a stranger:} \emph{Find the
mistake} (Round 2) puts a max/min error on an anonymous Student A. \textbf{Their world:} a
horse's jump at the riding ring (Round 4 \rung3) and a Service Club hoodie sale at the school
store (bonus). \textbf{Short clock:} every practice frame carries its minutes. Praise the
climb more than the answer.
\end{skillbox}

\begin{skillbox}[Warm-Up (5 min, on the slides as they walk in)]{forestbg}
Three quick wins, answers in one click. \textbf{(1)} $f(x)=x^2$ at $-3$, $0$, $2$: $9$, $0$,
$4$ --- the parent's table. \textbf{(2)} $(x-1)^2-4$ and \textbf{(3)} $(x+1)(x-3)$ both
multiply out to $x^2-2x-3$. The reveal lands the hook: \emph{the same quadratic in different
clothes --- today, what each form tells you for free.} Then the targets frame (30 seconds).
\end{skillbox}

\begin{skillbox}[The Lesson --- four rounds of definition, example, ladder, check (40 min)]{forestbg}
\begin{multicols}{2}
\textbf{Round 1 --- shifts (10 min).} Definition (parent dashed, $(x-1)^2-4$ solid, the arrow
from vertex to vertex) and Example 1 in 3 minutes: $(x-1)^2-4$ right 1 down 4; $(x+2)^2+3$ via
$x-(-2)$, left 2 up 3; the sign-rule block. Ladder (4 min): \rung1\ up 5, $(0,5)$;
\rung2\ $(4,-1)$; \rung3\ $(-3,2)$, axis $x=-3$. Show me $y=x^2-3$: answer \textbf{3}, down 3;
\textbf{1}/\textbf{2} think every number is a horizontal move.

\textbf{Round 2 --- stretch, compress, flip (10 min).} Definition (four parabolas, colour-coded)
and Example 2, $-2(x+1)^2+3$: flipped, narrower, vertex $(-1,3)$ a maximum, range $y\le3$ ---
``sign of $a$, size of $a$, then where.'' Ladder (4 min): \rung1\ up, narrower;
\rung2\ down, wider, $(2,4)$ max; \rung3\ $y=-\frac13(x+1)^2-5$. Find the mistake: Student A's
step 3 calls $(3,2)$ the lowest point of $-(x-3)^2+2$ --- it is the maximum; range $y\le2$.

\columnbreak
\textbf{Round 3 --- standard form (9 min).} Definition and Example 3 on $x^2-2x-3$: $y$-int
$-3$, axis $x=1$, vertex $(1,-4)$ --- ``the vertex we read off Round 1: same curve.'' Ladder
(4 min): \rung1\ $(0,1)$, $x=-2$; \rung2\ $(-3,-4)$; \rung3\ $(2,5)$, a maximum. Show me the
axis of $2x^2-12x+7$: answer \textbf{2}, $x=3$; \textbf{1} lost the minus, \textbf{3} divided
by $a$ not $2a$, \textbf{4} is $c$.

\textbf{Round 4 --- intercept form: the crux (11 min).} Definition (the anchor with all three
forms' free features marked) and Example 4, $(x+1)(x-3)$: zeros $-1$, $3$; axis $1$; vertex
$(1,-4)$. Ladder (5 min): \rung1\ $2$, $6$, $x=4$; \rung2\ $-5$, $1$, $(-2,-9)$;
\rung3\ the horse lands 12 ft out, top of the jump 4.5 ft. Show me the axis of $(x-4)(x+2)$:
answer \textbf{2}, $x=1$. \textbf{The frame not to rush:} anyone on \textbf{1} read the zeros as
$-4$ and $2$ --- ask a \textbf{1} and a \textbf{2} to explain before the reveal.
\end{multicols}
\end{skillbox}

\boxguard
\begin{skillbox}[Final Round (5 min)]{forestbg}
One problem per round, alone, 5-minute timer: left 4, down 1; down and wider; $x=4$; zeros $-6$
and $1$ (the crux's twin --- ``$6$ and $-1$'' is the bracket that lies). \textbf{Bonus:} the
hoodie profit's vertex $(20,2000)$ --- \$20 a hoodie, \$2000 profit. Students score themselves
out of 4; take the scores on fingers --- your exit read. Say the \emph{Watch out} aloud.
\end{skillbox}

\boxguard
\begin{skillbox}[Active Monitoring --- Watch For]{redbox}
\begin{itemize}[leftmargin=1.2em, nosep]
  \item \textbf{Round 1:} $(x+3)^2$ read as ``right 3''; $x^2-3$ read as a horizontal move.
  \item \textbf{Round 2:} calling the vertex a minimum when $a<0$; range written the wrong way.
  \item \textbf{Round 3 Show me:} $-\frac{b}{a}$ instead of $-\frac{b}{2a}$; a dropped minus.
  \item \textbf{Round 4 Show me:} zeros read with the printed signs (\emph{the target
        misconception}).
  \item \textbf{Attitude:} a student parked at \rung1\ with nothing written --- crouch, point at
        the example on the handout, ask for just $h$ and $k$.
\end{itemize}
Cold-call prompts (after the fingers, never before): ``Which way does $(x+2)$ move it, and how
do you know?'' \; ``How can you tell max from min before you find the vertex?''
\end{skillbox}

\begin{teachernote}[Slides]
Pacing: warm-up 5, rounds 10 / 10 / 9 / 11, Final Round 5, homework 10. In each round the
definition and example together take \textbf{at most 3 minutes} --- if you catch yourself
explaining past that, move to the ladder and explain at desks. Reveal a ladder when most of
the room has \rung2. If behind, cut Round 3's Show me, never Round 4's. Students write in the
handout's notes column, which prints without the answers.
\end{teachernote}

\begin{teachernote}[Homework]
The homework is a printed DeltaMath set --- the lesson builds without it, but the packet is
cover-only until \texttt{homework/main.pdf} is in place. Launch problem~1 aloud in the last 10
minutes, then circulate with the Watch For list in mind. The $+$-bracket zeros problem is the
formative check. Due the first class after two study halls; announce the date in class and on
TurtleNet.
\end{teachernote}
