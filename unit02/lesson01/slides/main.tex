% PARENT + TRANSFORMATIONS
\begin{frame}[t, plain]
  \forestheader{The parent $y=x^2$ and its four moves}
  \sectionlabel{Transformations}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \footnotesize
      \begin{itemize}\setlength{\itemsep}{5pt}
        \item $f(x)+k$: shift \textbf{up} $k$ (down if $k<0$)
        \item $f(x-h)$: shift \textbf{right} $h$ --- \emph{sign flips!}
        \item $a\,f(x)$: \textbf{narrower} if $|a|>1$, wider if $|a|<1$
        \item $-f(x)$: \textbf{reflect} over the $x$-axis
      \end{itemize}
      \vspace{4pt}
      Together: $a(x-h)^2+k$ has vertex $(h,k)$.
    \end{column}
    \begin{column}{0.5\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.36]
        \lgrid{-3}{5}{-5}{5}
        \begin{scope}\clip (-3,-5) rectangle (5,5);
          \draw[very thick, forest!45, domain=-2.2:2.2, samples=60, smooth] plot (\x,{(\x)*(\x)});
          \draw[very thick, forest, domain=-1.2:3.2, samples=70, smooth] plot (\x,{(\x-1)*(\x-1) - 4});
        \end{scope}
        \fill[forest!55] (0,0) circle (3pt);
        \fill[charcoal] (1,-4) circle (3pt) node[below right, font=\tiny]{$(1,-4)$};
        \draw[->, navy, thick] (0.2,-0.3) -- (0.8,-3.5);
      \end{tikzpicture}\\
      {\tiny parent (light) $\to$ right $1$, down $4$}
    \end{column}
  \end{columns}
\end{frame}

% VERTEX FORM
\begin{frame}[t, plain]
  \forestheader{Vertex form: $a(x-h)^2+k$}
  \sectionlabel[goldacc]{Reads out the vertex}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      For $f(x)=(x-1)^2-4$:
      \begin{itemize}\setlength{\itemsep}{4pt}
        \item $a=1,\ h=1,\ k=-4$
        \item \textbf{Vertex} $(h,k)=(1,-4)$; \textbf{axis} $x=1$
        \item $a>0 \Rightarrow$ opens \textbf{up}, \textbf{min} value $-4$
        \item Expand: $(x-1)^2-4 = x^2-2x-3$
      \end{itemize}
      \vspace{4pt}
      \begin{block}{Watch the sign}
        $(x-1)$ makes the square zero at $x=1$, so the vertex sits at $x=+1$ --- \emph{not} $-1$.
      \end{block}
    \end{column}
    \begin{column}{0.45\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.42]
        \lgrid{-3}{5}{-5}{3}
        \draw[navy, dashed, thick] (1,-5)--(1,3);
        \begin{scope}\clip (-3,-5) rectangle (5,3);
          \draw[very thick, forest, domain=-1.4:3.4, samples=75, smooth] plot (\x,{(\x-1)*(\x-1) - 4});
        \end{scope}
        \fill[charcoal] (1,-4) circle (3.5pt) node[below right, font=\tiny]{$(1,-4)$};
      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}

% STANDARD FORM
\begin{frame}[t, plain]
  \forestheader{Standard form: $ax^2+bx+c$}
  \sectionlabel{Find the hidden vertex}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      For $f(x)=x^2-6x+5$:
      \begin{itemize}\setlength{\itemsep}{4pt}
        \item \textbf{Axis:} $x=-\dfrac{b}{2a}=-\dfrac{-6}{2}=3$
        \item \textbf{Vertex:} $f(3)=9-18+5=-4 \Rightarrow (3,-4)$
        \item \textbf{$y$-intercept:} $(0,c)=(0,5)$
      \end{itemize}
      \vspace{4pt}
      \begin{block}{Why $-\tfrac{b}{2a}$?}
        The axis is the \emph{average} of two symmetric points --- the midpoint of the parabola.
      \end{block}
    \end{column}
    \begin{column}{0.45\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.42]
        \lgrid{-1}{7}{-5}{6}
        \draw[navy, dashed, thick] (3,-5)--(3,6);
        \begin{scope}\clip (-1,-5) rectangle (7,6);
          \draw[very thick, forest, domain=-0.2:6.2, samples=80, smooth] plot (\x,{(\x)*(\x) - 6*(\x) + 5});
        \end{scope}
        \fill[charcoal] (3,-4) circle (3.5pt);
        \fill[charcoal] (0,5) circle (3.5pt);
        \fill[charcoal] (1,0) circle (3.5pt);
        \fill[charcoal] (5,0) circle (3.5pt);
      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}

% INTERCEPT FORM
\begin{frame}[t, plain]
  \forestheader{Intercept form: $a(x-p)(x-q)$}
  \sectionlabel[goldacc]{Reads out the zeros}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      For $f(x)=(x+1)(x-3)$:
      \begin{itemize}\setlength{\itemsep}{4pt}
        \item \textbf{Zeros:} set each factor $=0 \Rightarrow x=-1,\ 3$
        \item \textbf{Axis:} halfway between, $x=\dfrac{-1+3}{2}=1$
        \item \textbf{Vertex:} $f(1)=(2)(-2)=-4 \Rightarrow (1,-4)$
      \end{itemize}
      \vspace{6pt}
      Same curve again --- the three costumes of $x^2-2x-3$.
    \end{column}
    \begin{column}{0.45\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.42]
        \lgrid{-3}{5}{-5}{3}
        \draw[navy, dashed, thick] (1,-5)--(1,3);
        \begin{scope}\clip (-3,-5) rectangle (5,3);
          \draw[very thick, forest, domain=-1.4:3.4, samples=75, smooth] plot (\x,{(\x+1)*(\x-3)});

  \end{columns}
\end{frame}

% ROADMAP
\begin{frame}[t, plain]
  \forestheader{Which form for which feature?}
  \sectionlabel{Summary}
  \begin{itemize}\setlength{\itemsep}{5pt}
    \item \textbf{Vertex form} $a(x-h)^2+k$ $\Rightarrow$ vertex $(h,k)$, axis $x=h$.
    \item \textbf{Standard form} $ax^2+bx+c$ $\Rightarrow$ $y$-intercept $(0,c)$, axis $x=-\tfrac{b}{2a}$.
    \item \textbf{Intercept form} $a(x-p)(x-q)$ $\Rightarrow$ zeros $p,q$, axis $x=\tfrac{p+q}{2}$.
  \end{itemize}
  \vspace{4pt}

    from standard form by hand.
