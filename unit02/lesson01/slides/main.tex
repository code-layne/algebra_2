% ═══ ROUND 1 — SHIFTS: VERTEX FORM y=(x-h)^2+k ═══════════════════════════════
\begin{frame}[t, plain]
  \forestheader{Definition: shifting the parent $y=x^2$}
  \sectionlabel{Round 1 --- slide it}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \begin{block}{Parent function}
        $y=x^2$ is the \textbf{parent function}: a parabola with its \textbf{vertex} at
        $(0,0)$.
      \end{block}
      \vspace{2pt}
      \begin{block}{Vertex form (shifts)}
        $y=(x-h)^2+k$ is the parent shifted $h$ units \textbf{right} and $k$ units
        \textbf{up}. Its vertex is $(h,k)$ and its \textbf{axis of symmetry} is the line
        $x=h$. A negative $h$ or $k$ means left or down.
      \end{block}
    \end{column}
    \begin{column}{0.40\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.34]
        \qgrid{-4}{5}{-5}{5}
        \begin{scope}\clip (-4,-5) rectangle (5,5);
          \draw[thick, slate, dashed, domain=-2.3:2.3, samples=50, smooth] plot (\x,{\x*\x});
          \draw[very thick, forest, domain=-2.1:4.1, samples=60, smooth]
            plot (\x,{(\x-1)*(\x-1)-4});
        \end{scope}
        \fdot{slate}{0}{0}
        \fdot{redacc}{1}{-4}
        \draw[->, thick, redacc] (0.2,-0.3) -- (0.9,-3.6);
      \end{tikzpicture}\\[-2pt]
      {\scriptsize dashed: $y=x^2$ \quad solid: $y=(x-1)^2-4$ \\ right 1, down 4: vertex
      $(1,-4)$}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Example 1: describe the shift, name the vertex}
  \sectionlabel{Watch --- 2 minutes}
  \begin{itemize}\setlength{\itemsep}{8pt}
    \item $y=(x-1)^2-4$: \; $h=1$, $k=-4$. \pause \; Right $1$, down $4$. Vertex $(1,-4)$, axis
          $x=1$.
    \pause
    \item $y=(x+2)^2+3$: \; rewrite $x+2$ as $x-(-2)$, so $h=-2$, $k=3$. \pause \; Left $2$,
          up $3$. Vertex $(-2,3)$, axis $x=-2$.
  \end{itemize}
  \pause
  \vspace{6pt}
  \begin{block}{The sign rule}
    Inside the bracket the sign is \emph{opposite} to the move: $x-1$ goes right, $x+2$ goes
    left. Outside the bracket it is not: $+3$ goes up.
  \end{block}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Your turn}
  \sectionlabel[goldacc]{4 minutes --- start at one star and climb}

    \item[\rung2] $y=(x-4)^2-1$
    \item[\rung3] $y=(x+3)^2+2$ \quad --- and its axis of symmetry
  \end{itemize}
  \vspace{4pt}
  \reveal{%
    \begin{block}{Check}
      \rung1\ up $5$; $(0,5)$ \qquad \rung2\ right $4$, down $1$; $(4,-1)$ \qquad
      \rung3\ left $3$, up $2$; $(-3,2)$; axis $x=-3$
    \end{block}}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Show me: how does $y=x^2-3$ move the parent?}
  \sectionlabel[goldacc]{Pick one --- 1 to 4 fingers on the count of three}
  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \opt1{right 3} \\[12pt]
      \opt3{down 3}
    \end{column}
    \begin{column}{0.45\textwidth}
      \opt2{left 3} \\[12pt]
      \opt4{up 3}
    \end{column}
  \end{columns}
  \vspace{10pt}
  \reveal{%
    \begin{block}{Answer: \textbf{3}}
      The $-3$ is \emph{outside} the square, so it is $k$: down $3$, vertex $(0,-3)$. Chose
      \textbf{1} or \textbf{2}? Those need a number \emph{inside} the bracket, like $(x-3)^2$.
    \end{block}}
\end{frame}

% ═══ ROUND 2 — STRETCH, COMPRESS, FLIP: a ════════════════════════════════════
\begin{frame}[t, plain]
  \forestheader{Definition: stretch, compress, flip --- that is $a$}
  \sectionlabel{Round 2 --- shape it}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \begin{block}{$y=a(x-h)^2+k$}
        \begin{itemize}\setlength{\itemsep}{2pt}
          \item $|a|>1$: a \textbf{vertical stretch} --- narrower.
          \item $0<|a|<1$: a \textbf{vertical compression} --- wider.
          \item $a<0$: a \textbf{reflection} over the $x$-axis --- it opens \emph{down}, and the
                vertex is a \textbf{maximum}.
          \item $a>0$: it opens up, and the vertex is a \textbf{minimum}.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.40\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.34]
        \qgrid{-4}{4}{-5}{5}
        \begin{scope}\clip (-4,-5) rectangle (4,5);
          \draw[very thick, forest, domain=-2.3:2.3, samples=50, smooth] plot (\x,{\x*\x});
          \draw[very thick, navy, domain=-1.6:1.6, samples=50, smooth] plot (\x,{2*\x*\x});
          \draw[very thick, goldacc, domain=-3.2:3.2, samples=50, smooth]
            plot (\x,{0.5*\x*\x});
          \draw[very thick, redacc, domain=-2.3:2.3, samples=50, smooth] plot (\x,{-\x*\x});
        \end{scope}
      \end{tikzpicture}\\[-2pt]
      {\scriptsize {\color{forest}$x^2$} \; {\color{navy}$2x^2$ narrower} \;
      {\color{goldacc}$\tfrac12x^2$ wider} \; {\color{redacc}$-x^2$ flipped}}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Example 2: describe $y=-2(x+1)^2+3$}
  \sectionlabel{Watch --- 2 minutes}
  \begin{itemize}\setlength{\itemsep}{6pt}
    \item $a=-2$: \; negative, so \textbf{reflected} --- opens down. \pause $|a|=2>1$, so
          \textbf{stretched} --- narrower.
    \pause
    \item $h=-1$, $k=3$: \; left $1$, up $3$. Vertex $(-1,3)$, axis $x=-1$.
    \pause
    \item Opens down, so $(-1,3)$ is a \textbf{maximum}: the highest point. Range $y\le3$.
  \end{itemize}
  \pause
  \vspace{6pt}
  \emph{Order to read it in:} the sign of $a$ (up or down), the size of $a$ (narrow or wide),
  then $h$ and $k$ (where).
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Find the mistake}
  \sectionlabel[goldacc]{2 minutes --- Student A made one mistake. Find it and fix it}
  \textbf{Student A} described $y=-(x-3)^2+2$:
  \begin{enumerate}\setlength{\itemsep}{4pt}
    \item ``Right $3$ and up $2$, so the vertex is $(3,2)$.''
    \item ``The axis of symmetry is $x=3$.''
    \item ``The vertex $(3,2)$ is the \emph{lowest} point, so the range is $y\ge2$.''
  \end{enumerate}
  \vspace{4pt}
  \reveal{%
    \begin{block}{The mistake: step 3}
      $a=-1<0$, so the parabola opens \emph{down}: $(3,2)$ is the \textbf{maximum} and the range
      is $y\le2$. Steps 1 and 2 are right --- check the sign of $a$ first.
    \end{block}}
\end{frame}

% ═══ ROUND 3 — STANDARD FORM: AXIS x = -b/(2a) ═══════════════════════════════
\begin{frame}[t, plain]
  \forestheader{Definition: reading standard form}
  \sectionlabel{Round 3 --- $ax^2+bx+c$}
  \begin{block}{Standard form $y=ax^2+bx+c$}
    \begin{itemize}\setlength{\itemsep}{3pt}
      \item The \textbf{$y$-intercept} is $c$ (put $x=0$).
      \item The \textbf{axis of symmetry} is \[ x=-\frac{b}{2a}. \]
      \item The vertex sits on the axis: put that $x$ back into the equation to get its
            $y$-value.
      \item $a$ means what it meant in Round 2: its sign says up or down.
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Example 3: read $y=x^2-2x-3$}
  \sectionlabel{Watch --- 2 minutes}
  $a=1$, $b=-2$, $c=-3$.
  \begin{itemize}\setlength{\itemsep}{6pt}
    \item $y$-intercept: $c=-3$, the point $(0,-3)$.
    \pause
    \item Axis: $x=-\dfrac{-2}{2(1)}=1$.
    \pause
    \item Vertex: $y=1^2-2(1)-3=-4$, so $(1,-4)$.
  \end{itemize}
  \pause
  \vspace{4pt}
  \begin{block}{Same curve}
    $(1,-4)$ is the vertex we read straight off $y=(x-1)^2-4$ in Round 1 --- the warm-up said
    they are the same quadratic.
  \end{block}
\end{frame}

    \item[\rung2] $y=x^2+6x+5$: the axis and the vertex

\end{frame}

% ═══ ROUND 4 — INTERCEPT FORM AND THE BRACKET THAT LIES (the crux) ════════════
\begin{frame}[t, plain]
  \forestheader{Definition: intercept form}
  \sectionlabel{Round 4 --- one curve, three forms}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \begin{block}{Intercept form $y=a(x-p)(x-q)$}
        The \textbf{zeros} ($x$-intercepts) are $x=p$ and $x=q$. The axis of symmetry is
        halfway between them: $x=\dfrac{p+q}{2}$.
      \end{block}
      \vspace{2pt}
      \begin{block}{The bracket that lies}
        In \emph{both} $(x-h)^2$ and $(x-p)(x-q)$, the number you read has the
        \emph{opposite} sign to the one printed: $(x+1)$ means $-1$.
      \end{block}
    \end{column}
    \begin{column}{0.40\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.34]
        \qgrid{-3}{5}{-5}{4}
        \begin{scope}\clip (-3,-5) rectangle (5,4);
          \draw[very thick, forest, domain=-1.8:3.8, samples=60, smooth]
            plot (\x,{(\x+1)*(\x-3)});
        \end{scope}
        \fdot{redacc}{-1}{0} \fdot{redacc}{3}{0} \fdot{navy}{1}{-4} \fdot{goldacc}{0}{-3}
        \draw[dashed, navy] (1,-5) -- (1,4);
      \end{tikzpicture}\\[-2pt]
      {\scriptsize $(x-1)^2-4$: {\color{navy}vertex $(1,-4)$} \\ $x^2-2x-3$:
      {\color{goldacc}$y$-int $-3$} \\ $(x+1)(x-3)$: {\color{redacc}zeros $-1$, $3$}}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Example 4: read $y=(x+1)(x-3)$}
  \sectionlabel{Watch --- 2 minutes}
  \begin{itemize}\setlength{\itemsep}{6pt}
    \item Zeros: $x+1=x-(-1)$, so $p=-1$; and $q=3$. \pause \; The graph crosses at $-1$ and
          $3$.
    \pause
    \item Axis: halfway, $x=\dfrac{-1+3}{2}=1$.
    \pause
    \item Vertex: $y=(1+1)(1-3)=-4$, so $(1,-4)$ --- the same vertex, the same curve.
  \end{itemize}
  \pause
  \vspace{4pt}
  \emph{Each form hands you something free:} vertex form the vertex, standard form the
  $y$-intercept, intercept form the zeros.
\end{frame}

\end{frame}

    \begin{column}{0.45\textwidth}

    \end{block}}
