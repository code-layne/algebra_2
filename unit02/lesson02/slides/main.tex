% ═══ ROUND 1 — FACTORING x^2+bx+c, GCF FIRST; THE SPECIAL PATTERNS AS CASES ════
\begin{frame}[t, plain]
  \forestheader{Definition: factoring $x^2+bx+c$}
  \sectionlabel{Round 1 --- un-multiplying}
  \begin{columns}[T]
    \begin{column}{0.62\textwidth}
      \begin{block}{Factoring and the product--sum search}
        To \textbf{factor} is to write a polynomial as a \emph{product}. Take out the
        \textbf{greatest common factor (GCF)} first. Then, for $x^2+bx+c$, find integers $p$ and
        $q$ with $p\cdot q=c$ and $p+q=b$: \; $x^2+bx+c=(x+p)(x+q)$.
      \end{block}
      \vspace{2pt}
      \begin{block}{Three cases of the search}
        \small
        \textbf{Difference of squares} ($b=0$): $x^2-k^2=(x+k)(x-k)$ --- the pair is $k,\,-k$.\\[2pt]
        \textbf{Perfect square} ($p=q$): $x^2+2kx+k^2=(x+k)^2$ --- a \textbf{double root}.\\[2pt]
        \textbf{Sum of squares} $x^2+k^2$: no pair has product $k^2$ and sum $0$ --- it does
        \textbf{not} factor.
      \end{block}
    \end{column}
    \begin{column}{0.34\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.36]
        \lgrid{-1}{7}{-2}{6}
        \begin{scope}\clip (-1,-2) rectangle (7,6);
          \draw[very thick, forest, domain=0.4:5.6, samples=60, smooth]
            plot (\x,{(\x-3)*(\x-3)});
        \end{scope}
        \fill[redacc] (3,0) circle (4pt) node[below, font=\tiny, charcoal]{$3$};
      \end{tikzpicture}\\[-2pt]
      {\scriptsize $y=x^2-6x+9=(x-3)^2$ \emph{touches} the axis once: a \textbf{double root}.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Example 1: factor $x^2-2x-3$, then $2x^2-4x-6$}
  \sectionlabel{Watch --- 2 minutes}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{(a)} $x^2-2x-3$: product $-3$, sum $-2$.
      \pause
      \begin{center}
      \renewcommand{\arraystretch}{1.15}
      \begin{tabular}{c|c}
        pair & sum \\ \hline
        $1,\,-3$ & $\mathbf{-2}$ \checkmark \\
        $-1,\,3$ & $2$
      \end{tabular}
      \end{center}
      \pause
      $x^2-2x-3=(x+1)(x-3)$ --- the warm-up, backwards.
    \end{column}
    \pause
    \begin{column}{0.48\textwidth}
      \textbf{(b)} $2x^2-4x-6$: every term has a $2$.
      \[ \begin{aligned} 2x^2-4x-6 &= 2\,(x^2-2x-3) \\ &= 2\,(x+1)(x-3) \end{aligned} \]
      \emph{GCF first} --- then the trinomial left inside is the one we just factored.
      \pause
      \par\vspace{6pt}
      \textbf{(c)} $x^2-25$: $b=0$, so the pair is $5,\,-5$: \; $(x+5)(x-5)$.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Your turn}
  \sectionlabel[goldacc]{6 minutes --- start at one star and climb}
  Factor completely.
  \begin{itemize}\setlength{\itemsep}{6pt}
    \item[\rung1] $x^2+7x+12$
    \item[\rung2] $x^2-49$
    \item[\rung3] $3x^2+3x-18$
  \end{itemize}
  \vspace{4pt}
  \reveal{%
    \begin{block}{Check}
      \rung1\ $(x+3)(x+4)$ \qquad \rung2\ $(x+7)(x-7)$ --- pair $7,\,-7$ \qquad

    \end{block}}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Show me: which one does \emph{not} factor?}
  \sectionlabel[goldacc]{Pick one --- 1 to 4 fingers on the count of three}
  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \opt1{$x^2-16$} \\[12pt]
      \opt3{$x^2+16$}
    \end{column}
    \begin{column}{0.45\textwidth}
      \opt2{$x^2+8x+16$} \\[12pt]
      \opt4{$x^2+6x-16$}
    \end{column}
  \end{columns}
  \vspace{10pt}
  \reveal{%
    \begin{block}{Answer: \textbf{3}}
      $x^2+16$ is a \emph{sum} of squares --- no two integers have product $16$ and sum $0$.
      The others: \textbf{1} $(x+4)(x-4)$, \textbf{2} $(x+4)^2$, \textbf{4} $(x+8)(x-2)$.
      Tempted by $(x+4)(x-4)$? Multiply it out: $x^2-16$.
    \end{block}}
\end{frame}

% ═══ ROUND 2 — ax^2+bx+c BY GROUPING ═══════════════════════════════════════
\begin{frame}[t, plain]
  \forestheader{Definition: factoring $ax^2+bx+c$ by grouping}
  \sectionlabel{Round 2 --- when $a\ne1$}
  \begin{block}{Grouping (the $a\cdot c$ method)}
    To factor $ax^2+bx+c$ with $a\ne1$ (after any GCF):
    \begin{enumerate}\setlength{\itemsep}{2pt}
      \item Find two integers whose product is $\mathbf{a\cdot c}$ and whose sum is $\mathbf{b}$.
      \item Use them to \textbf{split} the middle term $bx$ into two terms.
      \item \textbf{Group} the four terms in pairs, take the GCF out of each pair, and take out
            the common binomial.
    \end{enumerate}
  \end{block}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Example 2: factor $2x^2+5x-3$}
  \sectionlabel{Watch --- 2 minutes}
  $a\cdot c=2(-3)=-6$, \; $b=5$: \; the pair is $6$ and $-1$ \;($6\cdot(-1)=-6$, $6+(-1)=5$).
  \[
  \begin{aligned}
    2x^2+5x-3 &= 2x^2+6x-x-3 && \text{split } 5x \text{ as } 6x-x \\
              &= 2x(x+3)-1(x+3) && \text{GCF of each pair} \\
              &= (2x-1)(x+3)    && \text{take out } (x+3)
  \end{aligned}
  \]
  \pause
  \emph{The brackets must match.} If they come out as opposites, change the sign of the GCF you
  took out.
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Your turn}
  \sectionlabel[goldacc]{6 minutes --- start at one star and climb}
  Factor by grouping.
  \begin{itemize}\setlength{\itemsep}{6pt}
    \item[\rung1] $2x^2+7x+3$
    \item[\rung2] $3x^2-10x+8$
    \item[\rung3] $6x^2+x-2$
  \end{itemize}
  \vspace{2pt}
  \reveal{%
    \begin{block}{Check}
      \rung1\ pair $6,1$: $2x(x+3)+1(x+3)=(2x+1)(x+3)$ \\[1pt]
      \rung2\ pair $-6,-4$: $3x(x-2)-4(x-2)=(3x-4)(x-2)$ \\[1pt]
      \rung3\ pair $4,-3$: $2x(3x+2)-1(3x+2)=(2x-1)(3x+2)$
    \end{block}}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Find the mistake}
  \sectionlabel[goldacc]{3 minutes --- Student A gave up. Find the wrong step and finish it}
  \textbf{Student A} factored $3x^2+x-10$:
  \[
  \begin{aligned}
    \text{pair: } 6 \text{ and } {-5} &\qquad (6\cdot(-5)=-30,\; 6+(-5)=1) \\
    3x^2+6x-5x-10 &= 3x(x+2)-5(x-2) \\
                  &\text{``The brackets don't match, so it doesn't factor.''}
  \end{aligned}
  \]
  \reveal{%
    \begin{block}{The mistake}
      $-5x-10=-5(x+2)$, not $-5(x-2)$: taking out $-5$ flips \emph{both} signs. Then
      $3x(x+2)-5(x+2)=(3x-5)(x+2)$. \quad Student A's pair was right --- one sign sank it.
    \end{block}}
\end{frame}

% ═══ ROUND 3 — ZERO PRODUCT PROPERTY (the crux) ════════════════════════════
\begin{frame}[t, plain]
  \forestheader{Definition: the Zero Product Property}
  \sectionlabel{Round 3 --- from factoring to solving}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \begin{block}{Zero Product Property}
        If $A\cdot B=0$, then $A=0$ or $B=0$.
      \end{block}
      \vspace{2pt}
      \begin{block}{Solving by factoring}
        \begin{enumerate}\setlength{\itemsep}{1pt}
          \item Move every term to one side: $\;\ldots=0$.
          \item Factor completely.
          \item Set each factor equal to $0$ and solve.
        \end{enumerate}
        The solutions are the \textbf{roots}; on the graph of $y=ax^2+bx+c$ they are the
        $x$-intercepts (the \textbf{zeros}).
      \end{block}
    \end{column}
    \begin{column}{0.38\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.36]
        \lgrid{-3}{5}{-5}{3}
        \begin{scope}\clip (-3,-5) rectangle (5,3);
          \draw[very thick, forest, domain=-1.5:3.5, samples=70, smooth]
            plot (\x,{(\x+1)*(\x-3)});
        \end{scope}
        \fill[redacc] (-1,0) circle (4pt) node[above left, font=\tiny, charcoal]{$-1$};
        \fill[redacc] (3,0) circle (4pt) node[above right, font=\tiny, charcoal]{$3$};
      \end{tikzpicture}\\[-2pt]
      {\scriptsize $y=x^2-2x-3$: zeros $-1$ and $3$.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Example 3: solve $x^2=2x+3$}
  \sectionlabel{Watch --- 2 minutes}
  \[
  \begin{aligned}
    x^2 &= 2x+3 \\
    x^2-2x-3 &= 0            && \text{everything on one side} \\
    (x+1)(x-3) &= 0          && \text{factor (Round 1)} \\
    x+1=0 \quad\text{or}\quad x-3 &= 0 && \text{Zero Product Property} \\
    x=-1 \quad\text{or}\quad x &= 3
  \end{aligned}
  \]
  \pause
  \emph{Check:} $(-1)^2=1=2(-1)+3$ \checkmark \quad $3^2=9=2(3)+3$ \checkmark \quad --- the
  graph's $x$-intercepts.
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Your turn}
  \sectionlabel[goldacc]{6 minutes --- start at one star and climb}
  Solve.
  \begin{itemize}\setlength{\itemsep}{6pt}
    \item[\rung1] $(x-4)(x+6)=0$
    \item[\rung2] $x^2-3x=10$
    \item[\rung3] A new rug for your dorm room is $3$ ft longer than it is wide and covers
          $40$ ft$^2$. Solve $x(x+3)=40$ for the width. Which root do you throw out?
  \end{itemize}
  \vspace{2pt}
  \reveal{%
    \begin{block}{Check}
      \rung1\ $x=4$ or $x=-6$ --- the signs flip \\[1pt]
      \rung2\ $x^2-3x-10=0$, $(x-5)(x+2)=0$: $x=5$ or $x=-2$ \\[1pt]
      \rung3\ $x^2+3x-40=0$, $(x+8)(x-5)=0$: $x=-8$ or $x=5$. A width cannot be negative ---
      the rug is $5$ ft by $8$ ft.
    \end{block}}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Show me: solve $x^2=5x$}
  \sectionlabel[redacc]{The one that matters --- 1 to 4 fingers on the count of three}
  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \opt1{$x=5$} \\[12pt]
      \opt3{$x=0$ or $x=-5$}
    \end{column}
    \begin{column}{0.45\textwidth}
      \opt2{$x=0$ or $x=5$} \\[12pt]
      \opt4{$x=\sqrt5$}
    \end{column}
  \end{columns}
  \vspace{10pt}
  \reveal{%
    \begin{block}{Answer: \textbf{2}}
      $x^2-5x=0 \Rightarrow x(x-5)=0 \Rightarrow x=0$ or $x=5$. \; Chose \textbf{1}? You divided
      both sides by $x$ --- that \emph{throws away} $x=0$. Never divide by $x$: move it over and
      factor.
    \end{block}}
\end{frame}

% FINAL ROUND — all three rounds, self-scored /4. Item 4 is the crux's twin.
\begin{frame}[t, plain]
  \forestheader{Final Round}
  \sectionlabel[goldacc]{5 minutes --- on your own --- score yourself out of 4}
  \begin{enumerate}\setlength{\itemsep}{3pt}
    \item Factor $x^2-8x+15$. \qquad\qquad\;\; 2.\ Factor $3x^2-12$.
    \setcounter{enumi}{2}
    \item Factor $2x^2-x-6$. \qquad\qquad\;\;\, 4.\ Solve $(x-1)(x+2)=4$.
  \end{enumerate}
  {\small\textbf{Bonus (+1):} In the egg-drop contest, a capsule is launched up from a $32$-ft
  roof; its height is $h=-16t^2+16t+32$. When does it hit the ground?}
  \vspace{2pt}
  \reveal{%
    \begin{block}{Check --- one point each}
      1.\ $(x-3)(x-5)$ \quad 2.\ $3(x+2)(x-2)$ \quad 3.\ $(2x+3)(x-2)$ \quad
      4.\ $x^2+x-6=0$: $x=-3$ or $x=2$ \\[1pt]
      \textbf{Bonus:} $-16(t^2-t-2)=0$, $(t-2)(t+1)=0$: $t=2$ seconds ($t=-1$ is before the
      launch). \\[2pt]
      \textbf{Watch out:} the property is about \textbf{zero}. ``$x-1=4$ or $x+2=4$'' is wrong ---
      get it equal to $0$ first, and never divide by $x$.
    \end{block}}
\end{frame}
