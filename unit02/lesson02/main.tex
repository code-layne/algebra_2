\begin{tcolorbox}[colback=forestbg, colframe=forest, boxrule=0.9pt, arc=2mm,
    left=2mm, right=2mm, top=1.5mm, bottom=1.5mm]
    \textbf{Primary Objective:} Students \emph{produce} the factored form of a quadratic --- GCF
    first, $x^2+bx+c$ by the product--sum search (with the difference of squares and the perfect
    square as its special cases), $ax^2+bx+c$ by grouping --- and use the Zero Product Property
    to solve $ax^2+bx+c=0$, reading the roots as the graph's $x$-intercepts. Lesson~2.1
    \emph{read} zeros off a given factored form; 2.2 \emph{finds} that form.
    \\[2pt]
    \textbf{Standards (2023 VA SOL):} \textbf{A2.EO.3b} (factor completely over the integers),
    \textbf{A2.EO.3d} (difference of squares, perfect-square trinomials), \textbf{A2.EI.2b}
    (solve a quadratic equation algebraically --- here by factoring; real, factorable cases
    only), \textbf{A2.EI.2d} (verify solutions graphically: the roots are the $x$-intercepts).
    \\[2pt]
    \textbf{Lesson model:} \emph{Slides first.} The deck is the lesson and its printed handout is
    the student's notes, and students are \emph{solving the whole hour}: a warm-up on screen as
    they walk in; three rounds of a short \textbf{definition} and \textbf{example} ($\le$3 min),
    a \textbf{ladder} of three problems (\rung1\ \rung2\ \rung3), and a \textbf{Show me} or
    \textbf{Find the mistake}; a self-scored \textbf{Final Round}; and the homework started in
    class. \textbf{Homework source:} DeltaMath, online --- set \textbf{A2: Lesson 2.2: Solving by Factoring};
    nothing in the packet.
\end{tcolorbox}

\begin{skillbox}[Priority Ideas \& Skills]{goldbox}
\begin{tabularx}{\linewidth}{@{}X|X@{}}
\textbf{Priority Ideas \& Skills} & \textbf{Key Understandings (the ``why'')} \\[2pt]
\hline
\vspace{-6pt}
\begin{itemize}[leftmargin=1.1em, nosep, after=\vspace{2pt}]
  \item Take out the \textbf{GCF} first, then factor what remains completely.
  \item Factor $x^2+bx+c$: two integers with \textbf{product $c$} and \textbf{sum $b$};
        $x^2-k^2$ and $x^2+2kx+k^2$ are its special cases; $x^2+k^2$ does not factor.
  \item Factor $ax^2+bx+c$ by \textbf{grouping} (product $a\cdot c$, sum $b$).
  \item Solve by the \textbf{Zero Product Property}: $=0$, factor, each factor $=0$.
\end{itemize}
&
\vspace{-6pt}
\begin{itemize}[leftmargin=1.1em, nosep, after=\vspace{2pt}]
  \item Factoring \emph{un-multiplies}: $(x+p)(x+q)=x^2+(p+q)x+pq$ is why the search works ---
        and why $b=0$ forces the pair $k,-k$.
  \item A product is zero \emph{only} when a factor is zero --- the step that turns one
        quadratic into two linear equations.
  \item \textbf{Target misconception: the property is about zero.} Students divide both sides
        of $x^2=5x$ by $x$ (losing $x=0$), or set each factor of $(x-1)(x+2)=4$ equal to $4$.
        Both skip ``get it equal to zero first.''
  \item The roots are the $x$-intercepts --- the algebra lands on the 2.0--2.1 picture.
\end{itemize}
\end{tabularx}
\end{skillbox}

\begin{skillbox}[{Vocabulary, Concepts, \& Theorems --- one definition frame each}]{sky}
\renewcommand{\arraystretch}{1.3}
\begin{tabularx}{\linewidth}{@{}l X@{}}
\textbf{Factor / product--sum search} & To factor is to write a polynomial as a product. Take
                  out the \textbf{greatest common factor (GCF)} first. Then, for $x^2+bx+c$, find
                  integers $p$ and $q$ with $p\cdot q=c$ and $p+q=b$: $x^2+bx+c=(x+p)(x+q)$. \\
\textbf{Difference of squares} & ($b=0$) $x^2-k^2=(x+k)(x-k)$ --- the pair is $k,-k$. \\
\textbf{Perfect square} & ($p=q$) $x^2+2kx+k^2=(x+k)^2$ --- a \textbf{double root}: the graph
                          touches the axis once. \\
\textbf{Sum of squares} & $x^2+k^2$: no pair has product $k^2$ and sum $0$ --- it does not
                          factor. \\
\textbf{Grouping} & To factor $ax^2+bx+c$ with $a\ne1$: two integers with product $a\cdot c$ and
                    sum $b$ split $bx$; group in pairs; take the GCF out of each pair, then the
                    common binomial. \\
\textbf{Zero Product Property} & If $A\cdot B=0$, then $A=0$ or $B=0$. \\
\textbf{Roots / zeros} & The solutions of $ax^2+bx+c=0$; the $x$-intercepts of
                         $y=ax^2+bx+c$. \\
\end{tabularx}
\end{skillbox}

\begin{skillbox}[Engagement --- how this hour keeps everyone solving]{goldbox}
\textbf{Nobody waits:} every practice frame is a ladder --- \rung1\ everyone (a quick win),
\rung2\ everyone, \rung3\ early finishers --- so the fast are never idle and the slow still
finish two. \textbf{Everyone commits, safely:} \emph{Show me} is 1--4 fingers on the count of
three, all at once; nobody is singled out, and the spread of fingers is your read.
\textbf{Critique a stranger:} \emph{Find the mistake} puts the error on an anonymous Student A.
\textbf{Their world:} the dorm-rug rung and the egg-drop bonus. \textbf{Short clock:} every
practice frame carries its minutes --- set a visible timer. Praise the climb (``you got to
\rung2'') more than the answer.
\end{skillbox}

\begin{skillbox}[Warm-Up (5 min, on the slides as they walk in)]{forestbg}
Three quick wins; the answers are one click. \textbf{(1)} $(x+1)(x-3)=x^2-2x-3$ --- the move
today undoes. \textbf{(2)} product $-12$, sum $1$: $4$ and $-3$. \textbf{(3)} $y=(x+1)(x-3)$
crosses at $-1$ and $3$ (2.1). The reveal ends on \emph{Hold on to this}: given only
$x^2-2x-3$, how do you get the product back? That is Round~1. Then the targets frame (30
seconds --- point at ``you are solving the whole hour'').
\end{skillbox}

\begin{skillbox}[The Lesson --- three rounds of definition, example, ladder, check (40 min)]{forestbg}
\begin{multicols}{2}
\textbf{Round 1 --- $x^2+bx+c$, GCF first, and its special cases (13 min).} Definition and
Example 1 in 3 minutes: point at the three cases --- $b=0$ gives the pair $k,-k$; $p=q$ is the
tangent parabola, a double root; $x^2+k^2$ has no pair. Example: the pair table for
$x^2-2x-3$, then $2x^2-4x-6=2(x+1)(x-3)$, then $x^2-25=(x+5)(x-5)$ in one line. Ladder (6 min):
\rung1\ $(x+3)(x+4)$, \rung2\ $(x+7)(x-7)$, \rung3\ $3(x+3)(x-2)$ --- watch for $(3x+9)(x-2)$,
not \emph{completely} factored. Show me ``which does \emph{not} factor'': answer \textbf{3},
$x^2+16$. A \textbf{1} thinks $x^2+16=(x+4)(x-4)$ --- multiply it out together; a \textbf{2} or
\textbf{4} did not test the pair ($(x+4)^2$, $(x+8)(x-2)$).

\textbf{Round 2 --- grouping (13 min).} Definition and Example 2 ($2x^2+5x-3=(2x-1)(x+3)$) in 3
minutes, landing ``the brackets must match.'' Ladder (6 min): \rung1\ $(2x+1)(x+3)$,
\rung2\ $(3x-4)(x-2)$, \rung3\ $(2x-1)(3x+2)$. Find the mistake (3 min): Student A wrote
$-5x-10=-5(x-2)$ and quit; the fix is $-5(x+2)$, giving $(3x-5)(x+2)$. Ask ``what would you
tell Student A?'' --- not ``who got it wrong?''

\columnbreak
\textbf{Round 3 --- Zero Product Property: the crux (14 min).} Definition with the graph and
Example 3 ($x^2=2x+3$, roots $-1$, $3$, checked) in 3 minutes. Ladder (6 min):
\rung1\ $4$, $-6$ (the sign flip); \rung2\ $x^2-3x-10=0$: $5$, $-2$; \rung3\ the dorm rug,
$(x+8)(x-5)=0$: $5$ ft by $8$ ft, reject $-8$. Show me $x^2=5x$ (\textbf{the crux}): answer
\textbf{2}, $x=0$ or $5$. A \textbf{1} divided by $x$ and lost $x=0$ (the target misconception);
a \textbf{3} factored right but flipped a sign solving $x-5=0$; a \textbf{4} took a square root
of something. \textbf{The frame not to rush:} before revealing, ask a \textbf{1} and a
\textbf{2} to explain.
\end{multicols}
\end{skillbox}

\boxguard
\begin{skillbox}[Final Round (5 min)]{forestbg}
Four mixed problems covering all three rounds, alone, 5-minute timer: \textbf{1} $(x-3)(x-5)$;
\textbf{2} $3(x+2)(x-2)$ (GCF, then a difference of squares); \textbf{3} $(2x+3)(x-2)$;
\textbf{4} $(x-1)(x+2)=4\Rightarrow x^2+x-6=0$, $x=-3$ or $2$ (the crux's twin --- ``each factor
$=4$'' gives $5$ and $2$). \textbf{Bonus:} the egg-drop capsule, $-16(t-2)(t+1)=0$, lands at
$t=2$\,s. Students score themselves out of 4; ask for scores on fingers --- that is your exit
read. Say the \emph{Watch out} aloud as you reveal.
\end{skillbox}

\boxguard
\begin{skillbox}[Active Monitoring --- Watch For]{redbox}
\begin{itemize}[leftmargin=1.2em, nosep]
  \item \textbf{Round 1 \rung3:} no GCF --- not factored \emph{completely}.
  \item \textbf{Round 1 Show me:} $x^2+16=(x+4)(x-4)$.
  \item \textbf{Round 2:} the second group's sign; brackets that do not match (Find the mistake).
  \item \textbf{Round 3 Show me:} dividing by $x$ (\emph{the target misconception}).
  \item \textbf{Final Round 4:} setting each factor equal to $4$.
  \item \textbf{DeltaMath:} the $x^2=kx$ problem --- one root instead of two.
  \item \textbf{Attitude:} a student parked at \rung1\ with nothing written --- crouch, point at
        the example on the handout, ask for just the pair.
\end{itemize}
Cold-call prompts (after the fingers, never before): ``How do you know it is factored
\emph{completely}?'' \; ``What number does the property need on the right?''
\end{skillbox}

\begin{teachernote}[Slides]
Pacing: warm-up 5, rounds 13 / 13 / 14, Final Round 5, homework 10 --- 60 minutes. In each
round the definition and example together take \textbf{at most 3 minutes} --- if you catch
yourself explaining past that, move to the ladder and explain at desks. Reveal a ladder when
most of the room has \rung2; the \rung3\ students keep going. If behind, cut Example 1(c) and
shorten Round 1's Show me discussion --- never Round 3's crux. Students write in the handout's
notes column, which prints without the answers.
\end{teachernote}

\begin{teachernote}[Homework]

printed in the packet. Launch problem~1 aloud in the last 10 minutes (students on their
devices), then circulate with the Watch For list in mind. The $x^2=kx$ problem is the formative
check. Due the first class after two study halls; announce the date in class and on TurtleNet.
\end{teachernote}
