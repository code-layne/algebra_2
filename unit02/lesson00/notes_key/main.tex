\begin{objectivebox}
\textbf{By the end of this lesson, I will be able to\dots}
\begin{itemize}\setlength{\itemsep}{2pt}
  \item read the \emph{domain}, \emph{range}, \emph{$y$-intercept}, and \emph{$x$-intercept (zero)} of a line from its graph;
  \item find the \emph{slope} as a constant rate of change and classify the line as increasing, decreasing, or constant;
  \item state the intervals where the function is \emph{positive} or \emph{negative}, and justify the switch at the zero.
\end{itemize}
\end{objectivebox}

\begin{vocabbox}
\small
Fill in each term as we build it together.
\vspace{2pt}
\vocabans{Slope (rate of change)}{the constant change in output per unit change in input; $m=\frac{\Delta y}{\Delta x}=\frac{\text{rise}}{\text{run}}$}
\vocabans{$y$-intercept}{the point $(0,b)$ where the graph crosses the $y$-axis --- the output when $x=0$}
\vocabans{$x$-intercept (zero)}{the point $(a,0)$ where the graph crosses the $x$-axis; the input $a$ with $f(a)=0$}
\vocabans{Increasing / decreasing}{the graph rises ($m>0$) / falls ($m<0$) as you read it left to right}
\vocabans{Positive / negative interval}{the $x$-values where $f(x)>0$ (above the $x$-axis) / $f(x)<0$ (below it)}
\end{vocabbox}

\begin{hookbox}
A water tank drains at a steady rate. Its volume after $t$ minutes is $V(t) = -3t + 12$ gallons.
\begin{itemize}\setlength{\itemsep}{1pt}
  \item How much water did it start with (at $t=0$)? \ansline{$12$ gallons --- the $y$-intercept $(0,12)$.}
  \item How fast is it draining, and when is the tank empty? \ansline{Draining $3$ gal/min (slope $-3$); empty at $t=4$ min (the zero $(4,0)$).}
\end{itemize}
Every question about this tank is a question about a \emph{characteristic of its graph}. Let's name
them all.
\end{hookbox}

\begin{notesbox}{1. Linear Means a Constant Rate of Change}
Build a table for $f(x) = 2x - 4$:
\begin{center}
\renewcommand{\arraystretch}{1.25}
\begin{tabular}{|c|c|c|c|c|}
\hline
$x$ & $0$ & $1$ & $2$ & $3$ \\ \hline
$f(x)$ & $-4$ & $-2$ & $0$ & $2$ \\ \hline
\end{tabular}
\end{center}
Each step of $+1$ in $x$ changes the output by the \emph{same} amount, $\textcolor{keyred}{\mathbf{+2}}$. A function
with a \emph{constant} rate of change is called a \textbf{linear function}, and its graph is a
\ans{straight line}. We write it in the form $f(x) = mx + b$, where $m$ is the \ans{slope} and $b$ is
the value of $f(0)$.
\end{notesbox}

\boxguard[24]
\begin{notesbox}{2. Reading a Line: The Six Characteristics}
Here is the graph of $f(x) = 2x - 4$. We will read \emph{six} features straight off the picture.
\begin{center}
\begin{tikzpicture}[scale=0.52]
  \lgrid{-2}{6}{-6}{6}
  \draw[very thick, forest] (-1,-6)--(5,6);
  % rise/run triangle from (2,0) to (3,2)
  \draw[semithick, navy, dashed] (2,0)--(3,0)--(3,2);
  \node[navy, font=\scriptsize] at (2.5,-0.55){run $1$};
  \node[navy, font=\scriptsize, right] at (3.05,1){rise $2$};
  % intercept dots
  \fill[charcoal] (0,-4) circle (3pt) node[left, font=\scriptsize]{$(0,-4)$};
  \fill[charcoal] (2,0) circle (3pt)  node[above left, font=\scriptsize]{$(2,0)$};
\end{tikzpicture}
\end{center}

\begin{enumerate}[leftmargin=1.6em, itemsep=3pt]
  \item \textbf{Domain} (allowed inputs): the line runs left and right forever, so the domain is
        \ans{all real numbers, $(-\infty,\infty)$}.
  \item \textbf{Range} (resulting outputs): the line runs up and down forever, so the range is
        \ans{all real numbers, $(-\infty,\infty)$}.
  \item \textbf{$y$-intercept} (crosses the $y$-axis): the point $(0,\,\textcolor{keyred}{\mathbf{-4}})$. Find it by
        setting $x = 0$.
  \item \textbf{$x$-intercept / zero} (crosses the $x$-axis): find it by solving $2x - 4 = 0$.
        \begin{work}
          2x-4 &= 0 \\
            2x &= 4 \\
             x &= 2
        \end{work}
        So the zero is the point $(\textcolor{keyred}{\mathbf{2}},\,0)$. We call it the \emph{zero} because
        $f(\textcolor{keyred}{\mathbf{2}}) = 0$.
  \item \textbf{Slope} (rate of change): count $\dfrac{\text{rise}}{\text{run}} =
        \dfrac{\textcolor{keyred}{\mathbf{2}}}{\textcolor{keyred}{\mathbf{1}}} = \textcolor{keyred}{\mathbf{2}}$. It is the same between \emph{any} two
        points --- that is what ``constant rate of change'' means.
  \item \textbf{Sign of the function:} the graph is \emph{above} the $x$-axis (so $f(x)>0$) for
        $x \textcolor{keyred}{\mathbf{{}>2}}$, and \emph{below} it (so $f(x)<0$) for $x \textcolor{keyred}{\mathbf{{}<2}}$. The sign switches
        exactly at the \ans{zero ($x$-intercept)}.
\end{enumerate}
\end{notesbox}

\boxguard[24]
\begin{notesbox}{3. Rising, Falling, or Flat}
Read a line \emph{left to right}. Its direction is decided by the \emph{sign of the slope}.
\begin{center}
\begin{tikzpicture}[scale=0.42]
  % g(x) = -x + 3  (decreasing)
  \begin{scope}
    \lgrid{-2}{5}{-2}{5}
    \draw[very thick, forest] (-1,4)--(4,-1);
    \fill[charcoal] (0,3) circle (3pt);
    \fill[charcoal] (3,0) circle (3pt);
    \node[charcoal, font=\scriptsize] at (1.5,-2.8){$g(x) = -x + 3$};
  \end{scope}
  % h(x) = 2  (constant)
  \begin{scope}[xshift=9cm]
    \lgrid{-2}{5}{-2}{5}
    \draw[very thick, forest] (-1.5,2)--(4.5,2);
    \fill[charcoal] (0,2) circle (3pt);
    \node[charcoal, font=\scriptsize] at (1.5,-2.8){$h(x) = 2$};
  \end{scope}
\end{tikzpicture}
\end{center}
\begin{itemize}[leftmargin=1.5em, itemsep=2pt]
  \item $f(x) = 2x-4$ has slope $2 > 0$, so it \emph{rises}: it is \textbf{increasing} on
        \ans{all real numbers, $(-\infty,\infty)$}.
  \item $g(x) = -x+3$ has slope \ans{$-1$}, so it \emph{falls}: it is \textbf{decreasing}. Its
        $y$-intercept is $(0,\textcolor{keyred}{\mathbf{3}})$ and its zero is $(\textcolor{keyred}{\mathbf{3}},0)$.
  \item $h(x) = 2$ has slope \ans{$0$}, so it is \textbf{constant} (flat) --- neither increasing
        nor decreasing. Its range is the single value \ans{$2$ (that is, $\{2\}$)}, and it has \emph{no}
        $x$-intercept.
\end{itemize}
\end{notesbox}

\begin{practicebox}
The graph below is $p(x) = \tfrac{1}{2}x + 1$. Read each characteristic and write it on the line.
\begin{center}
\begin{tikzpicture}[scale=0.5]
  \lgrid{-5}{5}{-3}{4}
  \draw[very thick, forest] (-5,-1.5)--(5,3.5);
  \fill[charcoal] (0,1) circle (3pt) node[above left, font=\scriptsize]{$(0,1)$};
  \fill[charcoal] (-2,0) circle (3pt) node[below right, font=\scriptsize]{$(-2,0)$};
\end{tikzpicture}
\end{center}
\begin{enumerate}[leftmargin=1.6em, itemsep=3pt]
  \item Domain: \ans{all real numbers} \qquad Range: \ans{all real numbers}
  \item $y$-intercept: \ans{$(0,1)$} \qquad $x$-intercept (zero): \ans{$(-2,0)$}
  \item Slope $=$ \ans{$\tfrac{1}{2}$} \qquad Increasing or decreasing? \ans{increasing}
  \item Positive when $x$ \ans{$>-2$}; \quad negative when $x$ \ans{$<-2$}.
\end{enumerate}
\end{practicebox}
