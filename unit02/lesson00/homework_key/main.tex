  \item \textbf{Model it.} A punt sends a football up from a tee $4$ feet off the ground:
        $h(t)=-16t^2+48t+4$, with $t$ in seconds and $h$ in feet. Its graph turns at $t=1.5$.
    \begin{enumerate}[label=(\alph*), leftmargin=1.9em, itemsep=3pt]
      \item Find $h(1.5)$.
        \begin{work}
          h(1.5) &= -16(1.5)^2 + 48(1.5) + 4 \\
                 &= -36 + 72 + 4 \\
                 &= 40
        \end{work}
      \item The maximum height is \inblank{\blank{1.0cm}}{\gradedblank[id=q5a, type=short, points=1, answer={40}, check=numeric]{1.0cm-2pt}{12.5pt}} ft --- an \inblank{\blank{1.4cm}}{\gradedblank[id=q5b, type=short, points=1, answer={output}, check=text]{1.4cm-2pt}{12.5pt}} --- reached at $t=$
            \inblank{\blank{1.0cm}}{\gradedblank[id=q5c, type=short, points=1, answer={1.5}, check=numeric]{1.0cm-2pt}{12.5pt}} s --- an \inblank{\blank{1.4cm}}{\gradedblank[id=q5d, type=short, points=1, answer={input}, check=text]{1.4cm-2pt}{12.5pt}}.
      \item $h(1.5)=40$ means \rubricbox{\gradedblank[id=q5e, type=rubric, points=2, display={the ball is $40$ ft up, its highest point, $1.5$ s after the punt}]{\linewidth}{12.5pt}}
      \item The ball lands near $t=3.1$ s, but the function keeps going: as $t$ grows its outputs
            \inblank{\blank{1.3cm}}{\gradedblank[id=q5f, type=short, points=1, answer={fall}, check=text]{1.3cm-2pt}{12.5pt}}. Why does the model stop making sense after it lands? \rubricbox{\gradedblank[id=q5g, type=rubric, points=1, display={height cannot be negative}]{\linewidth}{12.5pt}}
    \end{enumerate}

  \item \textbf{Multiple choice (SOL-style).} A ball's height in feet after $t$ seconds is
        $h(t)=-16t^2+96t$; its graph has vertex $(3,144)$ and zeros $t=0$ and $t=6$. What is the
        ball's \emph{maximum height}?
    \begin{enumerate}[label=\Alph*., leftmargin=2.2em, itemsep=0pt, topsep=2pt]
      \item \letterbox{\gradedblank[id=q6a, type=choice, points=1, answer={B}]{0.4in}{12.5pt}}$3$ feet
      \item \ans{$144$ feet} \hfill \textcolor{keyred}{\textbf{$\leftarrow$ correct}}
      \item $6$ feet
      \item $96$ feet
    \end{enumerate}
    Say in one sentence how you ruled out one of the others.
    \gradedlines{2}{\gradedblank[id=q6b, type=rubric, points=2, display={The vertex $(3,144)$ gives height $144$ ft at time $t=3$ s, so the maximum is $144$. A is $3$ --- the \emph{input}, in seconds, not a height; C is the landing time, D a coefficient.}]{\linewidth}{\gradedlinesheight}}
\end{enumerate}
