% QUADRATIC = CONSTANT SECOND DIFFERENCE
\begin{frame}[t, plain]
  \forestheader{Quadratic means a constant second difference}
  \sectionlabel{Definition}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      For $f(x)=x^2-2x-3$, the outputs fall, bottom out, then rise --- \emph{not} a line.
      \\[6pt]
      The \emph{second} differences are constant ($+2$): the fingerprint of a quadratic
      $f(x)=ax^2+bx+c$. Its graph is a \textbf{parabola}.
    \end{column}
    \begin{column}{0.44\textwidth}
      \centering
      \renewcommand{\arraystretch}{1.2}
      \begin{tabular}{|c|c|c|c|c|}
        \hline
        $x$ & $0$ & $1$ & $2$ & $3$ \\ \hline
        $f(x)$ & $-3$ & $-4$ & $-3$ & $0$ \\ \hline
      \end{tabular}\\[6pt]
      {\footnotesize 1st diffs: $-1,+1,+3$ \quad 2nd diffs: \textbf{\color{forest}$+2$}}
    \end{column}
  \end{columns}
\end{frame}

% THE CHARACTERISTICS
\begin{frame}[t, plain]
  \forestheader{Reading a parabola: the characteristics}
  \sectionlabel[goldacc]{$f(x) = x^2 - 2x - 3$}
  \begin{columns}[T]
    \begin{column}{0.42\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.4]
        \lgrid{-3}{5}{-5}{6}
        \draw[navy, dashed, thick] (1,-5)--(1,6);
        \begin{scope}
          \clip (-3,-5) rectangle (5,6);
          \draw[very thick, forest, domain=-2.2:4.2, samples=80, smooth]
            plot (\x,{(\x)*(\x) - 2*(\x) - 3});
        \end{scope}
        \fill[charcoal] (1,-4) circle (3.5pt) node[below right, font=\tiny]{$(1,-4)$};
        \fill[charcoal] (-1,0) circle (3.5pt);
        \fill[charcoal] (3,0) circle (3.5pt);
        \fill[charcoal] (0,-3) circle (3.5pt);
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.58\textwidth}
      \footnotesize
      \begin{enumerate}\setlength{\itemsep}{3pt}
        \item \textbf{Vertex:} $(1,-4)$ --- a \textbf{minimum}; min \emph{value} $-4$
        \item \textbf{Axis of symmetry:} $x=1$
        \item \textbf{$y$-intercept:} $(0,-3)$; \textbf{zeros:} $x=-1,\,3$
        \item \textbf{Domain:} all reals \quad \textbf{Range:} $y\ge-4$
        \item \textbf{Decreasing} on $(-\infty,1)$, \textbf{increasing} on $(1,\infty)$
        \item \textbf{Negative} on $-1<x<3$; \textbf{end behavior:} both arms $\to+\infty$
      \end{enumerate}
    \end{column}
  \end{columns}
\end{frame}

% SYMMETRY as reasoning + decreasing vs negative
\begin{frame}[t, plain]
  \forestheader{Two ideas students fuse --- keep them apart}
  \sectionlabel[goldacc]{Key distinctions}
  \begin{itemize}\setlength{\itemsep}{7pt}
    \item \textbf{Max/min \emph{value}} is a $y$-value (the output at the vertex), \emph{not} a point.
    \item \textbf{Decreasing} (direction of travel) is \emph{not} \textbf{negative} (output sign):
          $f(x)=x^2-2x-3$ is decreasing on all of $(-\infty,1)$ but negative only on $(-1,3)$.
  \end{itemize}
  \vspace{2pt}
  \begin{block}{Symmetry is the reasoning tool}
    A parabola mirrors across its axis, so inputs the same distance from the axis share an output ---
    that is \emph{why} there are two matching zeros and two matching heights.
  \end{block}
\end{frame}
