% LINEAR = CONSTANT RATE OF CHANGE
\begin{frame}[t, plain]
  \forestheader{Linear means a constant rate of change}
  \sectionlabel{Definition}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      For $f(x) = 2x - 4$, every $+1$ in $x$ adds the same $+2$ to $y$.
      \\[6pt]
      That constant change \emph{is} the \textbf{slope}, and constant slope makes the graph a
      straight \textbf{line}: \[ f(x) = mx + b. \]
    \end{column}
    \begin{column}{0.44\textwidth}
      \centering
      \renewcommand{\arraystretch}{1.2}
      \begin{tabular}{|c|c|c|c|c|}
        \hline
        $x$ & $0$ & $1$ & $2$ & $3$ \\ \hline
        $f(x)$ & $-4$ & $-2$ & $0$ & $2$ \\ \hline
      \end{tabular}\\[6pt]
      {\footnotesize each step: \textbf{\color{forest}$+2$}}
    \end{column}
  \end{columns}
\end{frame}

% THE SIX CHARACTERISTICS
\begin{frame}[t, plain]
  \forestheader{Reading a line: six characteristics}
  \sectionlabel[goldacc]{$f(x) = 2x - 4$}
  \begin{columns}[T]
    \begin{column}{0.40\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.42]
        \lgrid{-2}{6}{-6}{6}
        \draw[very thick, forest] (-1,-6)--(5,6);
        \draw[semithick, navy, dashed] (2,0)--(3,0)--(3,2);
        \fill[charcoal] (0,-4) circle (3.5pt) node[left, font=\tiny]{$(0,-4)$};
        \fill[charcoal] (2,0) circle (3.5pt) node[above left, font=\tiny]{$(2,0)$};
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.58\textwidth}
      \footnotesize
      \begin{enumerate}\setlength{\itemsep}{3pt}
        \item \textbf{Domain:} all reals \quad \textbf{Range:} all reals
        \item \textbf{$y$-intercept:} $(0,-4)$
        \item \textbf{$x$-intercept / zero:} $(2,0)$, where $f(x)=0$
        \item \textbf{Slope:} $\tfrac{\text{rise}}{\text{run}}=\tfrac{2}{1}=2$
        \item \textbf{Increasing} (slope $>0$)
        \item \textbf{Positive} for $x>2$, \textbf{negative} for $x<2$
      \end{enumerate}
    \end{column}
  \end{columns}
\end{frame}

% POSITIVE vs INCREASING
\begin{frame}[t, plain]
  \forestheader{``Positive'' is not ``increasing''}
  \sectionlabel[goldacc]{The key distinction}
  \begin{itemize}\setlength{\itemsep}{7pt}
    \item \textbf{Positive / negative} asks where the graph \emph{sits}: above the $x$-axis
          ($f(x)>0$) or below it ($f(x)<0$). The sign switches at the \textbf{zero}.
    \item \textbf{Increasing / decreasing} asks which way the graph \emph{moves} as you read
          left to right --- the sign of the \textbf{slope}.
  \end{itemize}
  \vspace{2pt}
  \begin{block}{They are independent}
    A \emph{decreasing} line can still be \emph{positive}: $g(x)=-\tfrac12 x+2$ falls everywhere,
    yet $g(x)>0$ for all $x<4$.
  \end{block}
\end{frame}
