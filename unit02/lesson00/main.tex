\begin{tcolorbox}[colback=forestbg, colframe=forest, boxrule=0.9pt, arc=2mm,
    left=2mm, right=2mm, top=1.5mm, bottom=1.5mm]
    \textbf{Primary Objective:} Students can read the graph of a quadratic function --- a
    \emph{parabola} --- and name its key characteristics: the \emph{vertex} (the \emph{turning point},
    a \emph{maximum} or \emph{minimum}), the \emph{axis of symmetry}, the \emph{domain} and
    \emph{range}, the \emph{$y$-intercept} and \emph{zeros}, the intervals where it is
    \emph{increasing}/\emph{decreasing} and \emph{positive}/\emph{negative}, and its \emph{end
    behavior} --- and justify each feature from the graph, using \emph{symmetry} as the reasoning tool.
    This extends the Unit~1 ``read-a-graph'' toolkit to a curve that turns.
    \\[2pt]
    \textbf{Standards (2023 VA SOL):} \textbf{A2.F.2} --- investigate and analyze the characteristics
    of functions graphically (quadratics are polynomial functions). This Lesson~0 revisits
    \textbf{A2.F.2a} (domain, range, zeros, intercepts from a graph) and \textbf{A2.F.2c} (increasing/
    decreasing intervals) on a parabola, and \emph{introduces} \textbf{A2.F.2d} (location and value of
    the absolute maximum/minimum --- the vertex) and \textbf{A2.F.2g} (end behavior). \textbf{A2.F.2f}
    (determine $f(x)$ from a graph and interpret $x$ and $f(x)$ in context) is applied in the
    projectile model.
\end{tcolorbox}

\begin{skillbox}[Priority Ideas \& Skills]{goldbox}
\begin{tabularx}{\linewidth}{@{}X|X@{}}
\textbf{Priority Ideas \& Skills} & \textbf{Key Understandings (the ``why'')} \\[2pt]
\hline
\vspace{-6pt}
\begin{itemize}[leftmargin=1.1em, nosep, after=\vspace{2pt}]
  \item Identify the \textbf{vertex} (turning point) and classify it as a \textbf{max} or \textbf{min}; read the \textbf{max/min value}.
  \item Name the \textbf{axis of symmetry} and use \textbf{symmetry} to justify readings.
  \item Read \textbf{domain}, \textbf{range}, the \textbf{$y$-intercept}, and the \textbf{zeros} (up to two).
  \item Give the \textbf{increasing/decreasing} and \textbf{positive/negative} intervals.
  \item Describe the \textbf{end behavior} of both arms.
\end{itemize}
&
\vspace{-6pt}
\begin{itemize}[leftmargin=1.1em, nosep, after=\vspace{2pt}]
  \item A parabola \emph{turns} once; the vertex is its single highest or lowest point --- the absolute max or min.
  \item The \emph{sign of $a$} controls everything at once: opening direction, max vs.\ min, and end behavior.
  \item ``Max/min \emph{value}'' is an \emph{output} ($y$), not a point --- a frequent confusion.
  \item ``Decreasing'' (direction of travel) is \emph{not} ``negative'' (output sign); a parabola can decrease while positive.
  \item Symmetry is the \emph{reason} equal-distance inputs share an output; it explains the two zeros and the vertex.
\end{itemize}
\end{tabularx}
\end{skillbox}

\begin{skillbox}[{Vocabulary, Concepts, \& Theorems}]{sky}
\renewcommand{\arraystretch}{1.3}
\begin{tabularx}{\linewidth}{@{}l X@{}}
\textbf{Quadratic function} & A function $f(x)=ax^2+bx+c$ with $a\ne0$; its highest power is $x^2$ and its outputs have a \emph{constant second difference}. \\
\textbf{Parabola} & The symmetric U-shaped graph of a quadratic; opens \emph{up} if $a>0$, \emph{down} if $a<0$. \\
\textbf{Vertex (turning point)} & The point where the parabola turns --- its lowest point (opens up) or highest point (opens down). \\
\textbf{Axis of symmetry} & The vertical line through the vertex; the two halves of the parabola are mirror images across it. \\
\textbf{Maximum / minimum value} & The $y$-value at the vertex: the greatest output (max, opens down) or least output (min, opens up). \\
\textbf{Zeros ($x$-intercepts)} & Inputs where $f(x)=0$; a parabola has two, one, or no real zeros. \\
\textbf{End behavior} & What outputs do as $x\to\pm\infty$: both arms up ($a>0$) or both arms down ($a<0$). \\
\textbf{Even symmetry} & The parent $y=x^2$ satisfies $f(-x)=f(x)$: it is a mirror image across the $y$-axis. \\
\end{tabularx}
\end{skillbox}

\begin{fixedskillbox}[Activate Prior Knowledge \& Spiral Review]{forestbg}
    The Warm-Up rehearses the skills this lesson stands on: \textbf{(1)} squaring signed numbers
    correctly (the $-3^2$ vs.\ $(-3)^2$ trap); \textbf{(2)} evaluating a quadratic $f(x)=x^2-2x-3$ at
    several inputs; and \textbf{(3)} the Unit~1 read-a-graph toolkit (intercepts, opening direction) on
    a \emph{curve}. Debrief item~2 aloud: the equal outputs at inputs equidistant from the middle are
    the parabola's \emph{symmetry}, and the middle input is the axis of symmetry / vertex
    $x$-coordinate we formalize today.
\end{fixedskillbox}

\begin{skillbox}[Hook]{forestbg}
    A ball is thrown upward; its height after $t$ seconds is $h(t) = -16t^2 + 32t + 48$ feet. Show the
    parabola (rising, turning, falling). Pose the questions that name every feature we will study:
    \textbf{``How high was it released?''} (the $y$-intercept $(0,48)$),
    \textbf{``When and how high is its peak?''} (the vertex $(1,64)$ --- a \emph{maximum}),
    \textbf{``When does it hit the ground?''} (the positive zero $(3,0)$),
    \textbf{``When is it rising vs.\ falling?''} (increasing before $t=1$, decreasing after), and
    \textbf{``Why is it at the same height at $t=0.5$ and $t=1.5$?''} (symmetry about $t=1$).
    Land the point: \emph{every question about this flight is a question about a characteristic of the
    parabola.}
\end{skillbox}

\begin{skillbox}[Lesson]{forestbg}
\begin{multicols}{2}
\textbf{1. What makes a function quadratic.} Build a table for $f(x)=x^2-2x-3$ and read down the
outputs: they fall, bottom out, then rise --- \emph{not} a constant change, so \emph{not} a line. The
\emph{second} differences are constant ($+2$), the fingerprint of a quadratic. Name the form
$f(x)=ax^2+bx+c$ and the graph: a \textbf{parabola}.

\textbf{2. Opening direction.} The sign of $a$ decides the shape. $a>0$ opens \emph{up} (a bowl, vertex
is a \emph{minimum}); $a<0$ opens \emph{down} (a dome, vertex is a \emph{maximum}). Tie this to end
behavior immediately: both arms follow the opening.

\textbf{3. The vertex and axis of symmetry.} On $f(x)=x^2-2x-3$, mark the turning point $(1,-4)$: this
is the \textbf{vertex}, and the vertical line $x=1$ is the \textbf{axis of symmetry}. Stress language:
the \emph{minimum value} is the \emph{output} $-4$, not the point.

\columnbreak

\textbf{4. Intercepts and zeros.} Set $x=0$ for the \textbf{$y$-intercept} $(0,-3)$. The
\textbf{zeros} are where the curve crosses the $x$-axis, $x=-1$ and $x=3$. \textbf{Why can a parabola
have two zeros when a line has one?} Because it turns --- it can cross the axis on the way down and
again on the way up.

\textbf{5. Intervals.} Read left to right: $f$ is \textbf{decreasing} on $(-\infty,1)$ and
\textbf{increasing} on $(1,\infty)$ --- it turns at the vertex. Separately, $f$ is \textbf{negative}
\emph{between} the zeros ($-1<x<3$) and \textbf{positive} outside them. Hammer the distinction:
decreasing $\ne$ negative.

\textbf{6. Bring it home.} Return to the ball $h(t)=-16t^2+32t+48$ and label all features in context,
so the vocabulary describes a real quantity. Use symmetry to predict equal heights. Preview: Lesson~2.1
uses these features to \emph{graph} quadratics from vertex, standard, and intercept forms.
\end{multicols}
\end{skillbox}

\begin{skillbox}[Active Monitoring]{redbox}
Circulate during the notes practice and Tier work. Watch for and correct:
\begin{itemize}[leftmargin=1.2em, nosep]
  \item reporting the max/min \emph{value} as a point $(h,k)$ instead of the $y$-value $k$;
  \item fusing ``decreasing'' with ``negative'' --- a parabola is decreasing on one whole side yet only negative between its zeros;
  \item reading the vertex $x$-coordinate as the range, or giving the range as ``all reals'' for a parabola;
  \item saying only one arm rises --- forgetting that end behavior sends \emph{both} arms the same way.
\end{itemize}
Cold-call prompts: ``Is the vertex a max or a min --- and how does the sign of $a$ tell you?'' ``Name
the axis of symmetry, then predict a matching pair of outputs.'' ``Where is $f$ negative, and where is
it decreasing --- are those the same interval?''
\end{skillbox}

\begin{skillbox}[Group Work \& Differentiation]{redbox}
\textbf{Reading Parabolas} activity (tiered; mirrors the notes).
\begin{multicols}{3}
\textbf{Tier R --- Remediate.} From one pre-drawn upward parabola with integer vertex and zeros, read
the opening direction, vertex, axis of symmetry, zeros, $y$-intercept, and the minimum value.

\columnbreak
\textbf{Tier A --- Approaching.} For two parabolas (one up, one down), report the full feature list ---
vertex, max/min value, axis, zeros, range, increasing interval, end behavior --- and \emph{justify},
using the axis of symmetry, why two equidistant inputs share an output.

\columnbreak
\textbf{Tier E --- Extension.} Match the projectile $h(t)=-16t^2+32t+48$ to its graph, interpret every
feature \emph{in context}, reason about the meaningful domain, and use symmetry (no algebra) to find a
matching-height time.
\end{multicols}
\end{skillbox}

\begin{skillbox}[Individual Work \& Assessment]{redbox}
\textbf{Exit Ticket} (independent, no notes): from one graphed parabola $f(x)=x^2+2x-3$, read the
vertex, axis, zeros, and $y$-intercept; classify the vertex as max/min with its value and state the
range; \textbf{justify} using the sign of $a$; and answer an \textbf{SOL-style multiple-choice} item
that separates end behavior, range, and the negative interval. Collect and sort into ``got it /
point-vs-value slip / sign-of-$a$ slip / interval slip'' to decide whether Lesson~2.1 opens with a
quick re-teach.
\end{skillbox}

\begin{skillbox}[Reinforcement \& Extension]{goldbox}
\textbf{Homework:} identify a quadratic from a table's constant \emph{second} difference and read its
vertex from symmetry; read the full feature list from an upward and a downward parabola; and one
justification item on why both end-behavior arms go the same direction. \textbf{Extension:} interpret
every feature of a dolphin-leap model $h(t)=-16t^2+16t+32$ in context, including the meaningful domain.
\textbf{Preview:} Lesson~2.1, \emph{Graphing Quadratic Functions} --- using vertex, standard, and
intercept forms to find these characteristics from the equation and to transform the parent $y=x^2$.
\end{skillbox}

\begin{teachernote}[Warm-Up]
Item~1 pre-empts the year's most common error, $-3^2$ vs.\ $(-3)^2$. Item~2 is the bridge to the
lesson: the equal outputs at $x=-1$ and $x=3$ (both $0$) and at $x=-2,4$ (both $5$) are the parabola's
\emph{symmetry}, and the middle value $x=1$ is the axis of symmetry / vertex $x$-coordinate we name
today. Item~3 rehearses the Unit~1 read-a-graph toolkit on a curve instead of a line.
\end{teachernote}

\begin{teachernote}[Guided Notes]
The whole lesson is one move --- \emph{read a feature off the picture} --- now on a curve that turns.
Three predictable errors: (1) giving the min/max as a point instead of the $y$-value (item~3 in \S2);
(2) fusing ``decreasing'' with ``negative'' (the parabola is decreasing on $(-\infty,1)$ but only
negative on $(-1,3)$); and (3) writing the vertex $x$-coordinate as the range. \S3 exists to lock in
that the \emph{sign of $a$} controls opening direction, min-vs-max, and end behavior all at once. Keep
returning to the axis of symmetry as the justification tool: equal steps left and right share an
output.
\end{teachernote}

\begin{teachernote}[Group Activity]
Tier~A's justify item is the heart of the lesson: symmetry is the \emph{reason} equal-distance inputs
match, not a coincidence. Watch for two errors across tiers --- reporting the max/min as a point rather
than a $y$-value, and reading the vertex $x$-coordinate as the range. In Tier~E, item~2 (meaningful
domain) is the modeling payoff; accept any clear reason that negative time or negative height is
impossible. Item~3 is symmetry used as a shortcut --- no algebra required.
\end{teachernote}

\begin{teachernote}[Exit Ticket]
Sort responses into ``got it / point-vs-value slip (min as a point) / sign-of-$a$ slip / interval
slip.'' The telltale error is item~3D or a range of ``all reals'' --- both mean the student has not
connected $a>0$ to ``opens up, has a minimum, both arms rise.'' If several miss it, reopen Lesson~2.1
by shading above/below the axis and marking the vertex on one parabola.
\end{teachernote}

\begin{teachernote}[Homework]
Item~1 is the algebraic fingerprint of a quadratic --- constant \emph{second} differences (here $+2$,
so $a=1$) --- paired with reading the vertex from symmetry rather than from a graph. Item~4 is the
conceptual target: accept any answer tying the single turning point (plus symmetry) to same-direction
arms, and $a$'s sign to which direction. In the Extension, the usual slip is a domain of ``all reals'';
push students to the leap's start ($t=0$) and splashdown ($t=2$).
\end{teachernote}
