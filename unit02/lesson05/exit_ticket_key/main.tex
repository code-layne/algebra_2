No notes. Show your work in the space provided.

\begin{enumerate}[leftmargin=1.8em, itemsep=14pt]
  \item \textbf{Describe the association.} A scatterplot is shown.
        \begin{center}
        \begin{tikzpicture}[scale=0.45]
          \lgrid{-0.5}{7}{-0.5}{7}
          \dpt{1}{6}\dpt{2}{6}\dpt{3}{4}\dpt{4}{3}\dpt{5}{2}\dpt{6}{1}
        \end{tikzpicture}
        \end{center}
        Direction: \ans{negative} \; (positive / negative / none). \quad Is the correlation coefficient $r$
        closer to $-1$, $0$, or $1$? \ans{$-1$} \; Explain in one phrase.
        \ans{the points fall in a tight, nearly straight line.}

  \item \textbf{Interpret and predict.} A car's \emph{age} ($x$, years) and \emph{value} ($y$, in \$1000s)
        over ages $1$ to $10$ give the line of best fit
        \[ \hat y=-1.5x+24. \]
        \begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=6pt]
          \item Interpret the slope $-1.5$ in context.
                \ans{each additional year of age is associated with about \$1{,}500 less in value.}
          \item Predict the value at $x=6$ years: $\hat y={\color{keyred}\mathbf{15}}$ (that is
                \$\ans{15{,}000}). \; Interpolation or extrapolation? \ans{interpolation ($6$ is within $1$--$10$)}
        \end{enumerate}

  \item \textbf{Multiple choice (SOL-style).} A data set has correlation coefficient $r=-0.92$. Which best
        describes the linear association?
        \begin{enumerate}[label=(\Alph*), leftmargin=1.8em, itemsep=2pt]
          \item Weak negative
          \item Strong negative \quad\textcolor{keyred}{\textbf{$\leftarrow$ correct}}
          \item Strong positive
          \item No correlation
        \end{enumerate}
        \begin{itemize}[leftmargin=1.4em, nosep]
          \item The \emph{sign} ($-$) gives the direction (negative); the \emph{size} $|{-}0.92|=0.92$ is
                close to $1$, so the fit is \emph{strong}. Negative does not mean weak.
        \end{itemize}
\end{enumerate}

\begin{teachernote}
Item~1 checks the two-part read: direction from the sign of the trend (falling $\Rightarrow$ negative) and
strength from how tightly the points cluster ($r$ near $-1$). Item~2 is the interpret-and-predict target:
part (a) must carry \emph{units and context} (\$1{,}500 less \emph{per year of age}); part (b) tests
substitution and the interpolation call ($x=6$ is inside $1$--$10$). Item~3 is the classic
direction-vs-strength trap --- students who pick (A) read ``negative'' as ``weak.'' Sort exit tickets into
``got it / direction-vs-strength slip / $r$-reading slip / interpolation-extrapolation slip / causation
slip'' to decide whether the unit review needs a two-minute re-teach before the Unit~2 test.
\end{teachernote}
