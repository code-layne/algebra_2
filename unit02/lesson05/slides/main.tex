% ═══ ROUND 1 — LINEAR–QUADRATIC SYSTEMS BY SUBSTITUTION ═════════════════════
\begin{frame}[t, plain]
  \forestheader{Definition: a linear--quadratic system}
  \sectionlabel{Round 1 --- where a line meets a parabola}
  \begin{columns}[T]
    \begin{column}{0.6\textwidth}
      \begin{block}{Solution of a system}
        A \textbf{solution} of a \textbf{system} is an ordered pair $(x,y)$ that makes
        \emph{both} equations true --- a point where the graphs \textbf{intersect}.
      \end{block}
      \vspace{2pt}
      \begin{block}{Solving by substitution}
        \begin{enumerate}\setlength{\itemsep}{0pt}
          \item Both equations say $y=\ldots$: set the right sides equal.
          \item Move everything to one side; solve the quadratic.
          \item Put each $x$ into the \emph{line} to get its $y$; check the point in both.
        \end{enumerate}
      \end{block}
    \end{column}
    \begin{column}{0.36\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.33]
        \lgrid{-2}{5}{-5}{4}
        \begin{scope}\clip (-2,-5) rectangle (5,4);
          \draw[very thick, forest, domain=-1.9:3.9, samples=70, smooth]
            plot (\x,{\x*\x-2*\x-3});
          \draw[thick, navy] (-2,-5)--(5,2);
        \end{scope}
        \fill[redacc] (0,-3) circle (4pt) node[left, font=\tiny, charcoal]{$(0,-3)$};
        \fill[redacc] (3,0) circle (4pt) node[above left, font=\tiny, charcoal]{$(3,0)$};
      \end{tikzpicture}\\[-2pt]
      {\scriptsize $y=x^2-2x-3$ and $y=x-3$ \\ meet at \textbf{two} points.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Example 1: solve $y=x^2-2x-3$ and $y=x-3$}
  \sectionlabel{Watch --- 2 minutes}
  \[
  \begin{aligned}
    x^2-2x-3 &= x-3            && \text{set the } y\text{'s equal} \\
    x^2-3x &= 0                && \text{everything on one side --- the warm-up!} \\
    x(x-3) &= 0                && \text{factor} \\
    x=0 \quad\text{or}\quad x &= 3 && \text{Zero Product Property}
  \end{aligned}
  \]
  \pause
  Back into the line $y=x-3$: \; $x=0\Rightarrow y=-3$; \quad $x=3\Rightarrow y=0$.
  \quad Solutions: $\mathbf{(0,-3)}$ and $\mathbf{(3,0)}$.
  \pause
  \\[4pt]
  \emph{Check $(3,0)$ in the parabola:} $9-6-3=0$ \checkmark \quad --- the two red points on
  the graph.
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Your turn}
  \sectionlabel[goldacc]{5 minutes --- start at one star and climb}
  Solve each system. Answer with ordered pairs.
  \begin{itemize}\setlength{\itemsep}{6pt}
    \item[\rung1] $y=x^2$ \; and \; $y=x+2$
    \item[\rung2] $y=x^2+4x+1$ \; and \; $y=2x+4$
    \item[\rung3] $y=x^2-3$ \; and \; $x+y=3$ \quad (get $y$ alone first)
  \end{itemize}
  \vspace{2pt}
  \reveal{%
    \begin{block}{Check}
      \rung1\ $x^2-x-2=0$, $(x-2)(x+1)=0$: $(2,4)$ and $(-1,1)$ \\[1pt]
      \rung2\ $x^2+2x-3=0$, $(x+3)(x-1)=0$: $(1,6)$ and $(-3,-2)$ \\[1pt]
      \rung3\ $y=3-x$, so $x^2+x-6=0$, $(x+3)(x-2)=0$: $(2,1)$ and $(-3,6)$
    \end{block}}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Find the mistake}
  \sectionlabel[goldacc]{2 minutes --- Student A turned this in. What is missing? Finish it}
  \textbf{Student A} solved the system $y=x^2+2x-5$ and $y=4x-2$:
  \[
  \begin{aligned}
    x^2+2x-5 &= 4x-2 \\
    x^2-2x-3 &= 0 \\
    (x-3)(x+1) &= 0 \\
    &\text{``The solutions are } x=3 \text{ and } x=-1.\text{''}
  \end{aligned}
  \]
  \reveal{%
    \begin{block}{The mistake}
      Student A stopped at the $x$-values. A solution of a system is a \emph{point}: in the line,
      $y=4(3)-2=10$ and $y=4(-1)-2=-6$, so $(3,10)$ and $(-1,-6)$. \; Check in the parabola:
      $9+6-5=10$ \checkmark, \; $1-2-5=-6$ \checkmark.
    \end{block}}
\end{frame}

% ═══ ROUND 2 — HOW MANY SOLUTIONS? ══════════════════════════════════════════
\begin{frame}[t, plain]
  \forestheader{Definition: the number of solutions}
  \sectionlabel{Round 2 --- cross, touch, or miss}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \begin{block}{Number of solutions}
        A line and a parabola can \textbf{cross} at $2$ points, \textbf{touch} at $1$, or
        \textbf{miss} ($0$). After substitution, the one equation $ax^2+bx+c=0$ tells you which:
        its \textbf{discriminant} $D=b^2-4ac$ gives
        \[ D>0:\ 2 \qquad D=0:\ 1 \qquad D<0:\ 0. \]
      \end{block}
      \vspace{2pt}
      {\small Two parabolas work the same way: set them equal, collapse, and count.}
    \end{column}
    \begin{column}{0.38\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.26]
        \lgrid{-3}{5}{-7}{6}
        \begin{scope}\clip (-3,-7) rectangle (5,6);
          \draw[very thick, forest, domain=-2.2:4.2, samples=70, smooth]
            plot (\x,{\x*\x-2*\x-3});
          \draw[thick, redacc] (-3,5)--(5,5);
          \draw[thick, goldacc] (-3,-4)--(5,-4);
          \draw[thick, navy] (-3,-6)--(5,-6);
        \end{scope}
        \fill[redacc] (-2,5) circle (5pt);
        \fill[redacc] (4,5) circle (5pt);
        \fill[goldacc] (1,-4) circle (5pt);
        \node[right, font=\tiny, redacc] at (5,5) {$y=5$};
        \node[right, font=\tiny, goldacc] at (5,-4) {$y=-4$};
        \node[right, font=\tiny, navy] at (5,-6) {$y=-6$};
      \end{tikzpicture}\\[-2pt]
      {\scriptsize $y=x^2-2x-3$ against three lines: \\ cross, touch, miss.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Example 2: how many solutions --- $y=x^2-2x-3$ and each line?}
  \sectionlabel{Watch --- 2 minutes}
  \begin{center}
  \renewcommand{\arraystretch}{1.35}
  \begin{tabular}{c|c|c|c}
    line & set equal, one side $=0$ & $D=b^2-4ac$ & solutions \\ \hline
    $y=5$  & $x^2-2x-8=0$ & $4+32=36>0$  & $2$: \ $(-2,5)$, $(4,5)$ \\
    $y=-4$ & $x^2-2x+1=0$ & $4-4=0$      & $1$: \ $(1,-4)$ \\
    $y=-6$ & $x^2-2x+3=0$ & $4-12=-8<0$  & $0$
  \end{tabular}
  \end{center}
  \pause
  \vspace{4pt}
  Take $D$ of the \emph{collapsed} equation --- not of the parabola by itself. It matches the
  picture: the line $y=-4$ touches exactly at the vertex $(1,-4)$.
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Your turn}
  \sectionlabel[goldacc]{4 minutes --- start at one star and climb}
  How many solutions? Collapse to one equation and use $D$ --- don't solve.
  \begin{itemize}\setlength{\itemsep}{6pt}
    \item[\rung1] $y=x^2$ \; and \; $y=4$
    \item[\rung2] $y=x^2+1$ \; and \; $y=2x$
    \item[\rung3] $y=x^2$ \; and \; $y=-x^2+8$ \quad (two parabolas)
  \end{itemize}
  \vspace{4pt}
  \reveal{%
    \begin{block}{Check}
      \rung1\ $x^2-4=0$, $D=16$: \textbf{2} \qquad \rung2\ $x^2-2x+1=0$, $D=0$: \textbf{1} ---
      the line touches at $(1,2)$ \\[1pt]
      \rung3\ $2x^2-8=0$, $D=0-4(2)(-8)=64$: \textbf{2} --- they are $(2,4)$ and $(-2,4)$
    \end{block}}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Show me: how many solutions? $\;y=x^2-4x+7$ and $y=2x-2$}
  \sectionlabel[goldacc]{Pick one --- 1 to 4 fingers on the count of three}
  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \opt1{none} \\[12pt]
      \opt3{two}
    \end{column}
    \begin{column}{0.45\textwidth}
      \opt2{one} \\[12pt]
      \opt4{can't tell without a graph}
    \end{column}
  \end{columns}
  \vspace{10pt}
  \reveal{%
    \begin{block}{Answer: \textbf{2}}
      $x^2-4x+7=2x-2 \Rightarrow x^2-6x+9=0$, $D=36-36=0$: the line touches once, at $(3,4)$.
      \textbf{1}: you took $D$ of $x^2-4x+7$ alone ($-12$) --- collapse first. \textbf{3}: two is
      not automatic. \textbf{4}: $D$ tells you without a graph.
    \end{block}}
\end{frame}

% ═══ ROUND 3 — MODELING: WHICH FEATURE ANSWERS THE QUESTION (the crux) ═══════
\begin{frame}[t, plain]
  \forestheader{Definition: reading a quadratic model}
  \sectionlabel{Round 3 --- which feature answers the question?}
  \begin{columns}[T]
    \begin{column}{0.62\textwidth}
      \begin{block}{Feature $\to$ question}
        \footnotesize
        \renewcommand{\arraystretch}{1.25}
        \begin{tabular}{@{}l l@{}}
          \textbf{$y$-intercept} $h(0)$ & the \emph{starting} value \\
          \textbf{positive zero} ($h=0$) & \emph{when} it lands (throw out $t<0$) \\
          \textbf{vertex}, input $-\frac{b}{2a}$ & \emph{when / where} it is highest \\
          \textbf{vertex}, output $h\!\left(-\frac{b}{2a}\right)$ & \emph{how high / how much} \\
        \end{tabular}
      \end{block}
      \vspace{2pt}
      \begin{block}{Maximum or minimum value}
        The \textbf{maximum} (or \textbf{minimum}) \textbf{value} is the vertex's
        \textbf{output}. The vertex's \textbf{input} says \emph{when} or \emph{where} it happens.
      \end{block}
    \end{column}
    \begin{column}{0.34\textwidth}
      \centering
      \begin{tikzpicture}[x=0.55cm, y=0.02cm]
        \draw[linegray!45, thin, xstep=1, ystep=16] (0,0) grid (5.5,160);
        \draw[->, thick, charcoal] (0,0)--(5.8,0) node[below right, font=\tiny]{$t$};
        \draw[->, thick, charcoal] (0,0)--(0,168) node[left, font=\tiny]{$h$};
        \foreach \tk in {1,...,5} \node[below, font=\tiny] at (\tk,0) {$\tk$};
        \foreach \hk in {48,96,144} \node[left, font=\tiny] at (0,\hk) {$\hk$};
        \draw[very thick, forest, domain=0:5, samples=70, smooth]
          plot (\x,{-16*\x*\x+64*\x+80});
        \fill[redacc] (0,80) circle (3pt) node[right, font=\tiny, charcoal]{start};
        \fill[redacc] (2,144) circle (3pt) node[above, font=\tiny, charcoal]{max};
        \fill[redacc] (5,0) circle (3pt) node[above right, font=\tiny, charcoal]{lands};
      \end{tikzpicture}\\[-2pt]
      {\scriptsize $h(t)=-16t^2+64t+80$}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Example 3: a ball tossed up from an 80-ft roof}
  \sectionlabel{Watch --- 2 minutes}
  $h(t)=-16t^2+64t+80$, \ $h$ in feet, $t$ in seconds.
  \begin{itemize}\setlength{\itemsep}{5pt}
    \item \textbf{Where did it start?} \; $h(0)=80$ ft --- the $y$-intercept.
    \pause
    \item \textbf{When is it highest?} \; $t=-\dfrac{64}{2(-16)}=2$ s --- the vertex's
          \emph{input}.
    \pause
    \item \textbf{How high does it go?} \; $h(2)=-64+128+80=144$ ft --- the vertex's
          \emph{output}.
    \pause
    \item \textbf{When does it land?} \; $-16(t^2-4t-5)=0$, \ $(t-5)(t+1)=0$: \ $t=5$ s.
          \\ ($t=-1$ is before the toss --- throw it out.)
  \end{itemize}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Your turn}
  \sectionlabel[goldacc]{5 minutes --- start at one star and climb}
  A volleyball serve: $h(t)=-16t^2+16t+6$ \ (feet, seconds).
  \begin{itemize}\setlength{\itemsep}{5pt}
    \item[\rung1] How high was the ball when it was hit?
    \item[\rung2] When does it reach its highest point, and how high is that?
    \item[\rung3] The campus garden gets $40$ ft of fence on three sides; the greenhouse wall is
          the fourth. Its area is $A=x(40-2x)$ for width $x$. What width gives the most area,
          and how much?
  \end{itemize}
  \reveal{%
    \begin{block}{Check}
      \rung1\ $h(0)=6$ ft \qquad \rung2\ $t=\frac{-16}{-32}=0.5$ s; \ $h(0.5)=-4+8+6=10$ ft \\[1pt]
      \rung3\ $A=-2x^2+40x$, \ $x=\frac{-40}{-4}=10$ ft; \ $A=10\cdot20=200$ ft$^2$
    \end{block}}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Show me: $h(t)=-16t^2+32t+48$. How high does the ball go?}
  \sectionlabel[redacc]{The one that matters --- 1 to 4 fingers on the count of three}
  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \opt1{$1$ second} \\[12pt]
      \opt3{$3$ seconds}
    \end{column}
    \begin{column}{0.45\textwidth}
      \opt2{$64$ feet} \\[12pt]
      \opt4{$48$ feet}
    \end{column}
  \end{columns}
  \vspace{10pt}
  \reveal{%
    \begin{block}{Answer: \textbf{2}}
      Vertex at $t=-\frac{32}{-32}=1$, and $h(1)=-16+32+48=64$ ft. \; \textbf{1} is the vertex's
      \emph{input} --- \emph{when} it is highest, not how high. \textbf{3} is when it lands
      ($-16(t-3)(t+1)=0$). \textbf{4} is where it started. ``How high'' is an \emph{output}: its
      answer is in feet.
    \end{block}}
\end{frame}

% FINAL ROUND — one problem per round, self-scored /4. Item 4 is the crux's twin.
\begin{frame}[t, plain]
  \forestheader{Final Round}
  \sectionlabel[goldacc]{5 minutes --- on your own --- score yourself out of 4}
  \begin{enumerate}\setlength{\itemsep}{2pt}
    \item Solve the system $y=x^2-1$ and $y=x+1$.
    \item How many solutions: $y=x^2+2x+4$ and $y=2x+3$?
    \item A ball's height is $h(t)=-16t^2+48t+64$ (feet, seconds). When does it land?
    \item Same ball: how high does it go?
  \end{enumerate}
  {\small\textbf{Bonus (+1):} The art club sells hoodies at \$30 and sells 40. Each \$1 price
  increase loses one sale, so revenue is $R(x)=(30+x)(40-x)$ for $x$ one-dollar increases. What
  price brings in the most, and how much?}
  \reveal{%
    \begin{block}{Check --- one point each}
      \small
      1.\ $x^2-x-2=0$: $(2,3)$, $(-1,0)$ \quad 2.\ $x^2+1=0$, $D=-4$: none \quad
      3.\ $-16(t-4)(t+1)=0$: $t=4$ s \quad 4.\ $t=1.5$, $h(1.5)=100$ ft \\[1pt]
      \textbf{Bonus:} $R=-x^2+10x+1200$, $x=5$: price \$35, revenue $35\cdot35=\$1225$. \\[2pt]
      \textbf{Watch out:} the maximum \emph{value} is the vertex's \textbf{output} (how high, how
      much); \emph{when} or \emph{where} it happens is the \textbf{input}.
    \end{block}}
\end{frame}
