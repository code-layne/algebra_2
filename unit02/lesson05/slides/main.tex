% HOOK — four methods
\begin{frame}[t, plain]
  \forestheader{Four quadratics, four ``best methods''}
  \sectionlabel{Hook}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{itemize}\setlength{\itemsep}{4pt}
        \item $x^2-2x-3=0$ \ --- factor
        \item $x^2-6x+7=0$ \ --- won't factor
        \item $x^2+2x+5=0$ \ --- complex roots
        \item $2x^2-7x+3=0$ \ --- ?
      \end{itemize}
      Choosing a method every time is a hassle.
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{The big idea}
        What if \emph{one} formula solved \textbf{every} quadratic --- real or complex --- with no guessing?
      \end{block}
      It exists, from a move you know: \emph{completing the square}, done once.
    \end{column}
  \end{columns}

% THE FORMULA
\begin{frame}[t, plain]
  \forestheader{The quadratic formula}
  \sectionlabel[goldacc]{Definition}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Solves any $ax^2+bx+c=0$}
        \[ x=\frac{-b\pm\sqrt{b^2-4ac}}{2a}. \]
      \end{block}
      Complete the square on the \emph{general} equation and this pops out --- so it works \emph{every} time.
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Protocol}
      \begin{enumerate}\setlength{\itemsep}{2pt}
        \item Standard form $ax^2+bx+c=0$.
        \item Read $a,b,c$ \emph{with signs}.
        \item Substitute \& simplify.
      \end{enumerate}
      \[ x^2-2x-3=0:\ x=\frac{2\pm\sqrt{16}}{2}=3,-1. \]
    \end{column}
  \end{columns}
\end{frame}

% THE DISCRIMINANT
\begin{frame}[t, plain]
  \forestheader{The discriminant $b^2-4ac$ --- predict before you solve}
  \sectionlabel{Predict}
  \begin{columns}[T]
    \begin{column}{0.4\textwidth}
      The part under the radical decides everything:
      \begin{itemize}\setlength{\itemsep}{3pt}
        \item $>0$: two real roots
        \item $=0$: one repeated real root
        \item $<0$: two complex conjugates
      \end{itemize}
      \small Roots \emph{are} $x$-intercepts: 2, 1, or 0.
    \end{column}
    \begin{column}{0.6\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.34]
        \lgrid{-2}{4}{-5}{3}
        \begin{scope}\clip (-2,-5) rectangle (4,3);
          \draw[very thick, forest, domain=-1.2:3.2, samples=50, smooth] plot (\x,{(\x-1)*(\x-1)-4});
        \end{scope}
        \fill[charcoal] (-1,0) circle (3pt); \fill[charcoal] (3,0) circle (3pt);
        \node[font=\tiny, charcoal] at (1,-6.4){$>0$};
      \end{tikzpicture}
      \begin{tikzpicture}[scale=0.34]
        \lgrid{0}{4}{-1}{5}
        \begin{scope}\clip (0,-1) rectangle (4,5);
          \draw[very thick, forest, domain=0.05:3.95, samples=50, smooth] plot (\x,{(\x-2)*(\x-2)});
        \end{scope}
        \fill[charcoal] (2,0) circle (3pt);
        \node[font=\tiny, charcoal] at (2,-2.6){$=0$};
      \end{tikzpicture}
      \begin{tikzpicture}[scale=0.34]
        \lgrid{-3}{3}{-1}{7}
        \begin{scope}\clip (-3,-1) rectangle (3,7);
          \draw[very thick, forest, domain=-2.3:2.3, samples=50, smooth] plot (\x,{\x*\x+1});
        \end{scope}
        \fill[charcoal] (0,1) circle (3pt);
        \node[font=\tiny, charcoal] at (0,-2.6){$<0$};
      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}

% COMPLEX SOLUTIONS
\begin{frame}[t, plain]
  \forestheader{When $b^2-4ac<0$: complex solutions}
  \sectionlabel[goldacc]{Complex}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      A negative discriminant is \textbf{not} a dead end --- use $i$ (Lesson 2.4).
      \[ x^2+2x+5=0:\ b^2-4ac=-16. \]
      \[ x=\frac{-2\pm\sqrt{-16}}{2}=\frac{-2\pm4i}{2}=-1\pm2i. \]
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{Same answer as 2.4}
        Completing the square on $x^2+2x+5=0$ also gave $-1\pm2i$ --- a conjugate pair.
      \end{block}
      The parabola $y=x^2+2x+5$ never meets the $x$-axis --- exactly what $b^2-4ac<0$ predicted.
    \end{column}
  \end{columns}
\end{frame}

% CHOOSE A METHOD + MODEL
\begin{frame}[t, plain]
  \forestheader{Choose a method \& model}
  \sectionlabel{Apply}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Fastest tool}
      \begin{itemize}\setlength{\itemsep}{2pt}
        \item Factoring --- when it factors.
        \item Square roots --- for $x^2=k$.
        \item Formula --- \emph{always} works.
      \end{itemize}
    \end{column}
    \begin{column}{0.52\textwidth}
      \begin{block}{Model: when does it land?}
        $h(t)=-16t^2+24t+6$, solve $h=0$:
        \[ t=\frac{3\pm\sqrt{15}}{4}\approx1.72\text{ s}. \]
        Keep the positive root; reject the negative time.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
