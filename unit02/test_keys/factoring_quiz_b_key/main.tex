\qhead{1.\ Greatest Common Factor}{Factor completely: \quad $12x^3-8x^2+20x$}
\begin{work}
  \text{GCF} &= 4x \\
  12x^3-8x^2+20x &= 4x(3x^2)-4x(2x)+4x(5) \\
  &= 4x(3x^2-2x+5)
\end{work}

\qhead{2.\ Difference of Squares}{Factor completely: \quad $25x^2-36$}
\begin{work}
  25x^2-36 &= (5x)^2-(6)^2 \\
  a^2-b^2 &= (a+b)(a-b)
\end{work}
\vfill
\noindent\hfill Answer: \inblank{\slot{4.2cm}{$(5x+6)(5x-6)$}}{\gradedblank[id=q2, type=short, points=20, answer={(5x+6)(5x-6)}, check=expression]{4.2cm-2pt}{18pt}}

\qhead{3.\ Trinomial, $a=1$}{Factor completely: \quad $x^2+2x-35$}
\begin{work}
  \text{multiply to } {-35},\ \text{add to } 2 &: \quad 7 \text{ and } {-5} \\
  x^2+2x-35 &= (x+7)(x-5)
\end{work}
\vfill
\noindent\hfill Answer: \inblank{\slot{4.2cm}{$(x+7)(x-5)$}}{\gradedblank[id=q3, type=short, points=20, answer={(x+7)(x-5)}, check=expression]{4.2cm-2pt}{18pt}}

\qhead{4.\ Trinomial, $a>1$}{Factor completely: \quad $3x^2-7x-6$}
\begin{work}
  \text{Slip: } & x^2-7x-18 \\
  \text{Factor: } & (x-9)(x+2) \\
  \text{Slide: } & \left(x-\tfrac{9}{3}\right)\left(x+\tfrac{2}{3}\right) \\
  \text{Simplify: } & (x-3)\left(x+\tfrac{2}{3}\right) \\
  \text{Slide up: } & (x-3)(3x+2)
\end{work}
\vfill
\noindent\hfill Answer: \inblank{\slot{4.2cm}{$(x-3)(3x+2)$}}{\gradedblank[id=q4, type=short, points=20, answer={(x-3)(3x+2)}, check=expression]{4.2cm-2pt}{18pt}}

% ======================================================================
\qhead{5.\ Solving a Quadratic Equation by Factoring}{Solve by factoring: \quad $2x^2+x=15$}
\begin{work}
  & 2x^2+x-15 = 0 \\
  \text{Slip: } & x^2+x-30 \\
  \text{Factor: } & (x+6)(x-5) \\
  \text{Slide: } & \left(x+\tfrac{6}{2}\right)\left(x-\tfrac{5}{2}\right) \\
  \text{Slide up: } & (x+3)(2x-5) = 0 \\
  & x+3=0 \quad\text{or}\quad 2x-5=0
\end{work}
\vfill
\noindent\hfill Solutions: \inblank{\slot{4.2cm}{$x=-3$ or $x=\tfrac{5}{2}$}}{\gradedblank[id=q5, type=short, points=20, answer={-3, 5/2}, check=set, display={$x=-3$ or $x=5/2$}]{4.2cm-2pt}{18pt}}
