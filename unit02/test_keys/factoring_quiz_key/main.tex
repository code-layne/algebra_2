\qhead{1.\ Greatest Common Factor}{Factor completely: \quad $6x^3-15x^2+12x$}
\begin{work}
  \text{GCF} &= 3x \\
  6x^3-15x^2+12x &= 3x(2x^2)-3x(5x)+3x(4) \\
  &= 3x(2x^2-5x+4)
\end{work}
\vfill
\noindent\hfill Answer: \slot{4.2cm}{$3x(2x^2-5x+4)$}

% ======================================================================
\qhead{2.\ Difference of Squares}{Factor completely: \quad $9x^2-49$}
\begin{work}
  9x^2-49 &= (3x)^2-(7)^2 \\
  a^2-b^2 &= (a+b)(a-b)
\end{work}
\vfill
\noindent\hfill Answer: \slot{4.2cm}{$(3x+7)(3x-7)$}

% ======================================================================
\qhead{3.\ Trinomial, $a=1$}{Factor completely: \quad $x^2+3x-40$}
\begin{work}
  \text{multiply to } {-40},\ \text{add to } 3 &: \quad 8 \text{ and } {-5} \\
  x^2+3x-40 &= (x+8)(x-5)
\end{work}
\vfill
\noindent\hfill Answer: \slot{4.2cm}{$(x+8)(x-5)$}

% ======================================================================
\newpage
\qhead{4.\ Trinomial, $a>1$}{Factor completely: \quad $3x^2-10x-8$}
\begin{work}
  ac = -24,\ \text{add to } {-10} &: \quad -12 \text{ and } 2 \\
  3x^2-12x+2x-8 &= 3x(x-4)+2(x-4) \\
  &= (3x+2)(x-4)
\end{work}
\vfill
\noindent\hfill Answer: \slot{4.2cm}{$(3x+2)(x-4)$}

% ======================================================================
\qhead{5.\ Solving a Quadratic Equation by Factoring}{Solve by factoring: \quad $2x^2+5x=12$}
\begin{work}
  2x^2+5x-12 &= 0 \\
  ac=-24,\ \text{add to } 5 &: \quad 8 \text{ and } {-3} \\
  2x(x+4)-3(x+4) &= 0 \\
  (2x-3)(x+4) &= 0 \\
  2x-3=0 \quad&\text{or}\quad x+4=0
\end{work}
\vfill
\noindent\hfill Solutions: \slot{4.2cm}{$x=\tfrac{3}{2}$ or $x=-4$}
