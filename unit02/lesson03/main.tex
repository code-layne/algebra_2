\begin{tcolorbox}[colback=forestbg, colframe=forest, boxrule=0.9pt, arc=2mm,
    left=2mm, right=2mm, top=1.5mm, bottom=1.5mm]
    \textbf{Primary Objective:} Students solve a quadratic that will not factor when it can be
    written as ``a square $=$ a number,'' using the Square Root Property (isolate the square,
    keep the $\pm$, simplest radical form); meet the wall $x^2=-4$ and break it with the
    imaginary unit $i=\sqrt{-1}$, writing $\sqrt{-k}=i\sqrt k$ and complex numbers in standard
    form $a+bi$; and add, subtract, and multiply complex numbers, replacing $i^2$ with $-1$ and
    using conjugates to get a real product.
    \\[2pt]
    \textbf{Standards (2023 VA SOL):} \textbf{A2.EI.2b} (solve a quadratic equation
    algebraically --- here by square roots, real and imaginary roots), \textbf{A2.EI.2d} (verify
    and interpret solutions: real roots are $x$-intercepts; a parabola that misses the axis has
    no real roots), \textbf{A2.EO.4a} (explain the meaning of $i$), \textbf{A2.EO.4b} (identify
    equivalent radical expressions containing negative rationals and expressions in $a+bi$
    form), \textbf{A2.EO.4c} (add, subtract, and multiply complex numbers).
    \\[2pt]
    \textbf{Lesson model:} \emph{Slides first.} The deck is the lesson and its printed handout is
    the student's notes, and students are \emph{solving the whole hour}: a warm-up on screen as
    they walk in; three rounds of a short \textbf{definition} and \textbf{example} ($\le$3 min),
    a \textbf{ladder} of three problems (\rung1\ \rung2\ \rung3), and a \textbf{Show me} or
    \textbf{Find the mistake}; a self-scored \textbf{Final Round}; and the \textbf{homework}
    started in class. \textbf{Homework source:} DeltaMath, online --- set
    \textbf{A2: Lesson 2.3: Square Roots \& Complex Numbers} (not in the packet).
\end{tcolorbox}

\begin{skillbox}[Priority Ideas \& Skills]{goldbox}
\begin{tabularx}{\linewidth}{@{}X|X@{}}
\textbf{Priority Ideas \& Skills} & \textbf{Key Understandings (the ``why'')} \\[2pt]
\hline
\vspace{-6pt}
\begin{itemize}[leftmargin=1.1em, nosep, after=\vspace{2pt}]
  \item Solve $x^2=k$ and $(x-h)^2=k$ with the \textbf{Square Root Property}: isolate the square,
        then $\pm\sqrt{\ }$; answers in \textbf{simplest radical form}.
  \item Rewrite $\sqrt{-k}=i\sqrt k$ (\emph{$i$ out first}); write $a+bi$ and name its parts.
  \item Solve $x^2=-k$: two imaginary roots, $\pm i\sqrt k$.
  \item Add and subtract complex numbers by combining like parts; multiply by distributing, then
        $i^2=-1$; conjugates give $a^2+b^2$.
\end{itemize}
&
\vspace{-6pt}
\begin{itemize}[leftmargin=1.1em, nosep, after=\vspace{2pt}]
  \item A positive number has \emph{two} square roots --- the $\pm$ keeps both.
  \item Nothing real squares to a negative, so $y=x^2+4$ misses the axis; $i$ extends the
        numbers so $x^2=-4$ has solutions after all.
  \item Complex arithmetic is polynomial arithmetic plus one fact, $i^2=-1$.
  \item \textbf{Target misconception: $i^2$ is $-1$, so pull $i$ out of every negative radical
        \emph{before} multiplying.} Students write $\sqrt{-4}\cdot\sqrt{-9}=\sqrt{36}=6$
        (it is $-6$), or leave $+2i^2$ in a product, or treat $i^2$ as $+1$.
\end{itemize}
\end{tabularx}
\end{skillbox}

\begin{skillbox}[{Vocabulary, Concepts, \& Theorems --- one definition frame each}]{sky}
\renewcommand{\arraystretch}{1.3}
\begin{tabularx}{\linewidth}{@{}l X@{}}
\textbf{Square Root Property} & If $x^2=k$ and $k\ge0$, then $x=\pm\sqrt{k}$; $(x-h)^2=k$ gives
                                $x=h\pm\sqrt{k}$. Get the square alone \emph{before} you take the root. \\
\textbf{Simplest radical form} & No perfect-square factor left under the root:
                                 $\sqrt{12}=\sqrt4\cdot\sqrt3=2\sqrt3$. \\
\textbf{Imaginary unit} & $i=\sqrt{-1}$, so $i^2=-1$. For $k>0$, $\sqrt{-k}=i\sqrt{k}$ --- pull the
                          $i$ out first, then simplify. \\
\textbf{Complex number} & $a+bi$ (\textbf{standard form}), $a$ and $b$ real: $a$ is the
                          \textbf{real part}, $b$ the \textbf{imaginary part}. $b=0$: real;
                          $a=0$, $b\ne0$: \textbf{pure imaginary}. \\
\textbf{Operations} & Add or subtract: combine the real parts and the imaginary parts. Multiply:
                      distribute, then replace every $i^2$ with $-1$ and combine. \\
\textbf{Complex conjugates} & $a+bi$ and $a-bi$; their product $a^2+b^2$ is always real. \\
\end{tabularx}
\end{skillbox}

\begin{skillbox}[Engagement --- how this hour keeps everyone solving]{goldbox}
\textbf{Nobody waits:} every practice frame is a ladder --- \rung1\ everyone (a quick win),
\rung2\ everyone, \rung3\ early finishers. \textbf{Everyone commits, safely:} \emph{Show me} is
1--4 fingers on the count of three, all at once (Rounds 1 and 2). \textbf{Critique a stranger:}
Round 3's \emph{Find the mistake} puts the crux error on an anonymous Student A.
\textbf{Their world:} the campus-garden bed with its path (Round 1 \rung3) and the egg-drop
bonus. \textbf{Short clock:} every practice frame carries its minutes. Praise the climb more
than the answer.
\end{skillbox}

\begin{skillbox}[Warm-Up (5 min, on the slides as they walk in)]{forestbg}
Three quick wins; the answers are one click. \textbf{(1)} $5$ \emph{and} $-5$ square to $25$ ---
the $\pm$ of Round~1. \textbf{(2)} $\sqrt{72}=6\sqrt2$ --- simplest radical form. \textbf{(3)}
$(x+3)^2=x^2+6x+9$, not $x^2+9$ --- the distributing Round~3 needs. The reveal ends on
\emph{Hold on to this}: what number squares to $-4$? Leave it open --- Round~1 shows nothing
real does, Round~2 answers it. Then the targets frame (30 seconds).
\end{skillbox}

\begin{skillbox}[The Lesson --- three rounds of definition, example, ladder, check (40 min)]{forestbg}
\begin{multicols}{2}
\textbf{Round 1 --- the Square Root Property (13 min).} Definition and Example 1 in 3 minutes:
point at the graph of $y=x^2-6$ --- two intercepts, $\pm\sqrt6$, one each side, so two roots.
Example: $3x^2-5=31\Rightarrow x=\pm2\sqrt3$ (isolate first); $(x-3)^2=16\Rightarrow x=7$ or
$-1$. Ladder (7 min): \rung1\ $\pm7$; \rung2\ $x^2=20$, $\pm2\sqrt5$ (watch $\pm\sqrt{20}$ left
unsimplified); \rung3\ the garden bed, $(s+2)^2=50$, $s=-2+5\sqrt2\approx5.1$ ft, reject the
negative root. Show me $2x^2-8=42$: answer \textbf{3}, $\pm5$; \textbf{1} = dropped the $\pm$;
\textbf{2} = rooted before dividing by $2$; \textbf{4} = subtracted the $8$.

\textbf{Round 2 --- the imaginary unit (13 min).} Definition with the graph of $y=x^2+4$:
it never meets the axis, so \emph{no real} roots --- that is the warm-up's $-4$. Define $i$,
$i^2=-1$, $\sqrt{-k}=i\sqrt k$, and $a+bi$. Example 2: $\sqrt{-50}=5i\sqrt2$;
$5+\sqrt{-18}=5+3i\sqrt2$; $x^2+9=0\Rightarrow\pm3i$, checked. Ladder (7 min): \rung1\ $7i$,
$2i\sqrt5$; \rung2\ $2-5i$, real part $2$, imaginary part $-5$ (not $-5i$); \rung3\
$(x-2)^2=-9\Rightarrow x=2\pm3i$. Show me $x^2+25=0$: answer \textbf{3}, $\pm5i$; \textbf{1} =
ignored the sign ($(\pm5)^2=+25$); \textbf{2} = dropped the $\pm$; \textbf{4} = ``no solution''
--- correct it to ``no \emph{real} solution.''

\columnbreak
\textbf{Round 3 --- add, subtract, multiply: the crux (14 min).} Definition and Example 3 in 3
minutes: $(4+3i)-(1-5i)=3+8i$ (the minus hits both parts); $(2+3i)(4-i)=11+10i$ --- stop on the
$-3i^2$ line and ask what $i^2$ is; $(5+2i)(5-2i)=29$, real. Point at the \emph{Careful} line.
Ladder (8 min): \rung1\ $5+5i$; \rung2\ $3-3i+2i-2i^2=5-i$ (watch $1-i$: $i^2$ taken as $+1$);
\rung3\ $(1+2i)^2=-3+4i$ (watch $1+4i^2$ --- the middle term) and $(4+i)(4-i)=17$.
\textbf{Find the mistake --- the frame not to rush:} Student A wrote
$\sqrt{-4}\cdot\sqrt{-9}=\sqrt{36}=6$. Give 3 minutes; ask ``what would you tell Student A?''
before revealing. The fix: $\sqrt a\sqrt b=\sqrt{ab}$ needs $a,b\ge0$; pull $i$ out first,
$2i\cdot3i=6i^2=-6$. This is the target misconception.
\end{multicols}
\end{skillbox}

\boxguard
\begin{skillbox}[Final Round (5 min)]{forestbg}
Four mixed problems, one per round, alone, 5-minute timer: $4x^2-1=79\Rightarrow x=\pm2\sqrt5$;
$x^2+64=0\Rightarrow x=\pm8i$; $(2-3i)(2+3i)=13$; $\sqrt{-2}\cdot\sqrt{-18}=i\sqrt2\cdot
i\sqrt{18}=6i^2=-6$ (the crux's twin --- $+6$ means $\sqrt{36}$ again). \textbf{Bonus:} the
egg-drop capsule, $-16t^2+48=0$, $t=\sqrt3\approx1.7$\,s (reject $-\sqrt3$). Students score
themselves out of 4; ask for scores on fingers --- that is your exit read. Say the \emph{Watch
out} aloud as you reveal: $i^2=-1$, so pull the $i$ out before you multiply.
\end{skillbox}

\emph{Solving Quadratic Binomials (Level 3)}; \emph{Add/Subtract Complex Numbers};
\emph{Multiply Imaginary Numbers}; \emph{Multiply Complex Numbers}.
\textbf{Formative check:} \emph{Multiply Imaginary Numbers} --- the product of two negative
radicals (e.g.\ $\sqrt{-3}\cdot\sqrt{-12}$) --- sort into \emph{$-6$, correct} / \emph{$+6$, combined under one
root} / \emph{other}; if the middle pile is large, open 2.4 by re-doing it with $i$ pulled out.
\textbf{Preview:} 2.4 --- completing the square and the quadratic formula, which solve every
quadratic, real or complex answers.
\end{skillbox}

\boxguard
\begin{skillbox}[Active Monitoring --- Watch For]{redbox}
\begin{itemize}[leftmargin=1.2em, nosep]
  \item \textbf{Round 1 ladder and Show me:} dropping the $\pm$; rooting before isolating;
        $\sqrt{20}$ left unsimplified.
  \item \textbf{Round 2 \rung2:} naming the imaginary part $-5i$ instead of $-5$; ``no
        solution'' for $x^2=-25$.
  \item \textbf{Round 3 \rung2--\rung3:} stopping at $\ldots-2i^2$, or $i^2$ taken as $+1$;
        $(1+2i)^2=1+4i^2$ (lost middle term).
  \item \textbf{Round 3 Find the mistake / Final Round 4:} $\sqrt{-a}\cdot\sqrt{-b}=\sqrt{ab}$
        (\emph{the target misconception}).
  \item \textbf{Attitude:} a student parked at \rung1\ with nothing written --- crouch, point at
        the example on the handout, ask for just the first step.
\end{itemize}
Cold-call prompts (after the fingers, never before): ``Why does $x^2=25$ have two answers?''
\; ``What is $i^2$ --- and where did you use it?''
\end{skillbox}

\begin{teachernote}[Slides]
Pacing: warm-up 5, rounds 13 / 13 / 14, Final Round 5, homework 10 (60). In each round the
definition and example together take \textbf{at most 3 minutes} --- if you catch yourself
explaining past that, move to the ladder and explain at desks. Reveal a ladder when most of the
room has \rung2; the \rung3\ students keep going. If behind, cut Round 2's \rung3\ and shorten
its Show me, never Round 3's Find the mistake. Students write in the handout's notes column,
which prints without the answers.
\end{teachernote}

Launch problem~1 aloud in the last 10 minutes with students on their devices, then circulate

first class after two study halls; announce the date in class and on TurtleNet.
\end{teachernote}
