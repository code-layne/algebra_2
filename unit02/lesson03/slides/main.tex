% ═══ ROUND 1 — THE SQUARE ROOT PROPERTY ═════════════════════════════════════
\begin{frame}[t, plain]
  \forestheader{Definition: the Square Root Property}
  \sectionlabel{Round 1 --- undo the square}
  \begin{columns}[T]
    \begin{column}{0.6\textwidth}
      \begin{block}{Square Root Property}
        If $x^2=k$ and $k\ge0$, then $x=\pm\sqrt{k}$ --- a positive number has \emph{two}
        square roots. The same holds for a squared binomial: $(x-h)^2=k$ gives
        $x=h\pm\sqrt{k}$.
      \end{block}
      \vspace{2pt}
      \begin{block}{Isolate first; simplest radical form}
        Get the square alone \emph{before} you take the root. Write each answer in
        \textbf{simplest radical form} --- no perfect-square factor left under the root, as in
        $\sqrt{12}=\sqrt4\cdot\sqrt3=2\sqrt3$.
      \end{block}
    \end{column}
    \begin{column}{0.36\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.3]
        \lgrid{-4}{4}{-7}{4}
        \begin{scope}\clip (-4,-7) rectangle (4,4);
          \draw[very thick, forest, domain=-3.4:3.4, samples=60, smooth]
            plot (\x,{\x*\x-6});
        \end{scope}
        \fill[redacc] (-2.449,0) circle (4pt);
        \fill[redacc] (2.449,0) circle (4pt);
      \end{tikzpicture}\\[-2pt]
      {\scriptsize $y=x^2-6$ crosses at $x=\pm\sqrt6\approx\pm2.45$: \\ two roots, one on each
      side.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Example 1: solve $3x^2-5=31$, then $(x-3)^2=16$}
  \sectionlabel{Watch --- 2 minutes}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{(a)} Isolate the square, then root.
      \[
      \begin{aligned}
        3x^2-5 &= 31 \\
        3x^2 &= 36 && \text{add } 5 \\
        x^2 &= 12  && \text{divide by } 3 \\
        x &= \pm\sqrt{12}=\pm2\sqrt3
      \end{aligned}
      \]
    \end{column}
    \pause
    \begin{column}{0.48\textwidth}
      \textbf{(b)} The square is already alone.
      \[
      \begin{aligned}
        (x-3)^2 &= 16 \\
        x-3 &= \pm4 && \text{both roots} \\
        x &= 3\pm4
      \end{aligned}
      \]
      $x=7$ or $x=-1$.
    \end{column}
  \end{columns}
  \pause
  \vspace{4pt}
  \emph{Check (b):} $(7-3)^2=16$ \checkmark \quad $(-1-3)^2=16$ \checkmark \quad --- drop the
  $\pm$ and you lose a root.
\end{frame}

% ═══ ROUND 2 — THE IMAGINARY UNIT ═══════════════════════════════════════════
\begin{frame}[t, plain]
  \forestheader{Definition: the imaginary unit $i$}
  \sectionlabel{Round 2 --- a square that is negative}
  \begin{columns}[T]
    \begin{column}{0.6\textwidth}
      \begin{block}{The imaginary unit}
        The \textbf{imaginary unit} is $i=\sqrt{-1}$, so $i^2=-1$. For $k>0$,
        $\sqrt{-k}=i\sqrt{k}$ --- pull the $i$ out \emph{first}, then simplify.
      \end{block}
      \vspace{2pt}
      \begin{block}{Complex numbers}
        A \textbf{complex number} in \textbf{standard form} is $a+bi$, with $a$ and $b$ real:
        $a$ is the \textbf{real part} and $b$ the \textbf{imaginary part}. If $b=0$ it is real;
        if $a=0$ and $b\ne0$ it is \textbf{pure imaginary}.
      \end{block}
    \end{column}
    \begin{column}{0.36\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.3]
        \lgrid{-4}{4}{-2}{9}
        \begin{scope}\clip (-4,-2) rectangle (4,9);
          \draw[very thick, forest, domain=-2.4:2.4, samples=60, smooth]
            plot (\x,{\x*\x+4});
        \end{scope}
      \end{tikzpicture}\\[-2pt]
      {\scriptsize $y=x^2+4$ never touches the $x$-axis: \\ \textbf{no real roots}. But
      $x^2=-4$ has two \\ imaginary roots, $x=\pm2i$.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Example 2: three moves with $i$}
  \sectionlabel{Watch --- 2 minutes}
  \begin{itemize}\setlength{\itemsep}{8pt}
    \item \textbf{(a)} $\sqrt{-50}=i\sqrt{50}$ \pause $=i\cdot5\sqrt2=5i\sqrt2$ \quad
          \emph{($i$ out first, then simplify)}
    \pause
    \item \textbf{(b)} $5+\sqrt{-18}=5+i\sqrt{18}$ \pause $=5+3i\sqrt2$: \; real part $5$,
          imaginary part $3\sqrt2$
    \pause
    \item \textbf{(c)} $x^2+9=0$ \pause $\;\Rightarrow\; x^2=-9 \;\Rightarrow\;
          x=\pm\sqrt{-9}=\pm3i$
  \end{itemize}
  \pause
  \vspace{4pt}
  \emph{Check (c):} $(3i)^2=9i^2=9(-1)=-9$ \checkmark \quad $(-3i)^2=-9$ \checkmark \quad ---
  still two roots, still the $\pm$.
\end{frame}

% ═══ ROUND 3 — ADD, SUBTRACT, MULTIPLY (the crux) ═══════════════════════════
\begin{frame}[t, plain]
  \forestheader{Definition: operations with complex numbers}
  \sectionlabel{Round 3 --- arithmetic with $i$}
  \begin{block}{Add, subtract, multiply}
    To \textbf{add or subtract}, combine the real parts and combine the imaginary parts:
    $(a+bi)+(c+di)=(a+c)+(b+d)i$. To \textbf{multiply}, distribute as with binomials, then
    replace every $i^2$ with $-1$ and combine.
  \end{block}
  \vspace{4pt}
  \begin{block}{Complex conjugates}
    $a+bi$ and $a-bi$ are \textbf{complex conjugates}. Their product is always real:
    \[ (a+bi)(a-bi)=a^2-b^2i^2=a^2+b^2. \]
  \end{block}
  \vspace{4pt}
  {\small \textbf{Careful:} $\sqrt a\cdot\sqrt b=\sqrt{ab}$ only when $a,b\ge0$. With a negative
  under a root, pull the $i$ out first --- because $i^2=-1$.}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Example 3: subtract, multiply, conjugates}
  \sectionlabel{Watch --- 2 minutes}
  \textbf{(a)} $(4+3i)-(1-5i)=4+3i-1+5i=3+8i$ \quad \emph{(the minus hits both parts)}
  \pause
  \[
  \begin{aligned}
    \textbf{(b)}\quad (2+3i)(4-i) &= 8-2i+12i-3i^2 && \text{distribute} \\
                                  &= 8+10i-3(-1)   && i^2=-1 \\
                                  &= 11+10i        && \text{combine}
  \end{aligned}
  \]
  \pause
  \textbf{(c)} $(5+2i)(5-2i)=25-10i+10i-4i^2=25+4=29$ \quad \emph{(conjugates: a real answer)}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Find the mistake}
  \sectionlabel[redacc]{The one that matters --- 3 minutes --- find Student A's wrong step and fix it}
  \textbf{Student A} simplified $\sqrt{-4}\cdot\sqrt{-9}$:
  \[
  \begin{aligned}
    \sqrt{-4}\cdot\sqrt{-9} &= \sqrt{(-4)(-9)} \\
                            &= \sqrt{36} \\
                            &= 6 \qquad \text{``A negative times a negative is positive.''}
  \end{aligned}
  \]
  \reveal{%
    \begin{block}{The mistake}
      Line 1: $\sqrt a\cdot\sqrt b=\sqrt{ab}$ only works when $a,b\ge0$. Pull the $i$ out
      first: $\sqrt{-4}\cdot\sqrt{-9}=2i\cdot3i=6i^2=-6$. \quad The answer is $-6$, not $6$ ---
      because $i^2=-1$.
    \end{block}}
\end{frame}

  \end{enumerate}
  {\small\textbf{Bonus (+1):} At the egg drop, a capsule is let go $48$ ft above the ground; its
  height after $t$ seconds is $h=-16t^2+48$. When does it land? (Simplest radical form.)}
  \vspace{2pt}
  \reveal{%
    \begin{block}{Check --- one point each}
      1.\ $x^2=20$, $x=\pm2\sqrt5$ \quad 2.\ $x=\pm8i$ \quad 3.\ $4-9i^2=13$ \quad

    \end{block}}
