\begin{objectivebox}
\textbf{By the end of this lesson, I will be able to\dots}
\begin{itemize}\setlength{\itemsep}{2pt}
  \item graph the \emph{absolute value parent} $f(x)=|x|$ and read its vertex, axis, and minimum;
  \item use \emph{vertex form} $g(x)=a|x-h|+k$ to find the vertex $(h,k)$ and axis $x=h$;
  \item decide from $a$ whether a V \emph{opens up or down} (min vs.\ max) and is narrower or wider;
  \item describe any absolute value graph as \emph{shifts, a stretch, and a reflection} of $y=|x|$.
\end{itemize}
\end{objectivebox}

\begin{vocabbox}
\small
Fill in each term as we build it together.
\par\vspace{2pt}   % \par is required: \vocabans's \noindent is a no-op mid-paragraph
\vocabans{Absolute value parent function $f(x)=|x|$}{a V opening up, vertex at the origin; the rays $y=x$ ($x\ge 0$) and $y=-x$ ($x<0$)}
\vocabans{Vertex (turning point)}{the V's lowest point (opens up) or highest point (opens down)}
\vocabans{Axis of symmetry}{the vertical line $x=h$ through the vertex; the two rays mirror across it}
\vocabans{Vertex form $a|x-h|+k$}{vertex $(h,k)$, axis $x=h$; $a$ sets opening direction and width}
\vocabans{Reflection / stretch / compression}{$a<0$ flips it to open down; $|a|>1$ narrower, $0<|a|<1$ wider}
\end{vocabbox}

\begin{hookbox}
You drive toward a rest stop, reach it after $2$ hours, and keep going. Your \emph{distance from the
rest stop} (in tens of miles) is $d(t)=30\,|t-2|$.
\begin{itemize}\setlength{\itemsep}{1pt}
  \item Distance at $t=0$? \ans{$60$} \quad At $t=2$? \ans{$0$} \quad At $t=4$? \ans{$60$}
  \item You are \emph{closest} at $t=\ans{$2$}$, where the distance is \ans{$0$}.
\end{itemize}
Plot those points and the shape is a \textbf{V}. A distance is never negative, so the graph bottoms
out at one turning point --- the \emph{vertex} --- and is symmetric about the line through it. Every
absolute value function is this same V, moved and stretched.
\end{hookbox}

\boxguard[20]
\begin{notesbox}{3. Stretch, Compress, and Reflect: $y=a|x|$}
The number $a$ multiplies the \emph{outputs}: it scales the V and, if negative, flips it.
\begin{center}
\begin{tikzpicture}[scale=0.42]
  % y = 2|x|  (narrow)
  \begin{scope}
    \lgrid{-3}{3}{-1}{5}
    \draw[very thick, forest!40, dashed] (-3,3)--(0,0)--(3,3);
    \draw[very thick, forest] (-2.2,4.4)--(0,0)--(2.2,4.4);
    \node[charcoal, font=\scriptsize] at (0,-2.4){$y=2|x|$};
  \end{scope}
  % y = 1/2 |x|  (wide)
  \begin{scope}[xshift=9cm]
    \lgrid{-4}{4}{-1}{5}
    \draw[very thick, forest!40, dashed] (-3,3)--(0,0)--(3,3);
    \draw[very thick, forest] (-4,2)--(0,0)--(4,2);
    \node[charcoal, font=\scriptsize] at (0,-2.4){$y=\tfrac12|x|$};
  \end{scope}
  % y = -|x|  (down)
  \begin{scope}[xshift=19cm]
    \lgrid{-3}{3}{-4}{2}
    \draw[very thick, forest!40, dashed] (-2,2)--(0,0)--(2,2);
    \draw[very thick, forest] (-3,-3)--(0,0)--(3,-3);
    \node[charcoal, font=\scriptsize] at (0,-5.4){$y=-|x|$};
  \end{scope}
\end{tikzpicture}
\end{center}
\begin{itemize}[leftmargin=1.5em, itemsep=1pt]
  \item $|a|>1$ makes the V \ans{narrower}; \quad $0<|a|<1$ makes it \ans{wider}.
  \item $a<0$ \ans{reflects} the V, so it opens \ans{down} instead of up.
  \item The value of $a$ is the \emph{slope} of the \ans{right-hand} ray of the V.
\end{itemize}
\end{notesbox}

\boxguard[23]
\begin{notesbox}{4. Vertex Form: $g(x)=a|x-h|+k$}
All three moves at once. The vertex is $(h,k)$ and the axis of symmetry is $x=h$.
\[ \text{opens \textbf{up} if } a>0 \ (k \text{ is a } \textbf{minimum}); \qquad
   \text{opens \textbf{down} if } a<0 \ (k \text{ is a } \textbf{maximum}). \]
\textbf{Worked:} $g(x)=-2|x-1|+3$.
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabularx}{\linewidth}{@{}X X X X@{}}
Vertex $({\color{keyred}\mathbf{1}},\,{\color{keyred}\mathbf{3}})$ & Axis $x={\color{keyred}\mathbf{1}}$ & Opens \ans{down} & \ans{maximum} value $={\color{keyred}\mathbf{3}}$ \\
\end{tabularx}
\end{center}
Because $a=-2$, the V is \ans{narrower} (narrower / wider) than the parent and opens \ans{down}.
Evaluate at $x=0$ straight from the rule:
\begin{work}
  g(0) &= -2\,|\,0-1\,| + 3 \\
       &= -2\,|{-1}| + 3     \\
       &= -2(1) + 3          \\
       &= 1
\end{work}
\end{notesbox}

\begin{practicebox}
For $g(x) = \tfrac12\,|x+4| - 2$, fill in each feature.
\begin{enumerate}[leftmargin=1.7em, itemsep=3pt]
  \item Vertex $({\color{keyred}\mathbf{-4}},\,{\color{keyred}\mathbf{-2}})$; \quad axis of symmetry $x={\color{keyred}\mathbf{-4}}$.
  \item Opens \ans{up} (up / down), so $k$ is a \ans{min} (max / min) value of \ans{$-2$}.
  \item Compared to $y=|x|$, this graph is shifted \ans{left} $4$ and \ans{down} $2$, and is
        \ans{wider} (narrower / wider).
  \item Evaluate $g(0)$, one step per line.
        \begin{work}
          g(0) &= \tfrac12\,|0+4| - 2 \\
               &= \tfrac12(4) - 2      \\
               &= 2 - 2                \\
               &= 0
        \end{work}
\end{enumerate}
\end{practicebox}
