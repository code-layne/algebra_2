  % ---------------------------------------------------------------- column 1
  \begin{minipage}[t]{0.335\linewidth}\vspace{0pt}
  \cs{A.\ Three Forms of a Parabola}{%
    \textbf{Vertex form} $y=a(x-h)^2+k$\\
    vertex $(h,k)$, axis $x=h$. \ \emph{Sign flips:} $(x+3)^2\Rightarrow h=-3$.\par\vspace{2pt}
    \textbf{Standard form} $y=ax^2+bx+c$\\
    axis $x=-\frac{b}{2a}$; plug in for vertex $y$.\par\vspace{2pt}
    \textbf{Intercept form} $y=a(x-p)(x-q)$\\
    zeros $p,q$; axis $x=\frac{p+q}{2}$.\par\vspace{2pt}
    $a>0$: opens up, \textbf{minimum} $k$, range $y\ge k$.\\
    $a<0$: opens down, \textbf{maximum} $k$, range $y\le k$.\\
    $|a|>1$ narrower; $|a|<1$ wider.\\
    Increasing/decreasing switch at $x=h$.}
  \cs{B.\ Factoring (GCF first!)}{%
    $a^2-b^2=(a+b)(a-b)$\\
    $a^2+2ab+b^2=(a+b)^2$\\
    $a=1$: two numbers that \emph{multiply} to $c$, \emph{add} to $b$.\\
    $x^2-3x-28=(x-7)(x+4)$\par\vspace{2pt}
    \textbf{$a>1$, slip \& slide:} \ $3x^2+10x+8$\\
    Slip $x^2+10x+24=(x+4)(x+6)$\\
    Slide $\big(x+\frac43\big)\big(x+\frac63\big)$ $\to$ up \hl{$(3x+4)(x+2)$}\par\vspace{2pt}
    \textbf{Solve:} $=0$ $\to$ factor $\to$ each factor $=0$.}
  \end{minipage}\hfill
  % ---------------------------------------------------------------- column 2
  \begin{minipage}[t]{0.32\linewidth}\vspace{0pt}
  \cs{C.\ Square Roots \& $i$}{%
    Isolate the square, then $\pm$:\\
    $(x+2)^2=20 \Rightarrow x+2=\pm\sqrt{20}$\\
    \hl{$x=-2\pm2\sqrt5$}\par\vspace{2pt}
    $i=\sqrt{-1}$, \quad $i^2=-1$\\
    $\sqrt{-a}=i\sqrt{a}$: \ $\sqrt{-50}=5i\sqrt2$\\
    \textbf{Pull out $i$ BEFORE multiplying:}\\
    $\sqrt{-4}\cdot\sqrt{-25}=2i\cdot5i=10i^2=\mathbf{-10}$}
  \cs{C.\ Complex Arithmetic ($a+bi$)}{%
    Add/subtract: real with real, $i$ with $i$.\\
    $(3+2i)-(1-5i)=2+7i$\par\vspace{2pt}
    Multiply: FOIL, then $i^2\to -1$.\\
    $(2+i)(3-4i)=6-8i+3i-4i^2$\\
    \phantom{$(2+i)(3-4i)$}$=6-5i+4=$ \hl{$10-5i$}}
  \end{minipage}\hfill
  % ---------------------------------------------------------------- column 3
  \begin{minipage}[t]{0.32\linewidth}\vspace{0pt}
  \cs{D.\ Completing the Square}{%
    $x^2+8x-3=0$\\
    1.\ Move $c$: \ $x^2+8x=3$\\
    2.\ Add $\big(\frac b2\big)^2=16$ to \textbf{both} sides\\
    3.\ $(x+4)^2=19$\\
    4.\ $\pm\sqrt{\ }$: \ \hl{$x=-4\pm\sqrt{19}$}}
  \cs{D.\ Quadratic Formula}{%
    Standard form first: $ax^2+bx+c=0$\\[2pt]
    {\normalsize $x=\dfrac{-b\pm\sqrt{b^2-4ac}}{2a}$}\\[2pt]
    Simplify the root, then divide \emph{every} term.\\
    $\sqrt{-16}=4i$ $\to$ complex answers.}
  \cs{D.\ Discriminant $b^2-4ac$}{%
    \renewcommand{\arraystretch}{1.1}\setlength{\tabcolsep}{2pt}%
    \begin{tabular}{@{}lll@{}}
      $>0$ & two real & 2 $x$-int.\\
      $=0$ & one repeated real & 1 $x$-int.\\
      $<0$ & two complex & 0 $x$-int.
    \end{tabular}}
  \end{minipage}}
