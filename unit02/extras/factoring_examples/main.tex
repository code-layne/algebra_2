\newcommand{\sheet}{%
  \noindent\colorbox{forest}{\parbox{\dimexpr\linewidth-2\fboxsep}{%
    \vspace{2pt}\color{white}\textbf{Algebra 2 \hfill Factoring Quiz --- Worked Examples}\\[1pt]
    {\color{forestmid}\footnotesize\textbf{Use on the quiz. \quad GCF first, always. \quad
    Check: multiply your answer back out.}}\vspace{2pt}}}
  \par\vspace{8pt}\noindent
  \ex{0.325}{1.\ GCF}{Factor $10x^3+15x^2-5x$.}{%
    \text{GCF} &= 5x \\
    &\text{divide each term by } 5x}{$5x(2x^2+3x-1)$}\hfill
  \ex{0.325}{2.\ Difference of Squares}{Factor $16x^2-25$.}{%
    16x^2-25 &= (4x)^2-(5)^2 \\
    a^2-b^2 &= (a+b)(a-b)}{$(4x+5)(4x-5)$}\hfill
  \ex{0.325}{3.\ Trinomial, $a=1$}{Factor $x^2-2x-24$.}{%
    \text{multiply to } {-24} &:\ (-6)(4) \\
    \text{add to } {-2} &:\ -6+4}{$(x-6)(x+4)$}
  \par\vspace{8pt}\noindent
  \ex{0.43}{4.\ Trinomial, $a>1$ (grouping)}{Factor $2x^2+7x+6$.}{%
    ac=12,\ \text{add to } 7 &:\ 4 \text{ and } 3 \\
    \text{split } 7x &:\ 2x^2+4x+3x+6 \\
    \text{group} &:\ 2x(x+2)+3(x+2)}{$(2x+3)(x+2)$}\hfill
  \begin{minipage}[t]{0.555\linewidth}\vspace{0pt}
  \begin{fixedskillbox}[5.\ Solve by Factoring]{white}
  \small
  Solve $3x^2-x=10$.\par\vspace{2pt}
  \begin{minipage}[t]{0.6\linewidth}\vspace{0pt}
  $\begin{aligned}
    3x^2-x-10 &= 0 \\
    ac=-30,\ \text{add to } {-1} &:\ -6,\ 5 \\
    3x(x-2)+5(x-2) &= 0 \\
    (3x+5)(x-2) &= 0 \\
    3x+5=0 \ \text{ or }\ x-2 &= 0
  \end{aligned}$
  \end{minipage}\hfill
  \begin{minipage}[t]{0.37\linewidth}\vspace{0pt}
  \footnotesize\color{slate}
  \textbf{Step 1:} get $0$ on one side.\par\vspace{3pt}
  \textbf{Step 2:} factor.\par\vspace{3pt}
  \textbf{Step 3:} set each factor $=0$ and solve.
  \end{minipage}\par\vspace{4pt}
  \colorbox{forestbg}{\boldmath\bfseries $x=-\tfrac{5}{3}$ \ or \ $x=2$}
  \end{fixedskillbox}
  \end{minipage}}
