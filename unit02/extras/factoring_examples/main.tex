  \ex{0.325}{1.\ GCF}{Factor $10x^3+15x^2-5x$.}{%
    \text{GCF} &= 5x \\
    &\text{divide each term by } 5x}{$5x(2x^2+3x-1)$}\hfill
  \ex{0.325}{2.\ Difference of Squares}{Factor $16x^2-25$.}{%
    16x^2-25 &= (4x)^2-(5)^2 \\
    a^2-b^2 &= (a+b)(a-b)}{$(4x+5)(4x-5)$}\hfill
  \ex{0.325}{3.\ Trinomial, $a=1$}{Factor $x^2-2x-24$.}{%
    \text{multiply to } {-24} &:\ (-6)(4) \\
    \text{add to } {-2} &:\ -6+4}{$(x-6)(x+4)$}
  \par\vspace{8pt}\noindent
  \ex{0.43}{4.\ Trinomial, $a>1$ (slip \& slide)}{Factor $2x^2+7x+6$. \hfill $a\cdot c=2\cdot 6=12$}{%
    \text{Slip:}\ \ & x^2+7x+12 \\
    \text{Factor:}\ \ & (x+3)(x+4) \\
    \text{Slide:}\ \ & \left(x+\tfrac{3}{2}\right)\left(x+\tfrac{4}{2}\right) \\
    \text{Simplify:}\ \ & \left(x+\tfrac{3}{2}\right)(x+2) \\
    \text{Slide up:}\ \ & (2x+3)(x+2)}{$(2x+3)(x+2)$}\hfill
  \begin{minipage}[t]{0.555\linewidth}\vspace{0pt}
  \begin{fixedskillbox}[5.\ Solve by Factoring]{white}
  \small
  Solve $3x^2-x=10$.\par\vspace{2pt}
  \begin{minipage}[t]{0.66\linewidth}\vspace{0pt}
  $\begin{aligned}
    & 3x^2-x-10 = 0 \\
    \text{Slip:}\ \ & x^2-x-30 \\
    \text{Factor:}\ \ & (x-6)(x+5) \\
    \text{Slide:}\ \ & \left(x-\tfrac{6}{3}\right)\left(x+\tfrac{5}{3}\right) \\
    \text{Slide up:}\ \ & (x-2)(3x+5) = 0 \\
    & x-2=0 \ \text{ or }\ 3x+5=0
  \end{aligned}$
  \end{minipage}\hfill
  \begin{minipage}[t]{0.32\linewidth}\vspace{0pt}

  \textbf{Step 2:} factor (slip \& slide).\par\vspace{3pt}
  \textbf{Step 3:} set each factor $=0$ and solve.
  \end{minipage}\par\vspace{4pt}
