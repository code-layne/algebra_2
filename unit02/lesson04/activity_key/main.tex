\begin{headlinebox}{forestbg}
\small\textbf{Pieces, Graphs \& Staircases.} A piecewise function follows different rules on different
parts of its domain. Work with your partner. To \emph{evaluate}, first find the piece whose condition
your input satisfies, then apply that rule --- and watch the boundary ($<$ vs.\ $\le$). On a graph, a
\textbf{closed} dot $\bullet$ includes its endpoint and an \textbf{open} dot $\circ$ excludes it.
\end{headlinebox}

\vspace{0.06in}

\begin{tcolorbox}[colback=white, colframe=black!40, title=\textbf{Tier R --- Remediate},
    fonttitle=\bfseries, arc=1.5mm, left=3mm, right=3mm, top=2mm, bottom=2mm, breakable]
\begin{enumerate}[leftmargin=1.7em, itemsep=9pt]
  \item \textbf{Evaluate} $f(x)=\begin{cases} x+5, & x<0 \\ 2x, & x\ge 0 \end{cases}$ \quad at each input.
        \\[3pt]
        $f(-3)={\color{keyred}\mathbf{2}}$ \qquad $f(0)={\color{keyred}\mathbf{0}}$ \qquad
        $f(4)={\color{keyred}\mathbf{8}}$
  \item \textbf{Read the graph} of $g(x)=\begin{cases} x+1, & x<1 \\ 3, & x\ge 1 \end{cases}$.
        \begin{center}
        \begin{tikzpicture}[scale=0.5]
          \lgrid{-1}{4}{-1}{4}
          \draw[very thick, forest] (-1,0)--(1,2);
          \opendot{1}{2}
          \draw[very thick, forest] (1,3)--(4,3);
          \closeddot{1}{3}
        \end{tikzpicture}
        \end{center}
        $g(0)={\color{keyred}\mathbf{1}}$; \quad $g(1)={\color{keyred}\mathbf{3}}$; \quad
        $g(3)={\color{keyred}\mathbf{3}}$; \quad
        the \emph{closed} dot is at $({\color{keyred}\mathbf{1}},\,{\color{keyred}\mathbf{3}})$.
  \item \textbf{Open or closed?} When you graph a piece\ldots
        \begin{itemize}[leftmargin=1.5em, itemsep=2pt]
          \item defined for $x>3$, the endpoint at $x=3$ is drawn \ans{open} (open / closed).
          \item defined for $x\le 3$, the endpoint at $x=3$ is drawn \ans{closed} (open / closed).
        \end{itemize}
\end{enumerate}
\end{tcolorbox}

\begin{tcolorbox}[colback=white, colframe=black!40, title=\textbf{Tier A --- Approaching Proficiency},
    fonttitle=\bfseries, arc=1.5mm, left=3mm, right=3mm, top=2mm, bottom=2mm, breakable]
\begin{enumerate}[leftmargin=1.7em, itemsep=10pt]
  \item \textbf{Evaluate across the boundary.} For $h(x)=\begin{cases} -x+1, & x\le 2 \\ 2x-6, & x>2 \end{cases}$:
        \\[3pt]
        $h(0)={\color{keyred}\mathbf{1}}$ \quad $h(2)={\color{keyred}\mathbf{-1}}$ \quad
        $h(5)={\color{keyred}\mathbf{4}}$. \quad
        Is $h$ continuous at $x=2$? \ans{no} \; (top gives \ans{$-1$}, bottom gives \ans{$-2$}).
  \item \textbf{Read the graph} below.
        \begin{center}
        \begin{tikzpicture}[scale=0.5]
          \lgrid{-2}{5}{-3}{3}
          \draw[navy, dashed, semithick] (2,-3)--(2,3);
          \draw[very thick, forest] (-2,-2)--(2,2);
          \opendot{2}{2}
          \draw[very thick, forest] (2,-2)--(5,-2);
          \closeddot{2}{-2}
        \end{tikzpicture}
        \end{center}
        Domain \ans{all real numbers}; \; $f(1)={\color{keyred}\mathbf{1}}$; \;
        $f(2)={\color{keyred}\mathbf{-2}}$; \; increasing on \ans{$(-\infty,2)$}; \; constant on
        \ans{$[2,\infty)$}. \; Continuous at $x=2$? \ans{no}.
  \item \textbf{Write it as pieces.} Rewrite $|x-3|$ as a two-piece function.
        \[ |x-3|=\begin{cases} {\color{keyred}\mathbf{x-3}}, & x\ge 3 \\[3pt] {\color{keyred}\mathbf{3-x}}, & x<3 \end{cases} \]
  \item \textbf{Greatest integer.} $\lfloor 2.9\rfloor={\color{keyred}\mathbf{2}}$ \quad
        $\lfloor -1.2\rfloor={\color{keyred}\mathbf{-2}}$ \quad $\lfloor 4\rfloor={\color{keyred}\mathbf{4}}$.
\end{enumerate}
\end{tcolorbox}

\begin{tcolorbox}[colback=white, colframe=black!40, title=\textbf{Tier E --- Extension},
    fonttitle=\bfseries, arc=1.5mm, left=3mm, right=3mm, top=2mm, bottom=2mm, breakable]
\begin{enumerate}[leftmargin=1.7em, itemsep=10pt]
  \item \textbf{Model overtime pay.} A worker earns \$15/hr for up to $40$ hours, then \$22.50/hr
        (time-and-a-half) for every hour beyond $40$. Let $P(h)$ be the weekly pay for $h$ hours.
        \begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=5pt]
          \item Complete the rule:
                $P(h)=\begin{cases} {\color{keyred}\mathbf{15h}}, & 0\le h\le 40 \\[3pt] 600+{\color{keyred}\mathbf{22.5(h-40)}}, & h>40 \end{cases}$
          \item Find $P(40)$ and $P(45)$, and say what $P(45)$ means.
                \ansline{$P(40)=600$, \; $P(45)=600+22.5(5)=712.50$; \; $P(45)$ is the pay for a 45-hour week --- \$600 for the first 40 hours plus 5 overtime hours at \$22.50.}
        \end{enumerate}
  \item \textbf{Read a step (parking) graph.} A garage charges \$2 for each hour \emph{or part of an
        hour}. The cost $C(t)$ for $t$ hours parked is graphed.
        \begin{center}
        \begin{tikzpicture}[scale=0.5]
          \lgrid{0}{4}{0}{8}
          \draw[very thick, forest] (0,2)--(1,2); \opendot{0}{2} \closeddot{1}{2}
          \draw[very thick, forest] (1,4)--(2,4); \opendot{1}{4} \closeddot{2}{4}
          \draw[very thick, forest] (2,6)--(3,6); \opendot{2}{6} \closeddot{3}{6}
          \draw[very thick, forest] (3,8)--(4,8); \opendot{3}{8} \closeddot{4}{8}
        \end{tikzpicture}
        \end{center}
        $C(1.5)={\color{keyred}\mathbf{4}}$; \quad $C(3)={\color{keyred}\mathbf{6}}$. \quad Why does the
        graph \emph{jump} at each whole hour?
        \ansline{Any part of a new hour is charged a full \$2, so the cost is constant across each hour, then jumps \$2 at every whole hour.}
  \item \textbf{Reason about continuity.} A piecewise function's two pieces meet with \emph{no jump} at a
        boundary $x=a$ exactly when what is true about the two rules at $a$? Explain.
        \ansline{The two pieces meet with no jump exactly when both rules give the \emph{same value} at $x=a$.}
        \ansline{If the two values differ, the graph jumps there --- a discontinuity (as in items 1--2).}
\end{enumerate}
\end{tcolorbox}

\begin{teachernote}
Tier~A is the conceptual core: item~1 forces the boundary decision ($x=2$ belongs to the $\le$ piece, so
$h(2)=-1$) and then tests continuity by comparing the two rules' values there ($-1$ vs.\ $-2$ $\Rightarrow$
jump). Item~2 is the A2.F.2a ``graph with a discontinuity'' --- push students to read the \emph{closed}
dot for $f(2)$ and to name the \emph{constant} interval. Common slips: reading $\lfloor -1.2\rfloor$ as
$-1$ (it is $-2$ --- round \emph{down}); in E1, forgetting the \$600 base before overtime; in E2, thinking
$C(1.5)=3$ (any part of the 2nd hour already costs the full \$4). E3 is the payoff idea: pieces are
continuous at a boundary $\Leftrightarrow$ they share a value there.
\end{teachernote}
