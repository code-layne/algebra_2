\begin{tcolorbox}[colback=forestbg, colframe=forest, boxrule=0.9pt, arc=2mm,
    left=2mm, right=2mm, top=1.5mm, bottom=1.5mm]
    \textbf{Primary Objective:} Students solve quadratics that do not factor --- by
    \emph{completing the square} ($a=1$), then by the \emph{quadratic formula}, which is
    completing the square done once on $ax^2+bx+c=0$ --- simplifying to real or complex answers.
    They then use the \emph{discriminant} $b^2-4ac$ to predict, \emph{before} solving, two real /
    one repeated real / two complex solutions and the matching two / one / zero $x$-intercepts.
    Lesson~2.3 supplied $i$; 2.4 is where a negative radicand becomes an answer, not a dead end.
    \\[2pt]
    \textbf{Standards (2023 VA SOL):} \textbf{A2.EI.2b} (solve a quadratic equation in one
    variable over the set of complex numbers algebraically --- completing the square and the
    quadratic formula), \textbf{A2.EI.2d} (verify solutions and interpret them --- the real
    roots are the $x$-intercepts), \textbf{A2.F.2d} (the discriminant's link to the number of
    $x$-intercepts), and \textbf{A2.EI.2a} (a quadratic equation from a context --- the profit
    rung and the volleyball bonus).
    \\[2pt]
    \textbf{Lesson model:} \emph{Slides first.} The deck is the lesson and its printed handout is
    the student's notes, and students are \emph{solving the whole hour}: a warm-up on screen as
    they walk in; three rounds of a short \textbf{definition} and \textbf{example} ($\le$3 min),
    a \textbf{ladder} of three problems (\rung1\ \rung2\ \rung3), and a \textbf{Show me} or
    \textbf{Find the mistake}; a self-scored \textbf{Final Round}; and the homework started in
    class. \textbf{Homework source:} DeltaMath, online --- set \textbf{2.4 The Quadratic
    Formula}; nothing in the packet.
\end{tcolorbox}

\begin{skillbox}[Priority Ideas \& Skills]{goldbox}
\begin{tabularx}{\linewidth}{@{}X|X@{}}
\textbf{Priority Ideas \& Skills} & \textbf{Key Understandings (the ``why'')} \\[2pt]
\hline
\vspace{-6pt}
\begin{itemize}[leftmargin=1.1em, nosep, after=\vspace{2pt}]
  \item Complete the square on $x^2+bx+c=0$: move $c$, add $\left(\frac b2\right)^2$ to
        \textbf{both} sides, square-root with $\pm$.
  \item Write \textbf{standard form} first; read $a$, $b$, $c$ \textbf{with signs}.
  \item Apply the \textbf{quadratic formula}; simplify to radical or $p\pm qi$ form.
  \item Compute the \textbf{discriminant} and classify the solutions and $x$-intercepts
        without solving.
\end{itemize}
&
\vspace{-6pt}
\begin{itemize}[leftmargin=1.1em, nosep, after=\vspace{2pt}]
  \item Adding $\left(\frac b2\right)^2$ \emph{makes} a perfect square --- 2.2's pattern run
        backwards, and 2.1's vertex form.
  \item The formula is completing the square in general, so it works on \emph{every} quadratic.
  \item \textbf{Target misconception: a negative discriminant means ``no solution.''} It
        means no \emph{real} solutions and no $x$-intercepts; there are still two complex
        (conjugate) solutions. Close cousins: $-3^2$ for $(-3)^2$, and reading $b$ before the
        equation equals zero.
  \item Real roots are $x$-intercepts; complex roots are not on the $x$-axis.
\end{itemize}
\end{tabularx}
\end{skillbox}

\begin{skillbox}[{Vocabulary, Concepts, \& Theorems --- one definition frame each}]{sky}
\renewcommand{\arraystretch}{1.3}
\begin{tabularx}{\linewidth}{@{}l X@{}}
\textbf{Completing the square} & Adding $\left(\frac b2\right)^2$ to $x^2+bx$ makes a
        perfect-square trinomial, $x^2+bx+\left(\frac b2\right)^2=\left(x+\frac b2\right)^2$;
        completing the square is adding that number to \emph{both sides} of an equation so one
        side becomes a perfect square. \\
\textbf{Quadratic formula} & The solutions of $ax^2+bx+c=0$ ($a\ne0$) are
        $x=\frac{-b\pm\sqrt{b^2-4ac}}{2a}$ --- completing the square done once, on the
        letters; it solves every quadratic, with real or complex answers. \\
\textbf{Standard form} & $ax^2+bx+c=0$ --- everything on one side; read $a$, $b$, $c$ with
        their signs. \\
\textbf{Discriminant} & In $ax^2+bx+c=0$, the number under the radical, $D=b^2-4ac$. Its sign
        tells you the solutions before you solve: $D>0$ two real (two $x$-intercepts); $D=0$
        one repeated real (touches once); $D<0$ no real solutions --- two complex ones,
        $p\pm qi$ --- no $x$-intercepts. \\
\end{tabularx}
\end{skillbox}

\begin{skillbox}[Engagement --- how this hour keeps everyone solving]{goldbox}
\textbf{Nobody waits:} every practice frame is a ladder --- \rung1\ everyone (a quick win),
\rung2\ everyone, \rung3\ early finishers. \textbf{Everyone commits, safely:} \emph{Show me} is
1--4 fingers on the count of three, all at once. \textbf{Critique a stranger:} \emph{Find the
mistake} puts the error on an anonymous Student A. \textbf{Their world:} the garden-club
flower-bunch profit (Round~3 \rung3) and the volleyball-serve bonus. \textbf{Short clock:} every
practice frame carries its minutes. Praise the climb more than the answer --- and make a point of
Round~1's \rung3: whoever reached $-1\pm2i$ owns the crux later.
\end{skillbox}

\begin{skillbox}[Warm-Up (5 min, on the slides as they walk in)]{forestbg}
Three quick wins; the answers are one click. \textbf{(1)} $(x+3)^2=x^2+6x+9$ --- the perfect
square Round~1 builds on purpose. \textbf{(2)} $\sqrt{-16}=4i$ --- 2.3, needed whenever the
radicand goes negative. \textbf{(3)} $3x^2-x-4=0$: $a=3$, $b=-1$, $c=-4$ --- signs travel with
the numbers (Round~2). The reveal ends on \emph{Hold on to this}: given $x^2+6x=7$, what would
you add to get $(x+3)^2$? (Nine --- that is Round~1, and its \rung1.) Then the targets frame in
30 seconds.
\end{skillbox}

\begin{skillbox}[The Lesson --- three rounds of definition, example, ladder, check (40 min)]{forestbg}
\begin{multicols}{2}
\textbf{Round 1 --- completing the square (13 min).} Definition and Example 1 in 3 minutes:
$x^2-2x-3=0\Rightarrow(x-1)^2=4\Rightarrow x=3,-1$ --- the 2.2 roots, and 2.1's vertex form
$(x-1)^2-4$. Ladder (5 min): \rung1\ $(x+3)^2=16$: $1$, $-7$; \rung2\ $(x-2)^2=5$:
$2\pm\sqrt5$; \rung3\ $(x+1)^2=-4$: $-1\pm2i$ --- tell them to keep it. Watch for the forgotten
$\pm$. Show me $x^2+10x=3$: answer \textbf{2} ($+25$ both sides, $=28$); \textbf{1} = one side
only; \textbf{3} = squared $b$; \textbf{4} = halved, never squared.

\textbf{Round 2 --- the quadratic formula (13 min).} Definition (``completing the square on the
letters'') and Example 2 in 3 minutes: the anchor again, $\frac{2\pm4}{2}$, then
$2x^2+3x=1\to\frac{-3\pm\sqrt{17}}{4}$ --- point at \emph{standard form first}. Ladder (5 min):
\rung1\ $\frac{-5\pm\sqrt{13}}{2}$; \rung2\ $3x^2-2x-4=0$: $\frac{1\pm\sqrt{13}}{3}$ (watch
$\sqrt{52}=2\sqrt{13}$ and dividing \emph{every} term by 2); \rung3\ $2\pm3i$. Find the mistake:
Student A read $b=3$, $c=5$ straight off $2x^2=3x+5$ and got complex roots; standard form gives
$b=-3$, $c=-5$, $D=49$, $x=\frac52$ or $-1$. Ask ``what should Student A have done first?''

\columnbreak
\textbf{Round 3 --- the discriminant: the crux (14 min).} Definition in 90 seconds --- point
across the three parabolas $y=x^2-2x+c$ for $c=-3,1,5$: same $b$, and $D=16,0,-16$ walks the
curve up off the axis. Example 3: $D=17$, $0$, $-7$; say ``no \emph{real} solutions --- two
complex'' aloud every time. Ladder (5 min): \rung1\ $D=0$, one repeated real, touches;
\rung2\ $D=-23$, two complex, no $x$-intercepts; \rung3\ the garden club, $-x^2+12x-36=0$,
$D=0$: exactly one price, \$6. \textbf{The frame not to rush --- the crux Show me:} $x^2+2x+5$,
$D=-16$. Answer \textbf{2}, $-1\pm2i$, no $x$-intercepts. \textbf{1} = ``negative means no
solution'' (\emph{the target misconception}); \textbf{3} = complex roots placed on the
$x$-axis; \textbf{4} = $D$ mistaken for the answer. Before revealing, ask a \textbf{1} and a
\textbf{2} to argue --- then point back at Round~1's \rung3: they already found the solutions.
\end{multicols}
\end{skillbox}

\boxguard
\begin{skillbox}[Final Round (5 min)]{forestbg}
Four mixed problems, one per round, alone, 5-minute timer: \textbf{(1)} $(x+4)^2=25$: $1$,
$-9$; \textbf{(2)} $2x^2-6x+3=0$: $\frac{3\pm\sqrt3}{2}$; \textbf{(3)} $D=0$ for $9x^2-12x+4$:
one repeated real; \textbf{(4)} $x^2-6x+10=0$: $D=-4$, no real solutions but two, $3\pm i$ ---
the crux's twin (``zero solutions'' is the wrong answer to ``how many solutions?'').
\textbf{Bonus:} the volleyball serve, $-16t^2+24t-10=0$, $D=-64$: it never reaches 16 ft (it
peaks at 15). Students score themselves out of 4; collect the scores on fingers --- that is
your exit read. Say the \emph{Watch out} aloud as you reveal.
\end{skillbox}

\boxguard
\begin{skillbox}[Homework --- scored, started in class, due the first class after two study halls]{goldbox}
\textbf{Source: DeltaMath, online} --- set \textbf{2.4 The Quadratic Formula}; nothing goes in
the packet. Build the set to cover, in order: completing the square with $a=1$ (including one
with complex roots); the quadratic formula from standard form; at least two equations
\emph{not} already set to zero; formula answers in simplest radical and $p\pm qi$ form; the
discriminant --- classify the solutions and the number of $x$-intercepts. \textbf{Formative
check:} the last discriminant problem with $D<0$ that asks for the number of solutions ---
sort into \emph{two complex} / \emph{``no solution''} / \emph{other}; if the middle pile is
large, open 2.5 by re-solving $x^2+2x+5=0$. \textbf{Preview:} 2.5 --- when a parabola meets a
line (quadratic systems), and quadratics as models.
\end{skillbox}

\boxguard
\begin{skillbox}[Active Monitoring --- Watch For]{redbox}
\begin{itemize}[leftmargin=1.2em, nosep]
  \item \textbf{Round 1 ladder:} the missing $\pm$; $\left(\frac b2\right)^2$ added to one side only.
  \item \textbf{Round 2 ladder \rung2 and Find the mistake:} reading $a$, $b$, $c$ before
        standard form; $-b^2$ for $(-b)^2$; dividing only part of the numerator by $2a$.
  \item \textbf{Round 3 crux Show me and Final Round 4:} ``negative $D$ = no solution''
        (\emph{the target misconception}).
  \item \textbf{Attitude:} a student parked at \rung1\ with nothing written --- crouch, point at
        the example on the handout, ask for just the first step.
\end{itemize}
Cold-call prompts (after the fingers, never before): ``What does a negative discriminant rule
out --- and what does it not?'' \; ``Why must the equation equal zero before you read $b$?''
\end{skillbox}

\begin{teachernote}[Slides]
Pacing: warm-up 5, rounds 13 / 13 / 14, Final Round 5, homework 10 --- 60 minutes. In each round
the definition and example together take \textbf{at most 3 minutes}; each ladder 5; each check
2--3. Reveal a ladder when most of the room has \rung2; the \rung3\ students keep going. If
behind, trim Round~1's Show me to a 30-second finger read, or reveal Round~2's Find the mistake
after one minute --- never cut Round~3's crux Show me. Students write in the handout's notes
column, which prints without the answers.
\end{teachernote}

\begin{teachernote}[Homework]
The homework is the DeltaMath set \textbf{2.4 The Quadratic Formula}, online --- nothing is in
the packet. In the last 10 minutes students open it on a device; launch problem~1 aloud, then
circulate with the Watch For list in mind. The formative check is the $D<0$ ``how many
solutions'' problem (sort above). Due the first class after two study halls; announce the date
in class and on TurtleNet.
\end{teachernote}
