% ═══ ROUND 1 — COMPLETING THE SQUARE (a = 1) ═══════════════════════════════
\begin{frame}[t, plain]
  \forestheader{Definition: completing the square}
  \sectionlabel{Round 1 --- make a perfect square}
  \begin{block}{Completing the square}
    Adding $\left(\frac b2\right)^2$ to $x^2+bx$ makes a perfect-square trinomial:
    \[ x^2+bx+\left(\tfrac b2\right)^2=\left(x+\tfrac b2\right)^2. \]
    \textbf{Completing the square} is adding that number to \emph{both sides} of an equation so
    one side becomes a perfect square.
  \end{block}
  \vspace{2pt}
  \begin{block}{To solve $x^2+bx+c=0$ by completing the square}
    \begin{enumerate}\setlength{\itemsep}{1pt}
      \item Move the constant to the right: $\;x^2+bx=-c$.
      \item Add $\left(\frac b2\right)^2$ to \textbf{both} sides; write the left side as $\left(x+\frac b2\right)^2$.
      \item Take the square root of both sides --- remember $\pm$ --- and solve.
    \end{enumerate}
  \end{block}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Example 1: solve $x^2-2x-3=0$ by completing the square}
  \sectionlabel{Watch --- 2 minutes}
  \[
  \begin{aligned}
    x^2-2x &= 3 && \text{move the constant} \\
    x^2-2x+1 &= 3+1 && \left(\tfrac{-2}{2}\right)^2=1 \text{ on both sides} \\
    (x-1)^2 &= 4 && \text{a perfect square} \\
    x-1 &= \pm2 && \text{square root, } \pm \\
    x=3 \quad\text{or}\quad x &= -1
  \end{aligned}
  \]
  \pause
  \emph{Same roots as factoring in 2.2.} And $(x-1)^2-4=0$ is 2.1's vertex form: vertex
  $(1,-4)$.
\end{frame}

% ═══ ROUND 2 — THE QUADRATIC FORMULA ═══════════════════════════════════════
\begin{frame}[t, plain]
  \forestheader{Definition: the quadratic formula}
  \sectionlabel{Round 2 --- one formula for every quadratic}
  \begin{block}{The quadratic formula}
    The solutions of $ax^2+bx+c=0$ ($a\ne0$) are
    \[ x=\frac{-b\pm\sqrt{b^2-4ac}}{2a}. \]
    It is completing the square done once, on the letters --- so it solves \emph{every}
    quadratic, with real or complex answers.
  \end{block}
  \vspace{2pt}
  \begin{block}{Using it}
    \begin{enumerate}\setlength{\itemsep}{1pt}
      \item Write the equation in \textbf{standard form} $ax^2+bx+c=0$ --- everything on one side.
      \item Read $a$, $b$, and $c$ \textbf{with their signs}.
      \item Substitute in parentheses --- $(-3)^2=9$, not $-9$ --- and simplify.
    \end{enumerate}
  \end{block}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Example 2: the formula, twice}
  \sectionlabel{Watch --- 2 minutes}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \textbf{(a)} $x^2-2x-3=0$: \; $a=1$, $b=-2$, $c=-3$.
      \[
      \begin{aligned}
        x &= \frac{-(-2)\pm\sqrt{(-2)^2-4(1)(-3)}}{2(1)} \\
          &= \frac{2\pm\sqrt{16}}{2}=\frac{2\pm4}{2}
      \end{aligned}
      \]
      $x=3$ or $x=-1$ --- Round 1's answer.
    \end{column}
    \pause
    \begin{column}{0.49\textwidth}
      \textbf{(b)} $2x^2+3x=1$. \; Standard form first:
      \[ 2x^2+3x-1=0 \qquad a=2,\; b=3,\; c=-1 \]
      \[
      \begin{aligned}
        x &= \frac{-3\pm\sqrt{3^2-4(2)(-1)}}{2(2)} \\
          &= \frac{-3\pm\sqrt{17}}{4}
      \end{aligned}
      \]
      It does not factor --- the formula does not care.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Find the mistake}
  \sectionlabel[goldacc]{2 minutes --- Student A's answer is wrong. Find the step and fix it}
  \textbf{Student A} solved $2x^2=3x+5$:
  \[
  \begin{aligned}
    a=2,\quad b=3,\quad c&=5 \\
    x &= \frac{-3\pm\sqrt{3^2-4(2)(5)}}{2(2)}=\frac{-3\pm\sqrt{-31}}{4} \\
    &\text{``Two complex solutions.''}
  \end{aligned}
  \]
  \reveal{%
    \begin{block}{The mistake}
      Student A read $a$, $b$, $c$ \emph{before} writing standard form. First
      $2x^2-3x-5=0$, so $b=-3$, $c=-5$: \;
      $x=\dfrac{3\pm\sqrt{9+40}}{4}=\dfrac{3\pm7}{4}$, so $x=\dfrac52$ or $x=-1$ --- two
      \emph{real} solutions.
    \end{block}}
\end{frame}

% ═══ ROUND 3 — THE DISCRIMINANT (the crux) ═════════════════════════════════
\begin{frame}[t, plain]
  \forestheader{Definition: the discriminant}
  \sectionlabel{Round 3 --- predict before you solve}
  \begin{block}{The discriminant}
    In $ax^2+bx+c=0$, the number under the radical, $D=b^2-4ac$, is the
    \textbf{discriminant}. Its sign tells you the solutions \emph{before} you solve.
  \end{block}
  \vspace{2pt}
  \begin{columns}[T]
    \begin{column}{0.32\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.26]
        \lgrid{-2}{4}{-4}{5}
        \begin{scope}\clip (-2,-4) rectangle (4,5);
          \draw[very thick, forest, domain=-2:4, samples=60, smooth]
            plot (\x,{(\x-1)*(\x-1)-4});
        \end{scope}
        \fill[redacc] (-1,0) circle (5pt);
        \fill[redacc] (3,0) circle (5pt);
      \end{tikzpicture}\\[-1pt]
      {\scriptsize $y=x^2-2x-3$, \; $D=16$ \\ $D>0$: \textbf{two real} solutions ---
      \textbf{two} $x$-intercepts}
    \end{column}
    \begin{column}{0.32\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.26]
        \lgrid{-2}{4}{-4}{5}
        \begin{scope}\clip (-2,-4) rectangle (4,5);
          \draw[very thick, forest, domain=-1.3:3.3, samples=60, smooth]
            plot (\x,{(\x-1)*(\x-1)});
        \end{scope}
        \fill[redacc] (1,0) circle (5pt);
      \end{tikzpicture}\\[-1pt]
      {\scriptsize $y=x^2-2x+1$, \; $D=0$ \\ $D=0$: \textbf{one repeated real} solution ---
      touches \textbf{once}}
    \end{column}
    \begin{column}{0.32\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.26]
        \lgrid{-2}{4}{-4}{5}
        \begin{scope}\clip (-2,-4) rectangle (4,5);
          \draw[very thick, forest, domain=-0.1:2.1, samples=40, smooth]
            plot (\x,{(\x-1)*(\x-1)+4});
        \end{scope}
      \end{tikzpicture}\\[-1pt]
      {\scriptsize $y=x^2-2x+5$, \; $D=-16$ \\ $D<0$: \textbf{no real} solutions ---
      \textbf{two complex} ones, $p\pm qi$ --- \textbf{no} $x$-intercepts}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Example 3: predict, don't solve}
  \sectionlabel{Watch --- 2 minutes}
  \begin{itemize}\setlength{\itemsep}{8pt}
    \item $2x^2-5x+1=0$: \; $D=(-5)^2-4(2)(1)=25-8=17$ \pause
          $>0$: two real solutions, two $x$-intercepts.
    \pause
    \item $4x^2+4x+1=0$: \; $D=4^2-4(4)(1)=16-16=0$ \pause
          $=0$: one repeated real solution; the graph touches once.
    \pause
    \item $x^2-3x+4=0$: \; $D=(-3)^2-4(1)(4)=9-16=-7$ \pause
          $<0$: \textbf{no real} solutions --- but \textbf{two complex} solutions; no
          $x$-intercepts.
  \end{itemize}
  \pause
  \vspace{4pt}
  \emph{Always:} $b^2$ is never negative --- $(-5)^2=25$. Only $-4ac$ can pull $D$ below zero.
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Your turn}
  \sectionlabel[goldacc]{5 minutes --- start at one star and climb}
  Find $D$. Then say how many solutions, what kind, and how many $x$-intercepts.
  \begin{itemize}\setlength{\itemsep}{6pt}
    \item[\rung1] $x^2-6x+9=0$
    \item[\rung2] $3x^2-x+2=0$
    \item[\rung3] The garden club sells flower bunches at $x$ dollars each; its profit is
          $P=-x^2+12x-20$ dollars. Can the profit reach \$16? Use $D$.
  \end{itemize}
  \vspace{2pt}
  \reveal{%
    \begin{block}{Check}
      \rung1\ $D=36-36=0$: one repeated real solution; touches once \\[1pt]
      \rung2\ $D=1-24=-23$: no real --- two complex solutions; no $x$-intercepts \\[1pt]
      \rung3\ $-x^2+12x-36=0$: $D=144-4(-1)(-36)=0$ --- exactly one price, $x=6$: \$6 a bunch.
    \end{block}}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Show me: $x^2+2x+5=0$ has $D=-16$. Which is true?}
  \sectionlabel[redacc]{The one that matters --- 1 to 4 fingers on the count of three}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \opt1{No solution --- $D$ is negative.} \\[12pt]
      \opt3{Two complex solutions; the graph crosses the $x$-axis twice.}
    \end{column}
    \begin{column}{0.47\textwidth}
      \opt2{Two complex solutions, $-1\pm2i$; the graph has no $x$-intercepts.} \\[12pt]
      \opt4{$D=-16$, so the solutions are $x=\pm4i$.}
    \end{column}
  \end{columns}
  \vspace{8pt}
  \reveal{%
    \begin{block}{Answer: \textbf{2}}
      You solved it in Round 1: $x=-1\pm2i$. A negative $D$ means no \emph{real} solutions ---
      not \emph{no} solutions. \textbf{1} forgets the complex ones; \textbf{3} forgets that
      complex solutions are \emph{not} on the $x$-axis; \textbf{4} took $D$ for the answer ---
      $\sqrt D$ is only one piece of the formula.
    \end{block}}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Final Round}
  \sectionlabel[goldacc]{5 minutes --- on your own --- score yourself out of 4}
  \begin{enumerate}\setlength{\itemsep}{3pt}
    \item Complete the square: $x^2+8x=9$. \qquad 2.\ Use the formula: $2x^2=6x-3$.
    \setcounter{enumi}{2}
    \item Find $D$ for $9x^2-12x+4=0$ and say what it means.
    \item How many \emph{real} solutions does $x^2-6x+10=0$ have? How many solutions? Find them.
  \end{enumerate}
  {\small\textbf{Bonus (+1):} A volleyball serve's height is $h=-16t^2+24t+6$ feet after $t$
  seconds. Does it ever reach $16$ ft? Use the discriminant.}
  \reveal{%
    \begin{block}{Check --- one point each}
      1.\ $(x+4)^2=25$: $x=1$ or $-9$ \quad 2.\ $2x^2-6x+3=0$: $x=\frac{3\pm\sqrt3}{2}$ \quad
      3.\ $D=0$: one repeated real \\[1pt]
      4.\ $D=-4$: \textbf{no real} solutions, but \textbf{two} solutions: $x=3\pm i$ \quad
      \textbf{Bonus:} $-16t^2+24t-10=0$, $D=576-640=-64$: never --- it peaks at $15$ ft. \\[2pt]
      \textbf{Watch out:} a negative discriminant means no \emph{real} solutions and no
      $x$-intercepts --- there are still \textbf{two complex} solutions. And $(-3)^2=+9$.
    \end{block}}
\end{frame}
