% ═══ ROUND 1 — ADD AND SUBTRACT ═════════════════════════════════════════════
\begin{frame}[t, plain]
  \forestheader{Definition: adding and subtracting complex numbers}
  \sectionlabel{Round 1 --- combine like parts}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \begin{block}{Add or subtract}
        To add or subtract complex numbers, combine the \textbf{real parts} with each other and
        the \textbf{imaginary parts} with each other:
        \[ (a+bi)+(c+di)=(a+c)+(b+d)i. \]
        When you subtract, the minus sign reaches \emph{both} parts of the second number.
      \end{block}
      \vspace{2pt}
      \begin{block}{Standard form}
        Answer as $a+bi$, real part first. Rewrite $\sqrt{-16}$ as $4i$ first.
      \end{block}
    \end{column}
    \begin{column}{0.38\textwidth}
      \centering
      \textbf{It is combining like terms.}\\[8pt]
      \renewcommand{\arraystretch}{1.6}
      \begin{tabular}{r@{\;}l}
        $(5+3x)+(2-7x)$ & $=7-4x$ \\
        $(5+3i)+(2-7i)$ & $=7-4i$ \\
      \end{tabular}\\[8pt]
      {\scriptsize $i$ behaves like a variable here: \\ numbers with numbers, $i$'s with $i$'s.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Example 1: add, subtract, and simplify first}
  \sectionlabel{Watch --- 2 minutes}
  \begin{itemize}\setlength{\itemsep}{10pt}
    \item \textbf{(a)} $(7+2i)+(-3+5i)$ \pause $=(7-3)+(2+5)i=4+7i$
    \pause
    \item \textbf{(b)} $(6-4i)-(2-9i)$ \pause $=6-4i-2+9i$ \pause $=4+5i$ \quad
          \emph{(the minus turns $-9i$ into $+9i$)}
    \pause
    \item \textbf{(c)} $(3+\sqrt{-16})+(1-\sqrt{-4})$ \pause $=(3+4i)+(1-2i)$ \pause $=4+2i$
          \quad \emph{($i$ out first, then combine)}
  \end{itemize}
\end{frame}

\begin{frame}[t, plain]

  \begin{itemize}\setlength{\itemsep}{6pt}

      \rung3\ $(12+7i)-(5-3i)=12+7i-5+3i=7+10i$, so the code is $7$--$10$.
    \end{block}}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Show me: $(5-2i)-(3-6i)$}
  \sectionlabel[goldacc]{Pick one --- 1 to 4 fingers on the count of three}
  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \opt1{$2-8i$} \\[12pt]
      \opt3{$8-8i$}
    \end{column}
    \begin{column}{0.45\textwidth}
      \opt2{$2+4i$} \\[12pt]
      \opt4{$2-4i$}

      $5-2i-3+6i=2+4i$. \; \textbf{1} subtracted the $3$ but not the $-6i$ --- the minus reaches

      it is $-2i+6i=+4i$.
    \end{block}}
\end{frame}

% ═══ ROUND 2 — MULTIPLY ═════════════════════════════════════════════════════
\begin{frame}[t, plain]
  \forestheader{Definition: multiplying complex numbers}
  \sectionlabel{Round 2 --- distribute, then $i^2=-1$}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \begin{block}{Multiply}
        To multiply complex numbers, distribute every term of one to every term of the other,
        exactly as with binomials. Then replace every $i^2$ with $-1$ and combine into
        $a+bi$.
      \end{block}
      \vspace{2pt}
      \begin{block}{The one new fact}
        $i^2=-1$. So $-3i^2=-3(-1)=+3$ and $4i^2=-4$: an $i^2$ term is a \emph{real} number,
        and it changes sign.
      \end{block}
    \end{column}
    \begin{column}{0.4\textwidth}
      \centering
      \textbf{The box for $(2+3i)(4-i)$}\\[6pt]
      \renewcommand{\arraystretch}{1.7}
      \begin{tabular}{c|c|c|}
        \multicolumn{1}{c}{} & \multicolumn{1}{c}{$4$} & \multicolumn{1}{c}{$-i$} \\
        \cline{2-3}
        $2$  & $8$   & $-2i$ \\ \cline{2-3}
        $3i$ & $12i$ & $-3i^2$ \\ \cline{2-3}
      \end{tabular}\\[6pt]
      {\scriptsize Four products, then $-3i^2=+3$: \\ $8+10i+3=11+10i$.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Example 2: multiply}
  \sectionlabel{Watch --- 2 minutes}
  \textbf{(a)} $3i(4-5i)=12i-15i^2$ \pause $=12i+15=15+12i$ \quad \emph{(real part first)}
  \pause
  \[
  \begin{aligned}
    \textbf{(b)}\quad (2+3i)(4-i) &= 8-2i+12i-3i^2 && \text{distribute} \\
                                  &= 8+10i-3(-1)   && i^2=-1 \\
                                  &= 11+10i        && \text{combine}
  \end{aligned}
  \]
  \pause
  \textbf{(c)} $(1-3i)^2=(1-3i)(1-3i)=1-3i-3i+9i^2$ \pause $=1-6i-9=-8-6i$ \quad
  \emph{(write the square out twice --- keep the middle terms)}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Your turn}
  \sectionlabel[goldacc]{7 minutes --- start at one star and climb}
  Write each answer in standard form $a+bi$.
  \begin{itemize}\setlength{\itemsep}{6pt}
    \item[\rung1] $2i(5+i)$
    \item[\rung2] $(3+i)(2-4i)$
    \item[\rung3] $(2+5i)^2$
  \end{itemize}
  \vspace{4pt}
  \reveal{%
    \begin{block}{Check}
      \rung1\ $10i+2i^2=-2+10i$ \qquad \rung2\ $6-12i+2i-4i^2=6-10i+4=10-10i$ \\[1pt]
      \rung3\ $(2+5i)(2+5i)=4+10i+10i+25i^2=4+20i-25=-21+20i$ (not $4+25i^2$)
    \end{block}}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Show me: $(4+i)(2-3i)$}
  \sectionlabel[goldacc]{Pick one --- 1 to 4 fingers on the count of three}
  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \opt1{$5-10i$} \\[12pt]
      \opt3{$11$}
    \end{column}
    \begin{column}{0.45\textwidth}
      \opt2{$11-10i$} \\[12pt]
      \opt4{$8-10i-3i^2$}
    \end{column}
  \end{columns}
  \vspace{10pt}
  \reveal{%
    \begin{block}{Answer: \textbf{2}}
      $8-12i+2i-3i^2=8-10i+3=11-10i$. \; \textbf{1} used $i^2=+1$, so $-3i^2$ became $-3$.
      \textbf{3} multiplied first-by-first and last-by-last and lost the middle terms.
      \textbf{4} is right but unfinished --- not standard form until $i^2$ is gone.
    \end{block}}
\end{frame}

% ═══ ROUND 3 — CONJUGATES AND NEGATIVE RADICALS (the crux) ══════════════════
\begin{frame}[t, plain]
  \forestheader{Definition: complex conjugates}
  \sectionlabel{Round 3 --- products that come out real}
  \begin{block}{Complex conjugates}
    $a+bi$ and $a-bi$ are \textbf{complex conjugates} --- same real part, opposite imaginary
    part. Their product is always a real number, because the middle terms cancel and
    $-b^2i^2=+b^2$:
    \[ (a+bi)(a-bi)=a^2-abi+abi-b^2i^2=a^2+b^2. \]
  \end{block}
  \vspace{4pt}
  \begin{block}{Negative radicals: $i$ out first}
    $\sqrt a\cdot\sqrt b=\sqrt{ab}$ only when $a,b\ge0$. With a negative under a root, write it
    with $i$ \emph{before} you multiply: $\sqrt{-a}\cdot\sqrt{-b}=i\sqrt a\cdot i\sqrt b=-\sqrt{ab}$.
  \end{block}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Example 3: conjugates and negative radicals}
  \sectionlabel{Watch --- 2 minutes}
  \begin{itemize}\setlength{\itemsep}{10pt}
    \item \textbf{(a)} $(5+2i)(5-2i)=25-10i+10i-4i^2$ \pause $=25+4=29$ \quad
          \emph{(real --- the $i$ terms cancel)}
    \pause
    \item \textbf{(b)} The conjugate of $-3+7i$ is $-3-7i$; \pause their product is
          $(-3)^2+7^2=9+49=58$.
    \pause
    \item \textbf{(c)} $\sqrt{-4}\cdot\sqrt{-25}=2i\cdot5i$ \pause $=10i^2=-10$ \quad
          \emph{(not $\sqrt{100}=10$)}
  \end{itemize}
\end{frame}

  \forestheader{Your turn}
  \sectionlabel[goldacc]{8 minutes --- start at one star and climb}
  \begin{itemize}\setlength{\itemsep}{6pt}
    \item[\rung1] Multiply $(3+i)(3-i)$.
    \item[\rung2] Write the conjugate of $6-5i$, then multiply the two.
    \item[\rung3] Simplify $\sqrt{-3}\cdot\sqrt{-12}$ \; and \; $(2+\sqrt{-9})(2-\sqrt{-9})$.
  \end{itemize}
  \vspace{4pt}
  \reveal{%
    \begin{block}{Check}
      \rung1\ $9-i^2=9+1=10$ \qquad \rung2\ $6+5i$; \; $36+25=61$ \\[1pt]
      \rung3\ $i\sqrt3\cdot i\sqrt{12}=i^2\sqrt{36}=-6$ \; and \; $(2+3i)(2-3i)=4+9=13$
    \end{block}}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Find the mistake}
  \sectionlabel[redacc]{The one that matters --- 3 minutes --- find Student A's wrong step and fix it}
  \textbf{Student A} multiplied the conjugates $(6+3i)(6-3i)$:
  \[
  \begin{aligned}
    (6+3i)(6-3i) &= 36-18i+18i-9i^2 \\
                 &= 36-9 \\
                 &= 27 \qquad \text{``The $i$ terms cancel, so it's $36-9$.''}
  \end{aligned}
  \]
  \reveal{%
    \begin{block}{The mistake}
      Line 2: $-9i^2$ is \emph{not} $-9$. Because $i^2=-1$, $-9i^2=-9(-1)=+9$, so the product
      is $36+9=45$. \quad Conjugates always give $a^2+b^2$ --- a sum, never $a^2-b^2$.
    \end{block}}
\end{frame}
