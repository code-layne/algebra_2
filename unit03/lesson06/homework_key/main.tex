\begin{notesbox}{Practice}
Arms from the degree and leading coefficient. Intercepts from the factors. Cross, touch, or flatten
from the multiplicities. One test value per interval for above or below. Every problem below is those
four moves in some order.

\begin{enumerate}[leftmargin=1.9em, itemsep=8pt]
  \item \textbf{Predict from the equation.} Fill in the table without expanding anything.

    \vspace{3pt}
    \renewcommand{\arraystretch}{1.5}
    \begin{tabularx}{\linewidth}{|X|C{1.2cm}|C{1.4cm}|C{2.4cm}|X|C{1.4cm}|}
    \hline
    \rowcolor{forestbg}
    \textbf{Function} & \textbf{Deg.} & \textbf{Lead} & \textbf{Left / right arm} & \textbf{Zeros, and cross / touch / flatten at each} & \textbf{$y$-int} \\ \hline
    (a) $f(x)=(x-1)(x+4)^2$    & \ans{$3$} & \ans{$1$}  & \ans{down / up}   & \ans{$1$ cross; $-4$ touch} & \ans{$-16$} \\ \hline
    (b) $g(x)=-2(x+2)^3(x-1)$  & \ans{$4$} & \ans{$-2$} & \ans{down / down} & \ans{$-2$ flattens \& crosses; $1$ cross} & \ans{$16$} \\ \hline
    (c) $h(x)=x^2(x-5)$        & \ans{$3$} & \ans{$1$}  & \ans{down / up}   & \ans{$0$ touch; $5$ cross} & \ans{$0$} \\ \hline
    \end{tabularx}

  \item \textbf{Same zeros, different pictures.} One graph below is $y=x^2(x-3)$ and the other is
    $y=x(x-3)^2$. (Each horizontal gridline is $4$ units.)

      \node[font=\scriptsize\bfseries, charcoal] at (-1.4,2.5) {I};
    \end{tikzpicture}
    \hspace{0.4cm}
    \begin{tikzpicture}[scale=0.46]
      \lgrid{-2}{4}{-3}{2}
      \begin{scope}
        \clip (-2,-3) rectangle (4,2);
        \draw[very thick, forest, domain=-0.7:3.8, samples=160, smooth]
          plot (\x,{((\x)*(\x)*(\x) - 6*(\x)*(\x) + 9*(\x))/4});
      \end{scope}
      \foreach \x in {-1,1,2,3}
        \draw[charcoal] (\x,-0.15)--(\x,0.15) node[above=1pt, font=\tiny]{$\x$};
      \node[font=\scriptsize\bfseries, charcoal] at (-1.4,1.5) {II};

    \vspace{2pt}
    Graph I is \ans{$y=x^2(x-3)$} \qquad Graph II is \ans{$y=x(x-3)^2$}

    \vspace{2pt}
    What single feature did you use? \ans{which zero is the double one --- I touches at $0$ and
    crosses at $3$; II crosses at $0$ and touches at $3$}

  \item \textbf{The full five steps.} $f(x) = (x+2)(x-1)(x-4)$

    \vspace{2pt}
    (a) Degree \ans{$3$}, leading coefficient \ans{$1$}, end behavior \ans{down-left, up-right}

    \vspace{2pt}
    (b) Zeros: \ans{$-2,\,1,\,4$} \; Behavior at each: \ans{all crossings} \; $y$-intercept: \ans{$(0,8)$}

    \vspace{2pt}
    (c) Sign chart:

    \vspace{2pt}
    \renewcommand{\arraystretch}{1.5}
    \begin{tabularx}{\linewidth}{|C{2.4cm}|C{1.4cm}|C{2.4cm}|C{1.6cm}|X|}
    \hline
    \rowcolor{forestbg}
    \textbf{Interval} & \textbf{Test $x$} & \textbf{$f(\text{test }x)$} & \textbf{Sign} & \textbf{Above / below} \\ \hline
    $x<-2$   & $-3$ & \ans{$-28$} & \ans{$-$} & \ans{below} \\ \hline
    $-2<x<1$ & $0$  & \ans{$8$}   & \ans{$+$} & \ans{above} \\ \hline
    $1<x<4$  & $2$  & \ans{$-8$}  & \ans{$-$} & \ans{below} \\ \hline
    $x>4$    & $5$  & \ans{$28$}  & \ans{$+$} & \ans{above} \\ \hline
    \end{tabularx}

    \vspace{2pt}
    (d) On which intervals is $f(x)>0$? \ans{$(-2,1) \cup (4,\infty)$}

  \item \textbf{Factor first.} $f(x) = x^3 + 2x^2 - 8x$

    \vspace{2pt}
    (a) Factored completely: \ans{$x(x+4)(x-2)$} \quad (b) Zeros: \ans{$-4,\,0,\,2$}

    \vspace{2pt}
    (c) $y$-intercept: \ans{$(0,0)$} \quad (d) At most how many turning points? \ans{$2$}

    \vspace{2pt}
    (e) The graph is below the $x$-axis on $0<x<2$. Test $x=1$ to confirm: $f(1) = $ \ans{$-5$}

  \item \textbf{Read the equation off the graph.} (Each horizontal gridline is $4$ units.)
    \begin{center}
    \begin{tikzpicture}[scale=0.46]
      \lgrid{-3}{4}{-4}{3}
      \begin{scope}
        \clip (-3,-4) rectangle (4,3);
        \draw[very thick, forest, domain=-1.5:2.9, samples=170, smooth]
          plot (\x,{(2*(\x)*(\x)*(\x) - 6*(\x)*(\x) + 8)/4});
      \end{scope}
      \foreach \x in {-2,-1,1,2,3}
        \draw[charcoal] (\x,-0.15)--(\x,0.15) node[below=1pt, font=\tiny]{$\x$};
      \fill[charcoal] (-1,0) circle (2.5pt);
      \fill[charcoal] (2,0)  circle (2.5pt);
    \end{tikzpicture}
    \end{center}

    \vspace{2pt}
    (a) $x$-intercepts \ans{$-1$ and $2$}; the touch is at $x=$ \ans{$2$}; $y$-intercept
    \ans{$(0,8)$}

    \vspace{2pt}
    (b) Skeleton: $f(x)=a\,($\ans{$x+1$}$)($\ans{$x-2$}$)^2$ \quad Solve: $a=$ \ans{$2$}

    \vspace{2pt}
    (c) Equation: \ans{$f(x)=2(x+1)(x-2)^2$} \quad (d) Why can $a$ \emph{not} be found from an $x$-intercept?
    \ans{it gives $0 = 0$, true for every $a$}

  \item \textbf{All the characteristics.} Here is $p(x) = x^4 - 4x^2 = x^2(x-2)(x+2)$.
    (Each horizontal gridline is $2$ units.)
    \begin{center}
    \begin{tikzpicture}[scale=0.46]
      \lgrid{-3}{3}{-3}{4}
      \begin{scope}
        \clip (-3,-3) rectangle (3,4);
        \draw[very thick, forest, domain=-2.3:2.3, samples=170, smooth]
          plot (\x,{((\x)*(\x)*(\x)*(\x) - 4*(\x)*(\x))/2});
      \end{scope}
      \foreach \x in {-2,-1,1,2}
        \draw[charcoal] (\x,-0.15)--(\x,0.15) node[above=1pt, font=\tiny]{$\x$};
      \foreach \q in {-2,0,2} \fill[charcoal] (\q,0) circle (2.5pt);
    \end{tikzpicture}
    \end{center}
    \begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=4pt]
      \item Domain: \ans{all reals} \quad Range: \ans{$[-4,\infty)$}
      \item Zeros: \ans{$-2,\,0,\,2$} \; At $x=0$ the graph \ans{touches} the axis, because that zero has
        multiplicity \ans{$2$}.
      \item Relative maximum at \ans{$(0,0)$}; absolute minimum value about \ans{$-4$}, at
        $x\approx$ \ans{$\pm 1.4$}.
      \item Is the relative maximum also an absolute maximum? \ans{no} \; Why?
        \ans{both arms rise forever, so the graph goes far above $y=0$}
      \item $p(x)<0$ on \ans{$(-2,0) \cup (0,2)$} \quad (careful at $x=0$)
    \end{enumerate}

  \item \textbf{Modeling.} A $12$-inch by $12$-inch square of sheet metal has a square of side $x$
    cut from each corner; the sides fold up into an open box.
    \begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=4pt]
      \item $V(x) = $ \ans{$x(12-2x)^2$} \quad Expanded: $V(x)=4x^3-48x^2+144x$. Degree \ans{$3$}
      \item Feasible domain: \ans{$0<x<6$} \; Why is $x=7$ impossible? \ans{$12-2(7)=-2$ --- a
        negative side length}
      \item Complete the table.

        \vspace{2pt}
        \renewcommand{\arraystretch}{1.4}
        \begin{tabular}{|C{1.4cm}|C{1.4cm}|C{1.4cm}|C{1.4cm}|C{1.4cm}|C{1.4cm}|}
        \hline
        \rowcolor{forestbg}
        $x$ & $1$ & $2$ & $3$ & $4$ & $5$ \\ \hline
        $V(x)$ & $100$ & \ans{$128$} & $108$ & \ans{$64$} & $20$ \\ \hline
        \end{tabular}
      \item The largest volume is \ans{$128$} cubic inches, from a cut of \ans{$2$} inches.
        Is that a relative maximum, an absolute maximum on the feasible domain, or both?
        \ans{both}
      \item Interpret $V(4)$ in a sentence about boxes. \ans{cutting $4$-inch corners gives a box
        holding $64$ cubic inches}
    \end{enumerate}

  \item \textbf{Explain.} One or two sentences each.
    \begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=4pt]
      \item A classmate graphs $y=(x-5)^2(x+2)$ and reports that the graph is above the $x$-axis for
        $x>-2$ except at $x=5$, where it touches. Then they say ``so the graph changes from positive
        to negative at $x=5$.'' Fix the sentence and explain the rule they missed.
        \ansline{It stays positive on both sides of $5$ --- it touches and turns back. A sign change
        happens only at a zero of \emph{odd} multiplicity, and $(x-5)^2$ is even.}
      \item Why can a graph never tell you the exact degree of a polynomial --- only a minimum?
        Give both reasons you met this unit.
        \ansline{Imaginary zeros never appear on a graph (3.5), and a factor of even multiplicity can
        raise the degree without adding an intercept. Both add degree invisibly.}
    \end{enumerate}
\end{enumerate}
\end{notesbox}

\begin{extensionbox}
\begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=5pt]
  \item \textbf{Build to order.} \ans{$f(x)=-x(x+2)(x-3)^2$ --- degree $4$, $y$-intercept $(0,0)$.
        The touch needs an exponent of $2$, so the three zeros already force degree $4$; and both
        arms down requires an even degree with a negative lead, which degree $3$ could never give.}
  \item \textbf{Impossible?} \ans{No. Four $x$-intercepts force at least three turning points --- the
        graph must turn around between consecutive intercepts (and a touch is itself a turn). Two
        turns can produce at most three intercepts.}
  \item \textbf{Minimum degree, then count.} \ans{Multiplicities $1$, $1$, and $2$ add to $4$, and
        three turning points also need degree $\ge 4$, so the least degree is $4$ --- for example
        $(x+1)(x-4)(x-2)^2$. With degree exactly $4$ the real zeros already fill all four slots, so
        there are \emph{no} imaginary zeros.}
  \item \textbf{Two boxes, one volume.} \ans{$4x^3-48x^2+144x=108 \Rightarrow x^3-12x^2+36x-27=0$.
        The candidate $x=3$ works; dividing leaves $x^2-9x+9$, whose roots are
        $\frac{9\pm\sqrt{45}}{2} \approx 7.85$ and $1.15$. Reject $7.85$ (outside $0<x<6$), so the
        other feasible cut is $x \approx 1.15$ inches.}
\end{enumerate}
\end{extensionbox}

\begin{spiralbox}
\textbf{Where the unit ends.} You started Unit 3 by reading polynomial graphs you were handed
(Lesson 3.0). You then learned to multiply and factor them (3.1--3.2), divide them (3.3), find every
rational zero (3.4), and count the zeros a graph can never show you (3.5). Today all of it collapsed
into one procedure that runs both directions: equation $\rightarrow$ graph, and graph $\rightarrow$
equation. \textbf{Next: the Unit 3 test}, and after it, \textbf{Unit 4: Rational Functions}, where a
polynomial appears in a denominator for the first time --- and where dividing by zero puts brand-new
lines on the graph called \emph{asymptotes}.
\end{spiralbox}

\begin{teachernote}
\textbf{Grading priorities.} Problems 1, 3, and 5 are the assessable core --- predict from an
equation, run a sign chart, and read an equation off a graph. Problem 6 is the A2.F.2a/c/d/e language
practice; part (e) is the item most often missed, because $x=0$ must be \emph{excluded} from the
negative interval (the function is zero there, not negative). Problem 7(d) is worth a comment on
every paper: on the feasible domain the relative maximum \emph{is} the absolute maximum, which is why
it answers the question at all.

\vspace{3pt}
\textbf{Common errors.} Row (b) of problem 1 read as degree $3$ (the exponent $3$ mistaken for the
degree) or as leading coefficient $-2x$; ``crosses'' written at $x=-4$ in row (a); problem 5 solved
with $a=1$ because the $y$-intercept was never used; and problem 7(b) answered $0 \le x \le 6$ ---
the endpoints give a zero volume, so the strict inequality is the honest answer.

\vspace{3pt}
\textbf{Extension (d)} is the one to look at if a student wants more: it is a modeling question that
becomes a cubic equation, is cracked with the Rational Root Theorem from Lesson 3.4, and ends with a
feasibility judgement --- the whole unit in one problem. Accept $1.15$ or $\frac{9-\sqrt{45}}{2}$.
\end{teachernote}
