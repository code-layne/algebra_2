% ═══ ROUND 1 — THE FUNDAMENTAL THEOREM OF ALGEBRA ═════════════════════════════
\begin{frame}[t, plain]
  \forestheader{Definition: the Fundamental Theorem of Algebra}
  \sectionlabel{Round 1 --- counting zeros}
  \begin{block}{Fundamental Theorem of Algebra}
    A polynomial of degree $n\ge1$ has exactly $n$ \textbf{complex zeros}, counted with
    multiplicity --- the real zeros and the imaginary zeros together.
  \end{block}
  \vspace{2pt}
  \begin{block}{Conjugate pairs}
    If a polynomial has real coefficients and $a+bi$ is a zero, then its \textbf{conjugate}
    $a-bi$ is also a zero, so imaginary zeros always come in pairs.
  \end{block}
  \vspace{2pt}
  \begin{block}{Reading a graph}
    Each real zero is an $x$-intercept (a touch counts twice); imaginary zeros never show on
    the graph, so the number of imaginary zeros is the degree minus the number of real zeros.
  \end{block}
\end{frame}

  \sectionlabel[goldacc]{Pick one --- 1 to 4 fingers on the count of three}

  What is the third zero?\\[8pt]
  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \opt1{$-4$} \\[12pt]
      \opt3{$-2-i$}
    \end{column}
    \begin{column}{0.45\textwidth}
      \opt2{$2-i$} \\[12pt]
      \opt4{There isn't one --- I only see two.}
    \end{column}
  \end{columns}
  \vspace{8pt}

      Real coefficients, so $2+i$ brings its conjugate $2-i$; that makes $3$ zeros for degree

      flipped both signs; a conjugate flips only the $i$ part. \textbf{4} forgot the
      conjugate: degree $3$ means exactly three zeros.
    \end{block}}
\end{frame}

% ═══ ROUND 2 — WRITING A POLYNOMIAL FROM ITS ZEROS ═══════════════════════════
\begin{frame}[t, plain]
  \forestheader{Definition: factored form from the zeros}
  \sectionlabel{Round 2 --- zeros in, polynomial out}
  \begin{block}{Factored form}
    If a polynomial of degree $n$ has zeros $r_1,r_2,\dots,r_n$ (counted with multiplicity),
    then
    \[ f(x)=a(x-r_1)(x-r_2)\cdots(x-r_n), \]
    where $a$ is the leading coefficient. Each zero $r$ gives the factor $(x-r)$, so a zero at
    $x=3$ gives $(x-3)$ and a zero at $x=-3$ gives $(x+3)$.
  \end{block}
  \vspace{2pt}
  \begin{block}{The conjugate, and the last point}
    With real coefficients, every imaginary zero brings its conjugate, and the pair multiplies
    to a real quadratic: $(x-2i)(x+2i)=x^2+4$. To find $a$, substitute one more point the
    polynomial passes through.
  \end{block}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Example 2: build it, expand it, find $a$}
  \sectionlabel{Watch --- 3 minutes}
  Least degree, real coefficients, zeros include $3$ and $2i$, passes through $(0,24)$.
  Standard form.
  \[
  \begin{aligned}
    f(x) &= a(x-3)(x-2i)(x+2i) && \text{one factor per zero; add the conjugate } -2i \\
         &= a(x-3)(x^2+4) && -4i^2=+4 \\
         &= a(x^3-3x^2+4x-12) && \text{distribute} \\
    24 &= a(0-0+0-12) && \text{substitute } (0,24) \\
    a &= -2 && \text{divide by } -12 \\
    f(x) &= -2x^3+6x^2-8x+24 && \text{distribute the } -2
  \end{aligned}
  \]
\end{frame}

% ═══ ROUND 3 — GRAPHS FROM ZEROS AND END BEHAVIOR (the crux) ═════════════════
\begin{frame}[t, plain]
  \forestheader{Definition: reading a factored polynomial}
  \sectionlabel{Round 3 --- equation $\leftrightarrow$ graph}
  \begin{block}{Reading a factored polynomial}
    \small In $f(x)=a(x-r_1)^{m_1}(x-r_2)^{m_2}\cdots$, each factor $(x-r)$ puts an
    $x$-intercept at $x=r$ --- the number that makes the factor zero, the \emph{opposite} of
    the sign printed inside. An odd power \textbf{crosses} the $x$-axis; an even power
    \textbf{touches} it and turns back. The \textbf{end behavior} comes from the degree and the
    sign of $a$, and the \textbf{$y$-intercept} is $f(0)$. To \emph{match}, check the features
    one at a time; to \emph{write}, turn each intercept into a factor, square it at every touch,
    and use the $y$-intercept to find $a$.
  \end{block}
  \vspace{4pt}
  \graphABCD
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Example 3: match and defend}
  \sectionlabel{Watch --- 2 minutes}
  Which graph, A--D, is $f(x)=-(x+2)(x-1)^2$?\\[2pt]
  \graphABCD
  \[
  \begin{aligned}
    x &= -2 && \text{from } (x+2)\text{; power } 1\text{, so it crosses} \\
    x &= 1 && \text{from } (x-1)^2\text{; power } 2\text{, so it touches} \\
    \text{degree } 3,\ a &= -1 && \text{rises on the left, falls on the right} \\
    f(0) &= -(2)(-1)^2=-2 && \Rightarrow \text{Graph A, the only one with all four}
  \end{aligned}
  \]
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Find the mistake}
  \sectionlabel[redacc]{The one that matters --- 3 minutes --- find Student A's wrong step and fix it}
  \begin{columns}[T]
    \begin{column}{0.62\textwidth}
      \textbf{Student A} wrote an equation for the graph at the right:\\[4pt]
      \emph{``It crosses at $x=-1$ and touches at $x=3$, so I square that factor:
      $y=(x+1)(x+3)^2$. \; Check: at $x=0$, $y=(1)(3)^2=9$, which matches the
      $y$-intercept.''}
    \end{column}
    \begin{column}{0.34\textwidth}
      \centering
      \begin{tikzpicture}[x=0.42cm, y=0.12cm]
        \pgrid{-2}{4.5}{-6}{12}{2}
        \pcurve{-2}{4.5}{-6}{12}{((\x)+1)*((\x)-3)*((\x)-3)}
        \xtick{-1,3} \ytick{9}
      \end{tikzpicture}
    \end{column}
  \end{columns}
  \vspace{4pt}
  \reveal{%
    \begin{block}{The mistake}
      The zero $x=3$ needs the factor $(x-3)$; $(x+3)$ is zero at $x=-3$, where this graph has
      no intercept. Correct: $y=(x+1)(x-3)^2$. The $y$-intercept check could not catch it,
      because $3^2=(-3)^2=9$ --- check every intercept, not just $f(0)$.
    \end{block}}
\end{frame}
