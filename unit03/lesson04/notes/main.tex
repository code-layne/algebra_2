% ═══ ROUND 1 — DEGREE & LEADING COEFFICIENT ═══════════════════════════════════
\boxguard[20]
\begin{notesbox}{Round 1 --- Degree \& leading coefficient}
\small
\textbf{Degree and leading coefficient.} A polynomial in \textbf{standard form} lists its terms
from the highest power of $x$ down to the constant. The term with the highest power is the
\textbf{leading term}: its exponent is the \textbf{degree}, and its coefficient is the
\textbf{leading coefficient}. In \textbf{factored form}, the degree is the sum of the factors'
degrees, and the leading coefficient is the product of the factors' leading coefficients.

\vspace{6pt}
\textbf{Example.} Find the degree and the leading coefficient.\par
\quad (a) $f(x)=4x^2-7x^5+x-9$ \qquad\qquad (b) $g(x)=-2(x-3)(x+1)^2$
\begin{work}
  \text{(a)}\quad f(x) &= -7x^5+4x^2+x-9 && \text{highest power first} \\
                       &\Rightarrow \text{degree } 5,\ \text{leading coefficient } {-7} && \text{leading term } -7x^5 \\
  \text{(b)}\quad \text{degree} &= 1+2=3 && \text{add the degrees} \\
  \text{leading coefficient} &= -2(1)(1)=-2 && \text{multiply leading coefficients}
\end{work}

\vspace{6pt}

coefficient.

       &\Rightarrow \text{degree } 6,\ \text{leading coefficient } {-2}
\end{work}

\rung{2}\; $g(x)=(2x-1)(x+4)^3$
\begin{work}

% ═══ ROUND 2 — END BEHAVIOR ═══════════════════════════════════════════════════
\boxguard[20]
\begin{notesbox}{Round 2 --- End behavior}
\small
\textbf{End behavior.} The \textbf{end behavior} of a function is what $f(x)$ does at the far
ends of its graph: as $x\to\infty$ (the far right) and as $x\to-\infty$ (the far left). For a
polynomial, the leading term alone decides it. If the degree is \textbf{even}, both ends go the
\emph{same} way; if it is \textbf{odd}, they go \emph{opposite} ways. If the leading coefficient
is \textbf{positive}, the right end rises ($f(x)\to\infty$); if it is \textbf{negative}, the right
end falls ($f(x)\to-\infty$).
\begin{center}
\renewcommand{\arraystretch}{1.0}
\begin{tabular}{cccc}
\ebthumb{1.6*(\x)^2-1.5}{even, $a>0$: both up} &
\ebthumb{-1.6*(\x)^2+1.5}{even, $a<0$: both down} &
\ebthumb{0.45*(\x)^3}{odd, $a>0$: down, then up} &
\ebthumb{-0.45*(\x)^3}{odd, $a<0$: up, then down}
\end{tabular}
\end{center}

even or odd? Is the leading coefficient positive or negative?

\end{work}
\begin{work}
  \text{(b)}\quad \text{leading term} &= -2x^3 && \text{highest power, not first term} \\
  \text{as } x\to-\infty,\ g(x) &\to \infty && \text{odd: ends opposite} \\
  \text{as } x\to\infty,\ g(x) &\to -\infty && \text{negative: right end falls}
\end{work}

% ═══ ROUND 3 — SYMMETRY (the crux) ════════════════════════════════════════════
\boxguard[24]
\begin{notesbox}{Round 3 --- Symmetry: even, odd, or neither}
\small
\textbf{Even and odd functions.} A function is \textbf{even} if $f(-x)=f(x)$ for every $x$; its
graph is symmetric about the $y$-axis. A function is \textbf{odd} if $f(-x)=-f(x)$ for every $x$;
its graph is symmetric about the origin (it looks the same after a half-turn). A polynomial whose
terms all have even powers (a constant counts as $x^0$) is even, one whose terms all have odd
powers is odd, and one with a mix is \textbf{neither}.
\begin{center}
\begin{tikzpicture}[scale=0.3]
  \begin{scope}
    \lgrid{-3}{3}{-3}{4}
    \draw[navy, dashed, semithick] (0,-3)--(0,4);
    \begin{scope}\clip (-3,-3) rectangle (3,4);
      \draw[very thick, forest, domain=-2.5:2.5, samples=80, smooth] plot (\x,{0.5*(\x)^4-2*(\x)^2});
    \end{scope}
    \node[font=\scriptsize, charcoal, align=center] at (0,-4.4) {even: $y$-axis mirror};
  \end{scope}
  \begin{scope}[xshift=11cm]
    \lgrid{-3}{3}{-3}{4}
    \fill[navy] (0,0) circle (5pt);
    \begin{scope}\clip (-3,-3) rectangle (3,4);
      \draw[very thick, forest, domain=-2.7:2.7, samples=80, smooth] plot (\x,{0.4*(\x)^3-1.6*(\x)});
    \end{scope}
    \node[font=\scriptsize, charcoal] at (0,-4.4) {odd: origin half-turn};
  \end{scope}
  \begin{scope}[xshift=22cm]
    \lgrid{-3}{3}{-3}{4}
    \begin{scope}\clip (-3,-3) rectangle (3,4);
      \draw[very thick, forest, domain=-2.6:2.6, samples=80, smooth] plot (\x,{0.3*(\x)^3+1.5});
    \end{scope}
    \node[font=\scriptsize, charcoal] at (0,-4.4) {neither};
  \end{scope}
\end{tikzpicture}
\end{center}

\textbf{Example.} Decide algebraically: even, odd, or neither?\par
\quad (a) $f(x)=2x^4-x^2+3$ \qquad\qquad (b) $g(x)=x^3-4x$
\begin{work}
  \text{(a)}\quad f(-x) &= 2(-x)^4-(-x)^2+3 && \text{replace every $x$ with $-x$} \\
                        &= 2x^4-x^2+3=f(x) && \text{even powers lose the sign, so even} \\
  \text{(b)}\quad g(-x) &= (-x)^3-4(-x)=-x^3+4x && \text{odd powers keep the sign} \\
                        &= -(x^3-4x)=-g(x) && \text{every term flipped, so odd}
\end{work}

\rung{2}\; $h(x)=x^5-2x^3+x$
\begin{work}
  h(-x) &= -x^5+2x^3-x && \text{odd powers keep the sign} \\
        &= -(x^5-2x^3+x)=-h(x) \Rightarrow \text{odd}
\end{work}

\rung{3}\; $k(x)=x^4+x^3$
\begin{work}
  k(-x) &= x^4-x^3 && \text{not $k(x)=x^4+x^3$} \\
        & \ne -k(x)=-x^4-x^3 \Rightarrow \text{neither} && \text{a mix of even and odd powers}
\end{work}

\textbf{Find the mistake.} Student A described $f(x)=2x^2-x^5$:
\begin{center}
\begin{tabular}{@{}r@{\;}l@{}}
1. & $f(-x)=2(-x)^2-(-x)^5=2x^2+x^5$. That is not $f(x)$ or $-f(x)$, so $f$ is \emph{neither}. \\
2. & The leading term is $2x^2$, so the degree is $2$ and the leading coefficient is $2$. \\
3. & Even degree, positive: as $x\to-\infty,\ f(x)\to\infty$ and as $x\to\infty,\ f(x)\to\infty$. \\
\end{tabular}
\end{center}
Which step is wrong? Fix it.\par
\writelines{2}
\end{notesbox}

\begin{notesbox}{Closing Practice --- score yourself out of 4}
\small
\begin{enumerate}[leftmargin=1.9em, itemsep=4pt]
  \item Find the degree and the leading coefficient of $p(x)=(3x+1)^2(2-x)$.
  \begin{work}
    \text{degree} &= 2+1=3; \quad \text{leading coefficient} = 3^2\cdot(-1)=-9
  \end{work}
  \item Write the end behavior of $q(x)=6x^3-2x^4+x$ in arrow notation.
  \begin{work}
    \text{leading term } -2x^4 &\Rightarrow \text{as } x\to-\infty,\ q(x)\to-\infty \\
    &\phantom{{}\Rightarrow{}} \text{as } x\to\infty,\ q(x)\to-\infty
  \end{work}
  \item Even, odd, or neither: $r(x)=x^3+2$? Show the test.
  \begin{work}
    r(-x) &= -x^3+2, \ \text{not } r(x) \text{ and not } -r(x)=-x^3-2 \Rightarrow \text{neither}
  \end{work}
  \item \begin{minipage}[t]{0.74\linewidth}

  or neither.
  \begin{work}

    &\text{$s$ is even: a mirror in the $y$-axis}

\textbf{Bonus (+1).} The student council models the profit, in dollars, from spring-formal
tickets priced at $x$ dollars by $P(x)=40x^2-x^3$. What does the end behavior say happens to
the profit if the price gets very high?
\begin{work}
  \text{leading term } -x^3 &\Rightarrow \text{as } x\to\infty,\ P(x)\to-\infty \\
  &\phantom{{}\Rightarrow{}} \text{a very high price makes the profit negative: a loss}
\end{work}
