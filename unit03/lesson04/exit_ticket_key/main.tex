No notes. Show your steps; write every answer in standard $a+bi$ form.

\begin{enumerate}[leftmargin=1.8em, itemsep=14pt]
  \item \textbf{Rewrite with $i$.} Simplify, and write the second one in $a+bi$ form.
        \[ \sqrt{-72}=\ans{$6i\sqrt{2}$} \qquad\qquad 3-\sqrt{-16}=\ans{$3-4i$} \]

  \item \textbf{Multiply.} Use FOIL, then replace $i^2=-1$.
        $(3+2i)(4-i)$
        \begin{work}
          (3+2i)(4-i) &= 12-3i+8i-2i^2 \\
                      &= 12+5i-2(-1) \\
                      &= 14+5i
        \end{work}

  \item \textbf{SOL-style multiple choice.} Which is equal to $i^{38}$?
        \begin{enumerate}[label=\Alph*., leftmargin=1.9em, itemsep=1pt]
          \item $i$
          \item $-1$ \textcolor{keyred}{\textbf{$\leftarrow$ correct}}
          \item $-i$
          \item $1$
        \end{enumerate}
        \begin{itemize}[leftmargin=1.4em, nosep]
          \item[] $38\div4$ leaves remainder $2$, so $i^{38}=i^2=-1$. \textbf{D} ($i^{40}=1$) forgets the remainder; \textbf{A}/\textbf{C} use remainder $1$/$3$.
        \end{itemize}
\end{enumerate}
