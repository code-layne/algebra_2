\parthead{Part A --- Vocabulary (8 pts)}
Write the letter of the best description next to each term.

\vspace{4pt}
\noindent
\begin{minipage}[t]{0.40\textwidth}
\begin{enumerate}[leftmargin=2.2em, itemsep=7pt, label=\arabic*.]
  \item Degree \ans{B}
  \item Leading coefficient \ans{A}
  \item End behavior \ans{E}
  \item Multiplicity of a zero \ans{C}
  \item Turning point \ans{D}
  \item Factor Theorem \ans{F}
  \item Rational Root Theorem \ans{G}
  \item Fundamental Theorem of Algebra \ans{H}
\end{enumerate}
\end{minipage}%
\hfill
\begin{minipage}[t]{0.57\textwidth}
\small
\begin{description}[leftmargin=1.4em, itemsep=3pt]
  \item[(A)] The number multiplying the highest-power term; its sign decides the right-hand arm.
  \item[(B)] The greatest exponent on the variable once the polynomial is in standard form.
  \item[(C)] How many times the factor $(x-r)$ appears --- odd means the graph crosses at $r$, even means it touches.
  \item[(D)] A point where the graph changes from increasing to decreasing, or the reverse.
  \item[(E)] What the graph does at its far left and far right, as $x\to\pm\infty$.
  \item[(F)] $(x-r)$ is a factor of $f(x)$ exactly when $f(r)=0$.
  \item[(G)] Every rational zero of a polynomial with integer coefficients looks like $\pm\frac{p}{q}$, where $p$ divides the constant term and $q$ divides the leading coefficient.
  \item[(H)] A polynomial of degree $n\ge 1$ has exactly $n$ complex zeros, counted with multiplicity.
\end{description}
\end{minipage}

\parthead{Part B --- Multiple Choice (12 pts)}
Circle the best answer.

\begin{enumerate}[leftmargin=1.9em, itemsep=9pt]
  \item The end behavior of $f(x)=-2x^4+3x^2-1$ is
    \hfill (A) both arms up \quad \textbf{(B) both arms down} \ans{$\leftarrow$} \quad (C) left down, right up \quad (D) left up, right down
  \item $(2x-5)^2=$
    \hfill (A) $4x^2-25$ \quad (B) $4x^2+25$ \quad \textbf{(C) $4x^2-20x+25$} \ans{$\leftarrow$} \quad (D) $4x^2-10x+25$
  \item Factored completely, $x^3+27$ is
    \hfill \textbf{(A) $(x+3)(x^2-3x+9)$} \ans{$\leftarrow$} \quad (B) $(x+3)(x^2+3x+9)$ \quad (C) $(x+3)^3$ \quad (D) prime
  \item The remainder when $f(x)=x^3-4x^2+1$ is divided by $x-2$ is
    \hfill \textbf{(A) $-7$} \ans{$\leftarrow$} \quad (B) $-23$ \quad (C) $7$ \quad (D) $0$
  \item Which value is \emph{not} a possible rational zero of $f(x)=3x^3+2x^2-x+4$?
    \hfill (A) $-2$ \quad (B) $\frac{4}{3}$ \quad \textbf{(C) $\frac{3}{4}$} \ans{$\leftarrow$} \quad (D) $1$
  \item A polynomial with \emph{real} coefficients has degree $4$ and zeros $2$, $-1$, and $3i$. Its fourth zero must be
    \hfill \textbf{(A) $-3i$} \ans{$\leftarrow$} \quad (B) $3i$ \quad (C) $-3$ \quad (D) $\frac13$
\end{enumerate}

\begin{teachernote}
Item 1: even degree $\Rightarrow$ the arms match; $a=-2<0$ $\Rightarrow$ they both point down. Only two
facts are needed --- the $3x^2$ and the $-1$ are irrelevant. Item 2: (A) and (B) are the
$(a-b)^2=a^2-b^2$ error, (D) forgets to double the middle product. Item 4: Remainder Theorem,
$f(2)=8-16+1=-7$; (B) is $f(-2)$ --- the sign-flip error. Item 5: $p\mid 4$ and $q\mid 3$, so the
list is $\pm1,\pm2,\pm4,\pm\frac13,\pm\frac23,\pm\frac43$. $\frac34$ is the \emph{flipped} fraction.
Item 6: real coefficients force conjugate pairs; $3i$ cannot travel alone.
\end{teachernote}

\parthead{Part C --- Short Answer \& Computation (40 pts)}
Show your work.

\begin{enumerate}[leftmargin=1.9em, itemsep=10pt]
  \item \textbf{Read the graph.} The polynomial function $f$ is graphed below. Each gridline is one unit.
        \begin{center}
        \begin{tikzpicture}[scale=0.5]
          \lgrid{-3}{4}{-5}{5}
          \begin{scope}
            \clip (-3,-5) rectangle (4,5);
            \draw[very thick, forest, domain=-1.5:3.2, samples=180, smooth]
              plot (\x,{0 - (\x)*(\x)*(\x) + 3*(\x)*(\x) - 4});
          \end{scope}
          \foreach \x in {-2,-1,1,2,3}
            \draw[charcoal] (\x,-0.15)--(\x,0.15) node[below=1pt, font=\tiny]{$\x$};
          \dpt{-1}{0}
          \dpt{2}{0}
          \dpt{0}{-4}
        \end{tikzpicture}
        \end{center}
        (a) Degree: even or odd? \ans{odd} \quad Leading coefficient: positive or negative? \ans{negative}
        \\[4pt]
        (b) As $x\to-\infty$, $f(x)\to$ \ans{$+\infty$}; \quad as $x\to+\infty$, $f(x)\to$ \ans{$-\infty$}
        \\[4pt]
        (c) Zeros, each with its multiplicity, and whether the graph crosses or touches there:
        \ans{$x=-1$ (multiplicity 1, crosses); $x=2$ (multiplicity 2, touches)}
        \\[4pt]
        (d) $y$-intercept: \ans{$(0,-4)$}; \quad number of turning points: \ans{2}
        \\[4pt]
        (e) Relative maximum: \ans{$(2,0)$}; \quad relative minimum: \ans{$(0,-4)$};
        \\[4pt]
        \phantom{(e)} absolute maximum or minimum: \ans{none --- odd degree, so the arms run to $\pm\infty$}
        \\[4pt]
        (f) Interval(s) where $f$ is increasing: \ans{$(0,2)$}; \quad domain: \ans{all reals}; \quad range: \ans{all reals}

  \item \textbf{Operations.} Simplify. Write each answer in standard form.
        \\[4pt]
        (a) $(3x^3-5x+2)-(x^3+4x^2-5)$
        \ans{$=3x^3-5x+2-x^3-4x^2+5=2x^3-4x^2-5x+7$. (The subtraction hits \emph{every} term of the second polynomial: $-5$ becomes $+5$.)}
        \\
        (b) $(2x-3)(x^2+4x-1)$
        \ans{$=2x^3+8x^2-2x-3x^2-12x+3=2x^3+5x^2-14x+3$}
        \\
        (c) For your answer to (b), give the degree \ans{3}, the leading coefficient \ans{2},
        and the end behavior \ans{left down, right up}

  \item \textbf{Factor completely.} Every factor must be prime over the integers.
        \\[4pt]
        (a) $2x^3-18x$ \hspace{1.5cm} (b) $27x^3-8$ \hspace{1.5cm} (c) $x^3+3x^2-4x-12$
        \\[4pt]
        \ans{(a) GCF first: $2x(x^2-9)=2x(x-3)(x+3)$.\\[3pt]
        (b) Difference of cubes with $a=3x$, $b=2$: $(3x-2)\big((3x)^2+(3x)(2)+2^2\big)=(3x-2)(9x^2+6x+4)$; the quadratic factor is prime.\\[3pt]
        (c) Four terms $\Rightarrow$ grouping: $x^2(x+3)-4(x+3)=(x+3)(x^2-4)=(x+3)(x-2)(x+2)$.}

  \item \textbf{Division and the Remainder Theorem.}
        \\[4pt]
        (a) Divide using \emph{long division} and write the answer as $Q(x)+\dfrac{R}{D(x)}$:
        \[ (x^3+2x-9)\div(x-2) \]
        \ans{Write the dividend with a placeholder: $x^3+0x^2+2x-9$. Long division gives $Q(x)=x^2+2x+6$ with $R=3$, so the answer is $x^2+2x+6+\dfrac{3}{x-2}$. Check: $(x-2)(x^2+2x+6)+3=x^3+2x-12+3=x^3+2x-9$ \checkmark}
        \\
        (b) Use the \emph{Remainder Theorem} to find the remainder when $g(x)=x^3+2x^2-5x+6$ is divided
        by $x+3$. Is $x+3$ a factor of $g$? Explain in one sentence.
        \ans{$x+3=x-(-3)$, so the remainder is $g(-3)=-27+18+15+6=12$. Since the remainder is $12$ and not $0$, $x+3$ is \emph{not} a factor (the Factor Theorem requires $g(-3)=0$).}

  \item \textbf{Synthetic division.} You are told that $x=3$ is a zero of
        \[ f(x)=x^3-2x^2-5x+6. \]
        Use synthetic division to divide out that zero, factor $f$ completely, and list \emph{all} of
        its zeros.
        \ans{Synthetic division by $r=3$ on $1,\ -2,\ -5,\ 6$ gives the bottom row $1,\ 1,\ -2,\ 0$ --- remainder $0$, confirming the zero. The depressed polynomial is $x^2+x-2=(x+2)(x-1)$, so $f(x)=(x-3)(x+2)(x-1)$ and the zeros are $x=3,\ x=-2,\ x=1$.}

  \item \textbf{Rational Root Theorem.} For $f(x)=2x^3-x^2-8x+4$:
        \\[4pt]
        (a) List every possible rational zero.
        \ans{$p\mid 4$ gives $p=\pm1,\pm2,\pm4$; $q\mid 2$ gives $q=\pm1,\pm2$. The list (reduced, no repeats) is $\pm1,\ \pm2,\ \pm4,\ \pm\tfrac12$.}
        \\
        (b) Find all real zeros of $f$, showing the test and the division you used.
        \ans{Test $x=2$: $f(2)=16-4-16+4=0$, so $x=2$ is a zero. Synthetic division by $2$ on $2,\ -1,\ -8,\ 4$ gives $2,\ 3,\ -2,\ 0$, so the depressed polynomial is $2x^2+3x-2=(2x-1)(x+2)$. Hence $f(x)=(x-2)(2x-1)(x+2)$ and the zeros are $x=2,\ x=\tfrac12,\ x=-2$. Note $(2x-1)$ gives $x=\tfrac12$ --- not $2$.}

  \item \textbf{Solve over the complex numbers.} $x^3-3x^2+4x-12=0$
        \\[4pt]
        (a) Before solving: how many solutions must this equation have, counted with multiplicity, and
        which theorem says so? \ans{Exactly $3$, by the Fundamental Theorem of Algebra}
        \\[4pt]
        (b) Solve. Give every solution.
        \ans{Group: $x^2(x-3)+4(x-3)=(x-3)(x^2+4)=0$. So $x=3$, or $x^2=-4\Rightarrow x=\pm 2i$. The solutions are $x=3,\ 2i,\ -2i$ --- three of them, as (a) promised.}
        \\
        (c) How many of your solutions are real, and how many $x$-intercepts does the graph of
        $y=x^3-3x^2+4x-12$ therefore have? \ans{One real solution ($x=3$), so exactly one $x$-intercept --- a graph can never show the imaginary pair}

  \item \textbf{Build a polynomial from its zeros.} Write a polynomial function of \emph{least degree}
        with \emph{real} coefficients whose zeros are $-1$ and $2i$. Give your answer in factored form
        \emph{and} in standard form.
        \ans{Real coefficients force the conjugate $-2i$ to be a zero too, so there are three zeros and the least degree is $3$. Factored: $f(x)=(x+1)(x-2i)(x+2i)=(x+1)(x^2+4)$. Standard: $f(x)=x^3+x^2+4x+4$.}
\end{enumerate}

\begin{teachernote}
Item 1: (e) is the Unit 3 habit-breaker --- $(2,0)$ is a \emph{relative} max only, and an odd-degree
polynomial has \emph{no} absolute extrema; the range is all reals for the same reason. Accept
interval or inequality notation throughout. Item 1(c): full credit requires the multiplicity
\emph{and} the cross/touch, since the touch at $x=2$ is the only evidence of the squared factor.
Item 3: no credit for $(x+3)(x^2-4)$ or $2x(x^2-9)$ --- ``completely'' is the standard. Item 4(a):
the placeholder $0x^2$ is the graded step; a quotient of degree $1$ means it was skipped. Item 6(b):
accept a graph-based choice of which candidate to test, but the division must appear. Item 7(b):
watch for $\pm$ dropped on the square root, and for ``no real solutions'' written as the final
answer --- the question says \emph{over $\mathbb{C}$}.
\end{teachernote}

\parthead{Part D --- Extended Response (12 pts)}
Explain your reasoning in full sentences.

\begin{enumerate}[leftmargin=1.9em, itemsep=16pt]
  \item \textbf{The graph plan.} Consider $f(x)=-2(x+2)(x-1)^2(x-3)$. Do \emph{not} expand it.
        \textbf{(a)} State the degree and the leading coefficient, and explain how you got each one
        from the factored form. What end behavior do they force?
        \textbf{(b)} List the zeros with their multiplicities, and say whether the graph crosses or
        touches the $x$-axis at each.
        \textbf{(c)} Find the $y$-intercept.
        \textbf{(d)} Build a sign chart and state every interval on which $f(x)>0$.
        \textbf{(e)} At most how many turning points can this graph have, and why?
        \ans{(a) Degrees add: $1+2+1=4$, so the degree is $4$. Leading coefficients multiply:
        $-2\cdot 1\cdot 1\cdot 1=-2$. An even degree with a negative leading coefficient sends
        \emph{both} arms down: $f(x)\to-\infty$ at each end.\\[3pt]
        (b) $x=-2$ (multiplicity 1, crosses); $x=1$ (multiplicity 2, touches); $x=3$ (multiplicity 1,
        crosses).\\[3pt]
        (c) $f(0)=-2(2)(-1)^2(-3)=12$, so the $y$-intercept is $(0,12)$.\\[3pt]
        (d) Test one value in each interval, keeping only signs: $f(-3)<0$, $f(0)=12>0$, $f(2)=8>0$,
        $f(4)<0$. So $f(x)>0$ on $(-2,1)\cup(1,3)$ --- that is, for $-2<x<3$ except $x=1$ --- and
        $f(x)<0$ on $(-\infty,-2)\cup(3,\infty)$. The sign does \emph{not} flip at $x=1$; that is the
        fingerprint of the even multiplicity.\\[3pt]
        (e) At most $4-1=3$ turning points, since a degree-$n$ polynomial turns at most $n-1$ times.}

  \item \textbf{Modeling.} An open-top box is made from a $12$ in by $6$ in sheet of cardboard by
        cutting a square of side $x$ from each corner and folding up the sides.
        \textbf{(a)} Write the volume $V(x)$ in factored form, then expand it into standard form.
        \textbf{(b)} State the feasible domain of the model and justify it from the sheet's dimensions.
        \textbf{(c)} Find $V(1)$ and state its units.
        \textbf{(d)} Check that $x=6$ makes $V(x)=0$. Explain why $6$ is a genuine zero of the
        polynomial but not a possible cut length.
        \ans{(a) Cutting $x$ from both ends of each side leaves a base of $(12-2x)$ by $(6-2x)$ and a
        height of $x$, so $V(x)=x(12-2x)(6-2x)$. Expanding:
        $(12-2x)(6-2x)=4x^2-36x+72$, so $V(x)=4x^3-36x^2+72x$.\\[3pt]
        (b) Every dimension must be positive: $x>0$, $12-2x>0$ (so $x<6$), and $6-2x>0$ (so $x<3$).
        The tightest of these wins, so the feasible domain is $0<x<3$ --- the short side is what
        limits the cut.\\[3pt]
        (c) $V(1)=1(10)(4)=40$ cubic inches (in$^3$). Check against standard form:
        $4-36+72=40$ \checkmark\\[3pt]
        (d) $V(6)=6(12-12)(6-12)=6\cdot 0\cdot(-6)=0$, so $6$ really is a zero of the polynomial. But a
        $6$-inch cut from a $6$-inch side removes the entire width, and $6-2(6)=-6$ is a negative
        length --- there is no box at all. The polynomial does not know about cardboard; only the
        feasible domain $0<x<3$ describes a real box.}
\end{enumerate}

\begin{teachernote}
\textbf{Scoring Part D (6 pts each).} D1: 1 pt (a) degree $4$ \emph{and} lead $-2$ with both arms
down; 2 pts (b) all three zeros with multiplicities and cross/touch; 1 pt (c) $(0,12)$; 1 pt (d)
$(-2,1)\cup(1,3)$ with the no-flip at $x=1$ noted; 1 pt (e) $3$, justified by $n-1$. Give no credit
in (a) for ``degree 3'' from counting factors --- the exponents must be added. D2: 2 pts (a)
factored \emph{and} expanded; 2 pts (b) $0<x<3$ justified by $6-2x>0$ (a domain of $0<x<6$ earns 1);
1 pt (c) $40$ in$^3$ with cubic units; 1 pt (d) $x=6$ recognized as a real zero of $V$ but rejected
because the width $6-2x$ would be negative. Do not accept ``because it is too big'' without the
dimension.
\end{teachernote}
