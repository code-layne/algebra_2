\begin{headlinebox}{forestbg}
\small\textbf{Zeros Factory.} Factoring turns a quadratic into a product, and the Zero Product Property
turns that product into the \emph{roots} --- the parabola's $x$-intercepts. With your partner, run the
factory: factor, solve, and check each answer against the graph. \emph{Justify} where each root lands.
\end{headlinebox}

\vspace{0.06in}

\begin{tcolorbox}[colback=white, colframe=black!40, title=\textbf{Tier R --- Remediate},

\textbf{1. Zero Product Property.} Each equation is \emph{already} factored. Solve for both roots.
\[
  (x-2)(x+5)=0:\ x=\textcolor{keyred}{\mathbf{2,\,-5}} \qquad x(x-4)=0:\ x=\textcolor{keyred}{\mathbf{0,\,4}}
\]
\[
  (x+1)(x+6)=0:\ x=\textcolor{keyred}{\mathbf{-1,\,-6}} \qquad (x-7)(x-1)=0:\ x=\textcolor{keyred}{\mathbf{7,\,1}}
\]

Which equation has exactly these roots? Circle one, then \emph{factor it to justify}:
\[ {\color{keyred}\mathbf{x^2+2x-8=0}} \qquad x^2-2x-8=0 \qquad x^2+2x+8=0 \]
Justify: \ans{$x^2+2x-8=(x+4)(x-2)$, so roots $-4$ and $2$ --- matching the graph. ($x^2-2x-8$ gives $-2,4$; $x^2+2x+8$ does not factor over the integers.)}

\begin{tcolorbox}[colback=white, colframe=black!40, title=\textbf{Tier E --- Extension},
    fonttitle=\bfseries, arc=1.5mm, left=3mm, right=3mm, top=2mm, bottom=2mm, breakable]
\textbf{1. Leading coefficient $a\ne1$ (grouping).} Factor and solve.

\textbf{(a)} $3x^2+7x+2=0$
\begin{work}
  3x^2+6x+x+2 &= 0 \\
  3x(x+2)+1(x+2) &= 0 \\
  (3x+1)(x+2) &= 0 \\
  x &= -\frac{1}{3},\ -2
\end{work}
\textbf{(b)} $2x^2-5x-3=0$
\begin{work}
  2x^2-6x+x-3 &= 0 \\
  2x(x-3)+1(x-3) &= 0 \\
  (2x+1)(x-3) &= 0 \\
  x &= -\frac{1}{2},\ 3
\end{work}

\textbf{2. Special patterns.} Factor and give the roots. One has a \emph{double root} --- say which.

\textbf{(a)} $4x^2-25=0$
\begin{work}
  (2x-5)(2x+5) &= 0 \\
  x &= \frac{5}{2},\ -\frac{5}{2}
\end{work}
\textbf{(b)} $x^2+10x+25=0$
\begin{work}
  (x+5)^2 &= 0 \\
  x &= -5 \quad\text{(double root)}
\end{work}
Which has a double root, and what does that mean for the graph?
\ansline{The second: $(x+5)^2=0$ has the single (double) root $x=-5$, so the parabola \emph{touches} the $x$-axis at $(-5,0)$ instead of crossing.}

\tcbbreak
\textbf{3. Model it.} A rectangular patio is $3$ ft longer than it is wide, and its area is $40$ ft$^2$.
Let the width be $x$.
\begin{enumerate}[label=(\alph*), leftmargin=1.9em, itemsep=2pt]
  \item Write an equation for the area in terms of $x$. \ans{$x(x+3)=40$}
  \item Solve by factoring. \ansline{$x^2+3x-40=0 \Rightarrow (x+8)(x-5)=0 \Rightarrow x=-8$ or $x=5$.}
  \item One root is impossible here. Which, and why? State the patio's dimensions. \ansline{Reject $x=-8$ (a width cannot be negative). Width $x=5$ ft, length $x+3=8$ ft.}
\end{enumerate}
