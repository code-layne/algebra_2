% ═══ ROUND 1 — GCF & FACTORING BY GROUPING ══════════════════════════════════
\boxguard[20]
\begin{notesbox}{Round 1 --- GCF and factoring by grouping}
\small
The \textbf{greatest common factor (GCF)} of a polynomial is the largest number and the
\emph{lowest} power of each shared variable that divide every term; when the leading coefficient
is negative, we factor out a \emph{negative} GCF, which changes the sign of every term left
inside the parentheses.

\vspace{3pt}
\textbf{Factoring by grouping} factors a four-term polynomial by splitting it into two pairs,
taking the GCF out of each pair, and then factoring out the binomial the two pairs share.

\vspace{6pt}
\textbf{Example.} Factor.

(a) $-8x^5+12x^4-20x^3$
\begin{work}
  -8x^5+12x^4-20x^3 &= -4x^3(2x^2-3x+5) && \text{negative GCF, lowest power}
\end{work}
(b) $2x^3-6x^2-5x+15$
\begin{work}
  2x^3-6x^2-5x+15 &= (2x^3-6x^2)+(-5x+15) && \text{two pairs} \\
    &= 2x^2(x-3)-5(x-3) && \text{GCF of each pair} \\
    &= (x-3)(2x^2-5) && \text{shared binomial}
\end{work}

\vspace{4pt}
\textbf{Your turn.} \textit{Factor. Start at \rung{1} and climb.}

\rung{1}\; $12x^4y^2-18x^2y^3$
\begin{work}
  12x^4y^2-18x^2y^3 &= 6x^2y^2(2x^2-3y)
\end{work}

\rung{2}\; Factor out a negative GCF: \; $-5x^3+10x^2-15x$
\begin{work}
  -5x^3+10x^2-15x &= -5x(x^2-2x+3)
\end{work}

\rung{3}\; $2x^3+3x^2-10x-15$
\begin{work}
  2x^3+3x^2-10x-15 &= x^2(2x+3)-5(2x+3) \\
    &= (2x+3)(x^2-5)
\end{work}

\end{notesbox}

% ═══ ROUND 2 — SUM & DIFFERENCE OF CUBES ════════════════════════════════════
\boxguard[24]
\begin{notesbox}{Round 2 --- Sum and difference of cubes}
\small
A \textbf{sum of cubes} $a^3+b^3$ and a
\textbf{difference of cubes} $a^3-b^3$ factor by these identities:
\[
  a^3+b^3=(a+b)(a^2-ab+b^2) \qquad\qquad a^3-b^3=(a-b)(a^2+ab+b^2).
\]
The signs follow \textbf{SOAP}: the binomial has the \textbf{S}ame sign as the original, the
trinomial's middle term has the \textbf{O}pposite sign, and its last term is \textbf{A}lways
\textbf{P}ositive. The middle term is $ab$, never $2ab$, and the trinomial factor is prime.

\vspace{6pt}
\textbf{Example.} Factor.

(a) $x^3+8$
\begin{work}
  x^3+8 &= x^3+2^3 && a=x,\ b=2 \\
    &= (x+2)(x^2-2x+4) && \text{SOAP: same, opposite, always positive}
\end{work}
(b) $27x^3-64y^3$
\begin{work}
  27x^3-64y^3 &= (3x)^3-(4y)^3 && a=3x,\ b=4y \\
    &= (3x-4y)(9x^2+12xy+16y^2) && \text{middle } ab=12xy\text{, not } 2ab
\end{work}

\vspace{4pt}
\textbf{Your turn.} \textit{Find $a$ and $b$ first. Start at \rung{1} and climb.}

\rung{1}\; $x^3-125$
\begin{work}
  x^3-125 &= x^3-5^3 \\
    &= (x-5)(x^2+5x+25)
\end{work}

\rung{2}\; $8x^3+27$
\begin{work}
  8x^3+27 &= (2x)^3+3^3 \\
    &= (2x+3)(4x^2-6x+9)
\end{work}

\rung{3}\; In the art studio, a solid clay cube with edges $4x$ cm long has a small cube with
edges $5$ cm long cut out of one corner. The clay left has volume $64x^3-125$ cm$^3$. Factor it.
\begin{work}
  64x^3-125 &= (4x)^3-5^3 \\
    &= (4x-5)(16x^2+20x+25)
\end{work}

\textbf{Show me.} Factor $x^3-27$. \quad My answer: \blank{1.0cm}
\end{notesbox}

% ═══ ROUND 3 — FACTORING COMPLETELY (the crux) ══════════════════════════════
\boxguard[24]
\begin{notesbox}{Round 3 --- Factoring completely}
\small
A polynomial is \textbf{factored completely} when it is written as a
product in which no factor can be factored again over the integers; a factor that cannot be
factored is \textbf{prime}, and a sum of two squares such as $x^2+4$ is always prime.

\vspace{3pt}
\textbf{The order.} Take out the GCF first. Then count the terms --- two: squares or cubes;
three: a trinomial, or \textbf{quadratic form} $x^4+bx^2+c$, factored as though $x^2$ were the
variable; four: grouping. Then check every factor again.

\vspace{6pt}
\textbf{Example.} Factor completely.

(a) $2x^4-162$
\begin{work}
  2x^4-162 &= 2(x^4-81) && \text{GCF first} \\
    &= 2(x^2+9)(x^2-9) && \text{difference of squares} \\
    &= 2(x^2+9)(x+3)(x-3) && x^2-9 \text{ again; } x^2+9 \text{ is prime}
\end{work}
(b) $x^4-13x^2+36$
\begin{work}
  x^4-13x^2+36 &= (x^2-4)(x^2-9) && \text{quadratic form: } x^2 \text{ is the variable} \\
    &= (x+2)(x-2)(x+3)(x-3) && \text{check each factor again}
\end{work}

\vspace{4pt}
\textbf{Your turn.} \textit{Factor completely. Start at \rung{1} and climb.}

\rung{1}\; $5x^3-40$
\begin{work}
  5x^3-40 &= 5(x^3-8) \\
    &= 5(x-2)(x^2+2x+4)
\end{work}

\rung{2}\; $x^4-5x^2+4$
\begin{work}
  x^4-5x^2+4 &= (x^2-1)(x^2-4) \\
    &= (x+1)(x-1)(x+2)(x-2)
\end{work}

\rung{3}\; $x^6-64$ \; \textit{(squares first, then cubes)}
\begin{work}
  x^6-64 &= (x^3+8)(x^3-8) \\
    &= (x+2)(x^2-2x+4)(x-2)(x^2+2x+4)
\end{work}

\textbf{Find the mistake.} Student A was asked to factor $x^4-16$ completely and wrote
\[
  x^4-16=(x^2)^2-4^2=(x^2+4)(x^2-4). \qquad \text{``Two factors, so I'm done.''}
\]
What is wrong? Fix it.\par
\writelines{2}
\end{notesbox}

\begin{notesbox}{Closing Practice --- score yourself out of 4}
\small
Factor completely. One point each.
\begin{enumerate}[leftmargin=1.9em, itemsep=4pt]
  \item $x^3-6x^2+2x-12$
\begin{work}
  x^3-6x^2+2x-12 &= x^2(x-6)+2(x-6) \\
    &= (x-6)(x^2+2)
\end{work}
  \item $125x^3+8$
\begin{work}
  125x^3+8 &= (5x)^3+2^3 \\
    &= (5x+2)(25x^2-10x+4)
\end{work}
  \item $x^4-81$
\begin{work}
  x^4-81 &= (x^2+9)(x^2-9) \\
    &= (x^2+9)(x+3)(x-3)
\end{work}
  \item $3x^4-24x$
\begin{work}
  3x^4-24x &= 3x(x^3-8) \\
    &= 3x(x-2)(x^2+2x+4)
\end{work}
\end{enumerate}

\textbf{Bonus (+1).} The garden club's new raised bed is a box with volume
$V=x^3+3x^2-4x-12$ cubic feet. Factor $V$ completely to find its three dimensions.
\begin{work}
  x^3+3x^2-4x-12 &= x^2(x+3)-4(x+3) \\
    &= (x+3)(x^2-4) \\
    &= (x+3)(x+2)(x-2)
\end{work}
When $x=5$, the three dimensions are \blank{3.4cm} feet.
\end{notesbox}
