No notes. Show the factored form, then the roots.

\begin{enumerate}[leftmargin=1.8em, itemsep=12pt]
  \item \textbf{Factor and solve.} $x^2+2x-8=0$. State the factored form and the two roots (zeros).
        \ansline{Product $-8$, sum $2 \Rightarrow +4,-2$: $(x+4)(x-2)=0$, so $x=-4$ or $x=2$.}
  \item \textbf{GCF first.} Solve $3x^2-12x=0$. \emph{Do not divide by $x$.} Then say which two points are
        the graph's $x$-intercepts.
        \ansline{$3x(x-4)=0 \Rightarrow x=0$ or $x=4$; the $x$-intercepts are $(0,0)$ and $(4,0)$.}
  \item \textbf{SOL-style multiple choice.} A quadratic has zeros $x=-3$ and $x=4$. Which is its factored
        form?
        \begin{enumerate}[label=\Alph*., leftmargin=1.9em, itemsep=1pt]
          \item $(x-3)(x+4)$
          \item $(x+3)(x-4)$ \textcolor{keyred}{\textbf{$\leftarrow$ correct}}
          \item $(x+3)(x+4)$
          \item $(x-3)(x-4)$
        \end{enumerate}
        \begin{itemize}[leftmargin=1.4em, nosep]
          \item[] A zero of $-3$ comes from the factor $(x+3)$ (it makes the factor $0$), and a zero of $4$ from $(x-4)$ --- the sign \emph{flips}. \textbf{A} flips both; \textbf{C}/\textbf{D} give zeros of the wrong sign.
        \end{itemize}
\end{enumerate}
