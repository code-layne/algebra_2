No notes. Show the factored form, then the roots.

\begin{enumerate}[leftmargin=1.8em, itemsep=12pt]
  \item \textbf{Factor and solve.} $x^2+2x-8=0$. State the factored form and the two roots (zeros).
        \writelines{2}
  \item \textbf{GCF first.} Solve $3x^2-12x=0$. \emph{Do not divide by $x$.} Then say which two points are
        the graph's $x$-intercepts.
        \writelines{2}
  \item \textbf{SOL-style multiple choice.} A quadratic has zeros $x=-3$ and $x=4$. Which is its factored
        form?
        \begin{enumerate}[label=\Alph*., leftmargin=1.9em, itemsep=1pt]
          \item $(x-3)(x+4)$
          \item $(x+3)(x-4)$
          \item $(x+3)(x+4)$
          \item $(x-3)(x-4)$
        \end{enumerate}
\end{enumerate}
