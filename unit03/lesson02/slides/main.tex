% ═══ ROUND 1 — GCF & FACTORING BY GROUPING ══════════════════════════════════
\begin{frame}[t, plain]
  \forestheader{Definition: GCF and factoring by grouping}
  \sectionlabel{Round 1 --- take out what every term shares}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \begin{block}{Greatest common factor}
        The \textbf{greatest common factor (GCF)} of a polynomial is the largest number and the
        \emph{lowest} power of each shared variable that divide every term. When the leading
        coefficient is negative, factor out a \emph{negative} GCF --- it changes the sign of
        every term left inside.
      \end{block}
      \vspace{2pt}
      \begin{block}{Factoring by grouping}
        \textbf{Factoring by grouping} factors four terms: split them into two pairs, take the
        GCF out of each pair, then factor out the binomial the pairs share.
      \end{block}
    \end{column}
    \begin{column}{0.38\textwidth}
      \centering
      \textbf{Why the lowest power?}\\[6pt]
      $12x^3=6x^3\cdot 2$\\[2pt]
      $18x^5=6x^3\cdot 3x^2$\\[8pt]
      {\scriptsize $x^5$ does not divide $x^3$ --- so the GCF can only use $x^3$.}\\[10pt]
      \textbf{Grouping, in one line}\\[4pt]
      {\small $x^2(x+5)+3(x+5)=(x+5)(x^2+3)$}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Example 1: a negative GCF, then grouping}
  \sectionlabel{Watch --- 2 minutes}
  \textbf{(a)} $-8x^5+12x^4-20x^3$ \pause $=-4x^3(2x^2-3x+5)$ \quad
  \emph{(negative GCF, lowest power $x^3$; every sign inside flips)}
  \pause
  \[
  \begin{aligned}
    \textbf{(b)}\quad 2x^3-6x^2-5x+15 &= (2x^3-6x^2)+(-5x+15) && \text{two pairs} \\
                                      &= 2x^2(x-3)-5(x-3)      && \text{GCF of each pair} \\
                                      &= (x-3)(2x^2-5)         && \text{shared binomial}
  \end{aligned}
  \]
  \pause
  {\small Take out $-5$, not $5$, from the second pair --- that is what makes the two binomials
  match. If they do not match, check a sign.}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Show me: factor out the GCF of $-4x^3+8x^2-12x$}
  \sectionlabel[goldacc]{Pick one --- 1 to 4 fingers on the count of three}
  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \opt1{$-4x(x^2-2x+3)$} \\[12pt]
      \opt3{$-4(x^3-2x^2+3x)$}
    \end{column}
    \begin{column}{0.45\textwidth}
      \opt2{$-4x(x^2+2x-3)$} \\[12pt]
      \opt4{$-2x(2x^2-4x+6)$}
    \end{column}
  \end{columns}
  \vspace{10pt}
  \reveal{%
    \begin{block}{Answer: \textbf{1}}
      Distribute to check: $-4x\cdot x^2=-4x^3$, $-4x\cdot(-2x)=8x^2$, $-4x\cdot3=-12x$.
      \textbf{2} kept the old signs --- a negative GCF flips \emph{every} sign inside.
      \textbf{3} missed the $x$ every term shares. \textbf{4} is not the \emph{greatest}: $2$ is
      still common inside.
    \end{block}}
\end{frame}

% ═══ ROUND 2 — SUM & DIFFERENCE OF CUBES ════════════════════════════════════
\begin{frame}[t, plain]
  \forestheader{Definition: sum and difference of cubes}
  \sectionlabel{Round 2 --- two terms, both cubes}
  \begin{block}{The identities}
    A \textbf{sum of cubes} and a \textbf{difference of cubes} factor this way:
    \[
      a^3+b^3=(a+b)(a^2-ab+b^2) \qquad\qquad a^3-b^3=(a-b)(a^2+ab+b^2).
    \]
  \end{block}
  \vspace{2pt}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{block}{SOAP --- the signs}
        \textbf{S}ame sign in the binomial; \textbf{O}pposite sign on the trinomial's middle
        term; \textbf{A}lways \textbf{P}ositive last term.
      \end{block}
    \end{column}
    \begin{column}{0.41\textwidth}
      \small The middle term is $ab$, \textbf{never} $2ab$ --- this is not a perfect square, and
      the trinomial factor is prime. Find $a$ and $b$ first: $8x^3=(2x)^3$, $125=5^3$.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Example 2: one variable, then two}
  \sectionlabel{Watch --- 2 minutes}
  \[
  \begin{aligned}
    \textbf{(a)}\quad x^3+8 &= x^3+2^3 && a=x,\ b=2 \\
                            &= (x+2)(x^2-2x+4) && \text{same, opposite, always positive}
  \end{aligned}
  \]
  \pause
  \[
  \begin{aligned}
    \textbf{(b)}\quad 27x^3-64y^3 &= (3x)^3-(4y)^3 && a=3x,\ b=4y \\
                                  &= (3x-4y)(9x^2+12xy+16y^2) && \text{middle } ab=12xy\text{, not } 2ab
  \end{aligned}
  \]
  \pause
  {\small Check (a) by multiplying back: $x^3-2x^2+4x+2x^2-4x+8=x^3+8$ --- four of the six terms
  cancel.}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Show me: factor $x^3-27$}
  \sectionlabel[goldacc]{Pick one --- 1 to 4 fingers on the count of three}
  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \opt1{$(x-3)(x^2-3x+9)$} \\[12pt]
      \opt3{$(x-3)(x^2+3x+9)$}
    \end{column}
    \begin{column}{0.45\textwidth}
      \opt2{$(x-3)(x^2+6x+9)$} \\[12pt]
      \opt4{$(x-3)^3$}
    \end{column}
  \end{columns}
  \vspace{10pt}
  \reveal{%
    \begin{block}{Answer: \textbf{3}}
      $a=x$, $b=3$: same, opposite, always positive. \textbf{1} kept the minus in the middle ---
      SOAP says opposite. \textbf{2} used $2ab=6x$: that is $a^2+2ab+b^2$, a perfect square, not
      the cube identity. \textbf{4} cubed the binomial: $(x-3)^3$ has four terms.
    \end{block}}
\end{frame}

% ═══ ROUND 3 — FACTORING COMPLETELY (the crux) ══════════════════════════════
\begin{frame}[t, plain]
  \forestheader{Definition: factored completely}
  \sectionlabel{Round 3 --- keep going until nothing breaks}
  \begin{block}{Factored completely}
    A polynomial is \textbf{factored completely} when it is written as a product in which no
    factor can be factored again over the integers. A factor that cannot be factored is
    \textbf{prime}; a sum of two squares such as $x^2+4$ is always prime.
  \end{block}
  \vspace{2pt}
  \begin{block}{The order}
    \textbf{1.} GCF first. \; \textbf{2.} Count the terms --- two: squares or cubes; three: a
    trinomial, or \textbf{quadratic form} $x^4+bx^2+c$ (treat $x^2$ as the variable); four:
    grouping. \; \textbf{3.} Check every factor again.
  \end{block}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Example 3: more than one step}
  \sectionlabel{Watch --- 2 minutes}
  \[
  \begin{aligned}
    \textbf{(a)}\quad 2x^4-162 &= 2(x^4-81) && \text{GCF first} \\
                               &= 2(x^2+9)(x^2-9) && \text{difference of squares} \\
                               &= 2(x^2+9)(x+3)(x-3) && x^2-9\text{ again; } x^2+9\text{ is prime}
  \end{aligned}
  \]
  \pause
  \[
  \begin{aligned}
    \textbf{(b)}\quad x^4-13x^2+36 &= (x^2-4)(x^2-9) && \text{quadratic form in } x^2 \\
                                   &= (x+2)(x-2)(x+3)(x-3) && \text{check each factor}
  \end{aligned}
  \]
\end{frame}

  \forestheader{Your turn}
  \sectionlabel[goldacc]{8 minutes --- start at one star and climb}
  Factor completely.
  \begin{itemize}\setlength{\itemsep}{6pt}
    \item[\rung1] $5x^3-40$
    \item[\rung2] $x^4-5x^2+4$
    \item[\rung3] $x^6-64$ \; \emph{(squares first, then cubes)}
  \end{itemize}
  \vspace{4pt}
  \reveal{%
    \begin{block}{Check}
      \rung1\ $5(x^3-8)=5(x-2)(x^2+2x+4)$ \qquad \rung2\ $(x^2-1)(x^2-4)=(x+1)(x-1)(x+2)(x-2)$
      \\[1pt]
      \rung3\ $(x^3+8)(x^3-8)=(x+2)(x^2-2x+4)(x-2)(x^2+2x+4)$
    \end{block}}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Find the mistake}
  \sectionlabel[redacc]{The one that matters --- 3 minutes --- what is wrong?}
  \textbf{Student A} was asked to factor $x^4-16$ \emph{completely}:
  \[
    x^4-16=(x^2)^2-4^2=(x^2+4)(x^2-4) \qquad \text{``Two factors, so I'm done.''}
  \]
  \vspace{2pt}
  \reveal{%
    \begin{block}{The mistake}
      Student A stopped early. $x^2-4$ is \emph{still} a difference of squares:
      $x^2-4=(x+2)(x-2)$. So
      \[ x^4-16=(x^2+4)(x+2)(x-2). \]
      And leave $x^2+4$ alone --- it is a \emph{sum} of squares, so it is prime: not $(x+2)^2$,
      not $(x+2)(x-2)$.
    \end{block}}
\end{frame}
