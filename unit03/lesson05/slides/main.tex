% ═══ ROUND 1 — ZEROS AND MULTIPLICITY ═══════════════════════════════════════
\begin{frame}[t, plain]
  \forestheader{Definition: zeros and multiplicity}
  \sectionlabel{Round 1 --- where the graph meets the $x$-axis}
  \begin{columns}[T]
    \begin{column}{0.60\textwidth}
      \begin{block}{Zero}
        A \textbf{zero} of $f$ is an input $x$ with $f(x)=0$. On the graph it is an
        $x$-intercept; in factored form, each factor $(x-r)$ gives the zero $x=r$.
      \end{block}
      \vspace{2pt}
      \begin{block}{Multiplicity}
        The \textbf{multiplicity} of a zero is the number of times its factor appears --- the
        exponent on that factor. At a zero of \textbf{odd} multiplicity the graph
        \textbf{crosses} the $x$-axis; at a zero of \textbf{even} multiplicity it
        \textbf{touches} the axis and turns back.
      \end{block}
    \end{column}
    \begin{column}{0.36\textwidth}
      \centering
      \gAdef[0.36]\\[2pt]
      {\scriptsize $A(x)=(x+2)^{1}(x-1)^{2}$}
    \end{column}
  \end{columns}
\end{frame}

% ═══ ROUND 2 — TURNING POINTS AND EXTREMA ═══════════════════════════════════
\begin{frame}[t, plain]
  \forestheader{Definition: turning points and extrema}
  \sectionlabel{Round 2 --- where the graph turns}
  \begin{block}{Turning point}
    A \textbf{turning point} is a point where the graph changes from increasing to decreasing,
    or from decreasing to increasing. A polynomial of degree $n$ has \textbf{at most $n-1$}
    turning points.
  \end{block}
  \vspace{2pt}
  \begin{block}{Relative vs.\ absolute}
    The $y$-value at a turning point is a \textbf{relative maximum} if it is the highest point
    \emph{near there}, and a \textbf{relative minimum} if it is the lowest point near there. An
    \textbf{absolute maximum} (or \textbf{minimum}) is the greatest (or least) $y$-value of the
    \emph{whole} function. Odd degree: no absolute max or min. Even degree: one absolute
    extreme value --- a minimum when $a>0$, a maximum when $a<0$.
  \end{block}
\end{frame}

% ═══ ROUND 3 — INTERVALS, DOMAIN AND RANGE (the crux) ════════════════════════
\begin{frame}[t, plain]
  \forestheader{Definition: intervals, domain, and range}
  \sectionlabel{Round 3 --- read along the $x$-axis}
  \begin{block}{Increasing and decreasing}
    A function is \textbf{increasing} on an interval if its $y$-values rise as $x$ moves left
    to right, and \textbf{decreasing} if they fall. We name the interval by its
    \textbf{$x$-values}, in interval notation, with parentheses at the turning points.
  \end{block}
  \vspace{1pt}
  \begin{block}{Positive and negative}
    $f$ is \textbf{positive} where its graph is above the $x$-axis ($f(x)>0$) and
    \textbf{negative} where it is below ($f(x)<0$); these intervals start and stop at zeros.
  \end{block}
  \vspace{1pt}
  \begin{block}{Domain and range}
    The \textbf{domain} of every polynomial is $(-\infty,\infty)$. The \textbf{range} of an
    odd-degree polynomial is $(-\infty,\infty)$; an even-degree polynomial's range stops at its
    absolute extremum, such as $[-3,\infty)$.
  \end{block}
\end{frame}
