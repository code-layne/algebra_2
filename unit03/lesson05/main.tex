\begin{tcolorbox}[colback=forestbg, colframe=forest, boxrule=0.9pt, arc=2mm,
    left=2mm, right=2mm, top=1.5mm, bottom=1.5mm]
    \textbf{Primary Objective:} Students read a polynomial's graph \emph{between} its arms. From
    factored form they list each \textbf{zero} with its \textbf{multiplicity} and predict whether
    the graph \textbf{crosses} (odd) or \textbf{touches} (even) the $x$-axis, and they read zeros
    off a pre-drawn graph. They count \textbf{turning points} against the $n-1$ ceiling and tell a
    \textbf{relative} maximum or minimum from an \textbf{absolute} one. They write where a graph
    is \textbf{increasing}, \textbf{decreasing}, \textbf{positive}, and \textbf{negative} in
    interval notation --- with $x$-values --- and state the \textbf{domain} and \textbf{range}.
    \\[2pt]
    \textbf{Standards (2023 VA SOL):} \textbf{A2.F.2} --- investigate and analyze the
    characteristics of polynomial functions graphically: domain, range, zeros, and intercepts
    (\textbf{A2.F.2a}); intervals in which the function is increasing or decreasing
    (\textbf{A2.F.2c}); absolute maximum and minimum (\textbf{A2.F.2d}); relative maximum and
    minimum (\textbf{A2.F.2e}). \textbf{A2.F.2b} (compare and contrast) frames the odd- vs.\
    even-degree contrast for absolute extrema and range; end behavior (\textbf{A2.F.2g}) was
    Lesson~3.4 and is spiralled in the warm-up.
    \\[2pt]
    \textbf{Lesson model:} \emph{Packet first.} Students work in the printed packet --- a paper
    warm-up as they walk in, then the guided notes: three rounds of a short \textbf{definition}
    and \textbf{example} ($\le$3 min), a \textbf{ladder} of three problems (\rung1\ \rung2\
    \rung3), and a \textbf{Show me} or \textbf{Find the mistake}, closed by a self-scored
    \textbf{Closing Practice}. The projected deck mirrors the notes frame for frame and reveals
    every answer. The last ten minutes start the homework. \textbf{Homework source:} DeltaMath,
    online --- set \emph{A2: Lesson 3.5: Zeros, Extrema \& Intervals}.
\end{tcolorbox}

\begin{skillbox}[Priority Ideas \& Skills]{goldbox}
\begin{tabularx}{\linewidth}{@{}X|X@{}}
\textbf{Priority Ideas \& Skills} & \textbf{Key Understandings (the ``why'')} \\[2pt]
\hline
\vspace{-6pt}
\begin{itemize}[leftmargin=1.1em, nosep, after=\vspace{2pt}]
  \item List each zero of a factored polynomial with its multiplicity, and say crosses or touches.
  \item Read zeros off a pre-drawn graph; infer even multiplicity from a touch and a least degree.
  \item Count turning points; use ``at most $n-1$'' both ways (degree $\to$ turns, turns $\to$ degree).
  \item Name relative maxima/minima and any absolute maximum/minimum from a graph.
  \item Write increasing, decreasing, positive, and negative intervals in interval notation; state domain and range.
\end{itemize}
&
\vspace{-6pt}
\begin{itemize}[leftmargin=1.1em, nosep, after=\vspace{2pt}]
  \item A factor's exponent decides the sign change: an odd power changes sign at its zero
        (cross), an even power does not (touch).
  \item Every turn is a relative extremum, but only an even-degree graph has an absolute one ---
        the arms of an odd-degree graph run to $+\infty$ and $-\infty$. Flag (Round 2): calling every
        maximum ``the absolute maximum.''
  \item \textbf{Target misconception (the crux, Round 3):} writing an increasing or decreasing
        interval with $y$-values --- the turning point's $y$-coordinate --- instead of the
        $x$-values where the rise or fall happens. An interval answers \emph{where along the
        $x$-axis}.
\end{itemize}
\end{tabularx}
\end{skillbox}

\begin{skillbox}[{Vocabulary, Concepts, \& Theorems --- one definition each, as the notes and deck print them}]{sky}
\renewcommand{\arraystretch}{1.25}
\begin{tabularx}{\linewidth}{@{}>{\bfseries}l X@{}}
Zero & An input $x$ with $f(x)=0$; on the graph an $x$-intercept; each factor $(x-r)$ gives the zero $x=r$. \\
Multiplicity & The number of times a zero's factor appears (its exponent). Odd: the graph \textbf{crosses}; even: it \textbf{touches} and turns back. \\
Turning point & A point where the graph changes from increasing to decreasing or back. Degree $n$ $\Rightarrow$ \textbf{at most $n-1$}. \\
Relative max / min & The $y$-value at a turning point that is highest / lowest \emph{near there}. \\
Absolute max / min & The greatest / least $y$-value of the \emph{whole} function. Odd degree: none. Even degree: one --- a min when $a>0$, a max when $a<0$. \\
Increasing / decreasing & $y$-values rise / fall as $x$ moves left to right; named by \textbf{$x$-values} in interval notation, parentheses at turning points. \\
Positive / negative & Graph above ($f(x)>0$) / below ($f(x)<0$) the $x$-axis; intervals start and stop at zeros. \\
Domain / range & Domain of every polynomial $(-\infty,\infty)$. Range: odd degree $(-\infty,\infty)$; even degree stops at the absolute extremum, e.g.\ $[-3,\infty)$. \\
\end{tabularx}
\end{skillbox}

\begin{skillbox}[Warm-Up (5 min, on paper as they walk in)]{forestbg}
Three spiral items from Lesson~3.4, each a minute. \textbf{1.}~$f(x)=-2x^5+3x^2-1$: degree~$5$,
leading coefficient~$-2$, so as $x\to-\infty$, $f\to+\infty$ and as $x\to+\infty$,
$f\to-\infty$ (end behavior). \textbf{2.}~$g(x)=x^4-3x^2+7$ is \emph{even} --- only even powers,
so $g(-x)=g(x)$ (symmetry). \textbf{3.}~$(x-1)^2(x+3)=0$ gives $x=1$ and $x=-3$, and $(x-1)$
appears \emph{twice}; the last question --- does the graph cross at both? --- is left hanging
(answer: not at $x=1$). Reveal on the deck at 3 minutes and hand straight into Round~1: ``that
repeated factor is today's first idea.''
\end{skillbox}

\begin{skillbox}[The Lesson --- three rounds and Closing Practice (45 min)]{forestbg}
\begin{multicols}{2}
\textbf{Round 1 --- Zeros and multiplicity (13 min).} Read the definition with the display:
$A(x)=(x+2)(x-1)^2$ crosses at $-2$, touches at $1$ --- the warm-up's question, answered.
Example: $f(x)=(x+3)^2(x-1)(x-4)^3$ gives $-3$ (mult.~2, touches), $1$ (1, crosses), $4$ (3,
crosses), degree $6$. Ladder: \rung1\ $5$ (1, crosses), $-1$ (2, touches);
\rung2\ $0$ (2, touches), $3$ (3, crosses), $-6$ (1, crosses) --- the leading $-1$ is not a
zero; \rung3\ zeros $-3,-1,2$ read off the graph, $-1$ even (it touches), least degree
$1+2+1=4$. Show me $f(x)=(x+2)^3(x-4)^2$: \textbf{1} (crosses at $-2$, touches at $4$).
\textbf{2}~swapped odd/even; \textbf{3}~took the sign in the factor as the zero; \textbf{4}~thinks
any exponent above~1 bounces.

\textbf{Round 2 --- Turning points and extrema (13 min).} Point at ``at most $n-1$'' and at the
two words \emph{relative} (nearby) and \emph{absolute} (whole function). Example $f(x)=\tfrac14(x^2-4)^2$:
3 turns (the most degree 4 allows), rel.\ max $4$ at $x=0$ \emph{but not absolute} --- the arms
rise past it --- rel.\ and abs.\ min $0$ at $x=\pm2$. Ladder: \rung1\ $4$; $4$.
\rung2\ $C$: rel.\ max $4$ at $x=0$, rel.\ min $0$ at $x=2$, no absolute extrema (odd degree).
\rung3\ $M$: abs.\ max $4$ at $x=\pm2$, rel.\ min $0$ at $x=0$, not absolute (arms fall).
Show me on $A$: $(-1,4)$ is \textbf{2}, a relative max with no absolute max. \textbf{1}~is the
flagged slip (every max is ``the'' max); \textbf{3}~gave the $x$-value; \textbf{4}~hill vs.\ valley.

\columnbreak
\textbf{Round 3 --- Intervals, domain, range (13 min) --- the crux.} Say it twice: an interval
names \emph{$x$-values}, read along the axis. Example $E(x)=(x+1)^2(x-2)$: increasing
$(-\infty,-1)\cup(1,\infty)$, decreasing $(-1,1)$, positive $(2,\infty)$, negative
$(-\infty,-1)\cup(-1,2)$ (the touch point is $0$, not negative), domain and range
$(-\infty,\infty)$. Ladder: \rung1\ $K$ increasing $(-2,0)$, domain $(-\infty,\infty)$;
\rung2\ $S$ decreasing $(-\infty,3)$ (the flat spot at $0$ is not a turn), negative $(0,4)$,
range $[-3,\infty)$; \rung3\ barn warming $(0,2)\cup(4,6)$, below freezing $[0,2)\cup(2,5)$ ---
$h=0$ is in the domain and $T(0)=-5$, so the bracket. \textbf{Find the mistake (do not rush
it):} Student A wrote increasing $(-\infty,4)\cup(-2,\infty)$ from the turning points
$(-3,4)$ and $(1,-2)$ --- the $y$-values. Fix: increasing $(-\infty,-3)\cup(1,\infty)$,
decreasing $(-3,1)$.

\textbf{Closing Practice (6 min).} Alone, answers revealed together, score out of 4.
\textbf{1.}~$6$: mult.~2, touches; $-5$: mult.~3, crosses. \textbf{2.}~at most $5$; yes (even
degree). \textbf{3.}~$H$ increasing on $(-1,1)$ --- the crux's twin. \textbf{4.}~rel.\ min $-4$;
positive on $(-\infty,-2)$. \textbf{Bonus:} the library's $L$ decreases on $(1,\,2.5)$ --- not
$(12,40)$. Say the \emph{Watch out} aloud: intervals are $x$-values, never a turning point's
$y$-value.
\end{multicols}
\end{skillbox}

\begin{skillbox}[Active Monitoring --- Watch For]{redbox}
\begin{itemize}[leftmargin=1.2em, nosep]
  \item \textbf{Sign of the zero} (Round 1 \rung1--\rung2, Show me option 3): $(x+1)^2$ read as
        $x=1$. Ask: ``what makes this factor zero?''
  \item \textbf{Any exponent bounces} (Show me option 4): only an \emph{even} exponent touches;
        $(x-4)^3$ crosses, flattening as it goes.
  \item \textbf{Every max is the absolute max} (Round 2 \rung3, Show me option 1):
        point at the arms --- if an arm climbs past it, it is only relative.
  \item \textbf{$y$-values in an interval} (Round 3 ladder, Find the mistake, Closing Practice 3
        and bonus): the crux. Trace the $x$-axis with a finger from turn to turn.
  \item \textbf{Including a touch point in ``negative''} (Example 3, \rung3): $f=0$ there, so the
        interval breaks with a $\cup$.
  \item \textbf{Attitude:} a student parked at \rung1\ with nothing written --- crouch, point at
        the example in her notes, ask for just the first factor or the first turning point.
\end{itemize}
Cold-call prompts (after the fingers, never before): ``Which arm tells you there is no absolute
max?'' --- ``Read me the $x$-value where it stops rising.'' --- ``Why is $-1$ left out of the
negative interval?''
\end{skillbox}

\begin{teachernote}[Warm-Up]
Five minutes, on the warm-up page, alone. Items 1--2 are 3.4's end behavior and symmetry ---
anyone stuck on 1 needs the two questions (even/odd degree? sign of $a$?) said once. Reveal on
the deck at 3 minutes; do not answer item~3's last question --- Round~1's display answers it.
\end{teachernote}

\begin{teachernote}[Guided Notes \& Practice]
Three rounds of about 13 minutes: definition and example together \textbf{at most 3 minutes}
(students fill the example's work block as you click through it), the ladder about 8 (reveal
when most have \rung2), the check about 2. Then 6 minutes of Closing Practice, alone, scored out
of 4. If behind, trim Round 1's ladder to \rung1--\rung2 and Round 2's example to the extrema
lines --- never cut Round 3's Find the mistake: it is the crux. Every graph is pre-drawn; no one
sketches. Collect Closing Practice scores on fingers --- below 3 on item 3 means the crux did
not land; open 3.6 with it.
\end{teachernote}
