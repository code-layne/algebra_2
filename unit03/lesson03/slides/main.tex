% ═══ ROUND 1 — LONG DIVISION ═══════════════════════════════════════════════════
\begin{frame}[t, plain]
  \forestheader{Definition: polynomial long division}
  \sectionlabel{Round 1 --- divide, multiply, subtract, bring down}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \begin{block}{Polynomial long division}
        \textbf{Polynomial long division} divides $P(x)$ by a divisor $D(x)$ with the four steps
        of long division of numbers --- \emph{divide} the leading terms, \emph{multiply},
        \emph{subtract}, \emph{bring down} --- repeated until the remainder's degree is less
        than the divisor's. The answer is
        \[ \frac{P(x)}{D(x)} = Q(x) + \frac{R}{D(x)}, \]
        and it always checks: $P(x) = D(x)\cdot Q(x) + R$.
      \end{block}
    \end{column}
    \begin{column}{0.38\textwidth}
      \centering
      \textbf{Same statement as numbers}\\[6pt]
      \renewcommand{\arraystretch}{1.5}
      \begin{tabular}{r@{\;}l}
        $247 \div 5$ & $= 49$ R $2$ \\
        $\frac{247}{5}$ & $= 49 + \frac{2}{5}$ \\
        $247$ & $= 5\cdot 49 + 2$ \\
      \end{tabular}\\[6pt]
      {\scriptsize dividend $=$ divisor $\cdot$ quotient $+$ remainder}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Example 1: $(x^3 + 4x^2 - 3x - 5) \div (x + 2)$}
  \sectionlabel{Watch --- 2 minutes}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \small
      \[
      \renewcommand{\arraystretch}{1.15}
      \begin{array}{r@{\;}r@{\;}r@{\;}r@{\;}r}
        & & x^2 & +\,2x & -\,7 \\ \cline{2-5}
        x+2\,\big) & x^3 & +\,4x^2 & -\,3x & -\,5 \\
        & -(x^3 & +\,2x^2) & & \\ \cline{2-3}
        & & 2x^2 & -\,3x & \\
        & & -(2x^2 & +\,4x) & \\ \cline{3-4}
        & & & -7x & -\,5 \\
        & & & -(-7x & -\,14) \\ \cline{4-5}
        & & & & 9
      \end{array}
      \]
    \end{column}
    \begin{column}{0.48\textwidth}
      \small
      $x^3 \div x = x^2$; multiply $x^2(x+2)$; \emph{subtract} (the minus hits both terms);
      bring down $-3x$. Repeat. \pause
      \\[6pt]
      Stop at $9$: its degree, $0$, is less than the divisor's. \pause
      \[ \frac{x^3+4x^2-3x-5}{x+2} = x^2+2x-7+\frac{9}{x+2} \]
      \pause
      \textbf{Check:} $(x+2)(x^2+2x-7)+9 = x^3+4x^2-3x-14+9 = x^3+4x^2-3x-5$ \checkmark
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Show me: $(x^2 + 5x + 9) \div (x + 2)$}
  \sectionlabel[goldacc]{Pick one --- 1 to 4 fingers on the count of three}
  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \opt1{$x + 3$} \\[12pt]
      \opt3{$x + 7 - \dfrac{5}{x + 2}$}
    \end{column}
    \begin{column}{0.45\textwidth}
      \opt2{$x + 3 + \dfrac{3}{x + 2}$} \\[12pt]
      \opt4{$x + 3 + \dfrac{3}{x}$}
    \end{column}
  \end{columns}
  \vspace{8pt}
  \reveal{%
    \begin{block}{Answer: \textbf{2}}
      $(x+2)(x+3) = x^2 + 5x + 6$, so the remainder is $9 - 6 = 3$, over the divisor.
      \textbf{1} dropped the remainder. \textbf{3} subtracted only the $x^2$: the minus must hit
      $+2x$ too. \textbf{4} put the remainder over $x$, not over the divisor $x + 2$.
    \end{block}}
\end{frame}

% ═══ ROUND 2 — SYNTHETIC DIVISION ══════════════════════════════════════════════
\begin{frame}[t, plain]
  \forestheader{Definition: synthetic division}
  \sectionlabel{Round 2 --- the shortcut for $x - r$}
  \begin{columns}[T]
    \begin{column}{0.6\textwidth}
      \begin{block}{Synthetic division}
        \textbf{Synthetic division} divides by $x - r$ using only the coefficients. Write $r$
        --- the number that makes the divisor zero --- beside the dividend's coefficients,
        bring the first one down, then \emph{multiply by $r$ and add}, column by column. The
        last number is the \textbf{remainder}; the others are the \textbf{quotient}'s
        coefficients, one degree lower than the dividend.
      \end{block}
    \end{column}
    \begin{column}{0.36\textwidth}
      \centering
      \textbf{Which $r$? Flip the sign.}\\[6pt]
      \renewcommand{\arraystretch}{1.5}
      \begin{tabular}{c|c}
        divisor & $r$ \\ \hline
        $x - 2$ & $2$ \\
        $x + 2 = x - (-2)$ & $-2$ \\
        $x + 1$ & $-1$ \\
      \end{tabular}\\[6pt]
      {\scriptsize Only for $x - r$. For $2x + 3$ or $x^2 + 1$, use long division.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Example 2: the same division, faster}
  \sectionlabel{Watch --- 2 minutes}
  $(x^3 + 4x^2 - 3x - 5) \div (x + 2)$: \; $x + 2 = x - (-2)$, so $r = -2$.
  \vspace{4pt}
  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \centering
      \renewcommand{\arraystretch}{1.5}
      \begin{tabular}{r|rrrr}
        $-2$ & $1$ & $4$ & $-3$ & $-5$ \\
             &     & $-2$ & $-4$ & $14$ \\ \cline{2-5}
             & $1$ & $2$ & $-7$ & $9$
      \end{tabular}
    \end{column}
    \begin{column}{0.5\textwidth}
      \small
      Bring down $1$. \; $1\cdot(-2) = -2$; \; $4 + (-2) = 2$. \pause \\[3pt]
      $2\cdot(-2) = -4$; \; $-3 + (-4) = -7$. \; $-7\cdot(-2) = 14$; \; $-5 + 14 = 9$. \pause
      \\[6pt]
      Bottom row $1,\ 2,\ -7 \mid 9$: degree $3 \to 2$, and the last number is $R$.
      \[ x^2 + 2x - 7 + \frac{9}{x + 2} \]
      \emph{Same answer as Example 1.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Show me: $(x^3 - 4x^2 + x + 6) \div (x - 2)$}
  \sectionlabel[goldacc]{Pick one --- 1 to 4 fingers on the count of three}
  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \opt1{$x^2 - 6x + 13 - \dfrac{20}{x - 2}$} \\[12pt]
      \opt3{$x^2 - 2x - 3$}
    \end{column}
    \begin{column}{0.45\textwidth}
      \opt2{$x^3 - 2x^2 - 3x$} \\[12pt]
      \opt4{$x^2 - 6x + 13$}
    \end{column}
  \end{columns}
  \vspace{8pt}
  \reveal{%
    \begin{block}{Answer: \textbf{3}}
      $r = 2$: bottom row $1,\ -2,\ -3 \mid 0$, so $x^2 - 2x - 3$, remainder $0$.
      \textbf{1} used $-2$ for $x - 2$ --- the sign flips, so $x - 2$ uses $+2$.
      \textbf{4} used $-2$ and dropped the remainder. \textbf{2} forgot the quotient is one
      degree lower.
    \end{block}}
\end{frame}

% ═══ ROUND 3 — MISSING TERMS (the crux) ════════════════════════════════════════
\begin{frame}[t, plain]
  \forestheader{Definition: zero placeholders}
  \sectionlabel{Round 3 --- every power gets a column}
  \begin{block}{Zero placeholder}
    A \textbf{zero placeholder} is a $0$ written for every power of $x$ that is missing. Before
    you divide --- synthetic or long --- write the dividend (and a long-division divisor) in
    standard form with every power from the highest down to the constant, so every coefficient
    stays in its own column.
  \end{block}
  \vspace{6pt}
  \centering
  \renewcommand{\arraystretch}{1.4}
  \begin{tabular}{l|l}
    polynomial & coefficients \\ \hline
    $x^3 - 8 = x^3 + 0x^2 + 0x - 8$ & $1,\ 0,\ 0,\ -8$ \\
    $2x^4 - 3x^2 + 5 = 2x^4 + 0x^3 - 3x^2 + 0x + 5$ & $2,\ 0,\ -3,\ 0,\ 5$ \\
  \end{tabular}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Example 3: $(2x^4 - 3x^2 + 5) \div (x + 1)$}
  \sectionlabel{Watch --- 2 minutes}
  Zeros first: $2x^4 + 0x^3 - 3x^2 + 0x + 5$. \; $x + 1 = x - (-1)$, so $r = -1$.
  \vspace{4pt}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \centering
      \renewcommand{\arraystretch}{1.5}
      \begin{tabular}{r|rrrrr}
        $-1$ & $2$ & $0$ & $-3$ & $0$ & $5$ \\
             &     & $-2$ & $2$ & $1$ & $-1$ \\ \cline{2-6}
             & $2$ & $-2$ & $-1$ & $1$ & $4$
      \end{tabular}
    \end{column}
    \begin{column}{0.48\textwidth}
      \small
      Five coefficients, so five columns. \pause \\[4pt]
      Bottom row $2,\ -2,\ -1,\ 1 \mid 4$: degree $4 \to 3$. \pause
      \[ 2x^3 - 2x^2 - x + 1 + \frac{4}{x + 1} \]
    \end{column}
  \end{columns}
\end{frame}

    \end{block}}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Find the mistake}
  \sectionlabel[redacc]{The one that matters --- 3 minutes --- find Student A's wrong step and fix it}
  \textbf{Student A} divided $(x^3 - 3x^2 + 4) \div (x - 2)$:
  \vspace{4pt}
  \begin{columns}[T]
    \begin{column}{0.4\textwidth}
      \centering
      \renewcommand{\arraystretch}{1.5}
      \begin{tabular}{r|rrr}
        $2$ & $1$ & $-3$ & $4$ \\
            &     & $2$ & $-2$ \\ \cline{2-4}
            & $1$ & $-1$ & $2$
      \end{tabular}
    \end{column}
    \begin{column}{0.55\textwidth}
      ``So the answer is $x - 1 + \dfrac{2}{x - 2}$.''
    \end{column}
  \end{columns}
  \vspace{6pt}
  \reveal{%
    \begin{block}{The mistake}
      She skipped the $0$ for the missing $x$-term, so every number after the gap is in the
      wrong column. The coefficients are $1,\ -3,\ 0,\ 4$: \; $2 \mid 1,\ -3,\ 0,\ 4$ gives
      $1,\ -1,\ -2 \mid 0$, so the answer is $x^2 - x - 2$ --- it divides evenly.
    \end{block}}
\end{frame}
