\begin{notesbox}{Practice}
Standard form and placeholders first, every time. When a remainder is not $0$, write the answer as
quotient $+\;\dfrac{\text{remainder}}{\text{divisor}}$.

\begin{enumerate}[leftmargin=1.9em, itemsep=11pt]
  \item \textbf{Monomial divisors.}
    \begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=5pt]
      \item $\dfrac{16x^5 - 24x^3 + 8x^2}{8x^2} = $ \inblank{\blank{5.0cm}}{\gradedblank[id=q1a, type=short, points=1, answer={2x^3-3x+1}, display={$2x^3 - 3x + 1$}]{5.0cm-2pt}{10pt}}
      \item $\dfrac{18x^3y^2 - 27x^2y^3}{9x^2y^2} = $ \inblank{\blank{5.0cm}}{\gradedblank[id=q1b, type=short, points=1, answer={2x-3y}, display={$2x - 3y$}]{5.0cm-2pt}{10pt}}
      \item $\dfrac{20x^4 + 5x^3 - 10x^2}{5x^2} = $ \inblank{\blank{5.0cm}}{\gradedblank[id=q1c, type=short, points=1, answer={4x^2+x-2}, display={$4x^2 + x - 2$}]{5.0cm-2pt}{10pt}}
    \end{enumerate}

  \item \textbf{Long division.} Show the tableau on your own paper; record the results here.
    \begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=5pt]
      \item $(x^3 + 2x^2 - x - 2) \div (x - 1)$: \; quotient \inblank{\blank{3.0cm}}{\gradedblank[id=q2a, type=short, points=2, answer={x^2+3x+2}]{3.0cm-2pt}{10pt}}, remainder \inblank{\blank{1.2cm}}{\gradedblank[id=q2b, type=short, points=1, answer={0}, check=numeric]{1.2cm-2pt}{10pt}}
      \item $(2x^3 - 3x^2 + 4) \div (x - 2)$: \; quotient \inblank{\blank{3.0cm}}{\gradedblank[id=q2c, type=short, points=2, answer={2x^2+x+2}]{3.0cm-2pt}{10pt}}, remainder \inblank{\blank{1.2cm}}{\gradedblank[id=q2d, type=short, points=1, answer={8}, check=numeric]{1.2cm-2pt}{10pt}}

        \vspace{2pt}
        What placeholder did (b) need? \inblank{\blank{2.0cm}}{\gradedblank[id=q2e, type=short, points=1, answer={0x}, check=text]{2.0cm-2pt}{10pt}}
      \item $(x^4 - 2x^3 - 7x^2 + 8x + 12) \div (x^2 - x - 6)$: \; quotient \inblank{\blank{3.0cm}}{\gradedblank[id=q2f, type=short, points=2, answer={x^2-x-2}]{3.0cm-2pt}{10pt}},
        remainder \inblank{\blank{1.2cm}}{\gradedblank[id=q2g, type=short, points=1, answer={0}, check=numeric]{1.2cm-2pt}{10pt}}

        \vspace{2pt}
        Why could you \emph{not} use synthetic division on (c)? \inblank{\blank{5.0cm}}{\gradedblank[id=q2h, type=rubric, points=1, display={The divisor is quadratic, not of the form $x-r$.}]{5.0cm-2pt}{10pt}}
    \end{enumerate}

  \item \textbf{Synthetic division.} Give the quotient and the remainder.
    \begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=5pt]
      \item $(x^3 - 7x + 6) \div (x - 1)$: \; \inblank{\blank{5.0cm}}{\gradedblank[id=q3a, type=short, points=2, answer={x^2+x-6, remainder 0}, check=text, display={$x^2+x-6$, remainder $0$}]{5.0cm-2pt}{10pt}}
      \item $(3x^3 + 8x^2 - x - 10) \div (x + 2)$: \; \inblank{\blank{5.0cm}}{\gradedblank[id=q3b, type=short, points=2, answer={3x^2+2x-5, remainder 0}, check=text, display={$3x^2+2x-5$, remainder $0$}]{5.0cm-2pt}{10pt}}
      \item $(x^4 - 16) \div (x - 2)$: \; \inblank{\blank{5.0cm}}{\gradedblank[id=q3c, type=short, points=2, answer={x^3+2x^2+4x+8, remainder 0}, check=text, display={$x^3+2x^2+4x+8$, remainder $0$}]{5.0cm-2pt}{10pt}}

        \vspace{2pt}
        In Lesson 3.2 you factored $x^4-16$ completely. Does your quotient here agree with that
        answer? Explain in one line. \inblank{\blank{5.0cm}}{\gradedblank[id=q3d, type=rubric, points=2, display={Yes --- $x^4-16=(x-2)(x+2)(x^2+4)$, and $(x+2)(x^2+4)=x^3+2x^2+4x+8$.}]{5.0cm-2pt}{10pt}}
    \end{enumerate}

  \item \textbf{Remainder Theorem --- no dividing.} Let $f(x) = x^3 - 2x^2 + 3x - 4$.
    \begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=5pt]
      \item Remainder on division by $(x-1)$: \inblank{\blank{1.6cm}}{\gradedblank[id=q4a, type=short, points=1, answer={-2}, check=numeric]{1.6cm-2pt}{10pt}} \qquad by $(x+1)$: \inblank{\blank{1.6cm}}{\gradedblank[id=q4b, type=short, points=1, answer={-10}, check=numeric]{1.6cm-2pt}{10pt}}
            \qquad by $(x-2)$: \inblank{\blank{1.6cm}}{\gradedblank[id=q4c, type=short, points=1, answer={2}, check=numeric]{1.6cm-2pt}{10pt}}
      \item Are any of those three binomials a factor of $f$? \inblank{\blank{1.4cm}}{\gradedblank[id=q4d, type=short, points=1, answer={No}, check=text]{1.4cm-2pt}{10pt}} \; How can you tell?
            \inblank{\blank{4.6cm}}{\gradedblank[id=q4e, type=rubric, points=1, display={None of the three remainders is $0$ (Factor Theorem).}]{4.6cm-2pt}{10pt}}
    \end{enumerate}

  \item \textbf{Factor Theorem --- test, divide, finish.} Let $p(x) = x^3 + 2x^2 - 11x - 12$.
    \begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=5pt]
      \item Show that $x = -1$ is a zero: $p(-1) = $ \inblank{\blank{1.6cm}}{\gradedblank[id=q5a, type=short, points=1, answer={0}, check=numeric]{1.6cm-2pt}{10pt}}
      \item So \inblank{\blank{1.8cm}}{\gradedblank[id=q5b, type=short, points=1, answer={x+1}, display={$(x+1)$}]{1.8cm-2pt}{10pt}} is a factor. Divide it out: depressed polynomial \inblank{\blank{3.0cm}}{\gradedblank[id=q5c, type=short, points=2, answer={x^2+x-12}]{3.0cm-2pt}{10pt}}
      \item Factor completely: $p(x) = $ \inblank{\blank{5.0cm}}{\gradedblank[id=q5d, type=short, points=2, answer={(x+1)(x+4)(x-3)}]{5.0cm-2pt}{10pt}}
      \item List the zeros: \inblank{\blank{4.0cm}}{\gradedblank[id=q5e, type=short, points=1, answer={-1, -4, 3}, check=set, display={$-1,\ -4,\ 3$}]{4.0cm-2pt}{10pt}}
    \end{enumerate}

  \item \textbf{Back to a graph you already read.} On Lesson 3.0's exit ticket you read the graph of
        $h(x) = x^3 - 3x - 2$ --- it touches the $x$-axis at $x = -1$ and crosses at $x = 2$.
    \begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=5pt]
      \item Confirm with the Factor Theorem: $h(-1) = $ \inblank{\blank{1.4cm}}{\gradedblank[id=q6a, type=short, points=1, answer={0}, check=numeric]{1.4cm-2pt}{10pt}}
      \item Divide $h(x)$ by $(x+1)$: quotient \inblank{\blank{3.0cm}}{\gradedblank[id=q6b, type=short, points=2, answer={x^2-x-2}]{3.0cm-2pt}{10pt}}
      \item Factor the quotient and write $h$ completely: \inblank{\blank{5.0cm}}{\gradedblank[id=q6c, type=short, points=1, answer={(x+1)^2(x-2)}]{5.0cm-2pt}{10pt}}
      \item The zero $x=-1$ has multiplicity \inblank{\blank{1.0cm}}{\gradedblank[id=q6d, type=short, points=1, answer={2}, check=numeric]{1.0cm-2pt}{10pt}}, which is why the graph
            \inblank{\blank{2.2cm}}{\gradedblank[id=q6e, type=short, points=1, answer={touches}, check=text]{2.2cm-2pt}{10pt}} there instead of crossing.
    \end{enumerate}

  \item \textbf{Explain.} A classmate divides $f(x)$ by $(x - 5)$ and gets a remainder of $0$. In two
        or three sentences, tell them three things they now know about $f$ --- and say which
        statement in the division statement $P = D\cdot Q + R$ makes it true. \gradedlines{3}{\gradedblank[id=q7, type=rubric, points=4, display={They know that $(x-5)$ is a factor of $f$, that $f(5)=0$, and that $(5,0)$ is an $x$-intercept of the graph of $f$ --- and that the quotient they just found is the other factor. It follows from $P = D\cdot Q + R$: with $R=0$ the statement collapses to $f(x)=(x-5)\,Q(x)$, a product, so substituting $x=5$ makes the first factor zero and therefore $f(5)=0$.}]{\linewidth}{\gradedlinesheight}}
    \end{enumerate}
\end{notesbox}

\begin{extensionbox}
\textbf{Run the theorems backward.}
\begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=5pt]
  \item Find the value of $k$ that makes $(x - 2)$ a factor of $f(x) = x^3 + kx^2 - 4x + 4$, then
        factor the resulting polynomial completely. \ans{$f(2)=8+4k-8+4=4k+4=0$, so $k=-1$ and $f(x)=x^3-x^2-4x+4=(x-2)(x+2)(x-1)$.}
  \item A rectangular prism has volume $V(x) = x^3 + 6x^2 + 11x + 6$ cubic units and height
        $(x + 1)$ units. Find an expression for the area of its base, then give the two base
        dimensions as binomials. \ans{Base area $=V\div(x+1)=x^2+5x+6=(x+2)(x+3)$, so the base is $(x+2)$ by $(x+3)$.}
  \item Check your answer to (b) at $x = 2$: compute $V(2)$ directly, and multiply your three
        dimensions at $x=2$. Do they agree? \ans{$V(2)=8+24+22+6=60$; the box is $3\times 4\times 5=60$ --- yes.}
  \item Synthetic division evaluates as well as it divides. Use it to find $f(3)$ for
        $f(x) = 2x^4 - 5x^3 + x - 7$, then verify by substituting $3$ directly. \ansline{Coefficients $2,\ -5,\ 0,\ 1,\ -7$ with $r=3$ give the bottom row $2,\ 1,\ 3,\ 10,\ 23$, so $f(3)=23$. Directly: $162-135+3-7=23$. \checkmark}
\end{enumerate}
\end{extensionbox}

\begin{spiralbox}
\textbf{Coming next --- Lesson 3.4: The Rational Root Theorem.} Today you could crack any polynomial
\emph{once somebody told you a zero to test}. Tomorrow you generate that list yourself: the Rational
Root Theorem turns the constant term and the leading coefficient into a short list of candidate
zeros, synthetic division tests them, and the depressed polynomial finishes the job. Notice already
that a zero can be a fraction --- $2x^3-3x^2-11x+6$ has the zero $\tfrac12$ --- so the list will need
more than whole numbers.
\end{spiralbox}

\begin{teachernote}
\textbf{Grading focus.} Items 4, 6, and 7 carry the standard; 1--3 are procedure. In item~2(b) the
missing $x$ term is the point --- a student who divides $2x^3-3x^2+4$ without the placeholder gets a
quotient of the wrong degree, and the error is visible at a glance. Item~2(c) is the only long
division in the set that synthetic division cannot do, which is the argument for keeping both tools.

\vspace{3pt}
\textbf{Item 3(c) is worth two minutes of class debrief:} dividing $x^4-16$ by $(x-2)$ returns
$x^3+2x^2+4x+8$, which by grouping is $(x+2)(x^2+4)$ --- exactly the factorization students produced
in Lesson~3.2 by a completely different route. Same answer, two roads.

\vspace{3pt}
\textbf{Item 6 closes the longest loop in the unit.} Students read this graph on Lesson~3.0's exit
ticket, expanded $(x+1)^2(x-2)$ into $x^3-3x-2$ for Lesson~3.1 homework, could not factor it back in
Lesson~3.2, and now recover the factored form by division. Say that out loud when you review it.
\textbf{Item 7} is read for reasoning: full credit needs all three equivalent statements (factor,
$f(5)=0$, $x$-intercept at $5$), not just the first.
\end{teachernote}
