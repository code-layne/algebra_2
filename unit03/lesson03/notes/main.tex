% ═══ ROUND 1 — LONG DIVISION ═══════════════════════════════════════════════════
\boxguard[20]
\begin{notesbox}{Round 1 --- Polynomial long division}
\small
\textbf{Polynomial long division} divides a polynomial $P(x)$ by a divisor $D(x)$ with the same
four steps as long division of numbers --- \emph{divide} the leading terms, \emph{multiply},
\emph{subtract}, \emph{bring down} --- repeated until the remainder's degree is less than the
divisor's; the answer is written $\dfrac{P(x)}{D(x)} = Q(x) + \dfrac{R}{D(x)}$ (quotient plus
remainder over divisor), and it always checks: $P(x) = D(x)\cdot Q(x) + R$.

\vspace{6pt}
\textbf{Example.} Divide $(x^3 + 4x^2 - 3x - 5) \div (x + 2)$.
\[
\renewcommand{\arraystretch}{1.3}
\begin{array}{r@{\;}r@{\;}r@{\;}r@{\;}r}
  & & \kc{x^2} & \kc{+\,2x} & \kc{-\,7} \\ \cline{2-5}
  x+2\,\big) & x^3 & +\,4x^2 & -\,3x & -\,5 \\
  & \kc{-(x^3} & \kc{+\,2x^2)} & & \\ \cline{2-3}
  & & \kc{2x^2} & \kc{-\,3x} & \\
  & & \kc{-(2x^2} & \kc{+\,4x)} & \\ \cline{3-4}
  & & & \kc{-7x} & \kc{-\,5} \\
  & & & \kc{-(-7x} & \kc{-\,14)} \\ \cline{4-5}
  & & & & \kc{9}
\end{array}
\]
\begin{work}
  \frac{x^3+4x^2-3x-5}{x+2} &= x^2+2x-7+\frac{9}{x+2} \qquad \text{($Q + R/D$)} \\
  \text{Check: } (x+2)(x^2+2x-7)+9 &= x^3+4x^2-3x-5 \;\checkmark \qquad \text{($P = D\cdot Q + R$)}
\end{work}

\vspace{4pt}
\textbf{Your turn.} \textit{Start at \rung{1} and climb.} Write each answer as quotient $+$
remainder over divisor.

\begin{work}

    & & -4x^2 & -\,8x & \\

    & & & -2x & +\,5 \\
    & & & -(-2x & -\,3) \\ \cline{4-5}
    & & & & 8
  \end{array}

\end{work}

% ═══ ROUND 2 — SYNTHETIC DIVISION ══════════════════════════════════════════════
\boxguard[20]
\begin{notesbox}{Round 2 --- Synthetic division}
\small
\textbf{Synthetic division} is a shortcut for dividing by a divisor of the form $x - r$ that
uses only the coefficients: write $r$ --- the number that makes the divisor zero, so $x - 2$
uses $2$ and $x + 2 = x - (-2)$ uses $-2$ --- beside the dividend's coefficients, bring the first
coefficient down, then multiply by $r$ and add, column by column; the last number in the bottom
row is the remainder, and the numbers before it are the coefficients of the quotient, whose
degree is one less than the dividend's.

\vspace{6pt}
\noindent\begin{minipage}[t]{0.40\linewidth}
\textbf{Example.} The same division, faster:\\ $(x^3 + 4x^2 - 3x - 5) \div (x + 2)$.
\begin{center}
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{r|cccc}
  \sd{-2} & \sd{1} & \sd{4} & \sd{-3} & \sd{-5} \\
          &        & \sd{-2} & \sd{-4} & \sd{14} \\ \cline{2-5}
          & \sd{1} & \sd{2} & \sd{-7} & \sd{9}
\end{tabular}
\end{center}
\end{minipage}\hfill
\begin{minipage}[t]{0.58\linewidth}
\vspace{4pt}
\begin{work}
  r &= -2 \quad \text{(flip the sign)} \\
  Q &= x^2+2x-7 \quad \text{(degree $3\to2$)} \\
  R &= 9 \quad \text{(the last number)} \\
  \text{Answer: } & x^2+2x-7+\frac{9}{x+2}
\end{work}
\end{minipage}

\vspace{6pt}

% ═══ ROUND 3 — MISSING TERMS (the crux) ════════════════════════════════════════
\boxguard[20]
\begin{notesbox}{Round 3 --- Missing terms: zero placeholders}
\small
A \textbf{zero placeholder} is a $0$ written for every power of $x$ that is missing: before you
divide --- synthetic or long --- write the dividend (and a long-division divisor) in standard
form with every power from the highest down to the constant, so $x^3 - 8$ becomes
$x^3 + 0x^2 + 0x - 8$ with coefficients $1,\ 0,\ 0,\ -8$, and every coefficient stays in its own
column.

\vspace{6pt}
\noindent\begin{minipage}[t]{0.46\linewidth}
\textbf{Example.} $(2x^4 - 3x^2 + 5) \div (x + 1)$.
\begin{center}
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{r|ccccc}
  \sd{-1} & \sd{2} & \sd{0} & \sd{-3} & \sd{0} & \sd{5} \\
          &        & \sd{-2} & \sd{2} & \sd{1} & \sd{-1} \\ \cline{2-6}
          & \sd{2} & \sd{-2} & \sd{-1} & \sd{1} & \sd{4}
\end{tabular}
\end{center}
\end{minipage}\hfill
\begin{minipage}[t]{0.52\linewidth}
\vspace{4pt}
\begin{work}
  &2x^4 + 0x^3 - 3x^2 + 0x + 5 \\
  r &= -1 \quad \text{(flip the sign)} \\
  Q &= 2x^3-2x^2-x+1,\; R=4 \\
  \text{Answer: } & 2x^3-2x^2-x+1+\frac{4}{x+1}
\end{work}
\end{minipage}

\noindent\begin{minipage}{\linewidth}
\rung{3}\; Long division, zeros in \emph{both}: $(2x^3 + 5x - 4) \div (x^2 + 1)$
\begin{work}
  &\renewcommand{\arraystretch}{1.2}
  \begin{array}{r@{\;}r@{\;}r@{\;}r@{\;}r}
    & & & 2x & \\ \cline{2-5}
    x^2+0x+1\,\big) & 2x^3 & +\,0x^2 & +\,5x & -\,4 \\
    & -(2x^3 & +\,0x^2 & +\,2x) & \\ \cline{2-4}
    & & 0x^2 & +\,3x & -\,4
  \end{array}
  \qquad \text{degree $1 < 2$: stop.} \quad \text{Answer: } 2x+\frac{3x-4}{x^2+1}
\end{work}
\end{minipage}

\vspace{2pt}
\textbf{Find the mistake} (the one that matters). Student A divided $(x^3 - 3x^2 + 4) \div (x - 2)$:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{r|ccc}
  $2$ & $1$ & $-3$ & $4$ \\
      &     & $2$ & $-2$ \\ \cline{2-4}
      & $1$ & $-1$ & $2$
\end{tabular}
\qquad
\begin{minipage}{0.5\linewidth}
``So the answer is $x - 1 + \dfrac{2}{x-2}$.''
\end{minipage}
\end{center}
Which step is wrong? Fix it.
\writelines{2}
\end{notesbox}

\begin{notesbox}{Closing Practice --- score yourself out of 4}
\small
\begin{enumerate}[leftmargin=1.9em, itemsep=6pt]
  \item \textit{(Round 1)} Use long division: $(6x^2 + x - 10) \div (2x - 3)$.
  \begin{work}
    &\renewcommand{\arraystretch}{1.2}
    \begin{array}{r@{\;}r@{\;}r@{\;}r}
      & & 3x & +\,5 \\ \cline{2-4}
      2x-3\,\big) & 6x^2 & +\,x & -\,10 \\
      & -(6x^2 & -\,9x) & \\ \cline{2-3}
      & & 10x & -\,10 \\
      & & -(10x & -\,15) \\ \cline{3-4}
      & & & 5
    \end{array}
    \qquad \text{Answer: } 3x+5+\frac{5}{2x-3}
  \end{work}

  \item \textit{(Round 2)} Use synthetic division: $(x^3 + 2x^2 - 9x + 4) \div (x - 2)$.
  \begin{work}
    &\renewcommand{\arraystretch}{1.2}
    \begin{array}{r|rrrr}
      2 & 1 & 2 & -9 & 4 \\
        &   & 2 & 8 & -2 \\ \cline{2-5}
        & 1 & 4 & -1 & 2
    \end{array}
    \qquad \text{Answer: } x^2+4x-1+\frac{2}{x-2}
  \end{work}

  \item \textit{(Round 3)} $(x^4 - 16) \div (x + 2)$.
  \begin{work}
    &\renewcommand{\arraystretch}{1.2}
    \begin{array}{r|rrrrr}
      -2 & 1 & 0 & 0 & 0 & -16 \\
         &   & -2 & 4 & -8 & 16 \\ \cline{2-6}
         & 1 & -2 & 4 & -8 & 0
    \end{array}
    \qquad \text{Answer: } x^3-2x^2+4x-8 \quad (R=0)
  \end{work}

  \item \textit{(Mixed)} A classmate says $(2x^2 + x - 4) \div (x + 2) = 2x - 3 + \dfrac{2}{x+2}$.
        Use $P = D\cdot Q + R$ to decide whether she is right.
  \begin{work}
    (x+2)(2x-3)+2 &= 2x^2+x-6+2 = 2x^2+x-4 \qquad \text{(matches $P$: she is right)}
  \end{work}
\end{enumerate}
\textbf{Bonus (+1).} The dance floor for the spring formal is a rectangle with area
$2x^3 + x^2 - 13x + 6$ square feet, and one side is $x - 2$ feet long. How long is the other
side?
\begin{work}
  &\renewcommand{\arraystretch}{1.2}
  \begin{array}{r|rrrr}
    2 & 2 & 1 & -13 & 6 \\
      &   & 4 & 10 & -6 \\ \cline{2-5}
      & 2 & 5 & -3 & 0
  \end{array}
  \qquad \text{Answer: } 2x^2+5x-3 \text{ feet} \quad (R=0\text{: it divides evenly})
\end{work}
\end{notesbox}
