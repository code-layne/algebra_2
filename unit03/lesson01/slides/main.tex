% ═══ ROUND 1 — ADD AND SUBTRACT ═════════════════════════════════════════════
\begin{frame}[t, plain]
  \forestheader{Definition: adding \& subtracting polynomials}
  \sectionlabel{Round 1 --- collect like terms; the minus reaches every term}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \begin{block}{Polynomial, standard form, degree}
        \small A \textbf{polynomial} is a monomial or a sum of monomials. In \textbf{standard
        form} its terms run from the highest power down; its \textbf{degree} is that highest
        power (with two variables, add a term's exponents), and its \textbf{leading
        coefficient} is the number on that term.
      \end{block}
      \vspace{2pt}
      \begin{block}{Add and subtract}
        \small \textbf{Like terms} have the same variables to the same powers. To
        \textbf{add}, combine like terms. To \textbf{subtract}, change the sign of
        \textbf{every} term being subtracted, then add.
      \end{block}
    \end{column}
    \begin{column}{0.38\textwidth}
      \centering
      \textbf{A minus is $-1$ times everything}\\[8pt]
      $-(x^3+3x^2-6x+1)$\\[4pt]
      $=-x^3-3x^2+6x-1$\\[10pt]
      {\scriptsize Four terms inside, four signs changed. \\ Change only the first and the answer
      is wrong.}\\[8pt]
      {\scriptsize $3x^2$ and $3x^3$ are \emph{not} like terms.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Example 1: subtract, then name the degree}
  \sectionlabel{Watch --- 2 minutes --- fill in the example in your notes}
  \[ (4x^3-2x^2+5)-(x^3+3x^2-6x+1) \]
  \[
  \begin{aligned}
    &= 4x^3-2x^2+5-x^3-3x^2+6x-1 && \text{$-1$ times \emph{every} term}\\
    &= (4x^3-x^3)+(-2x^2-3x^2)+6x+(5-1) && \text{group like terms}\\
    &= 3x^3-5x^2+6x+4 && \text{standard form; degree } 3
  \end{aligned}
  \]
\end{frame}

  \forestheader{Your turn}
  \sectionlabel[goldacc]{8 minutes --- start at one star and climb}
  Answer in standard form.
  \begin{itemize}\setlength{\itemsep}{6pt}

    \item[\rung2] $(5x^3-x^2+4)-(2x^3+3x^2-x+6)$, and give its degree.
    \item[\rung3] $(3a^2b-4ab+b^2)-(a^2b+2ab-5b^2)$, and give its degree.
  \end{itemize}
  \vspace{4pt}
  \reveal{%
    \begin{block}{Check}
      \rung1\ $3x^2-2x+5$ \qquad \rung2\ $5x^3-x^2+4-2x^3-3x^2+x-6=3x^3-4x^2+x-2$; degree 3 \\[1pt]
      \rung3\ $3a^2b-4ab+b^2-a^2b-2ab+5b^2=2a^2b-6ab+6b^2$; degree $2+1=3$
    \end{block}}
\end{frame}

  \forestheader{Show me: $(6x^2-2x+1)-(2x^2-5x+4)$}

    \end{column}

    \end{block}}

% ═══ ROUND 2 — MULTIPLY ═════════════════════════════════════════════════════
\begin{frame}[t, plain]
  \forestheader{Definition: multiplying polynomials}
  \sectionlabel{Round 2 --- every term times every term}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \begin{block}{Multiply}
        \small To \textbf{multiply polynomials}, use the \textbf{distributive property}:
        multiply \textbf{every} term of the first by \textbf{every} term of the second, then
        combine like terms. Each small product multiplies the coefficients and adds the
        exponents on the same variable.
      \end{block}
      \vspace{2pt}
      \begin{block}{The box method}
        \small One factor's terms label the rows, the other's the columns; each cell holds one
        product. A binomial times a trinomial makes $2\times3=6$ products before you combine.
      \end{block}
    \end{column}
    \begin{column}{0.4\textwidth}
      \centering
      \textbf{The box for $(x+4)(x-3)$}\\[6pt]
      \renewcommand{\arraystretch}{1.6}
      \begin{tabular}{c|c|c|}
        \multicolumn{1}{c}{$\times$} & \multicolumn{1}{c}{$x$} & \multicolumn{1}{c}{$-3$} \\
        \cline{2-3}
        $x$  & $x^2$ & $-3x$ \\ \cline{2-3}
        $+4$ & $4x$  & $-12$ \\ \cline{2-3}
      \end{tabular}\\[6pt]
      {\scriptsize Four products, then combine: \\ $x^2-3x+4x-12=x^2+x-12$.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Example 2: multiply}
  \sectionlabel{Watch --- 2 minutes --- fill in the example in your notes}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \textbf{(a)} $-3x^2(2x^3-x+4)$ \\[3pt]
      $=-6x^5+3x^3-12x^2$ \quad {\small \emph{(coefficients multiply, exponents add)}}
      \\[10pt]
      \textbf{(b)} $(2x-1)(x^2+3x-4)$
      \[
      \begin{aligned}
        &= 2x^3+6x^2-8x-x^2-3x+4 && \text{6 products}\\
        &= 2x^3+5x^2-11x+4 && \text{combine}
      \end{aligned}
      \]
    \end{column}
    \begin{column}{0.4\textwidth}
      \centering
      \textbf{The box for (b)}\\[6pt]
      \renewcommand{\arraystretch}{1.6}
      \begin{tabular}{c|c|c|c|}
        \multicolumn{1}{c}{$\times$} & \multicolumn{1}{c}{$x^2$} & \multicolumn{1}{c}{$3x$}
          & \multicolumn{1}{c}{$-4$} \\ \cline{2-4}
        $2x$ & $2x^3$ & $6x^2$ & $-8x$ \\ \cline{2-4}
        $-1$ & $-x^2$ & $-3x$  & $+4$ \\ \cline{2-4}
      \end{tabular}\\[6pt]
      {\scriptsize Like terms sit on the diagonals: \\ $6x^2-x^2$ and $-8x-3x$.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Show me: $(2x+y)(x-3y)$}
  \sectionlabel[goldacc]{Pick one --- 1 to 4 fingers on the count of three}
  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \opt1{$2x^2-3y^2$} \\[12pt]
      \opt3{$2x^2-5xy-3y^2$}
    \end{column}
    \begin{column}{0.45\textwidth}
      \opt2{$2x^2-7xy-3y^2$} \\[12pt]
      \opt4{$2x^2-6xy-3y^2$}
    \end{column}
  \end{columns}
  \vspace{10pt}
  \reveal{%
    \begin{block}{Answer: \textbf{3}}
      $2x^2-6xy+xy-3y^2=2x^2-5xy-3y^2$. \; \textbf{1} multiplied first-by-first and
      last-by-last and lost the middle. \textbf{2} took $y\cdot x$ as $-xy$. \textbf{4} skipped
      the product $y\cdot x$ --- four products, every time.
    \end{block}}
\end{frame}

% ═══ ROUND 3 — THREE OR MORE FACTORS (the crux) ═════════════════════════════
\begin{frame}[t, plain]
  \forestheader{Definition: multiplying three or more factors}
  \sectionlabel{Round 3 --- two at a time}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \begin{block}{Two at a time}
        \small To \textbf{multiply three or more factors}, multiply any two, write that product
        in standard form, then multiply it by the next factor. A power of a binomial is
        repeated multiplication: $(x+2)^3=(x+2)(x+2)(x+2)$ --- the exponent does \textbf{not}
        distribute over a sum. The degree of a product is the sum of the factors' degrees.
      \end{block}
    \end{column}
    \begin{column}{0.4\textwidth}
      \begin{block}{Special products --- shortcuts for step 1}
        \small
        $(a+b)^2=a^2+2ab+b^2$ \\[3pt]
        $(a-b)^2=a^2-2ab+b^2$ \\[3pt]
        $(a+b)(a-b)=a^2-b^2$
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Example 3: three factors}
  \sectionlabel{Watch --- 2 minutes --- fill in the example in your notes}
  \begin{columns}[T]
    \begin{column}{0.44\textwidth}
      \textbf{(a)} $x(x+2)(x-3)$
      \[
      \begin{aligned}
        &= x(x^2-3x+2x-6) && \text{binomials}\\
        &= x(x^2-x-6) \\
        &= x^3-x^2-6x && \text{times } x
      \end{aligned}
      \]
    \end{column}
    \begin{column}{0.54\textwidth}
      \textbf{(b)} $(x+1)^3$
      \[
      \begin{aligned}
        &= (x+1)^2(x+1) && \text{three factors}\\
        &= (x^2+2x+1)(x+1) && (a+b)^2\\
        &= x^3+x^2+2x^2+2x+x+1 && \text{6 products}\\
        &= x^3+3x^2+3x+1 && \text{combine}
      \end{aligned}
      \]
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Find the mistake}
  \sectionlabel[redacc]{The one that matters --- 3 minutes --- find Student A's wrong step and fix it}
  \textbf{Student A} multiplied out $(x+2)^3$:
  \[
  \begin{aligned}
    (x+2)^3 &= x^3+2^3 \\
            &= x^3+8 \qquad \text{``Cube each term.''}
  \end{aligned}
  \]
  \reveal{%
    \begin{block}{The mistake}
      Line 1: an exponent does not distribute over a sum --- $(x+2)^3$ means
      $(x+2)(x+2)(x+2)$. Two at a time:
      $(x^2+4x+4)(x+2)=x^3+6x^2+12x+8$. \\[2pt]
      Test $x=1$: \; $(1+2)^3=27$, but $1+8=9$. The middle terms are not optional.
    \end{block}}
\end{frame}
