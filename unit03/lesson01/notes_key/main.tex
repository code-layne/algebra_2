% ═══ ROUND 1 — ADD AND SUBTRACT ═════════════════════════════════════════════
\boxguard[20]
\begin{notesbox}{Round 1 --- Adding \& subtracting polynomials}
\small
\textbf{Definition.} A \textbf{polynomial} is a monomial or a sum of monomials, each a real
number times variables raised to whole-number powers. It is in \textbf{standard form} when its
terms run from the highest power down; its \textbf{degree} is that highest power (in a term with
two variables, add the exponents), and its \textbf{leading coefficient} is the number on that
term. \textbf{Like terms} have the same variables raised to the same powers. To \textbf{add}
polynomials, combine like terms. To \textbf{subtract} a polynomial, change the sign of
\textbf{every} term being subtracted, then add.

\vspace{6pt}
\textbf{Example.} Subtract, write the result in standard form, and give its degree.
\[ (4x^3-2x^2+5)-(x^3+3x^2-6x+1) \]
\begin{work}
  &= 4x^3-2x^2+5-x^3-3x^2+6x-1 && \text{$-1$ times \emph{every} term}\\
  &= (4x^3-x^3)+(-2x^2-3x^2)+6x+(5-1) && \text{group like terms}\\
  &= 3x^3-5x^2+6x+4 && \text{standard form; degree } 3
\end{work}

\vspace{6pt}
\textbf{Your turn.} \textit{Start at \rung{1} and climb. Answer in standard form.}

\rung{1}\; $(2x^2+5x-3)+(x^2-7x+8)$
\begin{work}
  &= (2x^2+x^2)+(5x-7x)+(-3+8) && \text{group like terms}\\
  &= 3x^2-2x+5
\end{work}

\rung{2}\; $(5x^3-x^2+4)-(2x^3+3x^2-x+6)$, \; and give its degree.
\begin{work}
  &= 5x^3-x^2+4-2x^3-3x^2+x-6 && \text{$-1$ times every term}\\
  &= 3x^3-4x^2+x-2 && \text{degree } 3
\end{work}

\rung{3}\; $(3a^2b-4ab+b^2)-(a^2b+2ab-5b^2)$, \; and give its degree.
\begin{work}
  &= 3a^2b-4ab+b^2-a^2b-2ab+5b^2 && \text{$-1$ times every term}\\
  &= 2a^2b-6ab+6b^2 && \text{degree } 2+1=3
\end{work}

\textbf{Show me.} $(6x^2-2x+1)-(2x^2-5x+4)$ \quad My answer: \ans{2 \; ($4x^2+3x-3$)}
\end{notesbox}

% ═══ ROUND 2 — MULTIPLY ═════════════════════════════════════════════════════
\boxguard[24]
\begin{notesbox}{Round 2 --- Multiplying polynomials}
\small
\textbf{Definition.} To \textbf{multiply polynomials}, use the \textbf{distributive property}:
multiply \textbf{every} term of the first polynomial by \textbf{every} term of the second, then
combine like terms. Each small product multiplies the coefficients and adds the exponents on the
same variable. The \textbf{box method} keeps track: one factor's terms label the rows, the
other's label the columns, and each cell holds one product --- so a binomial times a trinomial
makes $2\times3=6$ products before you combine.

\vspace{4pt}
{\centering
\renewcommand{\arraystretch}{1.35}
\begin{tabular}{c|c|c|}
  \multicolumn{1}{c}{$\times$} & \multicolumn{1}{c}{$x$} & \multicolumn{1}{c}{$-3$} \\ \cline{2-3}
  $x$  & $x^2$ & $-3x$ \\ \cline{2-3}
  $+4$ & $4x$  & $-12$ \\ \cline{2-3}
\end{tabular}
\qquad
\raisebox{-0.9em}{$(x+4)(x-3)=x^2-3x+4x-12=x^2+x-12$}\par}

\vspace{6pt}
\textbf{Example.} Multiply. \quad (a)\; $-3x^2(2x^3-x+4)$ \qquad (b)\; $(2x-1)(x^2+3x-4)$
\begin{work}
  \text{(a)}\quad &= -6x^5+3x^3-12x^2 && \text{$-3x^2$ times each term; exponents add}\\
  \text{(b)}\quad &= 2x^3+6x^2-8x-x^2-3x+4 && \text{every term times every term: 6 products}\\
  &= 2x^3+5x^2-11x+4 && \text{combine like terms}
\end{work}

\vspace{6pt}
\textbf{Your turn.} \textit{Start at \rung{1} and climb. Answer in standard form.}

\rung{1}\; $4x(x^2-3x+5)$
\begin{work}
  &= 4x^3-12x^2+20x && \text{$4x$ times each term}
\end{work}

\rung{2}\; $(3x+2)(x-5)$
\begin{work}
  &= 3x^2-15x+2x-10 && \text{four products}\\
  &= 3x^2-13x-10
\end{work}

\rung{3}\; $(x+3)(2x^2-x+5)$
\begin{work}
  &= 2x^3-x^2+5x+6x^2-3x+15 && \text{six products}\\
  &= 2x^3+5x^2+2x+15
\end{work}

\textbf{Show me.} $(2x+y)(x-3y)$ \quad My answer: \ans{3 \; ($2x^2-5xy-3y^2$)}
\end{notesbox}

% ═══ ROUND 3 — THREE OR MORE FACTORS (the crux) ═════════════════════════════
\boxguard[24]
\begin{notesbox}{Round 3 --- Multiplying three or more factors}
\small
\textbf{Definition.} To \textbf{multiply three or more factors}, multiply \textbf{two at a
time}: multiply any two factors, write that product in standard form, then multiply it by the
next factor. A power of a binomial is repeated multiplication, $(x+2)^3=(x+2)(x+2)(x+2)$ ---
the exponent does \textbf{not} distribute over a sum. The \textbf{special products} are
shortcuts for the first step: $(a+b)^2=a^2+2ab+b^2$, \; $(a-b)^2=a^2-2ab+b^2$, \; and
$(a+b)(a-b)=a^2-b^2$. The degree of a product is the sum of the factors' degrees.

\vspace{6pt}
\textbf{Example.} Multiply. \quad (a)\; $x(x+2)(x-3)$ \qquad (b)\; $(x+1)^3$
\begin{work}
  \text{(a)}\quad &= x(x^2-3x+2x-6) && \text{multiply the two binomials first}\\
  &= x(x^2-x-6) = x^3-x^2-6x && \text{then times } x\\
  \text{(b)}\quad &= (x+1)^2(x+1) && \text{three factors of } (x+1)\\
  &= (x^2+2x+1)(x+1) && (a+b)^2 \text{ special product}\\
  &= x^3+x^2+2x^2+2x+x+1 && \text{six products}\\
  &= x^3+3x^2+3x+1 && \text{combine like terms}
\end{work}

\vspace{6pt}
\textbf{Your turn.} \textit{Start at \rung{1} and climb. Answer in standard form.}

\rung{1}\; $2x(x-4)(x+1)$
\begin{work}
  &= 2x(x^2-3x-4) && \text{binomials first}\\
  &= 2x^3-6x^2-8x && \text{then times } 2x
\end{work}

\rung{2}\; $(2x-1)(x+3)(x-2)$
\begin{work}
  &= (2x-1)(x^2+x-6) && \text{last two factors first}\\
  &= 2x^3+2x^2-12x-x^2-x+6 && \text{six products}\\
  &= 2x^3+x^2-13x+6
\end{work}

\rung{3}\; A cube-shaped storage ottoman for the dorm common room has edges $(x-3)$ inches
long. Write its volume, $(x-3)^3$, in standard form.
\begin{work}
  &= (x^2-6x+9)(x-3) && (a-b)^2 \text{ first}\\
  &= x^3-3x^2-6x^2+18x+9x-27 && \text{six products}\\
  &= x^3-9x^2+27x-27 \text{ cubic inches}
\end{work}

\textbf{Find the mistake.} Student A wrote:
\qquad $(x+2)^3 = x^3+2^3 = x^3+8$. \quad Which step is wrong? Explain, then fix it.
\ansline{Line 1: a cube does not distribute over a sum; $(x+2)^3=(x+2)(x+2)(x+2)$.}
\ansline{Fix: $(x^2+4x+4)(x+2)=x^3+6x^2+12x+8$. \; Test $x=1$: $3^3=27$, but $1+8=9$.}
\end{notesbox}

\begin{notesbox}{Closing Practice --- score yourself out of 4}
\small
Answer in standard form. \textit{One point each.}
\begin{enumerate}[leftmargin=1.9em, itemsep=4pt]
  \item $(7x^2-3x+2)-(4x^2+3x-5)$
\begin{work}
  &= 7x^2-3x+2-4x^2-3x+5 && \text{$-1$ times every term}\\
  &= 3x^2-6x+7
\end{work}
  \item $(x-4)(2x^2+x-3)$
\begin{work}
  &= 2x^3+x^2-3x-8x^2-4x+12 && \text{six products}\\
  &= 2x^3-7x^2-7x+12
\end{work}
  \item $(x-1)(x+1)(x+4)$
\begin{work}
  &= (x^2-1)(x+4) && (a-b)(a+b)=a^2-b^2\\
  &= x^3+4x^2-x-4
\end{work}
  \item $(x+3)^3$
\begin{work}
  &= (x^2+6x+9)(x+3) && (a+b)^2 \text{ first}\\
  &= x^3+3x^2+6x^2+18x+9x+27 && \text{six products}\\
  &= x^3+9x^2+27x+27
\end{work}
\end{enumerate}
\textbf{Bonus (+1).} The dance floor for the spring formal is a square, $(x+6)$ feet on a side.
The DJ's stage, $x$ feet by $(x-2)$ feet, sits on it. Write the floor area left for dancing in
standard form.
\begin{work}
  &= (x+6)^2-x(x-2) && \text{whole floor minus stage}\\
  &= x^2+12x+36-x^2+2x && \text{the minus reaches both stage terms}\\
  &= 14x+36 \text{ square feet}
\end{work}
\end{notesbox}
