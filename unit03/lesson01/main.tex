\begin{tcolorbox}[colback=forestbg, colframe=forest, boxrule=0.9pt, arc=2mm,
    left=2mm, right=2mm, top=1.5mm, bottom=1.5mm]
    \textbf{Primary Objective:} Students add, subtract, and multiply polynomials in one and two
    variables, writing every result in standard form: they subtract by changing the sign of every
    term subtracted, multiply every term by every term (distribution or the box method), and
    multiply three or more factors two at a time --- and they can explain why $(x+2)^3\ne x^3+8$.
    \\[2pt]
    \textbf{Standards (2023 VA SOL):} \textbf{A2.EO.3a} --- add, subtract, and multiply
    polynomials, including in two variables (all three rounds).
    \\[2pt]
    \textbf{Lesson model:} \emph{Packet first} (Units 3+). Students work on paper the whole hour:
    a \textbf{warm-up} page as they walk in; \textbf{guided notes} of three rounds --- a short
    \textbf{definition} and \textbf{example} ($\le$3 min), a \textbf{ladder} (\rung1\ \rung2\
    \rung3), and a \textbf{Show me} or \textbf{Find the mistake} --- then a self-scored
    \textbf{Closing Practice}; the deck projects the same pages and reveals the answers. Unit 3
    exception: the three rounds are three merged skills, climbing. \textbf{Homework source:}
    DeltaMath online, set \emph{A2: Lesson 3.1: Polynomial Operations}. \textbf{Extra credit:}
    two graded, optional pages at the back of the packet (\texttt{alg2-u03-l01-ec}).
\end{tcolorbox}

\begin{skillbox}[Priority Ideas \& Skills]{goldbox}
\begin{tabularx}{\linewidth}{@{}X|X@{}}
\textbf{Priority Ideas \& Skills} & \textbf{Key Understandings (the ``why'')} \\[2pt]
\hline
\vspace{-6pt}
\begin{itemize}[leftmargin=1.1em, nosep, after=\vspace{2pt}]
  \item Write a polynomial in standard form; name its degree (two variables: add a term's
        exponents) and leading coefficient.
  \item Add and subtract polynomials by combining like terms, distributing the $-1$ to every
        term subtracted.
  \item Multiply a monomial, binomial, or trinomial by a polynomial --- distribution or the box
        method --- in one and two variables.
  \item Multiply three or more factors two at a time, using the special products as tools.
\end{itemize}
&
\vspace{-6pt}
\begin{itemize}[leftmargin=1.1em, nosep, after=\vspace{2pt}]
  \item Every operation is the distributive property plus combining like terms --- nothing new,
        just bookkeeping.
  \item A minus sign before parentheses is a factor of $-1$; it reaches \emph{every} term
        (Round 1's sign error: flipping only the first).
  \item \textbf{Target misconception (the crux, Round 3):} \emph{``distributing the
        exponent''} --- $(x+2)^3=x^3+8$. A power of a sum is repeated multiplication; the
        middle terms come from the cross products. One substitution ($x=1$: $27\ne9$) kills it.
  \item This is the road from factored form (zeros) to standard form (degree, leading
        coefficient) used all unit.
\end{itemize}
\end{tabularx}
\end{skillbox}

\begin{skillbox}[{Vocabulary, Concepts, \& Theorems --- printed in the notes' definitions}]{sky}
\renewcommand{\arraystretch}{1.3}
\begin{tabularx}{\linewidth}{@{}l X@{}}
\textbf{Polynomial} & A monomial or a sum of monomials, each a real number times variables raised to whole-number powers. \\
\textbf{Standard form} & Terms written from the highest power down. \\
\textbf{Degree / leading coefficient} & The highest power (with two variables, the largest sum of a term's exponents) / the number on that term. \\
\textbf{Like terms} & The same variables raised to the same powers; only like terms combine. \\
\textbf{Subtracting a polynomial} & Change the sign of \emph{every} term being subtracted, then add. \\
\textbf{Distributive property / box method} & Multiply every term of one factor by every term of the other, then combine like terms; the box lays out one cell per product. \\
\textbf{Special products} & $(a+b)^2=a^2+2ab+b^2$, \; $(a-b)^2=a^2-2ab+b^2$, \; $(a+b)(a-b)=a^2-b^2$. \\
\textbf{Three or more factors} & Multiply two at a time; $(x+2)^3=(x+2)(x+2)(x+2)$; the degree of a product is the sum of the degrees. \\
\end{tabularx}
\end{skillbox}

\begin{skillbox}[Engagement --- how this hour keeps everyone solving]{goldbox}
\textbf{Nobody waits:} every round's practice is a ladder --- \rung1\ everyone (a quick win),
\rung2\ everyone, \rung3\ early finishers. \textbf{Everyone commits, safely:} the Round 1 and
Round 2 \emph{Show me} are 1--4 fingers on the count of three, all at once. \textbf{Critique a
stranger:} Round 3's \emph{Find the mistake} puts $(x+2)^3=x^3+8$ on an anonymous Student A.
\textbf{Their world:} a storage ottoman for the dorm common room (Round 3 \rung3) and the
spring-formal dance floor (Closing Practice bonus); the extra credit transfers to the student
council's canned-food-drive boxes. \textbf{Short clock:} every practice frame carries its
minutes. Praise the climb more than the answer.
\end{skillbox}

\begin{skillbox}[Warm-Up (5 min, on paper)]{forestbg}
\textbf{1. Exponent rules} --- $x^7$, $-10x^5$, $12x^2y$: the ``coefficients multiply, exponents
add'' move inside every product in Rounds 2--3. \textbf{2. Distribute} --- $4x^2+8x$,
$-6x+15$: Round 2 is this, done to every term. \textbf{3. Combine like terms} ---
$6x^2+2x+2$, then $-(x^2-4x+6)=-x^2+4x-6$: \textbf{all three} signs changed. \textbf{Handoff:}
``So what happens when you subtract a \emph{whole} polynomial?'' --- that is Round 1's
definition.
\end{skillbox}

\begin{skillbox}[The Lesson --- three rounds and Closing Practice (45 min)]{forestbg}
\begin{multicols}{2}
\textbf{Round 1 --- Adding \& subtracting (13 min).} Read the definition in one breath; point at
``\textbf{every} term.'' Example: $(4x^3-2x^2+5)-(x^3+3x^2-6x+1)=3x^3-5x^2+6x+4$, degree 3 ---
write the $-1$ step out in full. Ladder: \rung1\ $3x^2-2x+5$; \rung2\ $3x^3-4x^2+x-2$, degree 3;
\rung3\ $2a^2b-6ab+6b^2$, degree 3 (add a term's exponents). \textbf{Show me}
$(6x^2-2x+1)-(2x^2-5x+4)$: answer \textbf{2} ($4x^2+3x-3$). \textbf{1} = flipped only the first
sign (the subtraction-sign error --- re-teach on the spot); \textbf{3} = added; \textbf{4} =
missed the last term.

\textbf{Round 2 --- Multiplying (13 min).} Point at the box for $(x+4)(x-3)$: one cell per
product. Example: (a) $-6x^5+3x^3-12x^2$; (b) $(2x-1)(x^2+3x-4)=2x^3+5x^2-11x+4$ --- six
products, like terms on the box's diagonals. Ladder: \rung1\ $4x^3-12x^2+20x$; \rung2\
$3x^2-13x-10$; \rung3\ $2x^3+5x^2+2x+15$. \textbf{Show me} $(2x+y)(x-3y)$: answer \textbf{3}
($2x^2-5xy-3y^2$). \textbf{1} = first-and-last only (lost the middle); \textbf{2} = sign slip
on $y\cdot x$; \textbf{4} = skipped a product.

\columnbreak
\textbf{Round 3 --- Three or more factors (13 min) --- the crux.} Definition: two at a time;
``the exponent does \textbf{not} distribute over a sum.'' Example: (a)
$x(x+2)(x-3)=x^3-x^2-6x$; (b) $(x+1)^3=(x^2+2x+1)(x+1)=x^3+3x^2+3x+1$. Ladder: \rung1\
$2x^3-6x^2-8x$; \rung2\ $(2x-1)(x^2+x-6)=2x^3+x^2-13x+6$; \rung3\ the ottoman,
$(x-3)^3=x^3-9x^2+27x-27$ cubic inches. \textbf{Find the mistake:} Student A wrote
$(x+2)^3=x^3+2^3=x^3+8$. Wrong at line 1 --- a power of a sum is repeated multiplication;
correct $x^3+6x^2+12x+8$; test $x=1$: $27\ne9$. Do not rush this frame.

\textbf{Closing Practice (6 min).} 1.\ $3x^2-6x+7$ (Round 1); 2.\ $2x^3-7x^2-7x+12$ (Round 2);
3.\ $(x^2-1)(x+4)=x^3+4x^2-x-4$ (Round 3, special product); 4.\ $(x+3)^3=x^3+9x^2+27x+27$ (the
crux's twin). Bonus: $(x+6)^2-x(x-2)=14x+36$ square feet. Say the \emph{Watch out} aloud:
``$(x+3)^3$ is not $x^3+27$ --- and the minus reaches every term.''
\end{multicols}
\end{skillbox}

\begin{skillbox}[Extra Credit --- optional, graded (\texttt{alg2-u03-l01-ec}, 13 points)]{goldbox}
\begin{enumerate}[leftmargin=1.4em, nosep, itemsep=2pt]
  \item $(x+2)^3-(x-2)^3=12x^2+16$ (2 pts) --- \emph{combines} Round 3's cubes with Round 1's
        subtraction; the $x^3$ and $12x$ terms cancel only if the minus reaches every term.
  \item $(x+3)(x^2+bx+c)=x^3+5x^2+x-15$: $b=2$, $c=-5$ (1 + 1) --- \emph{reverse}: match
        coefficients after multiplying in general.
  \item SOL-style: $(2x-3)^2-(x^2-4x)=3x^2-8x+9$, \textbf{C} (1). A = squared as $4x^2+9$;
        B = flipped only the first sign; D = added.
  \item \emph{Model}: canned-food-drive box $V(x)=x(30-2x)(20-2x)$. (a) $4x^3-100x^2+600x$ (2);
        (b) $V(5)=1000$ (1); (c) rubric (2): 1000 cubic inches for 5-inch squares; $x=12$ gives a
        width of $-4$ in.
  \item \emph{Find the mistake} (rubric, 3): Student A took $(x-5)^2$ as $x^2+25$; correct
        $(x^2-10x+25)(x+1)=x^3-9x^2+15x+25$ --- the crux in a new form.
\end{enumerate}
\end{skillbox}

\begin{skillbox}[Active Monitoring --- Watch For]{redbox}
\begin{itemize}[leftmargin=1.2em, nosep]
  \item \textbf{Round 1 \rung2 and Show me:} only the first sign flipped ---
        $-(2x^3+3x^2-x+6)$ written as $-2x^3+3x^2-x+6$. Point at the warm-up's item 3.
  \item \textbf{Round 1 \rung3:} degree of $2a^2b$ given as 2 --- add the exponents.
  \item \textbf{Round 2 \rung2--\rung3:} four products where there should be six; draw the box.
        Show-me finger \textbf{1} = FOIL with the middle lost.
  \item \textbf{Round 3:} $(x-3)^3$ or $(x+3)^3$ ``cubed term by term'' --- ask for the
        $x=1$ test. Closing Practice item 4 is the exit read on the crux.
  \item \textbf{Attitude:} a student parked at \rung1\ with nothing written --- crouch, point at
        the example in her notes, ask for just the first product.
\end{itemize}
Cold-call prompts (after the fingers, never before): ``How many products should
$(x+3)(2x^2-x+5)$ make before you combine?'' \; ``Why is $(x+2)^3$ not $x^3+8$ --- prove it
with a number.''
\end{skillbox}
