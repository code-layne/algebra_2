\begin{notesbox}{Practice}
Every answer needs \emph{two} parts: the simplest form \textbf{and} the restrictions, taken from the
\textbf{original} denominator. Write each final answer in its box. Only what is inside a box is graded.

\begin{enumerate}[leftmargin=1.9em, itemsep=11pt]
  \item \textbf{Monomial factors.} Simplify and restrict.
    \begin{enumerate}[label=(\alph*), leftmargin=2.1em, itemsep=5pt]
      \item $\dfrac{14x^5}{21x^3}=\;$\inblank{\blank{2.2cm}}{\gradedblank[id=q1a, type=short, points=1, answer={2x^2/3}, display={$\frac{2x^2}{3}$}]{2.2cm-2pt}{18pt}} \quad $x\neq$ \inblank{\blank{1.0cm}}{\gradedblank[id=q1b, type=short, points=1, answer={0}, check=numeric]{1.0cm-2pt}{18pt}}
      \item $\dfrac{8a^2b^3}{12a^5b}=\;$\inblank{\blank{2.2cm}}{\gradedblank[id=q1c, type=short, points=1, answer={2b^2/(3a^3)}, display={$\frac{2b^2}{3a^3}$}]{2.2cm-2pt}{18pt}} \quad restrictions: \inblank{\blank{2.6cm}}{\gradedblank[id=q1d, type=short, points=1, answer={a!=0, b!=0}, check=set, display={$a\neq0,\ b\neq0$}]{2.6cm-2pt}{18pt}}
      \item $\dfrac{5x^2y}{15xy^3}=\;$\inblank{\blank{2.2cm}}{\gradedblank[id=q1e, type=short, points=1, answer={x/(3y^2)}, display={$\frac{x}{3y^2}$}]{2.2cm-2pt}{18pt}} \quad restrictions: \inblank{\blank{2.6cm}}{\gradedblank[id=q1f, type=short, points=1, answer={x!=0, y!=0}, check=set, display={$x\neq0,\ y\neq0$}]{2.6cm-2pt}{18pt}}
    \end{enumerate}

  \item \textbf{Binomial and quadratic factors.} Factor completely first. One of these has
        \emph{opposite} binomials --- watch the sign.
    \begin{enumerate}[label=(\alph*), leftmargin=2.1em, itemsep=7pt]
      \item $\dfrac{x^2-49}{x-7}=\;$\inblank{\blank{2.4cm}}{\gradedblank[id=q2a, type=short, points=2, answer={x+7}]{2.4cm-2pt}{18pt}} \quad $x\neq$ \inblank{\blank{1.2cm}}{\gradedblank[id=q2b, type=short, points=1, answer={7}, check=numeric]{1.2cm-2pt}{18pt}}
      \item $\dfrac{5x-20}{x^2-16}=\;$\inblank{\blank{2.4cm}}{\gradedblank[id=q2c, type=short, points=2, answer={5/(x+4)}, display={$\frac{5}{x+4}$}]{2.4cm-2pt}{18pt}} \quad $x\neq$ \inblank{\blank{1.8cm}}{\gradedblank[id=q2d, type=short, points=1, answer={4, -4}, check=set, display={$4,\ -4$}]{1.8cm-2pt}{18pt}}
      \item $\dfrac{x^2+7x+10}{x^2+3x-10}=\;$\inblank{\blank{2.4cm}}{\gradedblank[id=q2e, type=short, points=2, answer={(x+2)/(x-2)}, display={$\frac{x+2}{x-2}$}]{2.4cm-2pt}{18pt}} \quad $x\neq$ \inblank{\blank{1.8cm}}{\gradedblank[id=q2f, type=short, points=1, answer={2, -5}, check=set, display={$2,\ -5$}]{1.8cm-2pt}{18pt}}
      \item $\dfrac{3-x}{x^2-9}=\;$\inblank{\blank{2.4cm}}{\gradedblank[id=q2g, type=short, points=2, answer={-1/(x+3)}, display={$-\frac{1}{x+3}$}]{2.4cm-2pt}{18pt}} \quad $x\neq$ \inblank{\blank{1.8cm}}{\gradedblank[id=q2h, type=short, points=1, answer={3, -3}, check=set, display={$3,\ -3$}]{1.8cm-2pt}{18pt}}
      \item $\dfrac{2x^2-8x}{x^2-16}=\;$\inblank{\blank{2.4cm}}{\gradedblank[id=q2i, type=short, points=2, answer={2x/(x+4)}, display={$\frac{2x}{x+4}$}]{2.4cm-2pt}{18pt}} \quad $x\neq$ \inblank{\blank{1.8cm}}{\gradedblank[id=q2j, type=short, points=1, answer={4, -4}, check=set, display={$4,\ -4$}]{1.8cm-2pt}{18pt}}
      \item $\dfrac{x^2-6x+9}{x^2-9}=\;$\inblank{\blank{2.4cm}}{\gradedblank[id=q2k, type=short, points=2, answer={(x-3)/(x+3)}, display={$\frac{x-3}{x+3}$}]{2.4cm-2pt}{18pt}} \quad $x\neq$ \inblank{\blank{1.8cm}}{\gradedblank[id=q2l, type=short, points=1, answer={3, -3}, check=set, display={$3,\ -3$}]{1.8cm-2pt}{18pt}}
    \end{enumerate}

  \item \textbf{Match the algebra to the picture.} The graph below is $g(x)=\dfrac{x^2+2x-8}{x-2}$.
    \begin{center}
    \begin{tikzpicture}[scale=0.44]
      \lgrid{-7}{4}{-3}{8}
      \begin{scope}
        \clip (-7,-3) rectangle (4,8);
        \draw[very thick, forest] (-7,-3)--(4,8);
      \end{scope}
      \draw[forest, very thick, fill=white] (2,6) circle (4.2pt);
    \end{tikzpicture}
    \end{center}
    Factored form: \inblank{\blank{4.2cm}}{\gradedblank[id=q3a, type=short, points=1, answer={(x+4)(x-2)/(x-2)}, display={$(x+4)(x-2)/(x-2)$}]{4.2cm-2pt}{10pt}} \quad Simplest form: \inblank{\blank{2.0cm}}{\gradedblank[id=q3b, type=short, points=1, answer={x+4}]{2.0cm-2pt}{10pt}}, \; for $x\neq$ \inblank{\blank{1.0cm}}{\gradedblank[id=q3c, type=short, points=1, answer={2}, check=numeric]{1.0cm-2pt}{10pt}}
    \\[3pt]
    The hole is at \inblank{\blank{1.6cm}}{\gradedblank[id=q3d, type=short, points=1, answer={(2,6)}, check=text]{1.6cm-2pt}{10pt}}. \quad Does $g$ have a vertical asymptote? \inblank{\blank{1.2cm}}{\gradedblank[id=q3e, type=short, points=1, answer={No}, check=text]{1.2cm-2pt}{10pt}} \\*[3pt]
    Why is the graph a \emph{straight line} even though $g$ is a rational function? \rubricbox{\gradedblank[id=q3f, type=rubric, points=2, display={the only denominator factor cancels, so away from $x=2$ the expression \emph{is} the line $y=x+4$}]{\linewidth}{10pt}}

  \item \textbf{Justify.} Are $\dfrac{x^2-1}{x-1}$ and $x+1$ the same expression? Answer in terms of
        \emph{inputs}: where do they agree, where do they disagree, and what is the graph of each?
        \gradedlines{3}{\gradedblank[id=q4, type=rubric, points=3, display={They are \emph{equivalent} --- $\frac{(x-1)(x+1)}{x-1}=x+1$ --- and they agree at every input except $x=1$, where the first is undefined ($\frac00$) and the second equals $2$. So the graph of $x+1$ is the whole line, while the graph of $\frac{x^2-1}{x-1}$ is that same line with a hole punched out at $(1,2)$.}]{\linewidth}{\gradedlinesheight}}
\end{enumerate}
\end{notesbox}

\begin{spiralbox}
\textbf{Coming next --- Lesson 4.2: Multiplying \& Dividing Rational Expressions.} Today's four steps
do not go away; they become step one of everything that follows. To multiply, you factor
\emph{everything} first and then divide out across the whole product; to divide, you keep-change-flip
and then do the same. The one new wrinkle: when you flip a divisor, its \emph{numerator} moves to the
bottom --- so it starts generating restrictions too. Keep tonight's habit of listing restrictions
before you cancel; in 4.2 there will be three denominators to watch instead of one.
\end{spiralbox}
