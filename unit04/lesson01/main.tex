\begin{tcolorbox}[colback=forestbg, colframe=forest, boxrule=0.9pt, arc=2mm,
    left=2mm, right=2mm, top=1.5mm, bottom=1.5mm]
    \textbf{Primary Objective:} Students can write a rational expression in \emph{simplest form} ---
    factor the numerator and denominator completely, then divide out common \emph{factors} --- and can
    state \emph{every} domain restriction, taken from the \textbf{original} denominator, including the
    restrictions whose factor cancels. Students explain \emph{why} cancelling is legal
    ($\frac{c}{c}=1$), why it applies only to factors and never to terms, and why it never restores an
    input that was already excluded; they recognize \emph{opposite binomials} ($\frac{a-b}{b-a}=-1$),
    and they \emph{justify} that two rational expressions are \textbf{equivalent} by naming the inputs
    where the two forms disagree. Throughout, the algebra is tied back to Lesson~4.0's pictures: a
    restriction that cancels is a \emph{hole}, one that stays is a \emph{vertical asymptote}.
    \\[2pt]
    \textbf{Standards (2023 VA SOL):} \textbf{A2.EO.1b} --- justify and determine equivalent rational
    algebraic expressions with monomial and binomial factors (expressions limited to linear and
    quadratic) --- and \textbf{A2.EO.1d} --- represent and demonstrate equivalence of rational
    expressions written in different forms. Builds directly on \textbf{A2.EO.3b} (factor completely,
    Lesson~3.2) and on \textbf{A2.F.2h}/holes/domain restrictions from Lesson~4.0; it is the
    prerequisite for \textbf{A2.EO.1a} (4.2--4.3), \textbf{A2.F.2h} from the equation (4.5), and
    \textbf{A2.EI.4c} extraneous solutions (4.6).
\end{tcolorbox}

\begin{skillbox}[Priority Ideas \& Skills]{goldbox}
\begin{tabularx}{\linewidth}{@{}X|X@{}}
\textbf{Priority Ideas \& Skills} & \textbf{Key Understandings (the ``why'')} \\[2pt]
\hline
\vspace{-6pt}
\begin{itemize}[leftmargin=1.1em, nosep, after=\vspace{2pt}]
  \item \textbf{Factor completely} first --- GCF, difference of squares, trinomials (the Unit~3 toolkit, reused).
  \item \textbf{List every restriction} from the \textbf{original} denominator, \emph{before} cancelling.
  \item \textbf{Divide out common factors} --- monomial and binomial --- and write the \textbf{simplest form with its restrictions attached}.
  \item Handle \textbf{opposite binomials}: $\dfrac{a-b}{b-a}=-1$.
  \item \textbf{Justify equivalence} of two forms, and name the inputs where they disagree.
  \item Connect back to 4.0: a restriction that \textbf{cancels} $\Rightarrow$ \textbf{hole}; one that \textbf{stays} $\Rightarrow$ \textbf{vertical asymptote}.
\end{itemize}
&
\vspace{-6pt}
\begin{itemize}[leftmargin=1.1em, nosep, after=\vspace{2pt}]
  \item Cancelling is not crossing out symbols --- it is dividing out a factor equal to \textbf{one}, which is why it needs \emph{factors} and needs them nonzero.
  \item \emph{Factors are multiplied; terms are added.} Every ``illegal cancel'' of the year is this one confusion.
  \item \textbf{Cancelling never restores an input.} The restriction list belongs to the original expression, permanently.
  \item Two forms can be equivalent and still have different \emph{domains} --- equivalence is about values where \emph{both} are defined.
  \item An answer here has two parts. An expression with no restrictions is half an answer.
  \item Today's excluded values are 4.0's holes, and they will come back in 4.6 as extraneous solutions.
\end{itemize}
\end{tabularx}
\end{skillbox}

\begin{skillbox}[{Vocabulary, Concepts, \& Theorems}]{sky}
\renewcommand{\arraystretch}{1.3}
\begin{tabularx}{\linewidth}{@{}l X@{}}
\textbf{Rational expression} & \TallMath{\frac{P(x)}{Q(x)}} with $P,Q$ polynomials and $Q(x)\neq0$ --- a rational function written without the ``$f(x)=$.'' \\
\textbf{Domain restriction} & An excluded value: any input making the \emph{original} denominator zero. \\
\textbf{Simplest form} & Numerator and denominator share no common factor except $1$ --- \emph{with} the original restrictions still stated. \\
\textbf{Fundamental principle} & \TallMath{\frac{ac}{bc}=\frac{a}{b}\cdot\frac{c}{c}=\frac{a}{b}} for $b\neq0$, $c\neq0$. The whole justification for cancelling. \\
\textbf{Factor vs.\ term} & Factors are multiplied; terms are added or subtracted. Only common \emph{factors} may be divided out. \\
\textbf{Equivalent expressions} & Same value at every input for which \emph{both} are defined --- so the two forms may have different domains. \\
\textbf{Opposite binomials} & Same terms, every sign flipped ($x-5$ and $5-x$); one is $-1$ times the other, so their quotient is $-1$. \\
\textbf{Hole vs.\ wall (4.0)} & Factor cancels $\Rightarrow$ removable discontinuity; factor stays $\Rightarrow$ vertical asymptote. \\
\end{tabularx}
\end{skillbox}

\begin{fixedskillbox}[Activate Prior Knowledge \& Spiral Review]{forestbg}
    The Warm-Up rehearses exactly three skills, one per leg of the lesson: \textbf{(1)} \emph{factoring
    completely} --- $x^2-9$, $x^2-x-6$, and $3x^2-12$ (GCF first) --- the Lesson~3.2 toolkit, which is
    the engine of everything today; \textbf{(2)} \emph{what may be cancelled}, established
    numerically before any variable appears: reducing $\frac{18}{24}$ by writing both parts as
    $6\cdot(\ )$, and then refuting ``$\frac{3+6}{3}=6$'' by computing the true value $3$ --- so
    \emph{factors, not terms} is a fact students proved, not a rule they were handed; and \textbf{(3)}
    \emph{finding excluded values} for $\frac{x+1}{x^2-4}$ with the domain written as a union, straight
    from Lesson~4.0, including the reminder that a zero \emph{numerator} is harmless. Debrief all
    three, but do \emph{not} yet say that restrictions survive cancelling --- that is the hook's job.
\end{fixedskillbox}

\begin{skillbox}[Hook]{forestbg}
    Put two expressions on the board --- $A(x)=\dfrac{x^2-x-6}{x^2-9}$ and $B(x)=\dfrac{x+2}{x+3}$ ---
    and have students fill a four-column table of values at $x=0,1,4$, and $3$. The first three columns
    agree exactly ($\tfrac23$, $\tfrac34$, $\tfrac67$); the last does not: $A(3)=\tfrac00$ is undefined
    while $B(3)=\tfrac56$. Ask \textbf{``$B$ is just $A$ factored and cancelled --- so are they the same
    expression or not?''} and let the disagreement stand for a moment; both answers are defensible
    until the word \emph{equivalent} is defined. Then the question that runs the lesson:
    \textbf{``Cancelling is supposed to be legal. So what exactly did it change, and what did it
    not?''} Land the sentence students will hear all unit: \emph{cancelling changes how an expression
    looks, but it never gives back an input that was already thrown away.} Close the loop to yesterday
    by pointing at the missing point: that is the \emph{hole} they read off a graph in Lesson~4.0 ---
    today they are producing it with algebra.
\end{skillbox}

\begin{skillbox}[Lesson]{forestbg}
\begin{multicols}{2}
\textbf{1. Rational expressions and their restrictions.} A \textbf{rational expression} is a quotient
of polynomials. The restriction procedure is Lesson~4.0's, unchanged: set the \emph{denominator} equal
to zero and solve. Do the anchor ($x^2-9=0\Rightarrow x\neq3,-3$) and re-flag that a zero
\emph{numerator} is a zero of the expression, not a restriction. State the order rule explicitly:
\textbf{restrictions come before cancelling}, because after cancelling one of them is invisible.

\textbf{2. Why cancelling works.} Write the fundamental principle
$\frac{ac}{bc}=\frac ab\cdot\frac cc=\frac ab$ ($b,c\neq0$). Name what students are actually doing:
dividing out a factor equal to \textbf{one}. That single line explains both restrictions on the move
--- $c$ must be a \emph{factor}, and $c$ must be nonzero. Then kill the year's most durable error the
only way that works: \textbf{numerically}. A classmate claims $\frac{x+3}{3}=x$; test $x=6$ and get
$3\neq6$. \emph{Factors are multiplied; terms are added.}

\textbf{3. The four steps.} Factor completely $\rightarrow$ list restrictions from the \emph{original}
denominator $\rightarrow$ divide out shared factors $\rightarrow$ write the simplest form \emph{with}
its restrictions. Work the anchor to $\frac{x+2}{x+3}$, $x\neq3,-3$, and stop on the punch line: the
$(x-3)$ cancelled, \textbf{and $x=3$ is still excluded}. Then a monomial case
($\frac{12x^3y}{18x^5y^2}=\frac{2}{3x^2y}$, $x,y\neq0$ --- the A2.EO.1b ``monomial factors'' clause)
and a GCF case ($\frac{3x^2-12}{x^2+x-6}=\frac{3(x+2)}{x+3}$, $x\neq2,-3$).

\columnbreak

\textbf{4. Opposite binomials.} $x-5$ and $5-x$ do not cancel --- but $5-x=-1(x-5)$, so their quotient
is $-1$. Show the general form $\frac{a-b}{b-a}=-1$, and immediately contrast it with $x+3$ and $3+x$,
which \emph{are} equal: only \emph{subtraction} cares about order. Work
$\frac{x^2-16}{4-x}=-(x+4)$, $x\neq4$. Warn that the lost $-1$ is the most common single-point loss on
this topic.

\textbf{5. Equivalence --- and the graph it describes (A2.EO.1d).} Define \textbf{equivalent}: same
value at every input \emph{both} are defined for. This resolves the hook honestly --- $A$ and $B$
\emph{are} equivalent, and they still have different domains. Show the anchor's graph: a curve with a
vertical asymptote at $x=-3$ (factor stays) and a hole at $\left(3,\tfrac56\right)$ (factor cancels),
with the hole's height read off the \emph{simplified} form because the original has no value there at
all. Give both equivalence checks --- factor, or test an input --- and be precise about what a test
can do: one match proves nothing, one mismatch disproves everything (which is exactly why the $x=6$
test settled Section~2). Finish by naming where this returns: \textbf{4.2--4.3} (every operation opens
by factoring), \textbf{4.5} (the cancel test builds the graph), and \textbf{4.6} (an extraneous
solution is one of today's excluded values sneaking back in).
\end{multicols}
\end{skillbox}

\begin{skillbox}[Explicit Instruction: The Four-Step Simplification]{forestbg}
\begin{tabularx}{\linewidth}{@{}p{0.46\linewidth} X@{}}
\vspace{0pt}
\textbf{Steps (post these).}
\begin{enumerate}[leftmargin=1.3em, nosep, after=\vspace{2pt}]
  \item Factor the numerator completely.
  \item Factor the denominator completely.
  \item \textbf{Write the restrictions now} --- one for every denominator factor, from the
        \emph{original}.
  \item Divide out the factors that appear on both levels (watch for opposite binomials --- factor
        $-1$ out first).
  \item Write the simplest form, restrictions attached.
\end{enumerate}
&
\vspace{0pt}
\textbf{Worked, on the anchor.}
\[ \frac{x^2-x-6}{x^2-9}=\frac{(x-3)(x+2)}{(x-3)(x+3)} \]
Restrictions from the original bottom: $x\neq3$ and $x\neq-3$.
\[ =\frac{x+2}{x+3}, \qquad x\neq3,\;-3 \]
\textbf{Check the two questions students must answer aloud:} \emph{which factor did you divide out?}
and \emph{which restriction just went invisible?} A student who cannot answer the second has not
finished the problem.
\end{tabularx}
\end{skillbox}

\begin{skillbox}[Active Monitoring]{redbox}
Circulate during the guided practice and Tier work. Watch for and correct:
\begin{itemize}[leftmargin=1.2em, nosep]
  \item \textbf{cancelling terms} --- $\frac{x^2+9}{x+3}=x+3$, $\frac{x+3}{3}=x$, or crossing out matching symbols across a $+$ sign. Do not argue; hand the student a test value;
  \item \textbf{restrictions read off the simplified denominator}, which silently deletes the cancelled one --- the signature error of the lesson, and the direct ancestor of extraneous solutions in 4.6;
  \item \textbf{no restrictions at all} --- a simplified expression alone is half an answer;
  \item \textbf{opposite binomials treated as identical} ($\frac{3-x}{x^2-9}$ answered $\frac{1}{x+3}$ instead of $-\frac{1}{x+3}$);
  \item \textbf{incomplete factoring} --- skipping the GCF, or stopping before the difference of squares (the Unit 3 ``factor \emph{completely}'' habit);
  \item \textbf{cancelling a squared factor entirely} --- $\frac{(x+2)}{(x+2)^2}$ leaves one $(x+2)$ downstairs, so the restriction is a wall, not a hole.
\end{itemize}
Cold-call prompts: ``Which \emph{factor} did you divide out?'' ``Is that a factor or a term --- what is
it multiplied by?'' ``Where did the restrictions come from --- the original or the simplified
denominator?'' ``Does that restriction give a hole or a wall, and how do you know?''
\end{skillbox}

\begin{skillbox}[Group Work \& Differentiation]{redbox}
\textbf{Factor, Restrict, Divide Out} activity (tiered; mirrors the notes).
\begin{multicols}{3}
\textbf{Tier R --- Remediate.} Five scaffolded items building from monomial factors ($\frac{10x^4}{15x^2}$)
through one-binomial cases ($\frac{x^2-25}{x+5}$, $\frac{4x+8}{x^2-4}$) to a both-sides factoring
($\frac{x^2-x-12}{x^2-16}$), each with the factored line, the simplest form, and the restrictions asked
for separately. Closes with the lesson's core question in plain words: the factor $(x-4)$ cancelled ---
is $x=4$ still excluded, and why?

\columnbreak
\textbf{Tier A --- Approaching.} A four-row table (factored form / restrictions / simplest form) over
$\frac{6x^3y^2}{9xy^5}$, $\frac{2x^2-18}{x^2+6x+9}$ (GCF + perfect-square trinomial),
$\frac{25-x^2}{x^2-3x-10}$ (\emph{opposite binomials}), and $\frac{x^2+2x}{x^2+4x+4}$ (a squared factor
that only partly cancels). Then an \textbf{error analysis}: a student ``simplifies''
$\frac{x^2+9}{x+3}$ to $x+3$; groups disprove it at $x=1$ ($2.5$ vs.\ $4$), name the error using
\emph{factor} and \emph{term}, and recall from Unit~3 that $x^2+9$ is prime.

\columnbreak
\textbf{Tier E --- Extension.} \emph{Part 1:} three candidates for ``the same as'' $\frac{x-4}{x+1}$ ---
$\frac{x^2-16}{x^2+5x+4}$ (equivalent, extra restriction $x\neq-4$), $\frac{4-x}{x+1}$ (the trap: it is
the \emph{opposite}), $\frac{x^2-4x}{x^2+x}$ (equivalent, extra restriction $x\neq0$) --- with a written
justification of why differing domains do not break equivalence. \emph{Part 2:} build an expression
that simplifies to $\frac{x}{x-1}$ with restrictions $x\neq1,-5$. \emph{Part 3:} match a pre-drawn
line-with-a-hole to $\frac{x^2-2x-3}{x-3}$, give the hole $(3,4)$, and explain why its height cannot
come from the original.
\end{multicols}
\end{skillbox}

\begin{skillbox}[Individual Work \& Assessment]{redbox}
\textbf{Exit Ticket} (independent, no notes): \textbf{(1)} simplify $\frac{x^2-2x-15}{x^2-9}$ showing
the factored line, give the simplest form $\frac{x-5}{x-3}$ and \emph{both} restrictions, then sort
them --- hole at $x=-3$ (cancels), vertical asymptote at $x=3$; \textbf{(2)} \textbf{explain}, against a
classmate's claim that ``once the $(x+3)$ cancels, $x=-3$ is fine now'' --- full credit requires both
halves: no value exists at $x=-3$ in the \emph{original}, and the two forms agree everywhere else;
\textbf{(3)} an \textbf{SOL-style multiple-choice} item on $\frac{4-x^2}{x^2-x-2}$ whose distractors
each break exactly one idea (drop the $-1$ from the opposite binomials / read restrictions off the
simplified denominator / cancel the $x^2$ \emph{terms}). Collect and sort into ``got it / restriction
slip / sign slip / term-cancelling.'' The restriction pile decides whether 4.2 opens with a re-teach ---
it is the same error that manufactures extraneous solutions in 4.6.
\end{skillbox}

\begin{skillbox}[Reinforcement \& Extension]{goldbox}
\textbf{Homework:} three monomial-factor simplifications (including two-variable cases with their
$a\neq0$, $b\neq0$ restrictions); six binomial/quadratic simplifications spanning difference of
squares, GCF-then-factor, a trinomial-over-trinomial, a perfect-square trinomial (only \emph{one}
factor cancels), and one \emph{opposite-binomial} case, $\frac{3-x}{x^2-9}=-\frac{1}{x+3}$; a
graph-matching item where $g(x)=\frac{x^2+2x-8}{x-2}$ turns out to be the \textbf{line} $y=x+4$ with a
hole at $(2,6)$ and \emph{no} vertical asymptote; and a justification item on whether
$\frac{x^2-1}{x-1}$ and $x+1$ are ``the same.'' \textbf{Extension:} an error gallery --- three students,
three answers to $\frac{x^2-4}{x^2+4x+4}$ (correct / cancelled the $x^2$ terms / correct form but no
restrictions) --- and a \emph{design} task that previews 4.5: build an expression whose graph has a hole
at $x=2$ and a vertical asymptote at $x=-1$, then say which factor had to appear on both levels and
why. \textbf{Preview:} Lesson~4.2, \emph{Multiplying \& Dividing Rational Expressions} --- the same four
steps, now applied across a product, with the warning that flipping a divisor sends its \emph{numerator}
to the bottom, where it starts generating restrictions of its own.
\end{skillbox}
