\begin{notesbox}{Practice}
Show how you read each feature. For every graph, give the \emph{full} feature list.

\begin{enumerate}[leftmargin=1.9em, itemsep=12pt]
  \item \textbf{From a factored form.} Let $f(x) = (x-3)(x+1)^2$ (do \emph{not} expand).
    \begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=4pt]
      \item Its \textbf{degree} is \blank{1.0cm} and its \textbf{leading coefficient} is
            \blank{1.0cm}. (Multiply the leading term of each factor.)
      \item \textbf{End behavior:} as $x\to-\infty,\ f(x)\to\blank{1.0cm}$; as
            $x\to+\infty,\ f(x)\to\blank{1.0cm}$.
      \item \textbf{Zeros \& multiplicity:} $x=\blank{1.0cm}$ has multiplicity \blank{0.8cm}, so the
            graph \blank{1.6cm} there; $x=\blank{1.0cm}$ has multiplicity \blank{0.8cm}, so the graph
            \blank{1.6cm} there.
      \item \textbf{$y$-intercept:} \blank{1.8cm}.
    \end{enumerate}

  \item \textbf{A cubic graph.} The graph is $d(x) = x^3 - 3x^2 = x^2(x-3)$.
    \begin{center}
    \begin{tikzpicture}[scale=0.5]
      \lgrid{-2}{4}{-5}{5}
      \begin{scope}
        \clip (-2,-5) rectangle (4,5);
        \draw[very thick, forest, domain=-1.35:3.45, samples=90, smooth]
          plot (\x,{(\x)*(\x)*(\x) - 3*(\x)*(\x)});
      \end{scope}
      \fill[charcoal] (0,0) circle (3pt) node[above left, font=\scriptsize]{$(0,0)$};
      \fill[charcoal] (2,-4) circle (3pt) node[below right, font=\scriptsize]{$(2,-4)$};
      \fill[charcoal] (3,0) circle (3pt) node[above right, font=\scriptsize]{$(3,0)$};
    \end{tikzpicture}
    \end{center}
    Degree even/odd: \blank{1.8cm} \quad Both arms: \blank{2.6cm} \\[3pt]
    Zeros (with cross/touch): $x=0$ \blank{1.8cm}; \; $x=3$ \blank{1.8cm} \\[3pt]
    Turning points: \blank{0.8cm} \quad Relative max: \blank{1.6cm} \quad Relative min: \blank{1.6cm}
    \\[3pt]
    Increasing on: \blank{3.2cm} \quad Decreasing on: \blank{2.0cm}

  \item \textbf{A quartic graph.} The graph is $e(x) = x^4 - 5x^2 + 4 = (x-1)(x+1)(x-2)(x+2)$.
    \begin{center}
    \begin{tikzpicture}[scale=0.5]
      \lgrid{-3}{3}{-3}{6}
      \draw[navy, dashed, semithick] (0,-3)--(0,6);
      \begin{scope}
        \clip (-3,-3) rectangle (3,6);
        \draw[very thick, forest, domain=-2.45:2.45, samples=110, smooth]
          plot (\x,{(\x)^4 - 5*(\x)*(\x) + 4});
      \end{scope}
      \fill[charcoal] (0,4) circle (3pt) node[above right, font=\scriptsize]{$(0,4)$};
      \fill[charcoal] (-1,0) circle (3pt); \fill[charcoal] (1,0) circle (3pt);
      \fill[charcoal] (-2,0) circle (3pt); \fill[charcoal] (2,0) circle (3pt);
    \end{tikzpicture}
    \end{center}
    Degree even/odd: \blank{1.8cm} \quad Both arms: \blank{2.6cm} \quad Turning points: \blank{0.8cm}
    \\[3pt]
    Zeros (with cross/touch): \blank{5.0cm} \\[3pt]
    $y$-intercept: \blank{1.8cm} \quad Symmetry (even/odd/neither): \blank{2.2cm}

  \item \textbf{Explain.} Why does an \emph{odd}-degree polynomial have \emph{no} absolute maximum
        and \emph{no} absolute minimum? Refer to what its two arms do. \writelines{2}
\end{enumerate}
\end{notesbox}

\begin{extensionbox}
\textbf{Reading a model in context.} A start-up's monthly profit (in thousands of dollars) is
modeled by $P(t) = t^3 - 6t^2 + 9t$ for $0 \le t \le 3$ months.
\begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=5pt]
  \item Factor $P(t)$ completely and list its zeros with their multiplicities.
  \item The graph \emph{touches} the $t$-axis at $t=3$ instead of crossing. What does that
        \emph{mean} for the company's profit at $t=3$?
  \item On $0\le t\le 3$, the greatest profit occurs at the relative maximum. Read it from the model
        ($P(1)$) and interpret it in context.
\end{enumerate}
\writelines{3}
\end{extensionbox}
