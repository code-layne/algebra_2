\begin{tcolorbox}[colback=forestbg, colframe=forest, boxrule=0.9pt, arc=2mm,
    left=2mm, right=2mm, top=1.5mm, bottom=1.5mm]
    \textbf{Primary Objective:} Students can \emph{factor a polynomial completely} in one and two
    variables --- always pulling the \emph{GCF} out first, then choosing a method by the
    \emph{number of terms}: two terms (difference of squares, or the new \emph{sum and difference of
    cubes}), three terms (perfect-square trinomial, trinomial factoring, or \emph{quadratic form}),
    four terms (\emph{grouping}) --- and they can keep going until every factor is prime over the
    integers. They can \emph{verify} a factorization by multiplying it back out (Lesson~4.1 run
    forward) and can read the \emph{zeros} of a polynomial function off its completely factored form.
    This is Lesson~4.1 in reverse: yesterday factored $\rightarrow$ standard; today standard
    $\rightarrow$ factored.
    \\[2pt]
    \textbf{Standards (2023 VA SOL):} \textbf{A2.EO.3b} --- factor polynomials \emph{completely} in
    one and two variables with no more than four terms over the set of integers. \textbf{A2.EO.3d}
    --- represent and demonstrate equality of polynomial expressions written in different forms, and
    verify polynomial identities including the difference of squares, the \emph{sum and difference of
    cubes}, and perfect-square trinomials. \textbf{A2.F.2c} is revisited whenever a completely
    factored form is used to name a function's zeros (Lesson~4.0), and \textbf{A2.EO.3a} is reused
    every time a factorization is checked by multiplying.
\end{tcolorbox}

\begin{skillbox}[Priority Ideas \& Skills]{goldbox}
\begin{tabularx}{\linewidth}{@{}X|X@{}}
\textbf{Priority Ideas \& Skills} & \textbf{Key Understandings (the ``why'')} \\[2pt]
\hline
\vspace{-6pt}
\begin{itemize}[leftmargin=1.1em, nosep, after=\vspace{2pt}]
  \item Factor out the \textbf{GCF} first --- coefficients and the \emph{lowest} power of each shared variable --- including a negative leading coefficient.
  \item Choose a method by the \textbf{number of terms}: 2, 3, or 4.
  \item Apply the \textbf{difference of squares} and \textbf{perfect-square trinomial} identities, in one and two variables.
  \item Apply the new \textbf{sum and difference of cubes}: $a^3\pm b^3=(a\pm b)(a^2\mp ab+b^2)$.
  \item Factor a four-term polynomial by \textbf{grouping}.
  \item Recognize \textbf{quadratic form} ($x^4+bx^2+c$) and factor \textbf{completely}.
  \item \textbf{Verify} by multiplying back, and read the \textbf{zeros} off the factored form.
\end{itemize}
&
\vspace{-6pt}
\begin{itemize}[leftmargin=1.1em, nosep, after=\vspace{2pt}]
  \item Factoring is multiplication run backward --- so every answer is checkable by the work students did in Lesson~4.1.
  \item ``\textbf{Completely}'' is the whole lesson: a correct first step can still be an unfinished answer.
  \item The \emph{number of terms} is the single best clue to which tool to reach for.
  \item $a^2-b^2$ factors but $a^2+b^2$ does \emph{not} --- yet $a^3+b^3$ \emph{does}. Cubes behave differently from squares.
  \item Grouping works because a common \emph{binomial} factors out exactly the way a common monomial does.
  \item Standard form shows end behavior; factored form shows the \textbf{zeros and their multiplicities} --- which is why we want both.
\end{itemize}
\end{tabularx}
\end{skillbox}

\begin{skillbox}[{Vocabulary, Concepts, \& Theorems}]{sky}
\renewcommand{\arraystretch}{1.3}
\begin{tabularx}{\linewidth}{@{}l X@{}}
\textbf{Factor (verb)} & Rewrite a sum of terms as a \emph{product}. \\
\textbf{GCF} & The greatest common factor: the largest coefficient and the \emph{lowest} power of every variable shared by all terms. \\
\textbf{Prime (irreducible)} & A polynomial that cannot be factored further over the integers --- e.g.\ $x^2+9$ and $x^2+2x+4$. \\
\textbf{Factored completely} & Every factor is prime over the integers; nothing is left to pull apart. \\
\textbf{Difference of squares} & \TallMath{a^2-b^2=(a+b)(a-b)} \quad (but $a^2+b^2$ is \emph{prime}) \\
\textbf{Perfect-square trinomial} & \TallMath{a^2\pm 2ab+b^2=(a\pm b)^2} \\
\textbf{Sum of cubes} & \TallMath{a^3+b^3=(a+b)(a^2-ab+b^2)} \\
\textbf{Difference of cubes} & \TallMath{a^3-b^3=(a-b)(a^2+ab+b^2)} \\
\textbf{SOAP} & The sign pattern in both cube identities: \textbf{S}ame, \textbf{O}pposite, \textbf{A}lways \textbf{P}ositive. \\
\textbf{Factoring by grouping} & Split four terms into two pairs, GCF each pair, then factor out the shared \emph{binomial}. \\
\textbf{Quadratic form} & A polynomial like $x^4+bx^2+c$ that factors as if $x^2$ were the variable. \\
\textbf{Zero of a function} & An $x$ making $f(x)=0$; each linear factor $(x-r)$ contributes the zero $r$. \\
\end{tabularx}
\end{skillbox}

\begin{fixedskillbox}[Activate Prior Knowledge \& Spiral Review]{forestbg}
    The Warm-Up rehearses the four prerequisites this lesson stacks on: \textbf{(1)} \emph{greatest
    common factor} of numbers and monomials, and pulling it out (Algebra~1); \textbf{(2)} factoring
    $x^2+bx+c$ and the difference of squares from Lesson~3.2 --- the tools students already own;
    \textbf{(3)} a Lesson~4.1 expansion, $(x+2)(x^2-2x+4)$, which collapses to $x^3+8$ --- this is the
    \emph{sum of cubes discovered by multiplication}, and today's new identity should be introduced by
    calling back to it rather than by announcement; and \textbf{(4)} a short perfect-cubes table
    ($2^3$ through $5^3$), without which the cube identities are unusable. Debrief item~3 aloud:
    \textbf{``why did four of the six products vanish?''} --- the middle terms cancel in pairs, exactly
    as they do in a difference of squares.
\end{fixedskillbox}

\begin{skillbox}[Hook]{forestbg}
    Put Lesson~4.0's quartic back on the board --- students have already read this graph --- next to
    its equation:
    \[
      q(x) = x^4 - 5x^2 + 4 .
    \]
    \textbf{``You read four $x$-intercepts off this graph: $-2$, $-1$, $1$, and $2$. Where are they
    hiding in the equation?''} They are not visible in standard form; they are not \emph{anywhere} in
    standard form. \textbf{``What form would show them?''} A product of four factors:
    $(x-1)(x+1)(x-2)(x+2)$. Yesterday we went factored $\rightarrow$ standard by multiplying; today we
    go standard $\rightarrow$ factored, and the payoff is the zeros. Then set the day's real standard:
    write $x^4-16=(x^2-4)(x^2+4)$ on the board and ask \textbf{``is this factored?''} --- yes; \textbf{``is
    it factored \emph{completely}?''} --- no, because $x^2-4$ is still factorable. \emph{A correct step
    is not the same as a finished answer.} That distinction is what the word ``completely'' in today's
    standard is buying.
\end{skillbox}

\begin{skillbox}[Lesson]{forestbg}
\begin{multicols}{2}
\textbf{1. GCF first --- every time, no exceptions.} The GCF takes the largest common coefficient and
the \emph{lowest} exponent on each shared variable: $12x^5-18x^4+30x^3 = 6x^3(2x^2-3x+5)$. Do a
two-variable case, $15x^3y^2-25x^2y^3=5x^2y^2(3x-5y)$, and a negative-lead case,
$-4x^3+8x^2-12x=-4x(x^2-2x+3)$, where factoring out $-4x$ flips every sign inside. \textbf{Ask:}
\textbf{``Why does pulling the GCF out first make everything after it easier?''} (It shrinks the
degree and the numbers, and it often exposes an identity that was disguised: $2x^3-50x$ becomes
$2x(x^2-25)$, and only now is the difference of squares visible.)

\textbf{2. Count the terms --- that is your menu.} Build the decision tree on the board and leave it
up all period:
\begin{itemize}[leftmargin=1.1em, nosep]
  \item \textbf{2 terms:} difference of squares, or sum/difference of \emph{cubes}.
  \item \textbf{3 terms:} perfect-square trinomial, or ordinary trinomial factoring (grouping when $a\ne 1$); watch for quadratic form.
  \item \textbf{4 terms:} grouping.
\end{itemize}
Then the rule that outranks all of them: \textbf{look at every factor again.}

\textbf{3. Squares --- and the one that is prime.} $49x^2-64=(7x-8)(7x+8)$;
$9x^2-25y^2=(3x-5y)(3x+5y)$. Push it: $x^4-16=(x^2-4)(x^2+4)=(x-2)(x+2)(x^2+4)$. \textbf{Ask:}
\textbf{``Why did we stop at $x^2+4$?''} A \emph{sum} of squares is prime over the integers --- there
is no pair of integers whose product is $+4$ and whose sum is $0$. Perfect-square trinomials run the
4.1 identity backward: $x^2-14x+49=(x-7)^2$, $4x^2+20xy+25y^2=(2x+5y)^2$. Have students \emph{check
the middle term} ($2ab$) before committing.

\columnbreak

\textbf{4. Grouping --- four terms, two pairs.} $x^3+5x^2-4x-20$: GCF each pair,
$x^2(x+5)-4(x+5)$, then pull out the common \emph{binomial}: $(x+5)(x^2-4)$ --- and \emph{keep going}
to $(x+5)(x-2)(x+2)$. \textbf{The sign move:} when the third term is negative, factor a
\emph{negative} out of the second pair so the two binomials match; if they do not match, the sign is
almost always the reason. Do one that ends prime, $2x^3-6x^2-9x+27=(x-3)(2x^2-9)$, so students see
that ``completely'' sometimes means stopping.

\textbf{5. Cubes --- the new identity.} Return to the Warm-Up: students already multiplied
$(x+2)(x^2-2x+4)$ and got $x^3+8$. \textbf{``You proved today's formula before you knew it existed.''}
Generalize:
\[
  a^3+b^3=(a+b)(a^2-ab+b^2),\quad a^3-b^3=(a-b)(a^2+ab+b^2).
\]
Teach the sign pattern as \textbf{SOAP}: \emph{Same} sign in the binomial, \emph{Opposite} sign on the
middle term, \emph{Always Positive} at the end. Flag the two errors that account for nearly every
wrong answer: putting a $2$ on the middle term (it is $ab$, not $2ab$ --- this is \emph{not} a
perfect-square trinomial), and trying to factor $a^2-ab+b^2$ further (it is prime). Work
$27x^3-64=(3x-4)(9x^2+12x+16)$ and $8x^3+125y^3=(2x+5y)(4x^2-10xy+25y^2)$, identifying $a$ and $b$
\emph{out loud} first.

\textbf{6. Close the Hook.} $x^4-5x^2+4$ has three terms, so treat it as a quadratic in $x^2$:
$(x^2-1)(x^2-4)$ --- then finish, $(x-1)(x+1)(x-2)(x+2)$. Read the four zeros and match them to the
graph from Lesson~4.0. Then preview honestly: try $x^3-3x+2$ (the unit anchor) with today's tools ---
GCF, no; grouping, no; identities, no --- \emph{and it still factors.} That gap is exactly why
Lessons~4.3 and 4.4 exist.
\end{multicols}
\end{skillbox}

\begin{skillbox}[Active Monitoring]{redbox}
Circulate during the notes practice and Tier work. Watch for and correct:
\begin{itemize}[leftmargin=1.2em, nosep]
  \item skipping the GCF and diving straight at an identity --- $2x^3-50x$ ``does not factor'' until the $2x$ comes out;
  \item stopping early --- $(x^2-4)(x^2+4)$ or $(x+5)(x^2-4)$ turned in as final answers; require the words ``every factor is prime'' before a problem is done;
  \item taking the \emph{highest} power in the GCF instead of the lowest (writing $x^5$ out of $12x^5-18x^4+30x^3$);
  \item trying to factor $x^2+9$ (or $a^2+b^2$ generally) --- ask for the integer pair, and let the search fail;
  \item a $2$ in the cube identity's middle term, or a wrong SOAP sign --- send students to multiply their answer back out;
  \item mismatched binomials in grouping caused by not factoring a negative out of the second pair;
  \item forgetting to bring the GCF along into the final answer once the inside has been factored.
\end{itemize}
Cold-call prompts: ``What is always the first step?'' ``How many terms --- so which tool?'' ``Is that
factor prime, or is there one more step?'' ``Multiply your answer back: do you get what you started
with?''
\end{skillbox}

\begin{skillbox}[Group Work \& Differentiation]{redbox}
\textbf{Taking Polynomials Apart} activity (tiered; mirrors the notes).
\begin{multicols}{3}
\textbf{Tier R --- Remediate.} Two GCF problems (one two-variable), one difference of squares, one
$x^2+bx+c$ trinomial, and one four-term grouping with the two pairs already bracketed and the steps
laid out line by line. Each answer is checked by multiplying back.

\columnbreak
\textbf{Tier A --- Approaching.} GCF-then-identity problems that require two steps; a perfect-square
trinomial in two variables; a sum and a difference of cubes (with an ``identify $a$ and $b$ first''
line); an $a\ne1$ trinomial by grouping; a ``prime or not?'' sort ($x^2+9$ vs.\ $x^3+27$); and an
error-analysis item on $x^4-16=(x^2-4)(x^2+4)$ turned in as a final answer.

\columnbreak
\textbf{Tier E --- Extension.} Factor the Hook's $x^4-5x^2+4$ completely and match its four zeros to
the Lesson~4.0 graph; factor $x^6-64$ \emph{two ways} (difference of squares first vs.\ difference of
cubes first) and decide which route finishes --- the takeaway being ``squares before cubes''; verify
the difference-of-cubes identity by expanding it (A2.EO.3d); and hit the wall on $x^3-3x+2$, which
today's tools cannot crack, as a written preview of Lesson~4.3.
\end{multicols}
\end{skillbox}

\begin{skillbox}[Individual Work \& Assessment]{redbox}
\textbf{Exit Ticket} (independent, no notes): \textbf{(1)} $3x^3-27x$ --- a GCF-then-difference-of-squares
problem that is only right if students both start with the GCF \emph{and} finish; \textbf{(2)} a
four-term grouping, $x^3-2x^2-25x+50$, that must be carried past $(x-2)(x^2-25)$ to
$(x-2)(x-5)(x+5)$, followed by naming the function's three \emph{zeros} (the Lesson~4.0 link); and
\textbf{(3)} an \textbf{SOL-style multiple-choice} item on $8x^3-27$ whose distractors are exactly the
three cube-identity errors (a $2$ in the middle term, a flipped SOAP middle sign, and a flipped
binomial sign). Sort the tickets into ``got it / stopped early / GCF skipped / SOAP sign'' to decide
whether Lesson~4.3 opens with a five-minute re-teach.
\end{skillbox}

\begin{skillbox}[Reinforcement \& Extension]{goldbox}
\textbf{Homework:} GCF (including two-variable and a negative lead); two-term factoring across all
three identities plus a \emph{prime} case; three-term factoring (perfect square, two-variable
trinomial, $a\ne1$ by grouping); two four-term groupings; two multi-step ``factor completely''
problems ($2x^4-32$ and $x^4-10x^2+9$); a return to Lesson~4.0's graph of $x^4-5x^2+4$ --- factor it,
list its zeros, and confirm them against the intercepts already read; and one justification item on
why $x^2+9$ is prime while $x^3+27$ is not. \textbf{Extension:} verify the sum-of-cubes identity by
multiplying $(a+b)(a^2-ab+b^2)$ out in full, then apply it to a hollow-cube volume,
$x^3-27=(x-3)(x^2+3x+9)$, and interpret both factors geometrically. \textbf{Preview:} Lesson~4.3,
\emph{Dividing Polynomials} --- long and synthetic division and the Remainder \& Factor Theorems, the
tools that finally crack the polynomials today's identities cannot reach.
\end{skillbox}
