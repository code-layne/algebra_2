  \item \textbf{Solve completely.} \; $\dfrac{x}{x-1}+\dfrac{1}{x}=\dfrac{1}{x^2-x}$

        \vspace{3pt}
        \textbf{Step 1.} Factor: $x^2-x=\rule{2.0cm}{0.4pt}$. \; Restrictions: $x\neq\blank{2.0cm}$.
        \; LCD $=\blank{2.4cm}$

        \vspace{3pt}
        \textbf{Step 2.} Multiply every term and cancel:
        \; $\rule{1.6cm}{0.4pt}+\rule{1.6cm}{0.4pt}=\rule{1.2cm}{0.4pt}$

        \vspace{3pt}
        \textbf{Step 3.} Solve: $0=x^2+\rule{0.6cm}{0.4pt}\,x-\rule{0.6cm}{0.4pt}
        =(x+\rule{0.5cm}{0.4pt})(x-\rule{0.5cm}{0.4pt})$. \; Candidates \blank{2.2cm}

        \vspace{3pt}
        \textbf{Step 4.} Extraneous \blank{1.2cm} \quad Solution set \blank{2.0cm}

  \item \textbf{Justify it.} In one or two sentences, explain to a classmate why the candidate you threw
        away in item 1 is not a solution --- and why that does \emph{not} mean anyone made an algebra
        mistake. Use the words \textbf{denominator} and \textbf{extraneous}.
        \writelines{3}

        \[ \frac{x}{x+3}+\frac{3}{x-3}=\frac{18}{x^2-9}\;? \]

        \end{enumerate}
