            \begin{itemize}

                \begin{itemize}
                    \item Evaluating \& simplifying expressions
                    \item Solving equations \& inequalities
                    \item Linear \& quadratic functions \& regressions
                \end{itemize}
                \item \textbf{Unit 2}:  Linear, Absolute Value, \& Piecewise Functions
                \begin{itemize}
                    \item Linear functions \& transformations
                    \item Absolute value functions \& transformations
                    \item Piecewise functions
                    \item Absolute value equations \& inequalities
                \end{itemize}
                \item \textbf{Unit 3}:  Quadratic Functions \& Complex Numbers
                \begin{itemize}
                    \item Forms of quadratic functions \& transformations
                    \item Quadratic formula: the discriminant
                    \item Complex numbers \& operations
                    \item Solving quadratic equations
                    \item Systems of quadratics
                \end{itemize}
                \item \textbf{Unit 4}:  Polynomial Functions \& the Fundamental Theorem of Algebra
                \begin{itemize}
                    \item Polynomial functions \& factoring
                    \item Solving polynomial equations
                    \item \textit{Fundamental Theorem of Algebra}
                    \item \textit{Factor Theorem} \& multiplicity
                \end{itemize}
                \item \textbf{Unit 5}:  Rational Functions \& Function Operations
                \begin{itemize}
                    \item Simplifying rational expressions
                    \item Long \& synthetic division of polynomials
                    \item Direct, inverse, \& joint variations
                    \item Rational functions, discontinuities, \& asymptotes
                \end{itemize}
            \end{itemize}

                \item Unit 10:  Probability \& Statistics
            \end{itemize}
