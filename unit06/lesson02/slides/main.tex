% THE FORM
\begin{frame}[t, plain]
  \forestheader{One form, four jobs --- and only one of them matters to the range}
  \sectionlabel{The form}
  \begin{center}
    \Large $g(x) = a\,b^{\,x-h} + k$ \qquad from the parent \qquad $y=b^{x}$
  \end{center}
  \vspace{6pt}
  \begin{center}
  \footnotesize
  \renewcommand{\arraystretch}{1.35}
  \begin{tabular}{|l|l|l|}
  \hline
  & \textbf{What it does} & \textbf{Asymptote moves?} \\
  \hline
  $k$ (outside) & shifts the graph up or down & \textbf{YES} --- it becomes $y=k$ \\
  $h$ (inside, with $x$) & shifts the graph right or left & no \\
  $a$ (out front) & stretches / compresses; $a<0$ flips over the $x$-axis & no \\
  $-x$ (inside, on $x$) & flips the graph over the $y$-axis & no \\
  \hline
  \end{tabular}
  \end{center}
  \vspace{8pt}
  \begin{center}
    {\footnotesize $b^{\,x-3}$ moves \textbf{right} $3$ --- the same sign flip as $(x-3)^{2}$ and
    $\sqrt{x-3}$. Nothing new there.}
  \end{center}
\end{frame}

% K MOVES THE ASYMPTOTE
\begin{frame}[t, plain]
  \forestheader{$k$ moves the asymptote --- so $k$ moves the range}
  \sectionlabel[goldacc]{The one that counts}
  \begin{columns}[T]
    \begin{column}{0.38\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.30]
        \lgrid{-3}{4}{-6}{6}
        \begin{scope}\clip (-3,-6) rectangle (4,6); \expgraph{1}{0.693147}{0}{-4}{-3}{4} \end{scope}
        \hasym{-3}{4}{-4}
        \node[navy, font=\tiny, right] at (1.7,-4.7) {$y=-4$};
        \fill[charcoal] (0,-3) circle (4pt) node[left, font=\tiny]{$(0,-3)$};
        \fill[charcoal] (2,0) circle (4pt) node[above left, font=\tiny]{$(2,0)$};
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.58\textwidth}
      \footnotesize
      \textbf{$y=2^{x}-4$.} The whole picture slid down $4$ --- including the dashed line.
      \begin{itemize}\setlength{\itemsep}{3pt}
        \item Asymptote: $y=0 \;\longrightarrow\; y=-4$
        \item Range: $(0,\infty) \;\longrightarrow\; (-4,\infty)$
        \item $y$-intercept: $2^{0}-4=-3$
        \item $x$-intercept: $(2,0)$, since $2^{2}-4=0$
      \end{itemize}
      \vspace{3pt}
      \begin{block}{Two rules, stated precisely}
        Range is $(k,\infty)$ if $a>0$, $(-\infty,k)$ if $a<0$ --- \textbf{parenthesis} at $k$, never a
        bracket. \\[3pt]
        The graph has an $x$-intercept exactly when $a$ and $k$ have \textbf{opposite signs}.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% NO ZEROS EVER, REVISITED
\begin{frame}[t, plain]
  \forestheader{``No zeros, ever'' --- refined, not contradicted}
  \sectionlabel{Watch out}
  \begin{center}
    Lesson 6.0's claim was about $y=ab^{x}$, where $k=0$. \\[4pt]
    With $k=0$ the graph lives entirely on one side of the $x$-axis --- so of course it never crosses.
  \end{center}
  \vspace{8pt}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{No $x$-intercept}
        $y=2^{x}+4$ \quad ($a>0$, $k>0$) \\[3pt]
        $y=-3^{x}$ \quad ($k=0$) \\[3pt]
        $y=5(3)^{x}$ \quad ($k=0$)
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Has an $x$-intercept}
        $y=2^{x}-4$ \quad ($a>0$, $k<0$) \\[3pt]
        $y=-2^{x}+4$ \quad ($a<0$, $k>0$) \\[3pt]
        $y=\left(\frac12\right)^{x}-2$ \quad ($a>0$, $k<0$)
      \end{block}
    \end{column}
  \end{columns}
  \vspace{8pt}
  \begin{center}
    {\footnotesize A vertical shift is the only move that can push the asymptote across the $x$-axis.
    Nothing else can create a zero.}
  \end{center}
\end{frame}

% TWO MINUS SIGNS
\begin{frame}[t, plain]
  \forestheader{Two minus signs, two completely different flips}
  \sectionlabel[goldacc]{Reflections}
  \begin{columns}[T]
    \begin{column}{0.30\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.27]
        \lgrid{-3}{3}{-8}{3}
        \begin{scope}\clip (-3,-8) rectangle (3,3); \expgraph{-1}{0.693147}{0}{0}{-3}{3} \end{scope}
        \hasym{-3}{3}{0}
        \fill[charcoal] (0,-1) circle (4pt);
      \end{tikzpicture}\\[2pt]
      {\footnotesize $y=-2^{x}$}
    \end{column}
    \begin{column}{0.30\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.27]
        \lgrid{-3}{3}{-2}{9}
        \begin{scope}\clip (-3,-2) rectangle (3,9); \expgraph{1}{-0.693147}{0}{0}{-3}{3} \end{scope}
        \hasym{-3}{3}{0}
        \fill[charcoal] (0,1) circle (4pt);
      \end{tikzpicture}\\[2pt]
      {\footnotesize $y=2^{-x}$}
    \end{column}
    \begin{column}{0.38\textwidth}
      \footnotesize
      \textbf{$-2^{x}$: negative on the \emph{output}.}
      Flips over the $x$-axis. Range $(-\infty,0)$. Base is still $2$, so it is \textbf{still growth} ---
      the graph falls only because $a<0$. \\[6pt]
      \textbf{$2^{-x}$: negative on the \emph{input}.}
      Flips over the $y$-axis. Range still $(0,\infty)$. And
      $2^{-x}=\frac{1}{2^{x}}=\left(\frac12\right)^{x}$ --- so this one \textbf{is decay}. \\[6pt]
      Neither one moved the asymptote. Neither one is a vertical shift.
    \end{column}
  \end{columns}
\end{frame}

% GROWTH VS DECAY REMINDER
\begin{frame}[t, plain]
  \forestheader{Growth or decay is read off $b$ --- never off $a$}
  \sectionlabel{Compare \& contrast}
  \begin{center}
  \footnotesize
  \renewcommand{\arraystretch}{1.4}
  \begin{tabular}{|l|c|c|c|c|}
  \hline
  \textbf{Function} & \textbf{$b$} & \textbf{Family} & \textbf{Graph rises or falls} & \textbf{Range} \\
  \hline
  $y=2^{x}$ & $2$ & growth & rises & $(0,\infty)$ \\
  $y=-2^{x}$ & $2$ & \textbf{growth} & \textbf{falls} & $(-\infty,0)$ \\
  $y=\left(\frac12\right)^{x}$ & $\frac12$ & decay & falls & $(0,\infty)$ \\
  $y=-\left(\frac12\right)^{x}$ & $\frac12$ & \textbf{decay} & \textbf{rises} & $(-\infty,0)$ \\
  \hline
  \end{tabular}
  \end{center}
  \vspace{8pt}
  \begin{center}
    \Large $b$ picks the \textbf{family}. \quad $a$ picks the \textbf{side of the asymptote}.
  \end{center}
  \vspace{6pt}
  \begin{center}
    {\footnotesize Rows 2 and 4 are the ones students misname. ``It goes down, so it's decay'' is the
    error --- and it is exactly Lesson 6.0's warning, one lesson later.}
  \end{center}
\end{frame}

% READING BACKWARDS
\begin{frame}[t, plain]
  \forestheader{Backwards: from a graph to an equation, in this order}
  \sectionlabel[goldacc]{Graph $\to$ equation}
  \begin{columns}[T]
    \begin{column}{0.36\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.28]
        \lgrid{-3}{4}{-1}{12}
        \begin{scope}\clip (-3,-1) rectangle (4,12); \expgraph{3}{0.693147}{0}{2}{-3}{4} \end{scope}
        \hasym{-3}{4}{2}
        \node[navy, font=\tiny, left] at (-1.2,2.8) {$y=2$};
        \fill[charcoal] (0,5) circle (4pt) node[left, font=\tiny]{$(0,5)$};
        \fill[charcoal] (1,8) circle (4pt) node[right, font=\tiny]{$(1,8)$};
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.60\textwidth}
      \footnotesize
      Form: $y=a\,b^{x}+k$.
      \begin{enumerate}\setlength{\itemsep}{4pt}
        \item \textbf{Dashed line gives $k$:} \; $k=2$.
        \item \textbf{$y$-intercept is $a+k$:} \; $a+2=5$, so $a=3$.
        \item \textbf{One more point gives $b$:} \; $8=3b+2$, so $b=2$.
      \end{enumerate}
      \vspace{3pt}
      \[ y = 3(2)^{x}+2 \]
      Check at a point you did not use: $x=2$ gives $3(4)+2=14$. \\[4pt]
      \begin{block}{The error this order prevents}
        Skip step 1 and you write $a=5$. The $y$-intercept is $a+k$, not $a$ --- they agree only when
        $k=0$.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% ASYMPTOTE AS A REAL THING
\begin{frame}[t, plain]
  \forestheader{When $k$ is a physical fact: cooling coffee}
  \sectionlabel{Modeling}
  \begin{center}
    Coffee poured at $200^{\circ}$F into a $68^{\circ}$ room: \quad
    $T(t) = 68 + 132(0.85)^{t}$
  \end{center}
  \vspace{4pt}
  \begin{center}
  \footnotesize
  \renewcommand{\arraystretch}{1.35}
  \begin{tabular}{|l|c|c|c|c|c|c|}
  \hline
  $t$ (min) & $0$ & $4$ & $8$ & $12$ & $16$ & $20$ \\ \hline
  $T(t)$ ($^{\circ}$F) & $200$ & $136.9$ & $104.0$ & $86.8$ & $77.8$ & $73.1$ \\ \hline
  \end{tabular}
  \end{center}
  \vspace{6pt}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \footnotesize
      \begin{itemize}\setlength{\itemsep}{3pt}
        \item $k=68$ is \textbf{room temperature} --- the asymptote is a real thing.
        \item $a=132$ is the \emph{gap} the coffee starts with: $200-68$.
        \item Range $(68,\infty)$ algebraically; $(68,200]$ in the story.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{When does it hit $68^{\circ}$?}
        Never. That is what an asymptote \emph{means}. \\[3pt]
        {\scriptsize And ``when is it $100^{\circ}$?'' needs $132(0.85)^{t}=32$ --- no common base.
        Between $8$ and $12$ minutes is the best we can do today.}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

  \sectionlabel{Connections}
  \textbf{Today:} $g(x)=a\,b^{\,x-h}+k$ --- $k$ moves the asymptote to $y=k$ and takes the range with
  it; $h$ and $a$ do not; $-b^{x}$ flips over the $x$-axis while $b^{-x}$ flips over the $y$-axis and
  becomes decay; and a graph reads back to an equation as $k$, then $a$, then $b$.\\[8pt]
  \begin{itemize}\setlength{\itemsep}{3pt}
    \item \textbf{6.3} Solving with a common base --- today you \emph{read} the $x$-intercept of $y=2^{x}-4$ off the graph; next you solve $2^{x}=4$ for it.
    \item \textbf{6.4} Compound interest, half-life, and $e$ --- ending on an equation we cannot yet solve.
    \item \textbf{6.5} Exponential regression --- technology hands you $a$ and $b$; reading a graph backwards is how you judge them.
  \end{itemize}
