\begin{notesbox}{Practice: Distance, Cases \& Solution Sets}
Eight problems. Isolate the absolute value \emph{before} you split, and write every inequality
solution in set, interval, \emph{and} number-line form.

\begin{enumerate}[leftmargin=1.9em, itemsep=10pt]
  \item \textbf{Solve (already isolated).}
    \begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=4pt]
      \item $|x| = 10$: \quad $x = $ \ans{$10$ or $-10$}
      \item $|x - 5| = 2$: \quad $x = $ \ans{$7$ or $3$}
    \end{enumerate}

  \item \textbf{Isolate, then solve.}
    \begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=4pt]
      \item $|4x + 2| = 10$:
        \begin{work}
          |4x + 2| &= 10 \\
            4x + 2 &= 10 \quad\text{or}\quad 4x + 2 = -10 \\
                4x &= 8 \quad\text{or}\quad 4x = -12 \\
                 x &= 2 \quad\text{or}\quad x = -3
        \end{work}
        $x = $ \ans{$2$ or $-3$}
      \item $2|x - 4| - 1 = 9$:
        \begin{work}
          2|x - 4| - 1 &= 9 \\
              2|x - 4| &= 10 \\
               |x - 4| &= 5 \\
                 x - 4 &= 5 \quad\text{or}\quad x - 4 = -5 \\
                     x &= 9 \quad\text{or}\quad x = -1
        \end{work}
        $x = $ \ans{$9$ or $-1$}
    \end{enumerate}

  \item \textbf{How many solutions?} Write \emph{none}, \emph{one}, or \emph{two}, and give any
        solution(s). Each one is a statement about a \emph{distance}.
    \begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=4pt]
      \item $|x + 7| = -2$: \ans{none --- a distance is never $-2$}
      \item $|x - 3| = 0$: \ans{one: $x = 3$}
      \item $|x| = 5$: \ans{two: $x = 5$ or $x = -5$}
    \end{enumerate}

  \item \textbf{Solve the ``and'' inequality} $|x + 2| < 6$.
        \begin{work}
              |x + 2| &< 6 \\
          -6 < x + 2 &< 6 \\
              -8 < x &< 4
        \end{work}
        Set: \ans{$\{x \mid -8<x<4\}$} \qquad Interval: \ans{$(-8,4)$}
        \begin{center}
        \begin{tikzpicture}[x=0.44cm, y=0.5cm]
          \numline{-10}{6}
          \draw[very thick, keyred] (-8,0)--(4,0);
          \draw[keyred, thick, fill=white] (-8,0) circle (2.6pt);
          \draw[keyred, thick, fill=white] (4,0) circle (2.6pt);
        \end{tikzpicture}
        \end{center}

  \item \textbf{Solve the ``or'' inequality} $|2x - 1| \ge 7$.
        \begin{work}
          |2x - 1| &\ge 7 \\
            2x - 1 &\le -7 \quad\text{or}\quad 2x - 1 \ge 7 \\
                2x &\le -6 \quad\text{or}\quad 2x \ge 8 \\
                 x &\le -3 \quad\text{or}\quad x \ge 4
        \end{work}
        Set: \ans{$\{x \mid x\le-3 \text{ or } x\ge4\}$} \qquad Interval: \ans{$(-\infty,-3]\cup[4,\infty)$}
        \begin{center}
        \begin{tikzpicture}[x=0.44cm, y=0.5cm]
          \numline{-6}{7}
          \draw[very thick, keyred] (-6.5,0)--(-3,0);
          \draw[very thick, keyred] (4,0)--(7.5,0);
          \fill[keyred] (-3,0) circle (2.6pt);
          \fill[keyred] (4,0) circle (2.6pt);
        \end{tikzpicture}
        \end{center}

  \item \textbf{Model it.} A thermostat holds a room within $2^\circ$F of the $68^\circ$F setting.
    \begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=4pt]
      \item Write an absolute-value inequality for an acceptable temperature $T$. \ans{$|T - 68| \le 2$}
      \item Solve it and state the coolest and warmest acceptable temperatures.
        \begin{work}
                |T - 68| &\le 2 \\
          -2 \le T - 68 &\le 2 \\
              66 \le T &\le 70
        \end{work}
        Coolest \ans{$66^\circ$F} \quad Warmest \ans{$70^\circ$F}
      \item Is $66^\circ$F acceptable? \ans{yes} \; Which word in the sentence tells you?
            \ans{``within'' ($\le$)}
    \end{enumerate}

  \item \textbf{Explain.}
    \begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=4pt]
      \item How do you decide whether an absolute-value inequality becomes an ``and'' (one
            interval) or an ``or'' (two rays)? \ansline{The direction of the symbol. $<$ or $\le$ says the expression is \emph{closer} than $c$ to zero, which traps it between $-c$ and $c$ --- one interval. $>$ or $\ge$ says it is \emph{farther} than $c$, which throws it outside $[-c,c]$ --- two rays.}
      \item For what values of $k$ does $|x - 3| < k$ have \emph{no} solution? Justify from the
            meaning of absolute value. \ansline{For $k \le 0$. A distance is never negative, so no $x$ can be less than $0$ away from $3$; and $|x-3|<0$ is impossible for the same reason. (At $k=0$ it is still impossible, since nothing is \emph{strictly} less than $0$ away.)}
    \end{enumerate}

  \item \textbf{Multiple choice.} The solution set of $|x - 3| \le 1$ is
    \begin{enumerate}[label=\Alph*., leftmargin=2.2em, itemsep=1pt]
      \item $[2, 4]$ \textcolor{keyred}{\textbf{$\leftarrow$ correct}}
      \item $(-\infty, 2] \cup [4, \infty)$
      \item $[-4, -2]$
      \item no solution
    \end{enumerate}
    Explain why each of the \emph{other} three is false. \ansline{B is the ``or'' answer --- it solves $|x-3|\ge1$, the \emph{outside} case, but $\le$ means \emph{within} $1$ of $3$. C negates the endpoints; the distance is measured from $+3$, not $-3$. D is wrong because $c=1>0$, so there are solutions --- in fact a whole interval of them.}
\end{enumerate}
\end{notesbox}
