    % ── Linear equations ──────────────────────────────────────────────────────
    \item Solve $2(3x - 1) = 4x + 8$.

    \ansline{$6x - 2 = 4x + 8$}
    \ansline{$2x = 10$}
    Answer: \ans{$x = 5$}

    \vspace{0.2in}

    \item Solve $5 - x = 2x - 7$.

    \ansline{$12 = 3x$}
    Answer: \ans{$x = 4$}

    \vspace{0.2in}

    % ── Inequality ────────────────────────────────────────────────────────────
    \item Solve $-4x + 3 > 15$; interval notation.

    \ansline{$-4x > 12$}
    \ansline{Divide by $-4$; flip: $x < -3$}
    Answer: \ans{$(-\infty, -3)$}

    \vspace{0.2in}

    % ── Literal equations ─────────────────────────────────────────────────────
    \item Solve for $w$: $V = lwh$

    \ansline{Divide both sides by $lh$}
    Answer: \ans{$w = \dfrac{V}{lh}$}

    \vspace{0.2in}

    \item Solve for $y$: $3x - 2y = 12$

    \ansline{$-2y = 12 - 3x$}
    Answer: \ans{$y = \dfrac{3x - 12}{2}$}

    \columnbreak

    % ── System ────────────────────────────────────────────────────────────────
    \item Solve the system: $x - y = 1$, $\;2x + y = 8$

    \ansline{Add equations: $3x = 9 \Rightarrow x = 3$}
    \ansline{Sub back: $3 - y = 1 \Rightarrow y = 2$}
    Answer: \ans{$(3,\, 2)$}

    \vspace{0.2in}

    % ── Quadratics ────────────────────────────────────────────────────────────
    \item Solve $x^2 - 7x + 12 = 0$ by factoring.

    \ansline{Need: add to $-7$, multiply to $12$ $\Rightarrow$ $-3$ and $-4$}
    \ansline{$(x - 3)(x - 4) = 0$}
    Answer: \ans{$x = 3$ or $x = 4$}

    \vspace{0.2in}

    \item Solve $x^2 = 5x - 6$.

    \ansline{$x^2 - 5x + 6 = 0$}
    \ansline{$(x - 2)(x - 3) = 0$}
    Answer: \ans{$x = 2$ or $x = 3$}

    \vspace{0.2in}

    % ── Discriminant ──────────────────────────────────────────────────────────
    \item $2x^2 + x + 3 = 0$; number of real solutions?

    \ansline{$\Delta = 1^2 - 4(2)(3) = 1 - 24 = -23 < 0$}
    Answer: \ans{$0$ real solutions}

    \vspace{0.2in}

    \item Solve $x^2 - 6x + 9 = 0$.

    \ansline{$(x - 3)^2 = 0 \Rightarrow x - 3 = 0$}
    \ansline{$\Delta = 36 - 36 = 0$; one solution (double root)}
    Answer: \ans{$x = 3$ (one solution)}

\end{enumerate}
