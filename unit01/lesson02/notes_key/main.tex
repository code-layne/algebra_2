\begin{objectivebox}
\textbf{By the end of this lesson, I will be able to\dots}
\begin{itemize}\setlength{\itemsep}{2pt}
  \item read \emph{absolute value} as distance from zero and solve $|ax+b|=c$ with the two-case rule;

  \item solve absolute value \emph{inequalities} (``and'' for $<$, ``or'' for $>$);
  \item write a solution set in \emph{set}, \emph{interval}, and \emph{number-line} form, and model a tolerance situation.
\end{itemize}

\begin{vocabbox}
\small
Fill in each term as we build it together.
\vspace{2pt}
\vocabans{Absolute value (distance from zero)}{$|x|$ is how far $x$ is from $0$; it is never negative}
\vocabans{Two-case rule for $|X|=c$}{if $c\ge 0$, then $X=c$ or $X=-c$; if $c<0$, there is no solution}
\vocabans{``And'' case: $|X|<c$}{$-c<X<c$ --- the expression is trapped between $-c$ and $c$ (one interval)}
\vocabans{``Or'' case: $|X|>c$}{$X<-c$ or $X>c$ --- the expression falls outside $[-c,c]$ (two rays)}
\vocabans{Set / interval notation}{$\{x \mid -2\le x\le 3\}$ vs.\ $[-2,3]$; brackets include an endpoint, parentheses exclude it}
\end{vocabbox}

\begin{hookbox}
A cereal box is labeled $500$ g. Quality control rejects any box that is \emph{more than $5$ g off}
the label.
\begin{itemize}\setlength{\itemsep}{1pt}
  \item Lightest box that still passes? \ans{$495$ g} \qquad Heaviest? \ans{$505$ g}
  \item Write ``the weight $w$ is off from $500$ by \emph{at most} $5$'' with one symbol:
        $\big|\,{\color{keyred}\mathbf{w-500}}\,\big| \le {\color{keyred}\mathbf{5}}$
\end{itemize}
An absolute value is a \emph{distance}. Every problem today asks how far an expression is from a
target: ``exactly this far'' (equation) or ``within / beyond this far'' (inequality).
\end{hookbox}

\boxguard
\begin{notesbox}{1. Absolute Value Is a Distance}
$|x|$ is the distance of $x$ from $0$, so it is \textbf{never negative}. The question $|x|=5$ asks
\emph{which numbers are exactly $5$ units from zero} --- and there are \ans{two} of them:
\[ x = {\color{keyred}\mathbf{5}} \quad\text{or}\quad x = {\color{keyred}\mathbf{-5}}. \]
\begin{center}
\begin{tikzpicture}[scale=0.5]
  \numline{-6}{6}
  \fill[forest] (-5,0) circle (3.5pt); \fill[forest] (5,0) circle (3.5pt);
  \draw[<->, forest!70, thick] (0,0.6) to[bend left=18] node[above, font=\scriptsize]{$5$} (5,0.6);
  \draw[<->, forest!70, thick] (-5,0.6) to[bend left=18] node[above, font=\scriptsize]{$5$} (0,0.6);
\end{tikzpicture}
\end{center}
\textbf{Big idea:} a single distance corresponds to \emph{two} inputs --- one on each side of zero.
That is the whole engine of this lesson.
\end{notesbox}

\begin{notesbox}{2. Solving $|ax+b| = c$ --- The Two-Case Rule}
\textbf{Step 1: isolate} the absolute value. \textbf{Step 2:} if $c \ge 0$, split into two cases:
\[ ax+b = c \qquad\text{or}\qquad ax+b = -c. \]
\textbf{Try it:} $|2x - 3| = 7$.
\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabularx}{0.92\linewidth}{@{}X | X@{}}
\textbf{Case 1 (positive):} & \textbf{Case 2 (negative):} \\
$2x - 3 = 7$ & $2x - 3 = -7$ \\
$x = {\color{keyred}\mathbf{5}}$ & $x = {\color{keyred}\mathbf{-2}}$ \\
\end{tabularx}
\end{center}
\textbf{Verify} both in the original: $|2({\color{keyred}\mathbf{5}})-3| = 7$ \checkmark \; and
$|2({\color{keyred}\mathbf{-2}})-3| = 7$ \checkmark. Graphically, these are the two $x$-values where
$y=|2x-3|$ meets the line $y=7$.
\end{notesbox}

\begin{notesbox}{3. Isolate First --- and Two Special Cases}
You \emph{cannot} split until the bars stand alone. Solve $3|x+1| - 2 = 10$:
\begin{work}
  3|x+1| - 2 &= 10 \\
      3|x+1| &= 12 \\
       |x+1| &= 4 \\
         x+1 &= 4 \quad\text{or}\quad x+1 = -4 \\
           x &= 3 \quad\text{or}\quad x = -5
\end{work}
\textbf{Special cases} (explain each from the distance meaning):
\begin{itemize}[leftmargin=1.5em, itemsep=2pt]
  \item $|x - 4| = -3$ has \ans{no solution}, because a distance can never be \ans{negative}.
  \item $|x - 4| = 0$ has exactly \ans{one} solution: $x = {\color{keyred}\mathbf{4}}$ (distance $0$ means the expression \emph{is} zero).
\end{itemize}
\end{notesbox}

\boxguard
\begin{notesbox}{4. Inequalities --- The ``And'' Case ($<$, $\le$)}
$|x| < 3$ asks \emph{which numbers are within $3$ of zero} --- everything \textbf{between} $-3$ and
$3$. \textbf{``Less th\emph{AND}''}: $|X| < c \iff -c < X < c$ (one connected interval).

\textbf{Solve} $|2x - 1| \le 5$:
\begin{work}
        |2x - 1| &\le 5 \\
  -5 \le 2x - 1 &\le 5 \\
      -4 \le 2x &\le 6 \\
       -2 \le x &\le 3
\end{work}
Write the solution three ways:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabularx}{0.95\linewidth}{@{}l l X@{}}
\textbf{Set notation:} & $\{\,x \mid {\color{keyred}\mathbf{-2 \le x \le 3}}\,\}$ &
\textbf{Interval:} ${\color{keyred}\mathbf{[-2,3]}}$ \\
\end{tabularx}
\end{center}
\begin{center}
\begin{tikzpicture}[scale=0.5]
  \numline{-4}{5}
  \draw[very thick, forest] (-2,0)--(3,0);
  \fill[forest] (-2,0) circle (3.5pt); \fill[forest] (3,0) circle (3.5pt);
\end{tikzpicture}
\end{center}
{\footnotesize Shade \emph{between} the two bounds; use \textbf{closed} dots because of ``$\le$''.}
\end{notesbox}

\begin{notesbox}{5. Inequalities --- The ``Or'' Case ($>$, $\ge$)}
$|x| > 3$ asks \emph{which numbers are more than $3$ from zero} --- everything \textbf{outside}
$[-3,3]$. \textbf{``Great\emph{OR}''}: $|X| > c \iff X < -c$ \textbf{or} $X > c$ (two rays).

\textbf{Solve} $|x + 2| \ge 4$:
\begin{work}
  |x + 2| &\ge 4 \\
    x + 2 &\le -4 \quad\text{or}\quad x + 2 \ge 4 \\
        x &\le -6 \quad\text{or}\quad x \ge 2
\end{work}
Write the solution three ways:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabularx}{0.95\linewidth}{@{}l l X@{}}
\textbf{Set notation:} & $\{\,x \mid {\color{keyred}\mathbf{x \le -6 \ \text{or}\ x \ge 2}}\,\}$ &
\textbf{Interval:} ${\color{keyred}\mathbf{(-\infty,-6]\cup[2,\infty)}}$ \\
\end{tabularx}
\end{center}
\begin{center}
\begin{tikzpicture}[scale=0.5]
  \numline{-8}{4}
  \draw[very thick, forest, <-] (-8.4,0)--(-6,0);
  \draw[very thick, forest, ->] (2,0)--(4.4,0);
  \fill[forest] (-6,0) circle (3.5pt); \fill[forest] (2,0) circle (3.5pt);
\end{tikzpicture}
\end{center}
{\footnotesize Shade the two \emph{outside} rays; the interval uses $\cup$ (``union'').}
\end{notesbox}

\begin{practicebox}
Solve $|3x - 6| > 9$ and write the solution set in all three forms.
\begin{enumerate}[leftmargin=1.6em, itemsep=3pt]
  \item Split (``or''): $3x - 6 < {\color{keyred}\mathbf{-9}}$ or $3x - 6 > {\color{keyred}\mathbf{9}}$.
  \item Solve: $x < {\color{keyred}\mathbf{-1}}$ or $x > {\color{keyred}\mathbf{5}}$.
  \item Set: $\{\,x \mid {\color{keyred}\mathbf{x < -1 \ \text{or}\ x > 5}}\,\}$; \quad Interval: ${\color{keyred}\mathbf{(-\infty,-1)\cup(5,\infty)}}$.
\end{enumerate}
\end{practicebox}
