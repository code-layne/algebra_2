% ── I DO: section 1 ──────────────────────────────────────────────────────────
\boxguard[14]
\begin{notesbox}{1. Absolute Value Is a Distance --- So It Answers Twice}
\small
\begin{center}
\begin{minipage}[c]{0.50\linewidth}
\[ |x-h| \;=\; \text{the distance between } x \text{ and } h \]
A hiker at mile $x$ is \blank{1.8cm} miles from the tower. A distance is never \blank{1.7cm},
and \emph{two} different spots sit the same distance out --- one on each side.
\end{minipage}\hfill
\begin{minipage}[c]{0.46\linewidth}\centering
\begin{tikzpicture}[x=0.30cm, y=0.42cm]
  \trail
  \draw[very thick, redacc] (7,0.55)--(12,0.55);
  \draw[very thick, redacc] (12,0.55)--(17,0.55);
  \fill[redacc] (7,0) circle (2.8pt);
  \fill[redacc] (17,0) circle (2.8pt);
  \node[redacc, font=\tiny, above] at (9.5,0.6){$5$};
  \node[redacc, font=\tiny, above] at (14.5,0.6){$5$};
\end{tikzpicture}
\end{minipage}
\end{center}
\textbf{Exactly five miles out.} ``Distance from the tower is $5$'' is $|x-12|=5$. The bars
hide \emph{two} equations --- one for each side of the tower:
\begin{work}
  |x - 12| &= 5 \\
  x - 12 &= 5 \quad\text{or}\quad x - 12 = -5 \\
  x &= 17 \quad\text{or}\quad x = 7
\end{work}
Both markers from the hook, straight out of the algebra. \textbf{Verify} by putting each back
in: $|17-12| =$ \blank{0.9cm} and $|7-12| =$ \blank{0.9cm}. That is the \textbf{two-case rule}:
$|X| = c$ splits into $X = c$ \emph{or} $X = -c$.

\vspace{8pt}
\textbf{\textcolor{forest}{Isolate the Bars \emph{Before} You Split.}}
The two-case rule is a statement about $|X|$ \emph{alone}. Anything outside the bars gets undone
first --- treat the bars exactly as you would a variable.

\vspace{3pt}
\renewcommand{\arraystretch}{1.4}
\begin{tabularx}{\linewidth}{@{}l l X@{}}
\rowcolor{forestbg}
\textbf{$|X| = c$ when\dots} & \textbf{Split into} & \textbf{How many solutions --- and why} \\
$c > 0$ & $X =$ \blank{0.9cm} \; or \; $X =$ \blank{0.9cm} & \blank{1.0cm} --- two spots sit that far out \\
$c = 0$ & $X =$ \blank{0.9cm} & \blank{1.0cm} --- only one spot is \emph{zero} away \\
$c < 0$ & --- & \blank{1.4cm} --- a distance is never negative \\
\end{tabularx}

\vspace{4pt}
\textbf{Undo, then split.} Solve $3|x+1| - 2 = 10$.
\begin{work}
  3|x+1| - 2 &= 10 \\
  3|x+1| &= 12 \\
  |x+1| &= 4 \\
  x + 1 &= 4 \quad\text{or}\quad x + 1 = -4 \\
  x &= 3 \quad\text{or}\quad x = -5
\end{work}
Splitting at the \emph{first} line would give $3x+1-2=10$ --- a single answer, and a
\blank{1.4cm} one.
\end{notesbox}

\boxguard[34]
\begin{notesbox}{2. One Piece or Two --- and Three Ways to Write It}
\small
Same expression, same $5$, one symbol apart --- and the two answers look nothing alike.

\vspace{3pt}
\renewcommand{\arraystretch}{1.35}
\begin{tabularx}{\linewidth}{@{}X|X@{}}
\rowcolor{forestbg}
\textbf{In range: $|x-12| \le 5$} & \textbf{Out of range: $|x-12| \ge 5$} \\
\emph{Closer} to the tower than $5$: \textbf{between}. &
\emph{Farther} from the tower than $5$: \textbf{outside}. \\
One compound inequality, joined by \blank{1.0cm}: &
Two separate statements, joined by \blank{1.0cm}: \\
$-5 \le x-12 \le 5$, so \; \blank{1.1cm} $\le x \le$ \blank{1.1cm} &
$x-12 \le -5$ \; or \; $x-12 \ge 5$, so \; $x \le$ \blank{1.0cm} \, or \, $x \ge$ \blank{1.0cm} \\
\centering\arraybackslash
\begin{tikzpicture}[x=0.20cm, y=0.40cm, baseline=-2pt]
  \trail
  \draw[very thick, greenacc] (7,0)--(17,0);
  \fill[greenacc] (7,0) circle (3.6pt);
  \fill[greenacc] (17,0) circle (3.6pt);
\end{tikzpicture}
&
\centering\arraybackslash
\begin{tikzpicture}[x=0.20cm, y=0.40cm, baseline=-2pt]
  \trail
  \draw[very thick, redacc] (0,0)--(7,0);
  \draw[very thick, redacc] (17,0)--(20,0);
  \fill[redacc] (7,0) circle (3.6pt);
  \fill[redacc] (17,0) circle (3.6pt);
\end{tikzpicture}
\\
\blank{1.0cm} connected piece & \blank{1.0cm} separate pieces \\
Set: $\{x \mid$ \blank{2.4cm} $\}$ & Set: $\{x \mid$ \blank{2.4cm} $\}$ \\
Interval: \blank{2.0cm} & Interval: $(-\infty,$ \blank{0.8cm} $] \cup [$ \blank{0.8cm} $, \infty)$ \\
\end{tabularx}

\vspace{4pt}
\textbf{The endpoint carries the symbol.} A bracket \textbf{[ ]} and a \blank{1.3cm} dot mean
the endpoint is \emph{in} ($\le$, $\ge$); a parenthesis \textbf{( )} and an \blank{1.3cm} dot
mean it is \emph{out} ($<$, $>$). Next to $\infty$ you always write a \blank{2.0cm}, because
$\infty$ is not a number you can reach. Two rays need two intervals and a $\cup$.

\vspace{5pt}
\textbf{Tolerance --- how far from a desired value are we willing to be?} ``Within 5 miles of
the tower'' names a \textbf{desired value}, \blank{1.2cm}, and a \textbf{tolerance}, \blank{1.2cm}.
Every ``within so much of'' sentence is the same inequality:
\[ |\,x - \text{desired value}\,| \;\le\; \text{tolerance}. \]
A tolerance is never negative, and it is always the ``and'' case --- you stay \emph{between}.

\vspace{8pt}
\textbf{\textcolor{forest}{The Crux: Test One Hiker, and Watch the Two Disagree.}}
Students who treat $\le$ and $\ge$ as ``nearly the same'' get the \emph{opposite} set. Prove it
with one hiker.

\vspace{3pt}
\begin{center}
\fcolorbox{goldacc}{goldbg}{\parbox{0.94\linewidth}{\small
A hiker at \textbf{mile 3} is $|3-12| =$ \blank{0.9cm} miles from the tower. Is $9 \le 5$?
\blank{0.9cm} \quad Is $9 \ge 5$? \blank{0.9cm} \quad So mile $3$ belongs to \emph{exactly one}
of the two answers above --- \blank{2.4cm}. Picking the wrong symbol does not shift the answer a
little; it hands you the opposite set.}}
\end{center}

\vspace{4pt}
\textbf{The odd cases, from the same picture.} $|x-12| < -2$ asks for a distance less than a
negative number: \blank{2.2cm}. $|x-12| \ge -2$ asks for a distance at least $-2$, which every
mile satisfies: \blank{2.6cm}. Neither needs algebra --- only the meaning of \emph{distance}.
\end{notesbox}
