    \begin{enumerate}[label=\Alph*.]
        \item $x > 3$ \quad $(-\infty, 3)$
        \item $x > 3$ \quad $(3, \infty)$
        \item $x < 3$ \quad $(-\infty, 3)$
        \item $x \geq 3$ \quad $[3, \infty)$
    \end{enumerate}

    \vspace{0.1in}

    \item Solve for $b$: \quad $A = \dfrac{1}{2}bh$

    \item How many solutions does the equation $4x + 2 = 4x - 3$ have?

        \item Two solutions
        \item No solution
        \item Infinitely many solutions
    \end{enumerate}
