\begin{tcolorbox}[colback=forestbg, colframe=forest, boxrule=0.9pt, arc=2mm,
    left=2mm, right=2mm, top=1.5mm, bottom=1.5mm]
    \textbf{Primary Objective:} Students can read an absolute value as a \emph{distance} --- $|x|$ as
    distance from zero and $|x-h|$ as the distance between $x$ and $h$ --- and use that meaning to
    solve absolute value \emph{equations} and \emph{inequalities} in one variable. They isolate the
    absolute value, split $|ax+b|=c$ into the \emph{two cases} it requires, verify each solution, and
    recognize the special cases (no solution when $c<0$, one when $c=0$). For inequalities they read
    $|ax+b|<c$ as a ``between'' (\emph{and}) statement and $|ax+b|>c$ as an ``outside'' (\emph{or})
    statement, represent the solution set three ways --- \emph{set notation, interval notation, and a
    number line} --- and model and interpret a \emph{tolerance} situation. Lesson~1.1 wrote and
    solved \emph{linear} expressions; today those expressions live \emph{inside} the bars, and
    Lesson~1.3 turns this algebra into the V-shaped graph.
    \\[2pt]
    \textbf{Standards (2023 VA SOL):} This lesson delivers the whole of \textbf{A2.EI.1} --- represent,
    solve, and interpret absolute value equations and inequalities in one variable:
    \textbf{A2.EI.1a} (create an absolute value \emph{equation} to model a context),
    \textbf{A2.EI.1b} (solve an absolute value equation algebraically and verify graphically),
    \textbf{A2.EI.1c} (create an absolute value \emph{inequality} to model a context),
    \textbf{A2.EI.1d} (solve an absolute value inequality and represent the solution set with set
    notation, interval notation, and a number line), and \textbf{A2.EI.1e} (verify solutions
    algebraically/graphically, and explain and interpret solutions in context).
    \\[2pt]
    \textbf{Lesson model:} \emph{Experience first, formalize later.} Students work the activity in
    small groups using only prior knowledge; the teacher circulates with questions, cues, and prompts
    (not answers); the debrief attaches the formal vocabulary to what the groups already found; the
    practice set applies it to new contexts. No separate direct-instruction block.
\end{tcolorbox}

\begin{skillbox}[Learning Targets \& Key Understandings]{goldbox}
\begin{tabularx}{\linewidth}{@{}X|X@{}}
\textbf{Learning targets (student language)} & \textbf{Key understandings (the ``why'')} \\[2pt]
\hline
\vspace{-6pt}
\begin{itemize}[leftmargin=1.1em, nosep, after=\vspace{2pt}]
  \item I can \textbf{isolate} an absolute value and solve $|ax+b|=c$ with the \textbf{two-case}
        rule, then \textbf{verify} both answers.
  \item I can say when such an equation has \textbf{no solution} or \textbf{exactly one}, and why.
  \item I can solve $|ax+b|<c$ and $|ax+b|>c$ and write each solution set in \textbf{set},
        \textbf{interval}, and \textbf{number-line} form.
  \item I can \textbf{model} a ``within \dots{} of'' situation and interpret the endpoints in context.
\end{itemize}
&
\vspace{-6pt}
\begin{itemize}[leftmargin=1.1em, nosep, after=\vspace{2pt}]
  \item $|x-h|$ is a \emph{distance}, so it is never negative --- that single fact explains every
        case in the lesson.
  \item $|ax+b|=c$ asks ``what inputs sit \emph{distance} $c$ away?'' --- two answers, one on each
        side of the center.
  \item \textbf{``Less th\emph{AND}, great\emph{OR}.''} $<$ traps the expression \emph{between} the
        bounds (one interval); $>$ throws it \emph{outside} them (two rays). The target
        misconception is defaulting every inequality to ``between.''
  \item You \emph{must} isolate the bars first --- $2|x|+1=9$ is not solved until it reads $|x|=4$.
  \item Set, interval, and number line describe the \emph{same} set; fluency in all three is the
        A2.EI.1d skill.
\end{itemize}
\end{tabularx}
\end{skillbox}

\begin{skillbox}[{Vocabulary, Concepts, \& Theorems}]{sky}
\renewcommand{\arraystretch}{1.3}
\begin{tabularx}{\linewidth}{@{}l X@{}}
\textbf{Absolute value} & $|x|$ is the \emph{distance} of $x$ from $0$; $|x-h|$ is the distance between $x$ and $h$. Always $\ge 0$. \\
\textbf{Absolute value equation} & An equation with a variable inside absolute-value bars, e.g.\ $|ax+b|=c$. \\
\textbf{Two-case rule} & If $|X|=c$ and $c>0$, then $X=c$ \emph{or} $X=-c$; if $c=0$, one solution; if $c<0$, none. \\
\textbf{Compound inequality} & Two inequalities joined by \emph{and} (both must hold) or \emph{or} (either may hold). \\
\textbf{``And'' ($<$) case} & $|X|<c \iff -c<X<c$: the expression is trapped \emph{between} $-c$ and $c$. \\
\textbf{``Or'' ($>$) case} & $|X|>c \iff X<-c$ \emph{or} $X>c$: the expression falls \emph{outside} $[-c,c]$. \\
\textbf{Set / interval notation} & $\{x \mid -3<x<5\}$ vs.\ $(-3,5)$; brackets $[\,]$ include endpoints, parentheses $(\,)$ exclude. \\
\textbf{Tolerance} & An allowed distance from a target value, modeled by $|\text{measured}-\text{target}|\le \text{tol}$. \\
\end{tabularx}
\end{skillbox}

\begin{fixedskillbox}[Warm-Up --- Activate Prior Knowledge]{forestbg}
    The Warm-Up rehearses exactly what the activity leans on, each item planting a \emph{seed}
    without naming it: \textbf{(1)} evaluating absolute values, ending with ``how many numbers have
    an absolute value of $7$?'' --- the seed of the \emph{two-case rule}; \textbf{(2)} the distance
    between two numbers on a number line, ending at ``how far is $x$ from $6$?'' --- the seed of
    $|x-h|$, which the activity needs in item~1c; and \textbf{(3)} shading a number line with filled
    vs.\ open dots, closing with ``could a shaded picture ever come in \emph{two} separate pieces?''
    --- the seed of the ``or'' case. Debrief item~1 aloud, briefly: because $|x|$ measures distance,
    \emph{two} different numbers ($7$ and $-7$) share it. Do \emph{not} say ``two-case rule,''
    ``compound,'' or ``interval notation'' --- those words arrive in the debrief.
\end{fixedskillbox}

\boxguard
\begin{skillbox}[Experience \& Formalize --- The Activity: \emph{In Range}]{forestbg}
\textbf{Launch (2 min).} ``A straight stretch of parkway is marked in miles, $0$ to $20$. A ranger's
radio tower stands at \textbf{mile 12}, and a handheld radio reaches it only from \textbf{within 5
miles}.'' Groups of 3--4, one packet each but one shared conversation. Tell them: \emph{``Use what
you already know --- number lines, distance, solving. For every answer, be ready to show me where you
see it in the picture.''} All trail pictures are pre-drawn; students mark and shade them.

\begin{multicols}{2}
\textbf{What students do.} Part~1 builds the equation on one picture: mark the two hikers exactly
$5$ miles out ($7$ and $17$), say \emph{why} there are two, write the distance from mile $x$ to the
tower as $|x-12|$, turn ``exactly $5$ miles'' into $|x-12|=5$, and check both markers in it.
Part~2 turns the same picture into inequalities: shade where the radio \emph{can} reach and write it
as $7\le x\le 17$, restate it as $|x-12|\le 5$, then shade where it \emph{cannot} and get two
pieces. Item~2d is the \textbf{crux}: \emph{why} does ``can reach'' come out as one connected stretch
while ``cannot reach'' takes two? 2e verifies a single value ($|3-12|=9>5$), and 2f closes with
story-vs-math --- the parkway ends, so both out-of-range pieces get trimmed.

\columnbreak
\textbf{What the teacher does.} Circulate. Listen before speaking. When a group stalls, give a
\emph{question, cue, or prompt}, not an answer:
\begin{itemize}[leftmargin=1.1em, nosep]
  \item 1b: ``Put your finger on the tower. Which way can you walk from there? Does that give you
        one answer or two?''
  \item 1c: ``In the warm-up, how did you write the distance from $x$ to $6$? Same question, new
        landmark.''
  \item 2d: ``Point at the in-range stretch. Now point at everything else. Can you trace `everything
        else' without lifting your finger?''
  \item 2f: ``Where does the parkway stop? Is mile $-4$ a real answer to this question?''
\end{itemize}
While circulating, pick groups whose work goes on the board: a clean Part~1 marked picture, a group
with a good 2d explanation, and one good 2f answer.
\end{multicols}
\end{skillbox}

\boxguard
\begin{skillbox}[Debrief --- Formalize (what goes on the board, in a second color)]{forestbg}
Work through the displayed student work in order, and \textbf{attach each formal name to something a
group already wrote}. Students record each line in their QuickNotes as you go.
\begin{multicols}{2}
\begin{enumerate}[leftmargin=1.3em, nosep, itemsep=3pt]
  \item \textbf{Distance has a symbol.} On their picture: ``how far mile $x$ is from the tower''
        $\to$ \emph{$|x-12|$}; $|x|$ is distance from $0$, $|x-h|$ is distance between $x$ and $h$,
        and a distance is \emph{never negative}.
  \item \textbf{Two markers $\Rightarrow$ two cases.} ``$5$ down and $5$ up'' $\to$ the
        \emph{absolute value equation} $|x-12|=5$ and the \emph{two-case rule}: $X=c$ \emph{or}
        $X=-c$. Their substitution check in 1d is what \emph{verifying} a solution means (A2.EI.1b/e).
  \item \textbf{``Within'' is an inequality.} 2a--2b $\to$ $|x-12|\le 5$, and the shading $7\le
        x\le17$ is two conditions that must \emph{both} hold: a \emph{compound ``and''}. Write
        $-5\le x-12\le 5$ over their picture and solve it down to their own answer.
  \item \textbf{``Too far'' is the other one.} 2c/2d $\to$ $|x-12|>5$ gives $x<7$ \emph{or} $x>17$:
        a \emph{compound ``or''}, two \emph{rays}. Use the group's 2d wording verbatim --- you cannot
        trace it without lifting your finger, because the in-range piece sits between. Land the
        mnemonic: \textbf{less th\emph{AND}, great\emph{OR}}. Say it twice.
  \item \textbf{Isolate first} --- pose this one cold: $3|x+1|-2=10$ is not ready to split; get
        $|x+1|=4$ first. Then the two odd cases, also cold: $|x-4|=-3$ (none) and $|x-4|=0$ (one),
        both argued from distance.
  \item \textbf{Three notations, one set.} Their shading $\to$ $\{x\mid 7\le x\le 17\}$, $[7,17]$,
        and the number line. Closed dots/brackets for $\le$; open dots/parentheses for $<$; $\pm\infty$
        always a parenthesis. Their 2f trimming is the \emph{context} restricting the set.
\end{enumerate}
\columnbreak
\textbf{QuickNotes.} Fill the packet's QuickNotes box together, using $|x-3|$ with $c=2$ as the
compact picture: distance meaning, the two-case rule and its $c=0$/$c<0$ branches, isolate-first,
the ``and'' and ``or'' forms, and the three notations on $1\le x\le 5$. It is a summary of what the
groups discovered, not a new lecture.

\textbf{Why this order.} Every formal term lands on a picture the students already marked, so the
vocabulary is a \emph{label} rather than a definition to memorize. The special cases and
isolate-first go late and come from the teacher: they are the two things the activity deliberately
does \emph{not} carry (timebox), and both are easier to accept once ``a distance is never negative''
has already been said out loud three times.
\end{multicols}
\end{skillbox}

\boxguard
\begin{skillbox}[Application --- \emph{Roasting Tolerance} (worked together, 7 min)]{forestbg}
The one place the new vocabulary gets \emph{used} before students are on their own. A roaster fills
bags labeled $340$ g and rejects any bag more than $8$ g off. Students write $|w-340|\le 8$
(A2.EI.1c), solve it in the packet's \texttt{work} block to $332\le w\le 348$, then answer two
interpretation questions: is a $331$ g bag accepted (read it off the \emph{solved} range, do not
re-solve), and does tightening the rule to $5$ g make the acceptable range wider or narrower.
\textbf{Run it with the students holding the pen and you holding the questions:} ``What is the
target? What is the allowed distance from it? Which symbol says `no more than'?'' The last part is
the conceptual one --- a smaller tolerance is a \emph{narrower} interval around the same center,
which is the whole geometry of $|x-h|\le c$ in one sentence.
\end{skillbox}

\boxguard
\begin{skillbox}[Check Your Understanding --- the lesson's practice, \emph{not scored}]{redbox}
\textbf{There is no homework in this lesson}, so these six exercises are the entire practice set and
they carry the full A2.EI.1 spread: \textbf{(1)} an isolate-then-split equation ($3|x+1|-2=10$,
\texttt{work} block) with a verification; \textbf{(2)} an ``and'' inequality ($|2x-1|\le5$) in all
three notations; \textbf{(3)} an ``or'' inequality ($|3x-6|\ge9$) in all three notations --- the
deliberate pair against \#2; \textbf{(4)} the special cases the activity dropped, plus ``for which
$k$ does $|x-3|<k$ have no solution?''; \textbf{(5)} a thermostat tolerance model
($|T-68|\le2$) with the endpoint-interpretation follow-up; and \textbf{(6)} an SOL-style
multiple-choice item. Pairs for 1--3, then solo for 4--6. \textbf{They carry no point value} (the
cover's score column reads \textbf{NA}), so do not collect them for a grade. \textbf{Item~6 is the
formative check:} spot-check it on the way out and sort into ``got it / chose A (defaulted to
`between') / chose C (kept only one ray) / chose D (ignored the bound)'' to decide whether
Lesson~1.3 opens with a one-minute re-look at the out-of-range picture. Ten minutes will not cover
all six for every student --- that is fine; \#1--3 and \#6 are the priority, and \#4--5 are where
early finishers go.
\end{skillbox}

\boxguard
\begin{skillbox}[Watch For (while circulating)]{redbox}
\begin{itemize}[leftmargin=1.2em, nosep]
  \item giving only \emph{one} marker in 1a --- the seed of losing the negative case later;
  \item writing the distance in 1c as $x-12$ without bars, so mile $3$ comes out ``$-9$ miles away'';
  \item in 2c, shading only one side --- the target misconception, and exactly what 2d is for;
  \item mixing up ``and''/``or'' across Check Your Understanding 2, 3, and~6 (two rays for a $\le$,
        or one interval for a $\ge$);
  \item splitting before isolating in Check Your Understanding 1 --- insist on $|x+1|=4$ before any cases;
  \item calling $|x-4|=-3$ ``$x=1$ or $x=7$'' instead of \emph{no solution}, and answering ``$k<0$''
        on 4(c) --- at $k=0$ the inequality is still impossible, so the answer is $k\le0$;
  \item on \#5(c), reading ``within $2^\circ$'' as strict --- ``within'' includes the endpoints, so
        $66^\circ$ \emph{is} acceptable;
  \item open vs.\ closed dots and parentheses vs.\ brackets not matching $<$ vs.\ $\le$.
\end{itemize}
Cold-call prompts: ``Distance from what, and how far?'' ``Is this a \emph{between} or an
\emph{outside} --- how do you know from the symbol?'' ``Would $0$ be a solution? Check it.''
\end{skillbox}

\begin{skillbox}[Close \& Preview]{goldbox}
\textbf{Nothing is assigned.} This lesson has \textbf{no homework} --- the six unscored Check Your
Understanding exercises are the practice, they are done in class, and students keep the packet.
Close by naming what changed today rather than by assigning: ``an absolute value is a distance, and
that one fact decided every case we saw.'' Anything a student did not reach in \#4--5 is optional
review, not an assignment.
\textbf{Preview:} Lesson~1.3, \emph{Absolute Value Functions \& Transformations} --- the same
$|ax+b|$ expression becomes a \emph{graph}, and the two solutions found today are exactly where the
V-shape crosses the horizontal line $y=c$ (previewing vertex, axis of symmetry, and minimum).
\end{skillbox}

\begin{teachernote}[Warm-Up]
Item~1 plants the two-case idea: because $|x|$ is a \emph{distance}, two numbers share an absolute
value. Item~2 is the one the activity cashes in immediately --- if students cannot write ``how far
$x$ is from $6$'' as $|x-6|$, they will stall on 1c, so debrief that blank specifically. Item~3's
closing question (``could a shaded picture come in two separate pieces?'') is deliberately left
hanging; the answer is Part~2d. Five minutes; resist naming interval notation here even if a student
uses it.
\end{teachernote}

\begin{teachernote}[Experience \& Formalize]
\textbf{Activity (2pp).} Part~1 gets about 8 of the 20 minutes; Part~2 needs the other 12 because
2d is where the thinking happens. If a group finishes early, ask them to write the in-range
condition \emph{without} bars and \emph{with} bars and say which they would rather solve.
\textbf{Debrief + QuickNotes (13 min).} Do not re-teach from scratch --- every term should be
written \emph{on top of} a displayed student answer. The must-land moment is 2d: the out-of-range
set cannot be traced without lifting your finger, which is precisely why it needs an ``or.'' The
QuickNotes box is deliberately half a page; if time is short, fill the two-case rule and the and/or
forms together and let students finish it from the slides. \textbf{Application (7 min).} Worked
together --- see the Application box; it is where the coffee bag turns ``within $8$ g'' into
$|w-340|\le8$. \textbf{Check Your Understanding (10 min).} Six items and \textbf{no homework behind them}, so
this is the lesson's whole practice set --- expect most students to finish \#1--3 in class and to
use \#4--5 as the ``keep going'' bank (\#6 is the formative check, not optional). Items 1, 2, 3, and 5 carry \texttt{work} blocks: students
solve in the reserved space and the key prints the same steps there. \#3 exists to sit directly
against \#2 --- same three notations, opposite symbol --- so if you only debrief one, debrief that
pair side by side. \textbf{Not scored} --- spot-check item~6 rather than collecting the page.
\end{teachernote}
