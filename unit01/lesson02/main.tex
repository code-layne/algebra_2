                \item \textbf{Properties of real numbers, equality, \& inequality}
                \begin{itemize}
                    \item justification for steps in solving equations
                \end{itemize}
                \item \textbf{Linear equations in one variable}
                \begin{itemize}
                    \item multistep; distributive property
                    \item one solution, no solution, or infinite solutions
                \end{itemize}
                \item \textbf{Linear inequalities in one variable}
                \begin{itemize}
                    \item solve and graph on a number line
                    \item interval notation and set notation
                    \item flip the inequality when multiplying/dividing by a negative
                \end{itemize}
                \item \textbf{Literal equations}
                \begin{itemize}
                    \item rearrange a formula to solve for a specified variable
                    \item treat every other variable as a constant
                \end{itemize}
                \item \textbf{Systems of two linear equations}
                \begin{itemize}
                    \item substitution method
                    \item elimination (addition) method
                    \item graphical method; point of intersection
                    \item one solution, no solution, or infinite solutions
                \end{itemize}
            \end{enumerate}
        \end{minipage}
        \hfill
        \begin{minipage}[t]{0.48\linewidth}
            \begin{enumerate}[start=5]
                \item \textbf{Linear inequalities in two variables}
                \begin{itemize}
                    \item graph the boundary line (solid or dashed)
                    \item shade the solution region
                    \item systems of inequalities: double-shaded region
                \end{itemize}
                \item \textbf{Solving quadratic equations}
                \begin{itemize}
                    \item factoring ($|a|=1$ and $|a|>1$)
                    \item quadratic formula: \TallMath{x = \frac{-b \pm \sqrt{b^2-4ac}}{2a}}
                    \item completing the square (special cases)
                    \item zero-product property
                \end{itemize}
                \item \textbf{Discriminant and number of solutions - $b^2 - 4ac$}
                \begin{itemize}
                    \item $b^2 - 4ac > 0$: two distinct real solutions
                    \item $b^2 - 4ac = 0$: exactly one real solution
                    \item $b^2 - 4ac < 0$: no real solutions (Algebra 2: complex)
                \end{itemize}
            \end{enumerate}
        \end{minipage}

    \begin{skillbox}[{Vocabulary, Concepts, \& Theorems}]{greenbox}
        \renewcommand{\arraystretch}{1.35}
        \begin{tabularx}{\linewidth}{
                >{\bfseries}p{0.26\linewidth}
            X
        }
        \textbf{Concept} & \textbf{Definition / Notes} \\
        \hline
        \endhead

        {Equation}
        & {A mathematical statement asserting that two expressions are equal.} \\
        \hline

        {Inequality}
        & {A mathematical statement that one expression is greater than, less than, greater than
        or equal to, or less than or equal to another.} \\
        \hline

        {Solution}
        & {A value (or set of values) that makes an equation or inequality true.} \\
        \hline

        {Solution set}
        & {The complete collection of all solutions to an equation or inequality.} \\
        \hline

        {Properties of equality}
        & {Operations applied equally to both sides of an equation preserve equality
        (addition, subtraction, multiplication, division by nonzero).} \\
        \hline

        {Properties of inequality}
        & {Same operations as equality, \emph{except} multiplying or dividing by a negative
        number reverses the inequality symbol.} \\
        \hline

        {Interval notation}
        & {A compact way to write a solution set. Examples: $(-\infty, 3]$ means $x \leq 3$;
        $(2, 5)$ means $2 < x < 5$.} \\
        \hline

        {Literal equation}
        & {An equation with more than one variable (often a formula). Solved by isolating
        the target variable using properties of equality.} \\
        \hline

        {System of equations}
        & {Two or more equations with the same variables considered simultaneously.} \\
        \hline

        {Substitution method}
        & {Solve one equation for a variable, then substitute that expression into the
        other equation.} \\
        \hline

        {Elimination method}
        & {Add or subtract multiples of the equations to cancel one variable, then solve
        for the remaining variable.} \\
        \hline

        {Consistent system}
        & {A system with at least one solution (lines intersect or coincide).} \\
        \hline

        {Inconsistent system}
        & {A system with no solution (lines are parallel).} \\
        \hline

        {Quadratic equation}
        & {An equation of the form $ax^2 + bx + c = 0$ where $a \ne 0$.} \\
        \hline

        {Zero-product property}
        & {If $A \cdot B = 0$, then $A = 0$ or $B = 0$ (or both).
        Foundation for solving by factoring.} \\
        \hline

        {Quadratic formula}
        & \TallMath{x = \frac{-b \pm \sqrt{b^2 - 4ac}}{2a}} \quad solves any quadratic
        $ax^2 + bx + c = 0$. \\
        \hline

        {Discriminant}
        & {The expression $\Delta = b^2 - 4ac$ inside the quadratic formula.
        Determines the number and type of solutions.} \\
        \hline

        \end{tabularx}
    \end{skillbox}

    \begin{skillbox}[Lesson]{sky}
        \begin{multicols}{2}
            \begin{enumerate}
                \item Solve a multistep linear equation
                \begin{itemize}
                    \item $3(x-2)+4 = 2x+7$
                    \item \textbf{What do you do first?}
                    \item Distribute: $3x-6+4 = 2x+7$
                    \item Simplify: $3x-2 = 2x+7$
                    \item Isolate $x$: $x = 9$
                \end{itemize}

                \item Solve a linear inequality
                \begin{itemize}
                    \item $-2(x-1) \geq 4x-8$
                    \item $-2x+2 \geq 4x-8$
                    \item $10 \geq 6x \Rightarrow x \leq \tfrac{5}{3}$
                    \item \textbf{Why did the direction not flip here?}
                    \item Interval: $\left(-\infty, \tfrac{5}{3}\right]$
                \end{itemize}

                \item Solve a literal equation
                \begin{itemize}
                    \item Solve $A = \tfrac{1}{2}bh$ for $h$
                    \item \textbf{Treat $A$ and $b$ as constants.}
                    \item $2A = bh \Rightarrow h = \dfrac{2A}{b}$
                \end{itemize}
            \end{enumerate}
        \end{multicols}
    \end{skillbox}
    \begin{skillbox}[Lesson (cont.)]{sky}
        \begin{multicols}{2}
            \raggedcolumns
            \begin{enumerate}[start=4]
                \item Solve a system of equations
                \begin{itemize}
                    \item $x + 2y = 8$ and $3x - y = 3$
                    \item \textbf{Which method — substitution or elimination?}
                    \item Substitution: $x = 8 - 2y$
                    \item $3(8-2y)-y = 3 \Rightarrow 24-6y-y=3$
                    \item $-7y = -21 \Rightarrow y = 3$
                    \item $x = 8 - 2(3) = 2$
                    \item Solution: $(2, 3)$
                \end{itemize}
                \columnbreak
                \item Solve a quadratic equation
                \begin{itemize}
                    \item Factor $x^2 - 3x - 10 = 0$
                    \item \textbf{What two numbers add to $-3$ and multiply to $-10$?}
                    \item $(x-5)(x+2) = 0$
                    \item $x = 5$ or $x = -2$
                    \item Discriminant check: $\Delta = (-3)^2 - 4(1)(-10) = 9 + 40 = 49 > 0$
                    \item \textbf{Two real solutions — consistent with our answer.}
                \end{itemize}
            \end{enumerate}
        \end{multicols}
    \end{skillbox}

    \renewcommand{\arraystretch}{1.6}
    \begin{skillbox}[Explicit Instruction: Solving equations and inequalities with Desmos]{sky}
        \begin{multicols}{2}
            \raggedcolumns
            \begin{enumerate}[label=\arabic*.]
                \item \textbf{Equations:} Graph both sides as separate functions.
                The $x$-coordinate of the intersection is the solution.
                \begin{itemize}
                    \item $y_1 = $ left side \quad $y_2 = $ right side
                    \item Find the intersection point.
                \end{itemize}
                \item \textbf{Inequalities:} Graph the equation form first, then
                identify which region satisfies the inequality.
                \begin{itemize}
                    \item Graph $y = $ the expression.
                    \item Read off where $y \leq 0$, $y \geq 0$, etc.
                    \item Or graph the inequality directly to see the shaded region.
                \end{itemize}
                \item \textbf{Systems:} Graph both lines. The solution is the
                intersection point (or no solution if parallel; infinite if same line).
                \item \textbf{Quadratics:} Graph $y = ax^2+bx+c$.
                \begin{itemize}
                    \item Count the $x$-intercepts to predict the number of solutions.
                    \item Use the zeros to read off the exact solutions when integers.
                \end{itemize}
            \end{enumerate}
        \end{multicols}
    \end{skillbox}
