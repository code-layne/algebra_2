  \item \textbf{Solve, then classify:} $4(x-3)=2(2x-5)+2$.

        \vspace{2pt}
        Work:
        \begin{work}
          4(x-3) &= 2(2x-5)+2 \\
           4x-12 &= 4x-10+2   \\
           4x-12 &= 4x-8      \\
             -12 &= -8
        \end{work}

        Circle one: \quad one solution \quad$\cdot$\quad no solution \quad$\cdot$\quad
        infinitely many solutions

        \vspace{2pt}
        What in your last line told you which? \writeline

  \item \textbf{Solve by factoring:} $x^2-3x=10$.

        \vspace{2pt}
        Step 1 --- get one side to zero:
        \begin{work}
             x^2-3x &= 10 \\
          x^2-3x-10 &= 0
        \end{work}

        Step 2 --- factor and apply the zero product property:
        \begin{work}
          (x-5)(x+2) &= 0 \\
                   x &= 5 \quad\text{or}\quad x = -2
        \end{work}

        \textbf{Verify one of your solutions} in the original equation. \writeline

  \item \textbf{SOL-style multiple choice.} What is the solution set of \; $-3x+7>19$ \; ?
        \begin{work}
          -3x+7 &> 19 \\
            -3x &> 12 \\
              x &< -4
        \end{work}
        \begin{enumerate}[label=\Alph*., leftmargin=1.9em, itemsep=2pt]
          \item $x>-4$
          \item $x<-4$
          \item $x>4$
          \item $x<4$
        \end{enumerate}

        \vspace{2pt}
        In one sentence, what did the \emph{wrong} answer you were most tempted by get wrong?
        \writeline
\end{enumerate}
