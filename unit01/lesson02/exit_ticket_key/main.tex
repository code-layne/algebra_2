\begin{enumerate}[leftmargin=*, itemsep=1.8em]

    % ── Question 1 ──────────────────────────────────────────────────────────
    \item Solve $5 - 2(x + 3) = x + 1$. \textbf{Show every step.}

    \vspace{0.1in}
    \ansline{$5 - 2x - 6 = x + 1$}
    \ansline{$-2x - 1 = x + 1$}
    \ansline{$-3x = 2$}
    \ansline{$x = -\dfrac{2}{3}$}

    \vspace{0.05in}
    Answer: \ans{$x = -\dfrac{2}{3}$}

    \begin{teachernote}
        Common errors: (1) distributing $-2$ incorrectly as $5 - 2x + 6$ (sign on $-6$);
        (2) moving $x$ to the wrong side. Cold-call: ``What do you get after distributing?''
    \end{teachernote}

    % ── Question 2 ──────────────────────────────────────────────────────────
    \item Solve $3x + 4 \geq 10$. Write in interval notation.

    \vspace{0.1in}
    \ansline{$3x \geq 6$}
    \ansline{$x \geq 2$}

    \vspace{0.05in}
    Answer: \ans{$[2, \infty)$}

    \begin{teachernote}
        Watch for students who write $(2, \infty)$ (open bracket) --- the $\geq$ requires a closed bracket $[$.
        Also watch for students who reverse the inequality; dividing by $+3$ does not flip the symbol.
    \end{teachernote}

    % ── Question 3 (SOL-style MC) ────────────────────────────────────────────
    \item How many real solutions does $x^2 - 4x + 7 = 0$ have?

    \vspace{0.25in}
    \begin{tabularx}{\linewidth}{Y Y Y Y}
        \ans{\textbf{A.}\quad $0$}
        & \textbf{B.}\quad $1$
        & \textbf{C.}\quad $2$
        & \textbf{D.}\quad infinitely many
    \end{tabularx}

    \vspace{0.2in}
    \ans{$\Delta = (-4)^2 - 4(1)(7) = 16 - 28 = -12 < 0$} $\Rightarrow$ \ans{\textbf{A. 0 real solutions}}

    \begin{teachernote}
        \textbf{Distractor analysis:}
        \textbf{B} --- students who compute $\sqrt{-12}$ and think ``the square root gives one answer'';
        \textbf{C} --- students who ignore the sign of $\Delta$ and assume $\pm$ always gives two answers;
        \textbf{D} --- confusion with the concept of ``infinite solutions'' from linear equations.
        Reinforce: $\Delta < 0$ means no real solutions; in Algebra 2 we will revisit this with complex numbers.
    \end{teachernote}

\end{enumerate}
