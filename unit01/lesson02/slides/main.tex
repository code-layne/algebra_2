    \normalsize
    \begin{itemize}
      \setlength{\itemsep}{0.45em}
      \item Distribute, combine like terms, then isolate the variable
      \item \textbf{One solution:} variable remains after simplifying
      \item \textbf{No solution:} contradiction (e.g., $3 = 5$)

    \normalsize
    \begin{itemize}
      \setlength{\itemsep}{0.5em}
      \item Solve like an equation \textbf{except}: flip the symbol when multiplying
            or dividing both sides by a \textbf{negative} number
      \item Graph: closed dot ($\leq$, $\geq$) or open dot ($<$, $>$); arrow shows direction
      \item Interval notation: use brackets $[$ $]$ for $\leq$/$\geq$; parentheses $($ $)$ for $<$/$>$
    \end{itemize}
  \end{minipage}

    \vspace{0.4em}
    {\large\bfseries Systems of Two Linear Equations}\\[0.5em]
    \normalsize
    \begin{itemize}
      \setlength{\itemsep}{0.45em}
      \item \textbf{Substitution:} isolate one variable; substitute into the other equation
      \item \textbf{Elimination:} scale equations so a variable cancels; add the equations
      \item \textbf{Graphical:} plot both lines; solution is the intersection point
      \item \textbf{Number of solutions:} one (intersect), none (parallel), infinite (same line)
    \end{itemize}
  \end{minipage}
\end{frame}

      \setlength{\itemsep}{0.5em}

      {Explicit Instruction \quad\normalsize\itshape Solving Equations \& Inequalities --- Desmos};
  \end{tikzpicture}
  \vspace{1.05cm}\vspace{1em}\par
  \hspace{0.55cm}%
  \begin{minipage}{0.9\textwidth}
    \vspace{0.3em}
    \begin{columns}[T]
      \begin{column}{0.50\textwidth}
        \small\textbf{Equations}
        \begin{enumerate}
          \setlength{\itemsep}{0.4em}
          \item Graph $y_1 = $ left side, $y_2 = $ right side
          \item Find the intersection; the $x$-coordinate is the solution
          \item Verify by substituting back into the original equation
        \end{enumerate}
        \vspace{0.5em}
        \small\textbf{Inequalities}
        \begin{enumerate}
          \setlength{\itemsep}{0.4em}
          \item Graph the inequality directly in Desmos to see shading
          \item Or graph $y = $ expression and read where $y \leq 0$ / $y \geq 0$
        \end{enumerate}
      \end{column}
      \hspace{0.02\textwidth}
      \begin{column}{0.44\textwidth}
        \small\textbf{Systems}
        \begin{enumerate}
          \setlength{\itemsep}{0.4em}
          \item Enter both equations; Desmos plots both lines
          \item Click the intersection point to read the solution
          \item Parallel lines → no solution; same line → infinite solutions
        \end{enumerate}
        \vspace{0.5em}
        \small\textbf{Quadratics}
        \begin{enumerate}
          \setlength{\itemsep}{0.4em}
          \item Graph $y = ax^2+bx+c$
          \item Count x-intercepts: 2 roots, 1 root, or none
          \item Read integer roots directly; use formula for non-integers
        \end{enumerate}
      \end{column}
    \end{columns}
  \end{minipage}
\end{frame}
}
