% HOOK
\begin{frame}[t, plain]
  \forestheader{Solve all three. That is the whole instruction.}
  \sectionlabel{Hook}
  \begin{columns}[T]
    \begin{column}{0.44\textwidth}
      \centering
      \renewcommand{\arraystretch}{1.7}
      \begin{tabular}{c l}
        (a) & $3(x-2)=x-6$ \\
        (b) & $3(x-2)=3x-6$ \\
        (c) & $3(x-2)=3x-5$ \\
      \end{tabular}
    \end{column}
    \begin{column}{0.52\textwidth}
      \begin{itemize}\setlength{\itemsep}{6pt}
        \item (a) behaves: \; $2x=0$, so $x=0$.
        \item (b) leaves \; $-6=-6$. \; The $x$ is \textbf{gone}, and what is left is \textbf{true}.
        \item (c) leaves \; $-6=-5$. \; The $x$ is gone, and what is left is \textbf{false}.
        \item Nobody made a mistake. Those \emph{are} the answers.
      \end{itemize}
    \end{column}
  \end{columns}
  \begin{block}{An expression was a recipe. An equation is a \textbf{question}.}
    The question is ``which values make this true?'' --- so the answer is a \textbf{solution set},
    and it may hold \textbf{one} number, \textbf{no} numbers, or \textbf{every} number.
    \\[3pt]
    Same story, different family: \quad $x^2=9$ (two) \quad $x^2=0$ (one) \quad $x^2=-9$ (none).
    \textbf{Today the count becomes something you predict, not something you discover.}
  \end{block}
\end{frame}

% SECTION 1 — linear equations and the trichotomy
\begin{frame}[t, plain]
  \forestheader{Every step has a name, and the last line has a meaning}
  \sectionlabel{1. Linear equations}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \renewcommand{\arraystretch}{1.3}
      \begin{tabular}{@{}l l@{}}
        $5x+3(x-2)=2(x+9)$ & given \\
        $5x+3x-6=2x+18$ & distributive \\
        $8x-6=2x+18$ & like terms \\
        $6x-6=18$ & subtraction \\
        $6x=24$ & addition \\
        $x=4$ & division \\
      \end{tabular}
      \vspace{6pt}

      ``I moved it over'' is not a property. \\
      Then \textbf{check in the original}: \; $26=26$.
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Reading the last line}
        Variable \textbf{survives} \; $\Rightarrow$ \; one solution. \\[3pt]
        Variable \textbf{vanishes}, statement \textbf{true} \; $\Rightarrow$ \; all real numbers
        (\emph{identity}). \\[3pt]
        Variable \textbf{vanishes}, statement \textbf{false} \; $\Rightarrow$ \; no solution
        (\emph{contradiction}).
      \end{block}
      \begin{itemize}\setlength{\itemsep}{4pt}
        \item Properties of equality are the \emph{only} moves that keep the solution set the same.
        \item \$40 fee $+$ \$15/hour $=$ \$130 \; $\Rightarrow$ \; $h=6$ \textbf{hours} --- answer
          the question, not the letter.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% SECTION 2 — literal equations
\begin{frame}[t, plain]
  \forestheader{Same moves. Scarier alphabet.}
  \sectionlabel{2. Literal equations}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      % \arraystretch scales a strut off \baselineskip, which a \dfrac overshoots —
      % the rows then collide. Space these rows explicitly instead.
      \begin{tabular}{@{}l c l@{}}
        $A=\tfrac12 bh$ & $\to$ & $h=\dfrac{2A}{b}$ \\[14pt]
        $P=2\ell+2w$ & $\to$ & $w=\dfrac{P-2\ell}{2}$ \\[14pt]
        $y=mx+b$ & $\to$ & $x=\dfrac{y-b}{m}$ \\
      \end{tabular}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{The only real trap}
        $2\ell$ and $2w$ are \textbf{not} like terms, so $P\ne4\ell w$. \\[3pt]
        And when you divide, divide \textbf{every} term --- not just the one you were staring at.
      \end{block}
      \begin{itemize}\setlength{\itemsep}{5pt}
        \item If the target variable appears \emph{twice}, \textbf{factor it out} first:
          $ax+3x=b \;\Rightarrow\; x(a+3)=b$.
        \item \textbf{Unit 2 preview:} the three ``forms'' of a line are not three facts. They are
          one equation, solved for different things.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% SECTION 3 — inequalities
\begin{frame}[t, plain]
  \forestheader{Why the symbol flips --- derived, not decreed}
  \sectionlabel{3. Inequalities}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Watch it happen}
        Start with something true: \quad $2<6$ \\[4pt]
        Multiply both sides by $-1$: \quad $-2 \;{\color{goldacc}\mathbf{>}}\; -6$ \\[4pt]
        Multiplying by a negative \textbf{reflects the number line across zero}, and a reflection
        \textbf{reverses order}.
      \end{block}
      \centering
      $-3x+7>19 \;\Rightarrow\; -3x>12 \;\Rightarrow\;
      {\color{forest}\mathbf{x<-4}}$
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{itemize}\setlength{\itemsep}{5pt}
        \item Only \textbf{multiplying} and \textbf{dividing} by a negative flip it. Subtracting a
          negative does not.
        \item Endpoint: \textbf{open} $\circ$ for $<$ and $>$; \textbf{closed} $\bullet$ for $\le$
          and $\ge$.
        \item Ended with $3\ge x$? \textbf{Rewrite it as $x\le3$} before you shade --- that is how
          the picture ends up backwards.
        \item The solution set is \textbf{infinite}. That is why we draw it instead of listing it.
      \end{itemize}
    \end{column}
  \end{columns}
  \begin{block}{Context still has the last word}
    \$12 per ticket, at most \$75 \; $\Rightarrow$ \; $12t\le75$ \; $\Rightarrow$ \; $t\le6.25$
    \; $\Rightarrow$ \; \textbf{6 tickets}. The algebra gives a number; you still have to interpret it.
  \end{block}
\end{frame}

% SECTION 4 — quadratics, three methods
\begin{frame}[t, plain]
  \forestheader{Three methods, chosen by what the equation looks like}
  \sectionlabel{4. Quadratic equations}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{itemize}\setlength{\itemsep}{6pt}
        \item \textbf{Square roots} --- no $x$ term. \\
          $x^2=49 \Rightarrow x={\color{forest}\mathbf{\pm7}}$. \\
          {\small Writing only $7$ does not lose a step. It loses a \textbf{solution}.}
        \item \textbf{Factor $+$ zero product property} --- when it factors.
        \item \textbf{Quadratic formula} --- always works. \\
          $x^2-4x-7=0 \Rightarrow x=\dfrac{4\pm\sqrt{44}}{2}=2\pm\sqrt{11}$ \\
          {\small That radical simplification is Lesson 1.1 doing its job.}
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Step 0, every single time}
        Make one side \textbf{zero} first. \\[3pt]
        $x^2-3x=10 \;\Rightarrow\; x^2-3x-10=0 \;\Rightarrow\; (x-5)(x+2)=0$
      \end{block}
      \begin{itemize}\setlength{\itemsep}{5pt}
        \item $mn=0$ forces $m=0$ or $n=0$. \\
          $mn=8$ forces \textbf{nothing} --- eight has infinitely many factor pairs.
        \item In 1.1, factoring short of \emph{completely} cost you a step. Here it costs you a
          \textbf{solution}.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% SECTION 5 — the discriminant
\begin{frame}[t, plain]
  \forestheader{Counting the answers before you find any of them}
  \sectionlabel{5. The discriminant}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \centering
      \renewcommand{\arraystretch}{1.5}
      \begin{tabular}{@{}l c c c@{}}
        \textbf{Equation} & $b^2-4ac$ & \textbf{Sign} & \textbf{\# real} \\
        \hline
        $x^2-2x-3=0$ & $16$ & $+$ & $2$ \\
        $x^2-2x+1=0$ & $0$ & $0$ & $1$ \\
        $x^2-2x+3=0$ & $-8$ & $-$ & $0$ \\
      \end{tabular}
      \vspace{8pt}

      \raggedright
      Same $a$, same $b$ --- only $c$ moved. And only $c$ had to move.
    \end{column}
    \begin{column}{0.44\textwidth}
      \centering
      \begin{tikzpicture}[x=0.19cm, y=0.13cm]
        \foreach \k/\dx in {-3/0, 1/10, 3/20} {
          \draw[->, charcoal] ({-3.2+\dx},0) -- ({5.2+\dx},0);
          \draw[->, charcoal] ({0+\dx},-5.2) -- ({0+\dx},6.4);
          \draw[forest, very thick, domain=-1.7:3.7, smooth, variable=\x]
            plot ({\x+\dx},{\x*\x-2*\x+\k});
        }
      \end{tikzpicture}

      \vspace{4pt}
      \small crosses \textbf{twice} \; $\cdot$ \; \textbf{touches} \; $\cdot$ \; \textbf{misses}
    \end{column}
  \end{columns}
  \begin{block}{One fact, told twice}
    A real solution \textbf{is} an $x$-intercept. The discriminant is the algebra saying what the
    picture shows --- and it says it \emph{before} you solve anything.
  \end{block}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{What today actually was}
  \sectionlabel{Closing}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Errors that cost the most}
        Dividing by a negative and keeping the symbol. \\
        $(x-3)(x+2)=6 \Rightarrow x-3=6$. \\
        $x^2=25 \Rightarrow x=5$. \\[3pt]
        {\small The last two are the same mistake: an answer given up halfway.}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Habits to leave with}
        Name the property on every line. \\
        Read the \textbf{last} line before you panic. \\
        Zero \textbf{first}, then factor. \\
        Count with the discriminant, then solve. \\
        Substitute back --- that is what ``solution'' means.
      \end{block}
    \end{column}
  \end{columns}
