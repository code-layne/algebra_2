% DEBRIEF 1 — distance and the two cases
\begin{frame}[t, plain]
  \forestheader{Debrief: naming what you found (1)}
  \sectionlabel[redacc]{Part 1 --- exactly five miles out}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \small
      \begin{itemize}\setlength{\itemsep}{5pt}
        \item ``How far is mile $x$ from the tower?'' $\to$ \redink{$|x-12|$}: the
              \redink{distance} between $x$ and $12$. A distance is \redink{never negative}.
        \item ``Exactly $5$ miles out'' $\to$ the \redink{absolute value equation}
              $|x-12| = 5$.
        \item Your two markers $7$ and $17$ came from going $5$ \emph{down} and $5$ \emph{up}
              $\to$ the \redink{two-case rule}: if $|X| = c$ and $c>0$, then $X = c$ \emph{or}
              $X = -c$.
        \item You checked both by substituting --- that is how you \redink{verify} a solution.
      \end{itemize}
    \end{column}
    \begin{column}{0.40\textwidth}
      \centering
      \begin{tikzpicture}[x=0.22cm, y=0.5cm]
        \trail
        \fill[redacc] (7,0) circle (3pt);
        \fill[redacc] (17,0) circle (3pt);
        \draw[redacc, <->, semithick] (7,1.1)--(12,1.1) node[midway, above, font=\tiny]{$5$};
        \draw[redacc, <->, semithick] (12,1.1)--(17,1.1) node[midway, above, font=\tiny]{$5$};
      \end{tikzpicture}
      \vspace{4pt}
      {\footnotesize $|x-12| = 5 \;\Rightarrow\; x = 7$ or $x = 17$}
    \end{column}
  \end{columns}
\end{frame}

% DEBRIEF 2 — in range: "and"
\begin{frame}[t, plain]
  \forestheader{Debrief: in range --- ``less th\emph{AND}''}
  \sectionlabel[redacc]{Part 2a--b --- one connected piece}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \small
      \begin{itemize}\setlength{\itemsep}{5pt}
        \item ``Within $5$ miles'' $\to$ the \redink{absolute value inequality}
              $|x-12| \le 5$.
        \item Your shading ran from $7$ to $17$: the distance is \emph{less} than $5$, so the
              expression is \redink{trapped between} $-5$ and $5$:
              \[ -5 \le x-12 \le 5 \;\Longrightarrow\; 7 \le x \le 17 \]
        \item Two inequalities that must \emph{both} hold: a \redink{compound ``and''}. One
              connected \redink{interval}, $[7,17]$.
      \end{itemize}
    \end{column}
    \begin{column}{0.40\textwidth}
      \centering
      \begin{tikzpicture}[x=0.22cm, y=0.5cm]
        \trail
        \draw[very thick, greenacc] (7,0)--(17,0);
        \fill[greenacc] (7,0) circle (3pt);
        \fill[greenacc] (17,0) circle (3pt);
        \node[greenacc, font=\tiny, above] at (12,0.9){in range};
      \end{tikzpicture}
      \vspace{4pt}
      {\footnotesize $\{x \mid 7 \le x \le 17\} = [7,17]$}
    \end{column}
  \end{columns}
\end{frame}

% DEBRIEF 3 — out of range: "or"
\begin{frame}[t, plain]
  \forestheader{Debrief: out of range --- ``great\emph{OR}''}
  \sectionlabel[redacc]{Part 2c--d --- why it takes two pieces}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \small
      \begin{itemize}\setlength{\itemsep}{5pt}
        \item ``More than $5$ miles out'' $\to$ $|x-12| > 5$: the expression is thrown
              \redink{outside} $[-5,5]$.
              \[ x-12 < -5 \;\text{ or }\; x-12 > 5 \;\Longrightarrow\; x<7 \;\text{ or }\; x>17 \]
        \item Only \emph{one} of the two has to hold: a \redink{compound ``or''} --- \redink{two
              rays}, and the in-range stretch sits between them. That is your answer to 2d.
        \item \textbf{Less th\emph{AND}, great\emph{OR}.} The symbol decides, every time.
      \end{itemize}
    \end{column}
    \begin{column}{0.40\textwidth}
      \centering
      \begin{tikzpicture}[x=0.22cm, y=0.5cm]
        \trail
        \draw[very thick, redacc] (0,0)--(7,0);
        \draw[very thick, redacc] (17,0)--(20,0);
        \draw[redacc, thick, fill=white] (7,0) circle (3pt);
        \draw[redacc, thick, fill=white] (17,0) circle (3pt);
        \node[redacc, font=\tiny, above] at (3.5,0.9){out};
        \node[redacc, font=\tiny, above] at (18.5,0.9){out};
      \end{tikzpicture}
      \vspace{4pt}
      {\footnotesize On the parkway itself: $0 \le x < 7$ or $17 < x \le 20$.}
    \end{column}
  \end{columns}
\end{frame}

% DEBRIEF 4 — isolate first, special cases, three notations
\begin{frame}[t, plain]
  \forestheader{Debrief: isolate first, and the odd cases}
  \sectionlabel[goldacc]{Two examples from me, one rule from you}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \small
      \begin{itemize}\setlength{\itemsep}{5pt}
        \item \redink{Isolate the bars first.} $3|x+1|-2 = 10$ is not ready to split ---
              get $|x+1| = 4$, \emph{then} use the two cases.
        \item $|x-4| = -3$: \redink{no solution} --- nothing sits a negative distance away.
              $|x-4| = 0$: \redink{exactly one} --- only $4$ sits $0$ away.
        \item Same set, \redink{three notations}: \\
              $\{x \mid 7 \le x \le 17\}$ \quad $[7,17]$ \quad and a shaded number line.
      \end{itemize}

      \begin{enumerate}\setlength{\itemsep}{3pt}
        \item $|x-h|$ is the \textbf{distance} between $x$ and $h$ --- never negative.

        \item $|X| = c$, $c>0$: \; $X=c$ \textbf{or} $X=-c$ (two solutions). \\

        \item $|X| > c$: \; $X<-c$ \textbf{or} $X>c$ --- two rays (\textbf{or}).
        \item Write every set three ways: set, interval, number line.
      \end{enumerate}

% APPLICATION — worked together
\begin{frame}[t, plain]
  \forestheader{Application: Roasting Tolerance}
  \sectionlabel[goldacc]{We do this one together $\cdot$ 7 minutes}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \small
      A roaster fills bags labeled \textbf{340 grams}, and rejects any bag whose weight is
      \textbf{more than 8 grams off} the label.
      \begin{itemize}\setlength{\itemsep}{4pt}
        \item Target $340$, allowed distance $8$, ``no more than'' $\Rightarrow$ \;
              $|w - 340| \le 8$.
        \item Solve: \; $-8 \le w-340 \le 8 \;\Longrightarrow\; 332 \le w \le 348$.
        \item A $331$ g bag? \emph{Read it off the solved range} --- below $332$, so rejected
              (it is $9$ g off).
      \end{itemize}
      \vspace{2pt}
      \begin{block}{Tighten the rule to 5 grams}
        Same center, smaller allowed distance $\Rightarrow$ a \textbf{narrower} interval:
        $335 \le w \le 345$.
      \end{block}
    \end{column}
    \begin{column}{0.42\textwidth}
      \centering
      \begin{tikzpicture}[x=0.26cm, y=0.42cm]
        \draw[->, thick, charcoal] (329.4,0)--(350.6,0);
        \foreach \x in {330,...,350}{\draw[charcoal] (\x,0.14)--(\x,-0.14);}
        \foreach \x in {330,335,340,345,350}{\node[below, font=\tiny, charcoal] at (\x,-0.16){$\x$};}
        \draw[dashed, thin, charcoal] (340,-0.2)--(340,2.9);
        \draw[very thick, greenacc] (332,2.3)--(348,2.3)
          node[midway, above=0pt, font=\tiny, greenacc]{tolerance $8$};
        \draw[very thick, goldacc] (335,1.1)--(345,1.1)
          node[midway, above=0pt, font=\tiny, goldacc]{tolerance $5$};
        \fill[greenacc] (332,2.3) circle (2.2pt); \fill[greenacc] (348,2.3) circle (2.2pt);
        \fill[goldacc] (335,1.1) circle (2.2pt);  \fill[goldacc] (345,1.1) circle (2.2pt);
        \fill[charcoal] (340,0) circle (2.6pt);
      \end{tikzpicture}
      \vspace{4pt}
      {\footnotesize Both ranges are centered on the label, $340$.}
    \end{column}
  \end{columns}
\end{frame}
