% ── PRIORITY IDEAS & SKILLS ───────────────────────────────────────────────────
\renewcommand{\arraystretch}{1.6}
\begin{skillbox}[Priority Ideas \& Skills]{goldbox}
    \begin{minipage}[t]{0.48\linewidth}
        \begin{enumerate}
            \item \textbf{Linear equations in one variable}
            \begin{itemize}
                \item multistep; distributive property
                \item one solution, no solution, or infinite solutions
            \end{itemize}
            \item \textbf{Linear inequalities in one variable}
            \begin{itemize}
                \item flip symbol when dividing by a negative
                \item graph on number line; interval notation
            \end{itemize}
            \item \textbf{Literal equations}
            \begin{itemize}
                \item treat other variables as constants
                \item isolate the target variable
            \end{itemize}
            \item \textbf{Systems of two linear equations}
            \begin{itemize}
                \item substitution method
                \item elimination method
                \item graphical method
            \end{itemize}
        \end{enumerate}
    \end{minipage}
    \hfill
    \begin{minipage}[t]{0.48\linewidth}
        \begin{enumerate}[start=5]
            \item \textbf{Linear inequalities in two variables}
            \begin{itemize}
                \item solid vs.\ dashed boundary line
                \item shade the solution region
            \end{itemize}
            \item \textbf{Solving quadratic equations}
            \begin{itemize}
                \item factoring (zero-product property)
                \item quadratic formula: \TallMath{x = \frac{-b \pm \sqrt{b^2-4ac}}{2a}}
            \end{itemize}
            \item \textbf{Discriminant}
            \begin{itemize}
                \item $\Delta = b^2 - 4ac$
                \item $\Delta > 0$: two real solutions
                \item $\Delta = 0$: one real solution
                \item $\Delta < 0$: no real solutions
            \end{itemize}
        \end{enumerate}
    \end{minipage}
\end{skillbox}

% ── VOCABULARY, CONCEPTS & THEOREMS ──────────────────────────────────────────
\begin{skillbox}[{Vocabulary, Concepts, \& Theorems}]{greenbox}
    \renewcommand{\arraystretch}{1.35}
    \begin{tabularx}{\linewidth}{
            >{\bfseries}p{0.26\linewidth}
        X
    }
    \textbf{Concept} & \textbf{Definition / Notes} \\
    \hline
    \endhead
    {Equation}
    & {A mathematical statement asserting that two expressions are equal.} \\
    \hline

    {Inequality}
    & {A statement that one expression is $<$, $>$, $\leq$, or $\geq$ another.} \\
    \hline

    {Solution set}
    & {All values that satisfy an equation or inequality.} \\
    \hline

    {Properties of equality}
    & {Operations applied equally to both sides preserve equality.} \\
    \hline

    {Properties of inequality}
    & {Same as equality, except multiplying/dividing by a negative \emph{reverses} the symbol.} \\
    \hline

    {Interval notation}
    & {Compact solution form. Brackets $[$ $]$ for $\leq$/$\geq$; parentheses $($ $)$ for $<$/$>$.} \\
    \hline

    {Literal equation}
    & {A formula with more than one variable; solve by isolating the target variable.} \\
    \hline

    {System of equations}
    & {Two or more equations with the same variables considered simultaneously.} \\
    \hline

    {Substitution method}
    & {Isolate one variable and substitute that expression into the other equation.} \\
    \hline

    {Elimination method}
    & {Scale equations so one variable cancels when the equations are added.} \\
    \hline

    {Quadratic equation}
    & {$ax^2 + bx + c = 0$ where $a \ne 0$.} \\
    \hline

    {Zero-product property}
    & {If $A \cdot B = 0$, then $A = 0$ or $B = 0$.} \\
    \hline

    {Quadratic formula}
    & \TallMath{x = \dfrac{-b \pm \sqrt{b^2 - 4ac}}{2a}} \\
    \hline

    {Discriminant}
    & {$\Delta = b^2 - 4ac$. Positive: 2 real solutions; zero: 1; negative: 0 real solutions.} \\
    \hline

    \end{tabularx}
\end{skillbox}
