% Targets are deliberately spoiler-free (EFFL): they describe what students will be able to
% DO in plain language, without pre-naming the vocabulary the debrief will attach.
\begin{learningtargetbox}
\small
\begin{enumerate}[leftmargin=1.5em, itemsep=5pt]
  \item \ldots{} find every place that sits a given distance from a landmark --- and explain why
        there is usually more than one.
  \item \ldots{} describe all the spots that are \emph{close enough} to a target, and all the spots
        that are \emph{too far}, on a number line and in symbols.
  \item \ldots{} write those descriptions in more than one accepted form, and say which endpoints
        are included.
  \item \ldots{} turn a ``within so much of'' statement into math, and say what the answer means
        back in the situation.
\end{enumerate}
\end{learningtargetbox}
