    \vspace{0.4em}
    {\large\bfseries Linear Functions}\\[0.5em]
    \normalsize
    \begin{itemize}
      \setlength{\itemsep}{0.45em}
      \item Slope: $m = \dfrac{y_2 - y_1}{x_2 - x_1}$ \quad (rate of change)
      \item Forms: slope-intercept $y = mx+b$; \; standard $Ax+By=C$; \; point-slope $y-y_1=m(x-x_1)$
      \item Parallel: same slope; \quad Perpendicular: $m_1 m_2 = -1$
      \item Transformations of $y = x$: vertical shift ($+k$), vertical stretch/compression ($k \cdot$)
    \end{itemize}
  \end{minipage}
\end{frame}

  \hspace{0.55cm}%
  \begin{minipage}{0.9\textwidth}
    \vspace{0.4em}
    {\large\bfseries Exponential Functions: $y = ab^x$}\\[0.5em]
    \normalsize
    \begin{itemize}
      \setlength{\itemsep}{0.5em}
      \item $a$ = initial value ($y$-intercept when $x = 0$)
      \item $b$ = growth/decay factor \; ($b > 1$: growth; $0 < b < 1$: decay)
      \item Domain: all real numbers; \quad Range: $(0, \infty)$ when $a > 0$
      \item No $x$-intercept; the $x$-axis is a horizontal asymptote
      \item Transformations: $f(x)+k$ shifts vertically; $kf(x)$ stretches/compresses
    \end{itemize}
  \end{minipage}
\end{frame}
}

      \setlength{\itemsep}{0.45em}

      \item \textbf{Curve of best fit:} use technology to find the regression equation

            slope = rate of change; $y$-int = predicted value when $x = 0$

      \item \textbf{Conclusion:} describe relationship, direction, and strength from scatterplot
    \end{itemize}
