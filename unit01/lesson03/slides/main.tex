% SECTION 1 — functions and notation
\begin{frame}[t, plain]
  \forestheader{The vertical line test is the definition, drawn}
  \sectionlabel{1. Functions \& notation}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \small
      \begin{block}{A function}
        Every input gets \textbf{exactly one} output.
        \\[4pt]
        A vertical line is \emph{one input}. If it hits the graph twice, that input has two outputs
        --- promise broken.
      \end{block}
      \centering
      \begin{tikzpicture}[x=0.30cm, y=0.22cm]
        \draw[linegray!45, very thin, step=1] (-3,-4) grid (3,4);
        \draw[->, charcoal] (-3.3,0) -- (3.3,0);
        \draw[->, charcoal] (0,-4.3) -- (0,4.3);
        \draw[forest, very thick] (-3,-2) -- (3,4);
        \draw[goldacc, dashed, thick] (1,-4) -- (1,4);
      \end{tikzpicture}
      \hspace{6pt}
      \begin{tikzpicture}[x=0.30cm, y=0.22cm]
        \draw[linegray!45, very thin, step=1] (-3,-4) grid (3,4);
        \draw[->, charcoal] (-3.3,0) -- (3.3,0);
        \draw[->, charcoal] (0,-4.3) -- (0,4.3);
        \draw[forest, very thick] (1,-4) -- (1,4);
        \draw[goldacc, dashed, thick] (1,-4) -- (1,4);
      \end{tikzpicture}

      \small $y=x+1$ passes. \; $x=1$ fails.
    \end{column}
    \begin{column}{0.48\textwidth}
      \small
      \begin{block}{$f(x)$ --- talking to the machine}
        $f$ is the \textbf{name}, $x$ is the \textbf{input}, $f(x)$ is the \textbf{output}.
        \\[3pt]
        ``$f(3)=0$'' and ``$(3,0)$ is on the graph'' are the \textbf{same sentence}.
      \end{block}
      \begin{itemize}\setlength{\itemsep}{3pt}
        \item \textbf{Forward} (given $x$, find $f(x)$): arithmetic.
        \item \textbf{Backward} (given $f(x)$, find $x$): \textbf{that is Lesson 1.2}.
        \item The backward case $f(x)=0$ has a name: a \textbf{zero} --- and it is the
          $x$-intercept.
        \item So every equation you solved last lesson was finding a zero.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

  \sectionlabel{2. Linear characteristics}
  \begin{columns}[T]
    \begin{column}{0.42\textwidth}
      \centering
      \begin{tikzpicture}[x=0.36cm, y=0.24cm]
        \draw[linegray!45, very thin, step=1] (-2,-7) grid (5,4);

        \draw[forest, very thick, domain=-0.5:5, smooth, variable=\x] plot ({\x},{2*\x-6});
        \fill[goldacc] (3,0) circle (2.6pt);
        \fill[goldacc] (0,-6) circle (2.6pt);
      \end{tikzpicture}

      \vspace{4pt}
      \small $f(x)=2x-6$
    \end{column}
    \begin{column}{0.54\textwidth}
      \renewcommand{\arraystretch}{1.12}
      \begin{tabular}{@{}l l@{}}
        Domain & all real numbers \\
        Range & all real numbers \\
        \textbf{Zero} & $x=3$ \\
        $x$-intercept & $(3,0)$ \\
        $y$-intercept & $(0,-6)$ \\
        Slope & $2$ \\
      \end{tabular}

      \small Two of those rows are \textbf{the same fact twice}.
      \begin{block}{Slope is a rate --- and in context it has units}
        $V(t)=800-25t$: \; starts at $800$ \textbf{gallons}, loses $25$ \textbf{gallons per minute},
        empty at $t=32$.
        \\[2pt]
        Domain here is $0\le t\le32$ --- the algebra allows every real number, the
        \textbf{situation does not}, and the situation wins.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% SECTION 3 — forms and writing equations
\begin{frame}[t, plain]
  \forestheader{One line, three costumes}
  \sectionlabel{3. Forms \& writing equations}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \renewcommand{\arraystretch}{1.5}
      \begin{tabular}{@{}l@{\hspace{6pt}}l@{}}
        $y=mx+b$ & slope and $y$-intercept \\
        $y-y_1=m(x-x_1)$ & a slope and a point \\
        $Ax+By=C$ & both intercepts fast \\
      \end{tabular}
      \vspace{6pt}

      $y-4=2(x-5) \;\Rightarrow\; y=2x-6 \;\Rightarrow\; 2x-y=6$
      \\[4pt]
      \small Same line every time. This is Lesson 1.2's \textbf{literal equations} --- one equation
      solved for different things.
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{One strategy: get a slope, get a point}
        Slope $-3$ through $(2,5)$ \; $\Rightarrow$ \; $y=-3x+11$ \\
        $(1,2)$ and $(4,11)$ \; $\Rightarrow$ \; $m=3$, \; $y=3x-1$ \\
        Vertical through $(7,-2)$ \; $\Rightarrow$ \; $x=7$ \\
        Horizontal through $(7,-2)$ \; $\Rightarrow$ \; $y=-2$
      \end{block}
      \begin{itemize}\setlength{\itemsep}{4pt}
        \item $y=-2$ is a function. \; ${\color{goldacc}\mathbf{x=7}}$ \textbf{is not} --- one input,
          every output.
        \item \textbf{Parallel:} equal slopes. \; \textbf{Perpendicular:} opposite reciprocals
          (product $-1$).
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% SECTION 4 — transformations of the parent
\begin{frame}[t, plain]
  \forestheader{Two moves you will use for the rest of the year}
  \sectionlabel{4. Transforming $y=x$}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \centering
      \begin{tikzpicture}[x=0.23cm, y=0.18cm]
        \foreach \dx/\m/\b in {0/3/0, 11/1/3, 22/-1/0} {
          \draw[linegray!45, very thin, step=1] ({-3+\dx},-5) grid ({3+\dx},5);
          \draw[->, charcoal] ({-3.3+\dx},0) -- ({3.3+\dx},0);
          \draw[->, charcoal] ({0+\dx},-5.3) -- ({0+\dx},5.3);
          \draw[linegray, dashed, thick] ({-3+\dx},-3) -- ({3+\dx},3);
          \draw[forest, very thick, domain=-3:3, smooth, variable=\x]
            plot ({\x+\dx},{max(min(\m*\x+\b,5),-5)});
        }
      \end{tikzpicture}

      \vspace{5pt}
      \small $y=3x$ \hspace{1.3cm} $y=x+3$ \hspace{1.3cm} $y=-x$
    \end{column}
    \begin{column}{0.50\textwidth}
      \small
      \begin{block}{Each move touches exactly one feature}
        $k\,f(x)$ changes the \textbf{slope}; the $y$-intercept stays at $0$.
        \\[3pt]
        $f(x)+k$ changes the \textbf{$y$-intercept}; the slope stays at $1$.
        \\[3pt]
        $k<0$ \textbf{reflects} the line.
      \end{block}
      \begin{itemize}\setlength{\itemsep}{3pt}
        \item \textbf{Unit 2:} the same two moves on a V.
        \item \textbf{Unit 3:} the same two moves on a parabola.
        \item \textbf{Unit 7:} the same two moves on a growth curve.
        \item The parent changes. The moves do not.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% SECTION 5 — quadratic and exponential
\begin{frame}[t, plain]
  \forestheader{Same six questions. Very different answers.}
  \sectionlabel{5. Quadratic \& exponential}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \centering
      \begin{tikzpicture}[x=0.30cm, y=0.20cm]
        \draw[linegray!45, very thin, step=1] (-3,-5) grid (5,6);
        \draw[->, charcoal] (-3.3,0) -- (5.3,0);
        \draw[->, charcoal] (0,-5.3) -- (0,6.3);
        \draw[forest, very thick, domain=-2.1:4.1, smooth, variable=\x]
          plot ({\x},{\x*\x-2*\x-3});
        \fill[goldacc] (-1,0) circle (2.4pt);
        \fill[goldacc] (3,0) circle (2.4pt);
        \fill[goldacc] (1,-4) circle (2.4pt);
      \end{tikzpicture}

      \vspace{2pt}
      \small $g(x)=x^2-2x-3=(x-3)(x+1)$ \\
      zeros $-1,3$ \; $\cdot$ \; vertex $(1,-4)$ \; $\cdot$ \; \textbf{range $y\ge-4$}
      \\[3pt]
      The factored form \emph{displays} the zeros.
    \end{column}
    \begin{column}{0.48\textwidth}
      \centering
      \begin{tikzpicture}[x=0.34cm, y=0.085cm]
        \draw[->, charcoal] (-3.4,0) -- (4.4,0);
        \draw[->, charcoal] (0,-2) -- (0,19);
        \foreach \t in {4,8,12,16} {\draw[charcoal] (0.1,\t) -- (-0.1,\t);
          \node[left, font=\tiny] at (-0.12,\t) {$\t$};}
        \draw[forest, very thick, domain=-3.4:4.15, smooth, variable=\x]
          plot ({\x},{pow(2,\x)});
        \fill[goldacc] (0,1) circle (2.4pt);
      \end{tikzpicture}

      \vspace{2pt}
      \small $h(x)=2^x$ \; --- $y$-intercept $(0,1)$, \; \textbf{no $x$-intercept}, \;
      range $y>0$
      \\[3pt]
      In $y=ab^x$: \; $a$ is the \textbf{starting value} (because $b^0=1$), \; $b$ is the
      \textbf{multiplier}.
    \end{column}
  \end{columns}
  \begin{block}{The new answer is the \emph{range}}
    A line's range is everything. A parabola \textbf{turns around}, so its range stops. An
    exponential \textbf{multiplies}, so it shrinks toward zero forever without arriving --- which is
    why it has no zero at all.
  \end{block}
\end{frame}

% SECTION 6 — comparing the families
\begin{frame}[t, plain]
  \forestheader{Naming a family from a bare table}
  \sectionlabel{6. Compare \& contrast}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \centering
      \renewcommand{\arraystretch}{1.4}
      \begin{tabular}{@{}l|c|c|c|c|c@{}}
        $x$ & $0$ & $1$ & $2$ & $3$ & $4$ \\
        \hline
        $x$ & $0$ & $1$ & $2$ & $3$ & $4$ \\
        $x^2$ & $0$ & $1$ & $4$ & $9$ & $16$ \\
        $2^x$ & $1$ & $2$ & $4$ & $8$ & $16$ \\
      \end{tabular}
      \vspace{8pt}

      \raggedright
      \small \textbf{Linear:} add $1$ each step. \\
      \textbf{Quadratic:} differences $1,3,5,7$ --- \emph{those} go up by $2$. \\
      \textbf{Exponential:} multiply by $2$ each step.
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{The three signatures}
        Constant \textbf{difference} \; $\Rightarrow$ \; linear \\[3pt]
        Constant \textbf{second difference} \; $\Rightarrow$ \; quadratic \\[3pt]
        Constant \textbf{ratio} \; $\Rightarrow$ \; exponential
      \end{block}
      \begin{itemize}\setlength{\itemsep}{4pt}
        \item $5,8,11,14$ \; $\Rightarrow$ \; \textbf{linear}
        \item $3,6,12,24$ \; $\Rightarrow$ \; \textbf{exponential}
        \item $1,4,9,16$ \; $\Rightarrow$ \; \textbf{quadratic}
      \end{itemize}
      \small This is the only tool that names a family with no equation and no graph.
    \end{column}
  \end{columns}
\end{frame}
