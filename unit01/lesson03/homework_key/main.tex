    % ── Linear functions ──────────────────────────────────────────────────────
    \item $3x - 2y = 12$

    \ansline{Slope: rewrite as $y = \tfrac{3}{2}x - 6$; $m = \tfrac{3}{2}$}
    \ansline{$x$-int: set $y=0$: $3x=12$, $x=4$ $\Rightarrow (4,0)$}
    \ansline{$y$-int: set $x=0$: $-2y=12$, $y=-6$ $\Rightarrow (0,-6)$}
    Slope: \ans{$\tfrac{3}{2}$} \; $x$-int: \ans{$(4,0)$} \; $y$-int: \ans{$(0,-6)$}

    \vspace{0.2in}

    \item Line through $(-1,5)$ and $(3,-3)$.

    \ansline{$m = \dfrac{-3-5}{3-(-1)} = \dfrac{-8}{4} = -2$}
    \ansline{$y - 5 = -2(x+1) \Rightarrow y = -2x + 3$}
    Answer: \ans{$y = -2x + 3$}

    \vspace{0.2in}

    \item Parallel to $y = 3x - 4$ through $(2, 1)$.

    \ansline{Same slope: $m = 3$}
    \ansline{$y - 1 = 3(x - 2) \Rightarrow y = 3x - 5$}
    Answer: \ans{$y = 3x - 5$}

    \vspace{0.2in}

    \item $\{(1,2),(2,4),(1,6),(3,8)\}$ a function?

    \ansline{Input $1$ maps to both $2$ and $6$.}
    Answer: \ans{No --- input 1 has two different outputs.}

    \vspace{0.2in}

    \item $f(x) = -x^2 + 4x + 5$

    \ansline{Axis: $x = -\tfrac{4}{2(-1)} = 2$; vertex: $f(2) = -4+8+5=9$; vertex $(2,9)$}
    \ansline{Zeros: $-x^2+4x+5=0 \Rightarrow x^2-4x-5=0 \Rightarrow (x-5)(x+1)=0$}
    Vertex: \ans{$(2,9)$} \; Zeros: \ans{$x=5,\,-1$} \; $y$-int: \ans{$(0,5)$}

    \columnbreak

    \item $g(x) = 2 \cdot 3^x$

    \ansline{$g(0) = 2 \cdot 1 = 2$; \quad $g(2) = 2 \cdot 9 = 18$}
    $g(0) =$ \ans{$2$} \quad $g(2) =$ \ans{$18$} \quad Range: \ans{$(0, \infty)$}

    \vspace{0.2in}

    \item Ice cream regression.

    \ansline{Desmos $y_1 \sim mx_1 + b$: approximately $y = 4.6x - 189$}
    Equation: \ans{$y \approx 4.6x - 189$}

    \begin{teachernote}
        Regression coefficients depend on technology; accept answers near $m \approx 4.6$, $b \approx -189$.
    \end{teachernote}

    \vspace{0.2in}

    \item Predict at $x = 90$°F.

    \ansline{$y = 4.6(90) - 189 = 414 - 189 = 225$}
    Answer: \ans{approximately $225$ units} (extrapolation; use with caution)

    \vspace{0.2in}

    \item Interpret slope $-2.3$ (TV hours vs.\ GPA).

    \ans{For each additional hour of TV watched per day, GPA is predicted to decrease by approximately $2.3$ points.}

    \vspace{0.2in}

    \item Compare at $x = 10$: $f(10) = 20$; $g(10) = 100$; $h(10) = 2^{10} = 1{,}024$.

    \ansline{$f(10) = 20$, $g(10) = 100$, $h(10) = 1{,}024$}
    Answer: \ans{$h(x) = 2^x$ is greatest; exponential dominates for large $x$.}

\end{enumerate}
