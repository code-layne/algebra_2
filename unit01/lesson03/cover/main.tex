% ── PRIORITY IDEAS & SKILLS ───────────────────────────────────────────────────
\renewcommand{\arraystretch}{1.6}
\begin{skillbox}[Priority Ideas \& Skills]{goldbox}
    \begin{minipage}[t]{0.48\linewidth}
        \begin{enumerate}
            \item \textbf{Relations vs.\ functions}
            \begin{itemize}
                \item vertical line test
                \item domain and range
            \end{itemize}
            \item \textbf{Linear functions}
            \begin{itemize}
                \item slope, intercepts, zeros; interpret in context
                \item slope-intercept, standard, point-slope forms
                \item parallel and perpendicular lines
            \end{itemize}
            \item \textbf{Writing equations of lines}
            \begin{itemize}
                \item given graph, two points, or slope + point
                \item vertical lines $x=a$; horizontal lines $y=c$
            \end{itemize}
            \item \textbf{Quadratic functions}
            \begin{itemize}
                \item vertex, zeros, $y$-intercept, axis of symmetry
                \item domain and range; transformations
            \end{itemize}
        \end{enumerate}
    \end{minipage}
    \hfill
    \begin{minipage}[t]{0.48\linewidth}
        \begin{enumerate}[start=5]
            \item \textbf{Exponential functions}
            \begin{itemize}
                \item $y = ab^x$; initial value $a$, base $b$
                \item domain, range, $y$-intercept; transformations
            \end{itemize}
            \item \textbf{Comparing function families}
            \begin{itemize}
                \item linear: constant rate of change
                \item quadratic: symmetric parabola
                \item exponential: multiplicative growth
            \end{itemize}
            \item \textbf{Bivariate data \& regression}
            \begin{itemize}
                \item scatterplots; curve of best fit
                \item interpret slope and $y$-intercept in context
                \item use model to predict; evaluate validity
            \end{itemize}
        \end{enumerate}
    \end{minipage}
\end{skillbox}

% ── VOCABULARY, CONCEPTS & THEOREMS ──────────────────────────────────────────
\begin{skillbox}[{Vocabulary, Concepts, \& Theorems}]{sky}
    \renewcommand{\arraystretch}{1.35}
    \begin{tabularx}{\linewidth}{
            >{\bfseries}p{0.26\linewidth}
        X
    }
    \textbf{Concept} & \textbf{Definition / Notes} \\
    \hline
    \endhead

    {Relation}
    & {Any set of ordered pairs $(x, y)$.} \\
    \hline

    {Function}
    & {Every input maps to exactly one output. Passes the vertical line test.} \\
    \hline

    {Domain}
    & {All allowable inputs ($x$-values).} \\
    \hline

    {Range}
    & {All outputs ($y$-values) produced.} \\
    \hline

    {Slope}
    & {$m = \dfrac{y_2 - y_1}{x_2 - x_1}$; the rate of change of a linear function.} \\
    \hline

    {Slope-intercept form}
    & {$y = mx + b$.} \\
    \hline

    {Point-slope form}
    & {$y - y_1 = m(x - x_1)$.} \\
    \hline

    {Parallel lines}
    & {Same slope, different $y$-intercept.} \\
    \hline

    {Perpendicular lines}
    & {Slopes are negative reciprocals: $m_1 m_2 = -1$.} \\
    \hline

    {Vertex}
    & {The maximum or minimum point of a parabola.} \\
    \hline

    {Axis of symmetry}
    & {$x = -\dfrac{b}{2a}$; vertical line through the vertex.} \\
    \hline

    {Exponential function}
    & {$y = ab^x$; $a$ = initial value, $b$ = growth ($b>1$) or decay ($0<b<1$) factor.} \\
    \hline

    {Bivariate data}
    & {Data on two quantitative variables collected from the same subject.} \\
    \hline

    {Scatterplot}
    & {Graph of ordered pairs $(x,y)$; reveals the relationship between two variables.} \\
    \hline

    {Curve of best fit}
    & {A function (linear or quadratic) that best models the trend in a scatterplot.} \\
    \hline

    {Residual}
    & {Observed value $-$ predicted value.} \\
    \hline

    \end{tabularx}
\end{skillbox}
