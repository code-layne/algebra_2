    \begin{skillbox}[Priority Ideas \& Skills]{goldbox}
        \begin{minipage}[t]{0.48\linewidth}
            \begin{enumerate}
                \item \textbf{Relations vs.\ functions}
                \begin{itemize}
                    \item vertical line test; ordered pairs, tables, mappings, graphs
                    \item domain and range of a function
                \end{itemize}
                \item \textbf{Linear functions --- A.F.1}
                \begin{itemize}
                    \item slope, intercepts, zeros; interpret in context
                    \item equivalent forms: slope-intercept, standard, point-slope
                    \item transformations of $y = x$: shifts and stretches
                    \item parallel lines (same slope); perpendicular ($m_1 m_2 = -1$)
                \end{itemize}
                \item \textbf{Writing equations of lines}
                \begin{itemize}
                    \item given graph; given two points; given slope and a point
                    \item vertical lines $x = a$; horizontal lines $y = c$
                \end{itemize}
                \item \textbf{Quadratic functions --- A.F.2}
                \begin{itemize}
                    \item key features: vertex (max/min), zeros, $y$-intercept, axis of symmetry
                    \item domain and range; interpret in context
                    \item transformations $f(x)+k$ and $kf(x)$
                    \item connections: standard form $\leftrightarrow$ factored form $\leftrightarrow$ graph
                \end{itemize}
            \end{enumerate}
        \end{minipage}
        \hfill
        \begin{minipage}[t]{0.48\linewidth}
            \begin{enumerate}[start=5]
                \item \textbf{Exponential functions --- A.F.2}
                \begin{itemize}
                    \item form $y = ab^x$ (b is a natural number)
                    \item key features: $y$-intercept, domain, range
                    \item transformations $f(x)+k$ and $kf(x)$
                    \item interpret $a$ (initial value) and $b$ (growth/decay factor)
                \end{itemize}
                \item \textbf{Comparing function families}
                \begin{itemize}
                    \item linear: constant rate of change
                    \item quadratic: changing rate; symmetric parabola
                    \item exponential: multiplicative rate of change
                \end{itemize}
                \item \textbf{Bivariate data \& regression --- A.ST.1}
                \begin{itemize}
                    \item scatterplots; identify relationship (linear or quadratic)
                    \item curve of best fit using technology (Desmos regression)
                    \item interpret slope and $y$-intercept of a linear model in context
                    \item use model to predict; evaluate strength and validity
                    \item make conclusions from bivariate data analysis
                \end{itemize}
            \end{enumerate}
        \end{minipage}
    \end{skillbox}

    \begin{skillbox}[{Vocabulary, Concepts, \& Theorems}]{greenbox}
        \renewcommand{\arraystretch}{1.35}
        \begin{tabularx}{\linewidth}{
                >{\bfseries}p{0.26\linewidth}
            X
        }
        \textbf{Concept} & \textbf{Definition / Notes} \\
        \hline
        \endhead

        {Relation}
        & {Any set of ordered pairs $(x, y)$.} \\
        \hline

        {Function}
        & {A relation where every input ($x$) maps to exactly one output ($y$).
        Passes the vertical line test.} \\
        \hline

        {Domain}
        & {The set of all allowable inputs ($x$-values) for a function.} \\
        \hline

        {Range}
        & {The set of all outputs ($y$-values) produced by a function.} \\
        \hline

        {Slope}
        & {Rate of change: $m = \dfrac{\Delta y}{\Delta x} = \dfrac{y_2 - y_1}{x_2 - x_1}$.} \\
        \hline

        {$x$-intercept / zero}
        & {Where the graph crosses the $x$-axis; $y = 0$; also called a \emph{root} or \emph{zero}.} \\
        \hline

        {$y$-intercept}
        & {Where the graph crosses the $y$-axis; $x = 0$.} \\
        \hline

        {Slope-intercept form}
        & {$y = mx + b$; slope $m$, $y$-intercept $b$.} \\
        \hline

        {Standard form}
        & {$Ax + By = C$; useful for identifying intercepts by substitution.} \\
        \hline

        {Point-slope form}
        & {$y - y_1 = m(x - x_1)$; used when slope and one point are known.} \\
        \hline

        {Parallel lines}
        & {Same slope, different $y$-intercepts; never intersect.} \\
        \hline

        {Perpendicular lines}
        & {Slopes are negative reciprocals: $m_1 m_2 = -1$.} \\
        \hline

        {Vertex of a parabola}
        & {The highest (maximum) or lowest (minimum) point of a quadratic function.} \\
        \hline

        {Axis of symmetry}
        & {Vertical line through the vertex: $x = -\dfrac{b}{2a}$ for $f(x) = ax^2 + bx + c$.} \\
        \hline

        {Exponential function}
        & {$y = ab^x$ where $a \ne 0$, $b > 0$, $b \ne 1$. The base $b$ determines
        growth ($b > 1$) or decay ($0 < b < 1$).} \\
        \hline

        {Bivariate data}
        & {Data involving two quantitative variables collected on the same subject.} \\
        \hline

        {Scatterplot}
        & {A graph of ordered pairs $(x, y)$ used to reveal relationships between two variables.} \\
        \hline

        {Curve of best fit}
        & {A function (linear, quadratic, or other) that best models the trend in a scatterplot.} \\
        \hline

        {Linear regression}
        & {Technology-assisted process that finds the line $y = mx + b$ minimizing
        the sum of squared residuals.} \\
        \hline

        {Residual}
        & {The difference between an observed value and the predicted value from a model.} \\
        \hline

        \end{tabularx}
    \end{skillbox}

    \begin{skillbox}[Lesson]{sky}
        \begin{multicols}{2}
            \begin{enumerate}
                \item Key features of a linear function
                \begin{itemize}
                    \item $f(x) = -3x + 6$
                    \item Slope: $m = -3$ \quad $y$-int: $(0, 6)$ \quad zero: $(2, 0)$
                    \item Domain: all reals; Range: all reals
                    \item \textbf{What does slope $= -3$ mean in context?}
                \end{itemize}

                \item Writing an equation of a line
                \begin{itemize}
                    \item Through $(1, 4)$ and $(3, 10)$
                    \item $m = \dfrac{10-4}{3-1} = \dfrac{6}{2} = 3$
                    \item Point-slope: $y - 4 = 3(x - 1)$
                    \item Slope-intercept: $y = 3x + 1$
                \end{itemize}

                \item Key features of a quadratic
                \begin{itemize}
                    \item $f(x) = x^2 - 6x + 5 = (x-1)(x-5)$
                    \item Zeros: $x = 1$ and $x = 5$
                    \item Axis of symmetry: $x = 3$ \quad Vertex: $(3, -4)$
                    \item $y$-intercept: $(0, 5)$ \quad Opens: up (minimum)
                    \item Domain: all reals; Range: $[-4, \infty)$
                \end{itemize}
            \end{enumerate}
        \end{multicols}
    \end{skillbox}
    \begin{skillbox}[Lesson (cont.)]{sky}
        \begin{multicols}{2}
            \raggedcolumns
            \begin{enumerate}[start=4]
                \item Key features of an exponential
                \begin{itemize}
                    \item $f(x) = 3 \cdot 2^x$
                    \item $a = 3$ (initial value); $b = 2$ (growth factor)
                    \item $y$-intercept: $(0, 3)$
                    \item Domain: all reals; Range: $(0, \infty)$
                    \item No $x$-intercept (never crosses $x$-axis)
                    \item \textbf{What does doubling every step look like on a graph?}
                \end{itemize}
                \columnbreak
                \item Interpreting a linear regression model
                \begin{itemize}
                    \item Given data: (hours studied, test score)
                    \item Regression gives $y = 5.2x + 61$
                    \item Slope $5.2$: each additional hour predicts $+5.2$ points
                    \item $y$-int $61$: predicted score with zero hours of study
                    \item Predict score for 4 hours: $y = 5.2(4) + 61 = 81.8$
                    \item \textbf{Is this prediction reliable?  Why or why not?}
                \end{itemize}
            \end{enumerate}
        \end{multicols}
    \end{skillbox}

    \renewcommand{\arraystretch}{1.6}
    \begin{skillbox}[Explicit Instruction: Functions and Regression with Desmos]{sky}
        \begin{multicols}{2}
            \raggedcolumns
            \begin{enumerate}[label=\arabic*.]
                \item \textbf{Identifying function features:} Graph any function and use
                the cursor (or intersection/zero tools) to read key features directly.
                \begin{itemize}
                    \item Click a zero to read the $x$-intercept.
                    \item Minimum/maximum: look for the vertex label.
                    \item Domain restrictions: use $\{x \geq 0\}$ notation.
                \end{itemize}
                \item \textbf{Comparing function families:} Enter all three parent
                functions on the same screen and zoom out to see long-run behavior.
                \item \textbf{Regression workflow:}
                \begin{enumerate}[label=(\alph*)]
                    \item Enter data in a table (column $x_1$, column $y_1$).
                    \item Type the regression expression:
                          $y_1 \sim mx_1 + b$ (linear) or $y_1 \sim ax_1^2 + bx_1 + c$ (quadratic).
                    \item Read off the model coefficients from the sidebar.
                    \item Use the equation to make predictions.
                \end{enumerate}
            \end{enumerate}
        \end{multicols}
    \end{skillbox}
