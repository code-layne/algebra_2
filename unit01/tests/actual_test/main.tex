\parthead{Part B --- Writing Equations of Lines (20 pts)}
Write each line in slope-intercept form, $y=mx+b$. Show how you found $m$ and $b$; for 11 and 12, start in point-slope form, $y-y_1=m(x-x_1)$.

\begin{enumerate}[leftmargin=1.9em, itemsep=10pt, label=\arabic*., start=9]
  \item The line through $(2,\,3)$ and $(4,\,11)$.
    \begin{work}
      m &= \frac{11-3}{4-2} = \frac{8}{2} = 4 \\
      3 &= 4(2) + b \quad\Rightarrow\quad b = -5
    \end{work}
    \vspace{0.5cm}
    \hfill Equation: \inblank{\slot{3.6cm}{$y=4x-5$}}{\gradedblank[id=q9, type=short, points=5, answer={y=4x-5}, check=expression]{3.6cm-2pt}{13.5pt}}

  \item The line through $(-2,\,7)$ and $(4,\,-2)$.
    \begin{work}
      m &= \frac{-2-7}{4-(-2)} = \frac{-9}{6} = -\frac{3}{2} \\
      7 &= -\frac{3}{2}(-2) + b = 3 + b \quad\Rightarrow\quad b = 4
    \end{work}
    \vspace{0.5cm}
    \hfill Equation: \inblank{\slot{3.6cm}{$y=-\tfrac{3}{2}x+4$}}{\gradedblank[id=q10, type=short, points=5, answer={y=-(3/2)x+4}, check=expression, display={$y=-\tfrac{3}{2}x+4$}]{3.6cm-2pt}{13.5pt}}

  \item The line with slope $3$ that passes through $(2,\,1)$.
    \begin{work}
      y-1 &= 3(x-2) \\
      y-1 &= 3x-6 \\
      y &= 3x-5
    \end{work}
    \vspace{0.5cm}
    \hfill Equation: \inblank{\slot{3.6cm}{$y=3x-5$}}{\gradedblank[id=q11, type=short, points=5, answer={y=3x-5}, check=expression]{3.6cm-2pt}{13.5pt}}

  \item The line with slope $-\tfrac{1}{2}$ that passes through $(4,\,6)$.
    \begin{work}
      y-6 &= -\frac{1}{2}(x-4) \\
      y-6 &= -\frac{1}{2}x+2 \\
      y &= -\frac{1}{2}x+8
    \end{work}
    \vspace{0.5cm}
    \hfill Equation: \inblank{\slot{3.6cm}{$y=-\tfrac{1}{2}x+8$}}{\gradedblank[id=q12, type=short, points=5, answer={y=-(1/2)x+8}, check=expression, display={$y=-\tfrac{1}{2}x+8$}]{3.6cm-2pt}{13.5pt}}
\end{enumerate}
