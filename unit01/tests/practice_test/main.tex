\parthead{Part C --- Short Answer \& Computation (40 pts)}
Show your work. A wrong answer with a right idea can still earn credit; a right answer with no work
may not.

\begin{enumerate}[leftmargin=1.9em, itemsep=10pt]
  \item \textbf{Numbers and justification.}
        \\[4pt]
        (a) Label each number \textbf{R} (rational) or \textbf{I} (irrational).
        \\[3pt]
        \hspace*{1.2em}$\sqrt{64}$ \blank{1.0cm} \quad $\sqrt{10}$ \blank{1.0cm} \quad
        $-\tfrac73$ \blank{1.0cm} \quad $0.\overline{18}$ \blank{1.0cm} \quad $\sqrt[3]{5}$ \blank{1.0cm}
        \\[4pt]
        (b) Name the property that justifies each step.
        \\[3pt]
        \hspace*{1.2em}$6(x-4)=6x-24$ \blank{4.6cm}
        \\[3pt]
        \hspace*{1.2em}$9+(x+1)=(9+x)+1$ \blank{4.6cm}
        \\[3pt]
        \hspace*{1.2em}$5x+0=5x$ \blank{4.6cm}
        \\[4pt]
        (c) ``The set of whole numbers is closed under subtraction.'' Disprove it.
        \\[2pt]
        \writelines{2}

  \item \textbf{Translate and evaluate.}
        \\[4pt]
        (a) Write an expression for ``five more than the cube of a number $n$.'' \blank{3.2cm}
        \\[4pt]
        (b) Evaluate $\dfrac{|a-b|}{3}+\sqrt{c}$ \; for $a=4$, $b=-8$, $c=25$.
        \begin{work}
          \frac{|a-b|}{3}+\sqrt{c} &= \frac{|4-(-8)|}{3}+\sqrt{25} \\
                                   &= \frac{|12|}{3}+5 \\
                                   &= 4+5 \\
                                   &= 9
        \end{work}
        (c) Evaluate $\sqrt[3]{x}+4y$ \; for $x=-64$, $y=\tfrac34$.
        \begin{work}
          \sqrt[3]{x}+4y &= \sqrt[3]{-64}+4\left(\tfrac34\right) \\
                         &= -4+3 \\
                         &= -1
        \end{work}

  \item \textbf{Exponents and polynomial operations.}
        \\[4pt]
        (a) $(4a^3b^2)(-3a^4b^5)$
        \begin{work}
          (4a^3b^2)(-3a^4b^5) &= (4)(-3)\,a^{3+4}b^{2+5} \\
                              &= -12a^{7}b^{7}
        \end{work}
        (b) $\dfrac{30r^5s^3}{6r^5s^7}$ \; (no negative exponents)
        \begin{work}
          \frac{30r^5s^3}{6r^5s^7} &= 5r^{5-5}s^{3-7} \\
                                   &= 5r^{0}s^{-4} \\
                                   &= \frac{5}{s^{4}}
        \end{work}
        (c) $(3x-4)(2x+7)$
        \begin{work}
          (3x-4)(2x+7) &= 6x^2+21x-8x-28 \\
                       &= 6x^2+13x-28
        \end{work}
        (d) $(5x^2-3x+1)-(2x^2+x-6)$
        \begin{work}
          (5x^2-3x+1)-(2x^2+x-6) &= 5x^2-3x+1-2x^2-x+6 \\
                                 &= 3x^2-4x+7
        \end{work}

  \item \textbf{Factor completely.} GCF first, every time.
        \\[4pt]
        (a) $5x^2-45$
        \begin{work}
          5x^2-45 &= 5(x^2-9) \\
                  &= 5(x-3)(x+3)
        \end{work}
        (b) $x^2+3x-10$
        \begin{work}
          x^2+3x-10 &= (x+5)(x-2)
        \end{work}
        (c) $3x^2+14x+8$
        \begin{work}
          3x^2+14x+8 &= 3x^2+12x+2x+8 \\
                     &= 3x(x+4)+2(x+4) \\
                     &= (3x+2)(x+4)
        \end{work}
        (d) $2x^3-14x^2+24x$
        \begin{work}
          2x^3-14x^2+24x &= 2x(x^2-7x+12) \\
                         &= 2x(x-3)(x-4)
        \end{work}

  \item \textbf{Radicals and rational exponents.}
        \\[4pt]
        (a) $\sqrt{72}=$ \blank{2.4cm} \hspace{1.2cm}
        (b) $\sqrt[3]{54}=$ \blank{2.4cm}
        \\[4pt]
        (c) Write $\sqrt[3]{x^5}$ with a rational exponent. \blank{2.6cm}
        \\[4pt]
        (d) Write $16^{3/4}$ as a radical, then evaluate it. \blank{4.6cm}
        \\[4pt]
        (e) $\sqrt{18}+\sqrt{50}=$
        \begin{work}
          \sqrt{18}+\sqrt{50} &= 3\sqrt{2}+5\sqrt{2} \\
                              &= 8\sqrt{2}
        \end{work}

  \item \textbf{Linear equations and inequalities.}
        \\[4pt]
        (a) Solve $4(3x-2)=10x+6$.
        \begin{work}
          4(3x-2) &= 10x+6 \\
            12x-8 &= 10x+6 \\
             2x-8 &= 6 \\
               2x &= 14 \\
                x &= 7
        \end{work}
        (b) Classify each solution set as \textbf{one solution}, \textbf{no solution}, or
        \textbf{infinitely many}. You should not need to finish solving.
        \\[3pt]
        \hspace*{1.2em}$3x+7=3x-2$ \blank{3.6cm} \hspace{0.8cm}
        $4(x-3)=4x-12$ \blank{3.6cm}
        \\[4pt]
        (c) Solve $P=2\ell+2w$ for $w$.
        \begin{work}
              P &= 2\ell+2w \\
          P-2\ell &= 2w \\
          \frac{P-2\ell}{2} &= w
        \end{work}
        (d) Solve $-3x+2\le 14$, then graph the solution on the number line.
        \begin{work}
          -3x+2 &\le 14 \\
            -3x &\le 12 \\
              x &\ge -4
        \end{work}
        \begin{center}\numline{}\end{center}

  \item \textbf{Quadratic equations.}
        \\[4pt]
        (a) Solve by factoring: $x^2-x-12=0$
        \begin{work}
             x^2-x-12 &= 0 \\
           (x-4)(x+3) &= 0 \\
              x=4 \quad&\text{or}\quad x=-3
        \end{work}
        (b) Solve by square roots: $(x-3)^2=25$
        \begin{work}
          (x-3)^2 &= 25 \\
              x-3 &= \pm 5 \\
              x=8 \quad&\text{or}\quad x=-2
        \end{work}
        (c) Solve with the quadratic formula: $x^2-6x+2=0$ \; (exact answer)
        \begin{work}
          x &= \frac{6\pm\sqrt{(-6)^2-4(1)(2)}}{2(1)} \\
            &= \frac{6\pm\sqrt{36-8}}{2} \\
            &= \frac{6\pm\sqrt{28}}{2} \\
            &= \frac{6\pm 2\sqrt{7}}{2} \\
            &= 3\pm\sqrt{7}
        \end{work}
        (d) \emph{Without solving}, use the discriminant to say how many real solutions
        $4x^2-20x+25=0$ has.
        \\[2pt] \hspace*{1.2em}Discriminant: \blank{3.4cm} \quad
        Number of real solutions: \blank{4.0cm}

  \item \textbf{Read the graph.} The linear function $g$ is graphed below; each gridline is one unit.
        \begin{center}
        \gridplot{\draw[forest, very thick, domain=-1.4:5.4, smooth, variable=\x] plot ({\x},{2*\x-4});}
        \end{center}
        (a) $g(-1)=$ \blank{1.8cm} \hspace{1.2cm}
        (b) Solve $g(x)=0$: \; $x=$ \blank{1.8cm}
        \\[4pt]
        (c) That number in (b) is called the \blank{2.6cm} of $g$, and on the graph it is the
        point \blank{2.2cm}.
        \\[4pt]
        (d) Slope: \blank{1.8cm}; \quad $y$-intercept: \blank{2.0cm}; \quad
        equation of $g$: \blank{3.2cm}
        \\[4pt]
        (e) Is $g$ increasing or decreasing? \blank{2.6cm} \quad Domain: \blank{2.4cm}
        \\[4pt]
        (f) Write the equation of the line through $(0,5)$ \textbf{parallel} to $g$
        \blank{3.2cm}, and the line through $(0,-1)$ \textbf{perpendicular} to $g$
        \blank{3.2cm}.
\end{enumerate}

\begin{enumerate}[leftmargin=1.9em, itemsep=14pt]
  \item \textbf{Three families, five questions.} Each table below is generated by a linear, a

        \begin{center}
        \renewcommand{\arraystretch}{1.25}
        \begin{tabular}{@{}c@{\hspace{1.1cm}}c@{\hspace{1.1cm}}c@{}}

        \end{tabular}
        \\
        \end{tabular}
        \end{center}
        \textbf{(a)} Name the family of each table and give the numerical evidence --- constant
        difference, constant \emph{second} difference, or constant ratio.
        \\[2pt]
        \writelines{3}
        \textbf{(b)} Write a rule for each function.
        \\[4pt]
        \hspace*{1.2em}$A(x)=$ \blank{3.0cm} \quad $B(x)=$ \blank{3.0cm} \quad $C(x)=$ \blank{3.0cm}
        \\[6pt]
        \textbf{(c)} Which function is largest at $x=1$? At $x=4$? Show the values.
        \begin{work}
          A(1)=11, \quad B(1)&=6, \quad C(1)=4 \\
          A(4)=23, \quad B(4)&=162, \quad C(4)=19

        \\[3pt]

        \\[2pt]
        \hspace*{1.2em}\textbf{Cascade:} no activation fee, \$20 per month.
        \\[4pt]
        \textbf{(a)} Write a function for the total cost at each company after $m$ months.
        \\[4pt]
        \hspace*{1.2em}$N(m)=$ \blank{3.6cm} \hspace{1.4cm} $C(m)=$ \blank{3.6cm}
        \\[6pt]
        \textbf{(b)} Find the number of months at which the two costs are equal. Show every step.
        \begin{work}
          15m+25 &= 20m \\
              25 &= 5m \\
               5 &= m
        \end{work}
        Name the property of equality that justifies each step above.
        \\[3pt]
        \hspace*{1.2em}Step 1: \blank{5.4cm} \quad Step 2: \blank{5.4cm}
        \\[6pt]
        \textbf{(c)} Verify your answer by substituting it into \emph{both} functions.
        \begin{work}

        \\[2pt]

        \\[2pt]
        \writelines{2}
