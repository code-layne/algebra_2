% DEFINITION / EVALUATE
\begin{frame}[t, plain]
  \forestheader{Evaluating: which piece owns the input?}
  \sectionlabel{Evaluate}
  \[ f(x)=\begin{cases} 2x+1, & x<2\\ 5-x, & x\ge 2 \end{cases} \]
  \begin{itemize}\setlength{\itemsep}{5pt}
    \item $f(0)$: since $0<2$, use the \textbf{top}: $f(0)=1$.
    \item $f(2)$: since $2\ge 2$, use the \textbf{bottom}: $f(2)=3$.
    \item $f(5)=0$.
  \end{itemize}
  \vspace{2pt}
  \begin{block}{Boundary rule}
    The input $x=2$ belongs to the piece whose inequality \emph{includes} it (here the $\ge$ piece) ---
    not the strict-$<$ piece. First find the piece, then apply its rule.
  \end{block}
\end{frame}

\begin{frame}[t, plain]

  \sectionlabel[goldacc]{Bridge from 1.3}

      The V is really \textbf{two lines} pasted at the vertex:
      \[ |x|=\begin{cases} x, & x\ge 0\\ -x, & x<0 \end{cases} \]
      \begin{itemize}\setlength{\itemsep}{4pt}
        \item Check: $|-3|=-(-3)=3$.
        \item The vertex is the \emph{boundary} between the two rules.

% READING A GRAPH: OPEN/CLOSED, JUMP
\begin{frame}[t, plain]
  \forestheader{Reading a graph: open, closed, and jumps}
  \sectionlabel[goldacc]{Discontinuity}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \[ f(x)=\begin{cases} x+3, & x<1\\ -x+4, & x\ge 1 \end{cases} \]
      \begin{itemize}\setlength{\itemsep}{4pt}
        \item Closed $\bullet$ includes ($\le,\ge$); open $\circ$ excludes ($<,>$).
        \item At $x=1$ take the \emph{closed} dot: $f(1)=3$.
        \item Pieces \emph{disagree} at $x=1$ ($4$ vs.\ $3$) --- a \textbf{jump}.
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.46]
        \lgrid{-3}{5}{-1}{5}
        \draw[very thick, forest] (-3,0)--(1,4);
        \opendot{1}{4}
        \draw[very thick, forest] (1,3)--(4,0);
        \closeddot{1}{3}
      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}

% STEP FUNCTION
\begin{frame}[t, plain]
  \forestheader{Step functions: $\lfloor x\rfloor$}
  \sectionlabel{Greatest integer}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      $\lfloor x\rfloor=$ the greatest integer $\le x$ (\textbf{round down}):
      \begin{itemize}\setlength{\itemsep}{4pt}
        \item $\lfloor 3.7\rfloor=3$, \; $\lfloor 5\rfloor=5$.
        \item $\lfloor -0.4\rfloor=-1$, \; $\lfloor -2.5\rfloor=-3$ (down!).
        \item \emph{Constant} on each $[n,n+1)$, then jumps.
      \end{itemize}
      Pricing that ``rounds'' (parking, postage) is a step function.
    \end{column}
    \begin{column}{0.46\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.46]
        \lgrid{-1}{4}{-1}{4}
        \draw[very thick, forest] (-1,-1)--(0,-1); \closeddot{-1}{-1} \opendot{0}{-1}
        \draw[very thick, forest] (0,0)--(1,0);    \closeddot{0}{0}   \opendot{1}{0}
        \draw[very thick, forest] (1,1)--(2,1);    \closeddot{1}{1}   \opendot{2}{1}
        \draw[very thick, forest] (2,2)--(3,2);    \closeddot{2}{2}   \opendot{3}{2}
        \draw[very thick, forest] (3,3)--(4,3);    \closeddot{3}{3}   \opendot{4}{3}
      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}
