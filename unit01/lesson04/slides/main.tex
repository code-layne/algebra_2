% HOOK
\begin{frame}[t, plain]
  \forestheader{Hook: free for three months}
  \sectionlabel[goldacc]{A streaming service, then \$12 a month}
  \begin{itemize}\setlength{\itemsep}{5pt}
    \item Total paid after 2 months? After 3? After 5? \; \textbf{\$0, \$0, \$24.}
    \item Could one straight line describe the total for every month? \; \textbf{No} --- flat,

    \item So how many rules does the cost follow? \; \textbf{Two.}
  \end{itemize}
  \[ C(m)=\begin{cases} 0, & 0\le m\le 3 \\ 12(m-3), & m>3 \end{cases} \]
  \begin{block}{The question that carries today}
    One function, two rules --- a \textbf{piecewise-defined function}. Which rule does an input
    use? What does the graph do where the rule changes? And what happens when a rule ``rounds''?
  \end{block}
\end{frame}

% NOTES 1 — EVALUATE BY PIECE
\begin{frame}[t, plain]
  \forestheader{Notes 1: one function, several rules --- pick the piece first}
  \sectionlabel{Then apply its rule}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      $C(m)=\begin{cases} 0, & 0\le m\le 3 \\ 12(m-3), & m>3 \end{cases}$
      \begin{itemize}\setlength{\itemsep}{4pt}
        \item $C(2)$: $2\le3$, first piece: $0$.
        \item $C(3)$: the boundary. $3$ satisfies $0\le m\le3$, so first piece: $0$ --- month 3 is
              still free.
        \item $C(5)$: $5>3$, second piece: $12(2)=24$.
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      $f(x)=\begin{cases} 2x+1, & x<2 \\ 5-x, & x\ge2 \end{cases}$
      \begin{itemize}\setlength{\itemsep}{4pt}
        \item $f(0)=1$ (first piece)
        \item $f(2)=3$ --- the $\ge$ piece owns $2$
        \item $f(5)=0$
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{5pt}
  \begin{block}{Boundary rule}
    An input on the boundary belongs to the piece whose inequality \emph{includes} it --- the
    one with $\le$ or $\ge$, never the strict one.
  \end{block}
\end{frame}

% NOTES 2 — ABSOLUTE VALUE IS PIECEWISE
\begin{frame}[t, plain]
  \forestheader{Notes 2: absolute value is piecewise --- the V is two pieces}
  \sectionlabel{Lesson 1.3's trail, written with a brace}
  \begin{columns}[T]
    \begin{column}{0.40\textwidth}
      \[ |x|=\begin{cases} x, & x\ge0 \\ -x, & x<0 \end{cases} \]
      Check: $|-3|=-(-3)=3$.
    \end{column}
    \begin{column}{0.56\textwidth}
      $H(x)=2|x-3|+1$, boundary at the vertex $x=3$:
      \begin{align*}
        x\ge3:\quad H(x) &= 2(x-3)+1 = 2x-5 \\
        x<3:\quad H(x) &= 2(3-x)+1 = 7-2x
      \end{align*}
    \end{column}
  \end{columns}
  \vspace{4pt}
  \begin{block}{Meets at the vertex}
    Right ray slope $2$, left ray slope $-2$ --- and both rules give $H(3)=1$, so the two pieces
    \emph{meet}. That is the first ``meets'' example; the next slide shows a ``jumps.''
  \end{block}
\end{frame}

% NOTES 3 — MEETS OR JUMPS  (the target misconception)
\begin{frame}[t, plain]
  \forestheader{Notes 3: open or closed, meets or jumps}
  \sectionlabel[redacc]{Same story, two functions}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Total paid $C(m)$} \\[2pt]
      \centering
      \begin{tikzpicture}[scale=0.34]
        \mgrid
        \draw[very thick, forest] (0,0)--(3,0);
        \draw[very thick, forest] (3,0)--(7,4);
        \fdot{forest}{3}{0}
      \end{tikzpicture} \\[2pt]
      At $m=3$ both rules give $0$: the pieces \textbf{meet}. Constant on $[0,3]$, increasing after.
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Price this month $P(m)$} \\[2pt]
      \centering
      \begin{tikzpicture}[scale=0.34]
        \mgrid
        \draw[very thick, redacc] (0,0)--(3,0);
        \draw[very thick, redacc] (3,1)--(7,1);
        \fdot{redacc}{3}{0}
        \hdot{redacc}{3}{1}
      \end{tikzpicture} \\[2pt]
      At $m=3$ the rules give $0$ and $12$: the graph \textbf{jumps}. Range $\{0,12\}$.
    \end{column}
  \end{columns}
  \vspace{2pt}
  \begin{block}{Careful --- exactly one piece owns the boundary}
    ``$P(3)=12$, because the price is \$12 after the free months''? No: $3$ satisfies
    $0\le m\le3$, so $P(3)=0$ --- the closed dot says so. The open dot is not a point of the graph.
  \end{block}
\end{frame}

% NOTES 4 — STEP FUNCTIONS
\begin{frame}[t, plain]
  \forestheader{Notes 4: step functions --- the greatest-integer function}
  \sectionlabel{And your turn}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      $\lfloor x\rfloor$ = the greatest integer $\le x$: it \textbf{rounds down}.
      \begin{itemize}\setlength{\itemsep}{3pt}
        \item $\lfloor3.7\rfloor=3$, \; $\lfloor5\rfloor=5$, \; $\lfloor-0.4\rfloor=-1$
        \item $\lfloor-2.5\rfloor=-3$ --- down means \emph{more negative}
        \item Constant $n$ on $[n,n+1)$: closed left, open right, then a jump
        \item Age in whole years; full hours parked
      \end{itemize}
    \end{column}
    \begin{column}{0.40\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.42]
        \lgrid{-1.5}{4.5}{-1.5}{4}
        \draw[very thick, forest] (-1,-1)--(0,-1); \fdot{forest}{-1}{-1} \hdot{forest}{0}{-1}
        \draw[very thick, forest] (0,0)--(1,0);    \fdot{forest}{0}{0}   \hdot{forest}{1}{0}
        \draw[very thick, forest] (1,1)--(2,1);    \fdot{forest}{1}{1}   \hdot{forest}{2}{1}
        \draw[very thick, forest] (2,2)--(3,2);    \fdot{forest}{2}{2}   \hdot{forest}{3}{2}
        \draw[very thick, forest] (3,3)--(4,3);    \fdot{forest}{3}{3}   \hdot{forest}{4}{3}
      \end{tikzpicture}
    \end{column}
  \end{columns}
  \vspace{3pt}
  \textbf{Guided Practice, together:} $p(x)=-2x$ for $x\le-1$, $x+3$ for $x>-1$ --- evaluate
  at $-3,-1,0,2$; both rules give $2$ at $-1$, so it meets; read the intervals and the minimum
  off the graph.
\end{frame}

% INDIVIDUAL PRACTICE LAUNCH
\begin{frame}[t, plain]
  \forestheader{Individual Practice: on your own}
  \sectionlabel[goldacc]{15 minutes $\cdot$ silent $\cdot$ last page of your notes}
  \begin{enumerate}\setlength{\itemsep}{6pt}
    \item \textbf{Across the boundary.} Evaluate $h$ at $0$, $2$, $5$ --- then does $h$ meet or
          jump at $2$?
    \item \textbf{From the graph.} Domain, $f(1)$, $f(2)$ off the closed dot, the intervals ---
          then \emph{write} the rule.
    \item \textbf{Two pieces and the staircase.} $|x-3|$ as two pieces; then round down:
          $\lfloor2.9\rfloor$, $\lfloor-1.2\rfloor$, $\lfloor4\rfloor$.
  \end{enumerate}
  \vspace{5pt}
  \begin{block}{The rule}
    Find the piece first; at a boundary, ask which inequality \emph{includes} the input. Stuck?
    Look \emph{up the page} at the notes --- not at your neighbour. Done early? Change problem~1's
    second rule to $2x-5$: does $h$ meet at $2$ now?
  \end{block}
\end{frame}

% DEBRIEF
\begin{frame}[t, plain]
  \forestheader{Debrief: correct your own page in a second colour}
  \sectionlabel[redacc]{10 minutes}
  \begin{enumerate}\setlength{\itemsep}{5pt}
    \item \textbf{1 --- the piece first.} $0\le2$: $h(0)=1$. \; $2\le2$: $h(2)=-1$. \; $5>2$:
          $h(5)=4$. The other rule gives $-2$ at $2$, so $h$ \emph{jumps}. \checkmark
    \item \textbf{The must-land moment.} Two dots at $x=2$ in problem~2. The \emph{filled} one is
          $f(2)=-2$; the open one is where the slanted piece stops. The story cannot decide it ---
          the inequality does. Same as $P(3)=0$.
    \item \textbf{Meets or jumps --- one test.} Same value at the boundary means meet; different
          means jump. Guided Practice met ($2$ and $2$); problems~1 and~2 jump.
    \item \textbf{3 --- two pieces, rounding down.} $|x-3|$ is $x-3$ on the right, $3-x$ on the
          left. $\lfloor-1.2\rfloor=-2$: down means more negative. Constant on $[4,5)$.
  \end{enumerate}
\end{frame}
