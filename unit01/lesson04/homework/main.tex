\begin{notesbox}{Practice}
Show your work. To evaluate a piecewise function, first find the piece whose condition the input
satisfies --- at a boundary, the piece whose inequality \emph{includes} it --- then apply that
rule. On a graph, a closed dot includes its endpoint and an open dot excludes it.

\begin{enumerate}[leftmargin=1.9em, itemsep=10pt]
  \item \textbf{Evaluate.} $f(x)=\begin{cases} x+4, & x\le -1 \\ 2x, & x>-1 \end{cases}$
    \\[3pt]
    $f(-3)=$ \blank{1.1cm} \quad $f(-1)=$ \blank{1.1cm} \quad $f(0)=$ \blank{1.1cm} \quad
    $f(5)=$ \blank{1.1cm} \quad Which piece owns $x=-1$, and why? \blank{3.4cm}

  \boxguard[18]
  \item \textbf{Meets or jumps?} Two functions, one number apart:
    \[ g(x)=\begin{cases} x+2, & x<1 \\ 5-x, & x\ge1 \end{cases} \qquad\qquad
       k(x)=\begin{cases} x+2, & x<1 \\ 4-x, & x\ge1 \end{cases} \]
    \vspace{-8pt}
    \begin{center}
    \renewcommand{\arraystretch}{1.4}
    \begin{tabularx}{\linewidth}{@{}X|X@{}}
    \textbf{(a) $g$} & \textbf{(b) $k$} \\
    \hline
    At $x=1$: first rule \blank{0.8cm}, second rule \blank{0.8cm} & At $x=1$: first rule \blank{0.8cm}, second rule \blank{0.8cm} \\
    So the graph \blank{1.2cm} (meets / jumps) & So the graph \blank{1.2cm} (meets / jumps) \\
    \end{tabularx}
    \end{center}
    (c) Say in one sentence what decides whether two pieces meet at a boundary. \writelines{2}

  \boxguard[20]
  \item \textbf{Read the graph.} The graph shows \emph{all} of $f$. \\[2pt]
    \begin{minipage}[c]{0.60\linewidth}
    $f(-1)=$ \blank{0.8cm}; \; $f(1)=$ \blank{0.8cm} (read the \blank{1.0cm} dot); \\[3pt]
    domain \blank{1.6cm}; \; increasing on \blank{1.6cm}; \; decreasing on \blank{1.6cm}; \\[3pt]
    continuous at $x=1$? \blank{0.9cm} --- the pieces \blank{1.2cm} there.
    \end{minipage}\hfill
    \begin{minipage}[c]{0.36\linewidth}\centering
    \begin{tikzpicture}[scale=0.36]
      \lgrid{-3.5}{4.5}{-1.5}{5}
      \draw[very thick, forest] (-3,-1)--(1,3);
      \fdot{forest}{-3}{-1} \hdot{forest}{1}{3}
      \draw[very thick, forest] (1,4)--(4,1);
      \fdot{forest}{1}{4} \fdot{forest}{4}{1}
    \end{tikzpicture}
    \end{minipage}

  \boxguard[16]
  \item \textbf{The staircase.} Round \emph{down} each time.
    \\[3pt]
    $\lfloor 4.2\rfloor=$ \blank{0.9cm} \quad $\lfloor -3.1\rfloor=$ \blank{0.9cm} \quad
    $\lfloor 6\rfloor=$ \blank{0.9cm} \quad $\lfloor -0.5\rfloor=$ \blank{0.9cm} \quad
    For every $x$ in $[3,4)$, $\lfloor x\rfloor=$ \blank{0.9cm} \\[3pt]
    For which inputs is $\lfloor x\rfloor=x$ exactly? \blank{2.6cm} \; Is $\lfloor x\rfloor$
    \emph{ever} bigger than $x$? \blank{0.9cm}

  \boxguard[24]
  \item \textbf{Model it.} A phone plan costs \$30 for up to $5$ GB of data, then \$10 for each
        GB over $5$. Let $C(g)$ be the monthly cost for $g$ GB.
    \begin{enumerate}[label=(\alph*), leftmargin=1.9em, itemsep=4pt]
      \item Complete the rule:
            $C(g)=\begin{cases} \text{\blank{1.4cm}}, & 0\le g\le5 \\[3pt] 30+\text{\blank{2.4cm}}, & g>5 \end{cases}$
      \item Find $C(3)$ and $C(8)$, and say what $C(8)$ means.
        \begin{work}
          C(3) &= 30 \\
          C(8) &= 30 + 10(8-5) \\
               &= 30 + 30 \\
               &= 60
        \end{work}
        $C(8)$ means: \blank{7.0cm}
      \item Suppose the company also adds a \$10 \emph{surcharge} the moment you pass $5$ GB.
            Rewrite the second piece: $C(g)=$ \blank{3.0cm} for $g>5$. Do the pieces still meet at
            $g=5$? \blank{0.9cm} \; The graph now \blank{1.2cm} from \$\blank{0.8cm} to
            \$\blank{0.8cm}.
    \end{enumerate}

  \item \textbf{Multiple choice (SOL-style).} For
        $f(x)=\begin{cases} 2x+1, & x<2 \\ x-4, & x\ge2 \end{cases}$, what is $f(2)$?
    \begin{enumerate}[label=\Alph*., leftmargin=2.2em, itemsep=1pt]
      \item $5$
      \item $-2$
      \item $3$
      \item $f(2)$ is undefined
    \end{enumerate}
    Say in one sentence how you ruled out one of the others. \writelines{2}
\end{enumerate}
\end{notesbox}

\begin{extensionbox}
\textbf{Build and reason.}
\begin{enumerate}[label=(\alph*), leftmargin=1.9em, itemsep=6pt]
  \item A function is the constant $0$ for $x<0$, the line $y=2x$ for $0\le x\le2$, and the
        constant $4$ for $x>2$. Write it with a brace, then decide whether it is continuous at
        $x=0$ and at $x=2$.
        \par\writelines{2}
  \item A simple tax charges $10\%$ on income up to \$10{,}000 and $20\%$ on the amount above
        \$10{,}000. Complete
        $T(x)=\begin{cases} 0.10x, & 0\le x\le 10000 \\[2pt] 1000+\text{\blank{2.6cm}}, & x>10000 \end{cases}$
        and explain why $T$ is continuous at \$10{,}000. \par\writelines{2}
\end{enumerate}
\end{extensionbox}
