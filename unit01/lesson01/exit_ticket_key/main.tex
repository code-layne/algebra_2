No notes. Show enough work that a wrong answer can still earn a right idea.

\begin{enumerate}[leftmargin=1.9em, itemsep=10pt]
  \item \textbf{Evaluate} $2x^2-5x+|x-8|$ at $x=-3$. Substitute first, then work outward from the
        grouping symbols.

        \vspace{2pt}
        Substitute, then simplify one line at a time:
\begin{work}
  2x^2-5x+|x-8| &= 2(-3)^2-5(-3)+|{-3}-8| \\
                &= 2(9)+15+|{-11}|        \\
                &= 18+15+11
\end{work}
        Value: \ans{$44$}

  \item \textbf{Factor completely:} $3x^2-27$.

        \vspace{2pt}
        Step 1 --- pull out the GCF: \ans{$3(x^2-9)$}

        \vspace{6pt}
        Step 2 --- keep going: \ans{$3(x+3)(x-3)$}

        \vspace{6pt}
        \textbf{Check by expanding.} Multiply your factors back out and show you land on $3x^2-27$.
        \ansline{$(x+3)(x-3)=x^2-9$, and $3(x^2-9)=3x^2-27$.}

  \item \textbf{SOL-style multiple choice.} Which expression is equivalent to
        \TallMath{\frac{15x^4y^{-1}}{5x^2y^3}} \; ?
        \begin{enumerate}[label=\Alph*., leftmargin=1.9em, itemsep=2pt]
          \item $3x^2y^4$
          \item \TallMath{\frac{3x^2}{y^4}} \quad \textcolor{keyred}{\textbf{$\leftarrow$ correct}}
          \item \TallMath{\frac{3x^2}{y^2}}
          \item \TallMath{\frac{3x^6}{y^4}}
        \end{enumerate}

        \vspace{2pt}
        In one sentence, what did the \emph{wrong} answer you were most tempted by get wrong?
        \ansline{Answers vary --- A subtracts the $y$ exponents backwards, C loses the negative sign
        on $y^{-1}$, D adds the $x$ exponents instead of subtracting.}
\end{enumerate}
