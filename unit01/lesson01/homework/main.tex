\begin{notesbox}{Practice}
Show your work. Keep standard form in integers with $A \ge 0$.

\begin{enumerate}[leftmargin=1.9em, itemsep=11pt]
  \item \textbf{Slope from two points.} Compute each slope.
    \begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=4pt]
      \item $(2,3)$ and $(5,9)$:
        \begin{work}
          m &= \frac{9 - 3}{5 - 2} \\
            &= \frac{6}{3}         \\
            &= 2
        \end{work}
      \item $(-1,4)$ and $(3,-4)$:
        \begin{work}
          m &= \frac{-4 - 4}{3 - (-1)} \\
            &= \frac{-8}{4}            \\
            &= -2
        \end{work}
    \end{enumerate}

  \item \textbf{Write in slope-intercept form.}
    \begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=4pt]
      \item Slope $4$, $y$-intercept $(0,-3)$: \quad $y = \blank{3.0cm}$
      \item Slope $-3$, through $(2,1)$ --- use point-slope first, then simplify:
        \begin{work}
          y - 1 &= -3(x - 2) \\
          y - 1 &= -3x + 6   \\
              y &= -3x + 7
        \end{work}
    \end{enumerate}

  \item \textbf{Two points, all three forms.} Write the line through $(1,-1)$ and $(3,5)$.
    \begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=4pt]
      \item Slope: $m = \blank{2.0cm}$ \qquad Point-slope: \blank{4.2cm}
      \item Slope-intercept: \blank{3.2cm} \qquad Standard: \blank{3.2cm}
    \end{enumerate}

  \item \textbf{Convert.} A line is given in standard form as $4x - 2y = 10$.
    \begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=4pt]
      \item Rewrite it in slope-intercept form by solving for $y$:
        \begin{work}
          4x - 2y &= 10       \\
             - 2y &= -4x + 10 \\
                y &= 2x - 5
        \end{work}
      \item Its slope is \blank{1.8cm} and its $y$-intercept is $(0,\blank{1.0cm})$.
    \end{enumerate}

  \item \textbf{Transform the parent $y = x$.} Describe the stretch/reflection and the shift that
        produce each line.
    \begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=4pt]
      \item $y = 3x$: \blank{6.0cm}
      \item $y = -x + 4$: \blank{6.0cm}
    \end{enumerate}

  \item \textbf{Write from a graph.} Read two lattice points off the line, then write its equation
        in slope-intercept form.
    \begin{center}
    \begin{tikzpicture}[scale=0.5]
      \lgrid{-2}{4}{-4}{4}
      \draw[very thick, forest] (-1.5,4)--(2.5,-4);
      \fill[charcoal] (0,1) circle (3pt);
      \fill[charcoal] (1,-1) circle (3pt);
    \end{tikzpicture}
    \end{center}
    Slope $= \blank{1.8cm}$; \quad $y$-intercept $= \blank{1.8cm}$; \quad equation
    $y = \blank{3.0cm}$.

  \item \textbf{Explain.} Two \emph{different} lines are parallel. What must be true about their
        slopes, and what must be different about their equations in slope-intercept form?
        \writelines{3}
\end{enumerate}
\end{notesbox}

\begin{extensionbox}
\textbf{Parallel, perpendicular, and a model.}
\begin{enumerate}[label=(\alph*), leftmargin=1.9em, itemsep=6pt]
  \item A line is $y = \tfrac{1}{3}x - 2$. Write the line \emph{parallel} to it through $(0,4)$, and
        the line \emph{perpendicular} to it through $(0,4)$.
  \item A gym charges a \$20 joining fee plus \$15 per month: $C(m) = 15m + 20$. Interpret the slope
        and the $y$-intercept in context, and find the total cost after $6$ months.
\end{enumerate}
\writelines{3}
\end{extensionbox}
