\begin{enumerate}[leftmargin=*, itemsep=0pt]

    % ── Evaluate ──────────────────────────────────────────────────────────────
    \item Evaluate $3x - 2y$ when $x = 4$, $y = -1$.

    \ansline{$3(4) - 2(-1) = 12 + 2$}
    \ansline{$= 14$}
    Answer: \ans{$14$}

    \vspace{0.2in}

    \item Evaluate $x^2 + 2x - 5$ when $x = -3$.

    \ansline{$(-3)^2 + 2(-3) - 5 = 9 - 6 - 5$}
    \ansline{$= -2$}
    Answer: \ans{$-2$}

    \vspace{0.2in}

    % ── Simplify expressions ──────────────────────────────────────────────────
    \item Simplify: $5x^2 - 3x + 2x^2 + 7x$

    \ansline{$(5x^2 + 2x^2) + (-3x + 7x)$}
    \ansline{$= 7x^2 + 4x$}
    Answer: \ans{$7x^2 + 4x$}

    \vspace{0.2in}

    \item Simplify: $\dfrac{(x^2)^3}{x^4}$

    \ansline{$\dfrac{x^6}{x^4} = x^{6-4}$}
    \ansline{$= x^2$}
    Answer: \ans{$x^2$}

    \vspace{0.2in}

    \item Simplify: $\sqrt{75}$

    \ansline{$\sqrt{25 \cdot 3} = \sqrt{25}\cdot\sqrt{3}$}
    \ansline{$= 5\sqrt{3}$}
    Answer: \ans{$5\sqrt{3}$}

    \columnbreak

    % ── Factor ────────────────────────────────────────────────────────────────
    \item Factor: $x^2 - 5x + 6$

    \ansline{need two numbers: add to $-5$, multiply to $6$}
    \ansline{$-2$ and $-3$}
    Answer: \ans{$(x-2)(x-3)$}

    \vspace{0.2in}

    \item Factor: $x^2 - 16$

    \ansline{difference of squares: $a^2 - b^2 = (a+b)(a-b)$}
    \ansline{$x^2 - 4^2$}
    Answer: \ans{$(x+4)(x-4)$}

    \vspace{0.2in}

    % ── Polynomial operations ─────────────────────────────────────────────────
    \item Multiply: $(x + 3)(x - 5)$

    \ansline{$x^2 - 5x + 3x - 15$}
    \ansline{$= x^2 - 2x - 15$}
    Answer: \ans{$x^2 - 2x - 15$}

    \vspace{0.2in}

    \item Simplify: $3\sqrt{8} + \sqrt{2}$

    \ansline{$3\sqrt{4\cdot 2} + \sqrt{2} = 3(2\sqrt{2}) + \sqrt{2}$}
    \ansline{$= 6\sqrt{2} + \sqrt{2} = 7\sqrt{2}$}
    Answer: \ans{$7\sqrt{2}$}

    \vspace{0.2in}

    \item Factor: $2x^2 + 5x + 3$

    \ansline{slip-and-slide: $x^2+5x+6=(x+2)(x+3)$}
    \ansline{$\Rightarrow\!\left(x+\tfrac{2}{2}\right)\!\left(x+\tfrac{3}{2}\right)=(x+1)(2x+3)$}
    Answer: \ans{$(x+1)(2x+3)$}

\end{enumerate}
