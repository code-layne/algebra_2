    \normalsize
    \begin{tabularx}{0.88\textwidth}{@{} >{\bfseries}p{0.34\textwidth} X @{}}
      Product rule        & $x^a \cdot x^b = x^{a+b}$ \\
      Power rule          & $(x^a)^b = x^{a \cdot b}$ \\
      Power of a product  & $(x \cdot y)^a = x^a \cdot y^a$ \\
      Quotient rule       & $\dfrac{x^a}{x^b} = x^{a-b}$ \\
      Zero power rule     & $x^0 = 1$ \\
      Negative exponent   & $x^{-a} = \dfrac{1}{x^a}$ \\
    \end{tabularx}

        \small
        \textbf{Method 1: Largest perfect square}
        \begin{enumerate}
          \setlength{\itemsep}{0.3em}
          \item Find the largest perfect square factor (and variable pairs)
          \item Rewrite as $\sqrt{a \cdot b}$; apply $\sqrt{ab} = \sqrt{a}\cdot\sqrt{b}$
          \item Pull out $\sqrt{a}$; leave the rest inside
        \end{enumerate}
        \vspace{0.4em}
        \textbf{Method 2: Prime factorization}
        \begin{enumerate}
          \setlength{\itemsep}{0.3em}
          \item Write the full prime factorization
          \item Group factors in pairs; each pair exits as one factor
          \item Anything unpaired stays inside
        \end{enumerate}
      \end{column}
    \end{columns}
  \end{minipage}
\end{frame}
