% SLOPE
\begin{frame}[t, plain]
  \forestheader{Slope is the rate of change}
  \sectionlabel{Definition}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      For any two points on a line,
      \[ m = \frac{y_2 - y_1}{x_2 - x_1} = \frac{\text{rise}}{\text{run}}. \]
      Constant rate $\Rightarrow$ \emph{any} two points give the same $m$.
    \end{column}
    \begin{column}{0.44\textwidth}
      Using $(1,4)$ and $(3,0)$:
      \[ m = \frac{0 - 4}{3 - 1} = \frac{-4}{2} = -2. \]
      {\footnotesize Subtract $y$'s and $x$'s in the \textbf{same order}.}
    \end{column}
  \end{columns}
\end{frame}

% THREE FORMS
\begin{frame}[t, plain]
  \forestheader{Three forms, one line}
  \sectionlabel[goldacc]{$y = 2x - 4$}
  \begin{itemize}\setlength{\itemsep}{7pt}
    \item \textbf{Slope-intercept:} $y = mx + b$ \; --- reads off slope $2$ and $y$-intercept $(0,-4)$.
    \item \textbf{Point-slope:} $y - y_1 = m(x - x_1)$ \; --- from $(2,0)$: $y - 0 = 2(x-2)$.
    \item \textbf{Standard:} $Ax + By = C$ \; --- $2x - y = 4$ (integers, $A \ge 0$).
  \end{itemize}
  \vspace{2pt}
  \begin{block}{Same line, different outfits}
    Distribute the point-slope form: $y = 2(x-2) = 2x - 4$. All three are equal.
  \end{block}
\end{frame}

% WRITING EQUATIONS
\begin{frame}[t, plain]
  \forestheader{Writing a line: match the givens}
  \sectionlabel{The main event}
  \begin{itemize}\setlength{\itemsep}{7pt}
    \item \textbf{Slope $+$ $y$-intercept} $\to$ drop into $y = mx + b$.
    \item \textbf{Slope $+$ any point} $\to$ point-slope, then simplify.
    \item \textbf{Two points} $\to$ find $m$ first, then point-slope.
  \end{itemize}
  \vspace{2pt}
  \begin{block}{Worked: through $(1,4)$ and $(3,0)$}
    $m = -2$; \; $y - 0 = -2(x - 3)$; \; so $y = -2x + 6$, or $2x + y = 6$.
  \end{block}
\end{frame}

      \begin{itemize}\setlength{\itemsep}{5pt}
        \item vertical \textbf{stretch} by $2$ \; ($kf(x)$): slope $1 \to 2$;
        \item \textbf{shift down} $4$ \; ($f(x)+k$): intercept $0 \to -4$.
      \end{itemize}
      {\footnotesize A negative slope also \textbf{reflects} over the $x$-axis.}
    \end{column}
  \end{columns}
  \vspace{2pt}
  \begin{block}{Big idea}
    $kf(x)$ and $f(x)+k$ will graph \emph{every} function family this year.
  \end{block}
\end{frame}
