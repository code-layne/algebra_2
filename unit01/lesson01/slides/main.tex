% DEBRIEF 1 — slope, and slope-intercept form
\begin{frame}[t, plain]
  \forestheader{Debrief: naming what you found (1)}
  \sectionlabel[redacc]{Summit: \$12 over 4 climbs}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \small
      \begin{itemize}\setlength{\itemsep}{5pt}
        \item ``\$12 over 4 climbs'' $\to$ the \redink{slope}
              $m=\dfrac{\Delta y}{\Delta x}=\dfrac{y_2-y_1}{x_2-x_1}=\dfrac{18-6}{4-0}=3$.
        \item Why can you use \emph{any} two points? Because the rate is \redink{constant} ---
              that is what made it a line back in Lesson 1.0.
        \item Your $C(n) = 3n + 6$ is \redink{slope-intercept form}, $y = mx + b$: a
              \redink{rate} and a \redink{starting value}, each sitting where you can read it.
      \end{itemize}
    \end{column}
    \begin{column}{0.40\textwidth}

% DEBRIEF 2 — point-slope: the crux
\begin{frame}[t, plain]
  \forestheader{Same line, different outfit}
  \sectionlabel[goldacc]{Priya's rule --- your answer to 2d}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \small
      Priya never found Basecamp's walk-in fee. She started from the receipt she \emph{had}:
      \[ D - 24 = 2(n - 3) \]
      Your group started from the fee:
      \[ D = 2n + 18 \]
      \vspace{-6pt}
      \begin{block}{Same line?}
        At $n = 7$ both give \$32. That is evidence --- expanding hers is proof.
      \end{block}
    \end{column}
    \begin{column}{0.50\textwidth}
      \small
      \begin{align*}
        D - 24 &= 2(n-3) \\
        D - 24 &= 2n - 6 \\
             D &= 2n + 18
      \end{align*}
      \vspace{-4pt}
      \begin{itemize}\setlength{\itemsep}{4pt}
        \item Priya's form is \redink{point-slope}: $y - y_1 = m(x - x_1)$.
        \item It is built from a slope and \redink{any point at all} --- which matters, because the
              $y$-intercept is often the one point you \emph{do not} have.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% DEBRIEF 3 — standard form
\begin{frame}[t, plain]
  \forestheader{Debrief: a third outfit --- and it is mine, not yours}
  \sectionlabel[redacc]{Standard form: $Ax + By = C$}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \small
      \begin{itemize}\setlength{\itemsep}{5pt}
        \item Nothing in the activity produced this one, so here it is cold: \redink{standard form}
              $Ax + By = C$, with $A$, $B$, $C$ integers and $A \ge 0$.
        \item It hides the slope, but it hands you both \redink{intercepts} quickly --- set $x=0$,
              then set $y=0$.
        \item Three outfits, one line. Expand point-slope to get slope-intercept; clear fractions
              and move the $x$ across to get standard.
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \small
      Basecamp, all three ways:
      \begin{align*}
        D - 24 &= 2(n - 3) &&\text{point-slope}\\
             D &= 2n + 18  &&\text{slope-intercept}\\
        2n - D &= -18      &&\text{standard}
      \end{align*}
      \vspace{-4pt}
      \begin{block}{Read it off}
        In $2n - D = -18$: set $n=0$ and the walk-in fee \$18 falls straight out.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% DEBRIEF 4 — the parent line and the special cases
\begin{frame}[t, plain]
  \forestheader{Debrief: every line is $y=x$ in disguise}
  \sectionlabel[redacc]{Transformations, and the four special cases}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \small
      \begin{itemize}\setlength{\itemsep}{4pt}
        \item Start with the \redink{parent} $y = x$. Multiply by $k$: the line gets steeper,
              flatter, or flips --- that is the \redink{slope}. Add $k$: the whole line slides up
              or down --- that is the \redink{$y$-intercept}.
        \item \redink{Parallel} lines: equal slopes. \redink{Perpendicular} lines: opposite
              reciprocals, $m_1m_2 = -1$.
        \item Horizontal $y = c$: slope \redink{$0$}. \; Vertical $x = a$: slope
              \redink{undefined} --- the one line $y = mx + b$ cannot write.
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.34]
        \lgrid{-4}{5}{-6}{6}
        \draw[very thick, navy, dashed] (-4,-4)--(5,5) node[above left, font=\tiny, navy]{$y=x$};
        \draw[very thick, forest] (-1,-6)--(5,6) node[below right, font=\tiny, forest]{$y=2x-4$};
        \fill[redacc] (0,-4) circle (4pt);
        \fill[redacc] (2,0) circle (4pt);
      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}

% QUICKNOTES SUMMARY
\begin{frame}[t, plain]
  \forestheader{QuickNotes: three ways to write one line}
  \sectionlabel{Fill the box in your packet}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \footnotesize
      \begin{enumerate}\setlength{\itemsep}{3pt}
        \item \textbf{Slope:} $m=\dfrac{y_2-y_1}{x_2-x_1}$, from \emph{any} two points.
        \item \textbf{Slope-intercept:} $y = mx+b$ --- the rate and the starting value.
        \item \textbf{Point-slope:} $y-y_1 = m(x-x_1)$ --- a slope and any point.
        \item \textbf{Standard:} $Ax+By=C$ --- integers, $A\ge0$, tidy intercepts.
      \end{enumerate}
    \end{column}
    \begin{column}{0.50\textwidth}
      \footnotesize
      \begin{enumerate}\setlength{\itemsep}{3pt}
        \setcounter{enumi}{4}
        \item[5.] \textbf{Parent $y=x$:} $\times k$ changes the slope; $+k$ changes the intercept.
        \item[6.] \textbf{Parallel} = equal slopes. \; \textbf{Perpendicular} = opposite
              reciprocals.
        \item[7.] $y=c$ has slope $0$; \; $x=a$ has \emph{undefined} slope.
      \end{enumerate}
      \vspace{2pt}
      \begin{block}{The one to remember}
        Different-looking equations are \textbf{not} different lines. That was Priya.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% APPLICATION — worked together
\begin{frame}[t, plain]
  \forestheader{Application: the drone descent}
  \sectionlabel[goldacc]{We do this one together $\cdot$ 7 minutes}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \small
      A drone comes down at a steady rate. At $t = 2$~s it is $46$~m up; at $t = 6$~s it is $30$~m
      up.
      \begin{itemize}\setlength{\itemsep}{4pt}
        \item Find the rate. What does the \textbf{sign} tell you?
        \item Write it in \textbf{point-slope} from $(2,46)$, then expand. What does the number
              $54$ mean --- and did anyone measure it?
        \item Put the same line in \textbf{standard} form.
      \end{itemize}
      \vspace{2pt}
      \begin{block}{Now change one number}
        Same descent, but from $60$~m. In $h = mt + b$, \textbf{which number moves} --- and does
        the graph tilt or slide?
      \end{block}
    \end{column}
    \begin{column}{0.42\textwidth}
      \centering
      \begin{tikzpicture}[x=0.35cm, y=0.08cm]
        \draw[xstep=1, ystep=10, linegray!45, thin] (0,0) grid (13,60);
        \draw[->, thick, charcoal] (0,0)--(13.6,0) node[right, font=\tiny]{$t$};
        \draw[->, thick, charcoal] (0,0)--(0,64) node[above, font=\tiny]{$h$};
        \foreach \y in {20,40,60} \node[charcoal, font=\tiny, left] at (0,\y){\y};
