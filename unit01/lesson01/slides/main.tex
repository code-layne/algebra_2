% HOOK
\begin{frame}[t, plain]
  \forestheader{Hook: the bill that hides the monthly fee}
  \sectionlabel[goldacc]{Work backwards}
  \begin{itemize}\setlength{\itemsep}{5pt}
    \item Two months of bills: \textbf{4~GB cost \$52}, \textbf{10~GB cost \$70}.
    \item $6$ more GB for \$$18$ more \; $\Rightarrow$ \; \$$3$ per GB.
    \item So the flat fee is $52 - 4(3) = \$40$.
  \end{itemize}
  \vspace{6pt}
  \begin{block}{The point}
    Nobody handed you the starting value --- you \emph{worked back} to it. Today: build the equation
    of a line from a graph, from two points, or from a slope and \emph{any} point at all.
  \end{block}
\end{frame}

% NOTES 1 — SLOPE
\begin{frame}[t, plain]
  \forestheader{Notes 1: slope from any two points}
  \sectionlabel{Same order, top and bottom}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \[ m=\frac{\Delta y}{\Delta x}=\frac{y_2-y_1}{x_2-x_1} \]
      The phone plan gives the points $(4,52)$ and $(10,70)$:
      \[ m=\frac{70-52}{10-4}=\frac{18}{6}=3 \]
      \$3 per GB. \; \emph{Any} two points work --- the rate is constant (Lesson 1.0).
    \end{column}
    \begin{column}{0.42\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.40]
        \lgrid{-3}{5}{-6}{6}
        \draw[very thick, forest] (-2,-6)--(4,6);
        \draw[thick, navy, dashed] (1,0)--(2,0)--(2,2);
        \node[navy, font=\tiny] at (1.5,-0.8){run};
        \node[navy, font=\tiny, right] at (2.1,1){rise};
        \fill[charcoal] (0,-2) circle (4pt);
        \fill[charcoal] (3,4) circle (4pt);
      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}

% NOTES 2 — THREE FORMS
\begin{frame}[t, plain]
  \forestheader{Notes 2: three ways to write one line}
  \sectionlabel{The forms}
  \begin{itemize}\setlength{\itemsep}{5pt}
    \item \textbf{Slope-intercept} $y=mx+b$ --- hands you the rate and the starting value.
    \item \textbf{Point-slope} $y-y_1=m(x-x_1)$ --- built from a slope and \emph{any} point.
    \item \textbf{Standard} $Ax+By=C$ --- integers, $A\ge0$; tidy for reading intercepts.
  \end{itemize}
  \vspace{5pt}
  \begin{block}{Why point-slope exists}
    In the phone plan the $y$-intercept was the one number nobody gave you. Point-slope lets you
    write the line \emph{immediately}, from the point you already have:
    \; $y-52 = 3(x-4)$.
  \end{block}
\end{frame}

% NOTES 3 — CONVERT
\begin{frame}[t, plain]
  \forestheader{Notes 3: same line, different outfit}
  \sectionlabel[redacc]{Expand, then tidy}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Point-slope $\to$ slope-intercept}
      \begin{align*}
        y-52 &= 3(x-4) \\
        y-52 &= 3x-12 \\
           y &= 3x+40
      \end{align*}
      And there is the \$40 monthly fee.
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{$\to$ standard}
      \begin{align*}
             y &= 3x+40 \\
        -3x+y &= 40 \\
         3x-y &= -40
      \end{align*}
      Integers, and $A \ge 0$.
    \end{column}
  \end{columns}
  \vspace{4pt}
  \textbf{Check any conversion:} at $x=10$, slope-intercept gives \$70 --- and so did the bill.
\end{frame}

% NOTES 4 — PARENT & SPECIAL CASES
\begin{frame}[t, plain]
  \forestheader{Notes 4: the parent $y=x$, and four special cases}
  \sectionlabel{And your turn}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{itemize}\setlength{\itemsep}{3pt}
        \item Multiply by $k$: changes the \textbf{slope} (steeper, flatter, flipped).
        \item Add $k$: changes the \textbf{$y$-intercept}.
        \item \textbf{Parallel}: equal slopes. \; \textbf{Perpendicular}: opposite reciprocals,
              $m_1m_2=-1$.
        \item $y=c$: slope $0$. \; $x=a$: slope \emph{undefined} --- the one line $y=mx+b$ cannot
              write.
      \end{itemize}
    \end{column}
    \begin{column}{0.44\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.38]
        \lgrid{-4}{5}{-6}{6}
        \draw[very thick, navy, dashed] (-4,-4)--(5,5) node[above left, font=\tiny, navy]{$y=x$};
        \draw[very thick, forest] (-1,-6)--(5,6) node[below right, font=\tiny, forest]{$y=2x-4$};
        \fill[charcoal] (0,-4) circle (4pt);
      \end{tikzpicture}
    \end{column}
  \end{columns}
  \vspace{3pt}
  \textbf{Guided Practice, together:} slope $-\tfrac12$ through $(2,-5)$ --- point-slope, then
  slope-intercept, then the parallel and perpendicular slopes.
\end{frame}

% INDIVIDUAL PRACTICE LAUNCH
\begin{frame}[t, plain]
  \forestheader{Individual Practice: on your own}
  \sectionlabel[goldacc]{15 minutes $\cdot$ silent $\cdot$ last page of your notes}
  \begin{enumerate}\setlength{\itemsep}{6pt}
    \item \textbf{From two points.} $(-1,4)$ and $(3,-4)$: slope, then point-slope using $(3,-4)$,
          then slope-intercept.
    \item \textbf{From standard form.} Rewrite $3x-4y=12$ as $y=mx+b$. Name the slope and the
          $y$-intercept.
    \item \textbf{From a rate and a start.} Slope $\tfrac23$, crossing at $(0,5)$: slope-intercept,
          then \textbf{standard} --- integers, $A \ge 0$.
  \end{enumerate}
  \vspace{5pt}
  \begin{block}{The rule}
    Stuck? Look \emph{up the page} at the notes --- not at your neighbour. Done early? Redo
    problem~1 starting from $(-1,4)$ instead, and see whether you get the same line.
  \end{block}
\end{frame}

% DEBRIEF
\begin{frame}[t, plain]
  \forestheader{Debrief: correct your own page in a second colour}
  \sectionlabel[redacc]{10 minutes}
  \begin{enumerate}\setlength{\itemsep}{5pt}
    \item \textbf{1a --- same order, top and bottom.}
          $m=\dfrac{-4-4}{3-(-1)}=\dfrac{-8}{4}=-2$.
    \item \textbf{1b, the must-land moment.} From $(3,-4)$: $y+4=-2(x-3)$. From $(-1,4)$:
          $y-4=-2(x+1)$. \emph{Both} expand to $y=-2x+2$ --- same line, different outfit. That is
          why point-slope exists: the intercept is the point you did not have.
    \item \textbf{2 --- divide \emph{every} term.} $-4y=-3x+12 \Rightarrow y=\tfrac34x-3$, so
          $m=\tfrac34$ and $(0,-3)$. Check: $x=4$ gives $y=0$ in both forms.
    \item \textbf{3 --- clear the fraction, then fix the sign.} $3y=2x+15 \Rightarrow -2x+3y=15
          \Rightarrow 2x-3y=-15$. Standard form wants $A \ge 0$.
  \end{enumerate}
\end{frame}
