\begin{enumerate}

    \item Evaluate $3x^2 - 2y + 4z$ when $x = -2$, $y = 5$, $z = -3$.

    \begin{itemize}
        \item \ans{$3(-2)^2 - 2(5) + 4(-3)$}
        \item \ans{$= 3(4) - 10 - 12$}
        \item \ans{$= 12 - 10 - 12$}
        \item \ans{$= -10$}
    \end{itemize}

    \vspace{0.1in}

    \item Which expression is the equivalent, fully simplified form of
    \TallMath{\frac{x^3 \cdot x^5}{x^2}}?

    \begin{enumerate}[label=\Alph*.]
        \item \ans{$x^6$} \quad \textcolor{keyred}{\textbf{$\leftarrow$ correct}}
        \item $x^8$
        \item \TallMath{\frac{x^8}{x^2}}
        \item $x^{15}$
    \end{enumerate}

    \vspace{0.2in}

    \item Find the fully factored form of each expression.

    \begin{enumerate}[label=\alph*.]
        \item $x^2 + 9x + 20$

        \begin{itemize}
            \item What two numbers add to $9$ and multiply to $20$? \ans{$4$ and $5$}
            \item \ans{$(x+4)(x+5)$}
        \end{itemize}

        \vspace{0.2in}

        \item $6x^2 + 11x + 3$ \quad (slip-and-slide: $|a| > 1$)

        \begin{itemize}
            \item \ans{$= x^2 + 11x + 18$}
            \item What two numbers add to $11$ and multiply to $18$? \ans{$9$ and $2$}
            \item \ans{$= (x + 9)(x + 2)$}
            \item \ans{$= \left(x + \tfrac{9}{6}\right)\left(x + \tfrac{2}{6}\right)$}
            \item \ans{$= \left(x + \tfrac{3}{2}\right)\left(x + \tfrac{1}{3}\right)$}
            \item \ans{$= (2x + 3)(3x + 1)$}
        \end{itemize}

        \vspace{0.1in}
    \end{enumerate}

    \item Which expression is the equivalent, fully simplified form of $\sqrt{180}$?

    \begin{enumerate}[label=\Alph*.]
        \item \ans{$6\sqrt{5}$} \quad \textcolor{keyred}{\textbf{$\leftarrow$ correct}}
        \item $3\sqrt{20}$
        \item $\sqrt{36 \cdot 5}$
        \item $2\sqrt{45}$
    \end{enumerate}

\end{enumerate}
