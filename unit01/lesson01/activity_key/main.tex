% ─────────────────────────────────────────────
%  TIER R — Remediate
% ─────────────────────────────────────────────
\begin{tcolorbox}[colback=white,colframe=black!40,boxrule=0.7pt,arc=2mm,
    title={\bfseries Tier R — Remediate},fonttitle=\small]

\begin{enumerate}

    \item Evaluate $2x + 3y$ when $x = 4$ and $y = -2$.
    \begin{itemize}
        \item \ans{$2(4) + 3(-2)$}
        \item \ans{$= 8 - 6$}
        \item \ans{$= 2$}
    \end{itemize}

    \vspace{0.1in}

    \item Simplify $x^4 \cdot x^3$.
    \begin{itemize}
        \item \ans{$x^{4+3} = x^7$} \quad (product rule)
    \end{itemize}

    \vspace{0.1in}

    \item Factor $x^2 + 5x + 6$.
    \begin{itemize}
        \item What two numbers add to $5$ and multiply to $6$? \ans{$2$ and $3$}
        \item \ans{$(x+2)(x+3)$}
    \end{itemize}

    \vspace{0.1in}

    \item Which expression is the fully simplified form of $\sqrt{48}$?
    \begin{enumerate}[label=\Alph*.]
        \item \ans{$4\sqrt{3}$} \quad \textcolor{keyred}{\textbf{$\leftarrow$ correct}}
        \item $2\sqrt{12}$
        \item $\sqrt{16 \cdot 3}$
        \item $6\sqrt{2}$
    \end{enumerate}
    \begin{itemize}
        \item \ans{$\sqrt{48} = \sqrt{16 \cdot 3} = 4\sqrt{3}$}
    \end{itemize}

\end{enumerate}
\end{tcolorbox}

\vspace{0.2in}

% ─────────────────────────────────────────────
%  TIER A — Approaching Proficiency
% ─────────────────────────────────────────────
\begin{tcolorbox}[colback=white,colframe=black!40,boxrule=0.7pt,arc=2mm,
    title={\bfseries Tier A — Approaching Proficiency},fonttitle=\small]

\begin{enumerate}

    \item Evaluate $3x^2 - 2y + 4z$ when $x = -2$, $y = 5$, $z = -3$.
    \begin{itemize}
        \item \ans{$3(-2)^2 - 2(5) + 4(-3)$}
        \item \ans{$= 3(4) - 10 - 12$}
        \item \ans{$= 12 - 10 - 12$}
        \item \ans{$= -10$}
    \end{itemize}

    \vspace{0.1in}

    \item Which is the simplified form of \TallMath{\dfrac{x^3 \cdot x^5}{x^2}}?
    \begin{enumerate}[label=\Alph*.]
        \item \ans{$x^6$} \quad \textcolor{keyred}{\textbf{$\leftarrow$ correct}}
        \item $x^8$
        \item \TallMath{\dfrac{x^8}{x^2}}
        \item $x^{15}$
    \end{enumerate}
    \begin{itemize}
        \item \ans{$x^{3+5} = x^8$, then $x^{8-2} = x^6$}
    \end{itemize}

    \vspace{0.1in}

    \item Fully factored forms:
    \begin{enumerate}[label=\alph*.]
        \item $x^2 + 9x + 20$
        \begin{itemize}
            \item Add to $9$, multiply to $20$: \ans{$4$ and $5$}
            \item \ans{$(x+4)(x+5)$}
        \end{itemize}

        \vspace{0.1in}

        \item $6x^2 + 11x + 3$ \quad (slip-and-slide)
        \begin{itemize}
            \item \ans{$\to x^2 + 11x + 18$}
            \item Add to $11$, multiply to $18$: \ans{$9$ and $2$}
            \item \ans{$= (x+9)(x+2)$}
            \item \ans{$= \left(x + \tfrac{9}{6}\right)\!\left(x + \tfrac{2}{6}\right)$}
            \item \ans{$= \left(x + \tfrac{3}{2}\right)\!\left(x + \tfrac{1}{3}\right)$}
            \item \ans{$= (2x+3)(3x+1)$}
        \end{itemize}
    \end{enumerate}

    \vspace{0.1in}

    \item Which is the simplified form of $\sqrt{180}$?
    \begin{enumerate}[label=\Alph*.]
        \item \ans{$6\sqrt{5}$} \quad \textcolor{keyred}{\textbf{$\leftarrow$ correct}}
        \item $3\sqrt{20}$
        \item $\sqrt{36 \cdot 5}$
        \item $2\sqrt{45}$
    \end{enumerate}
    \begin{itemize}
        \item \ans{$\sqrt{180} = \sqrt{36 \cdot 5} = 6\sqrt{5}$}
    \end{itemize}

\end{enumerate}
\end{tcolorbox}

% ─────────────────────────────────────────────
%  TIER E — Extension
% ─────────────────────────────────────────────
\begin{tcolorbox}[colback=white,colframe=black!40,boxrule=0.7pt,arc=2mm,
    title={\bfseries Tier E --- Extension},fonttitle=\small]

\begin{enumerate}

    \item Evaluate \TallMath{\dfrac{2a^3 - b}{a^2 + b^2}} when $a = -3$, $b = 4$.
    \begin{itemize}
        \item \ans{$\dfrac{2(-3)^3 - 4}{(-3)^2 + 4^2}$}
        \item \ans{$= \dfrac{2(-27) - 4}{9 + 16}$}
        \item \ans{$= \dfrac{-54 - 4}{25}$}
        \item \ans{$= \dfrac{-58}{25}$}
    \end{itemize}

    \vspace{0.1in}

    \item Simplify \TallMath{\dfrac{(2x^3)^2 \cdot x^{-1}}{4x^5}}.
    \begin{itemize}
        \item \ans{$(2x^3)^2 = 4x^6$} \quad (power of a product)
        \item \ans{$\dfrac{4x^6 \cdot x^{-1}}{4x^5} = \dfrac{4x^5}{4x^5}$}
        \item \ans{$= x^0 = 1$}
    \end{itemize}

    \vspace{0.1in}

    \item Factor completely.
    \begin{enumerate}[label=\alph*.]
        \item $4x^2 - 25$
        \begin{itemize}
            \item difference of squares: $a^2 - b^2 = (a+b)(a-b)$
            \item \ans{$(2x+5)(2x-5)$}
        \end{itemize}

        \vspace{0.1in}

        \item $10x^2 - 13x - 3$ \quad (slip-and-slide)
        \begin{itemize}
            \item \ans{$\to x^2 - 13x - 30$}
            \item Add to $-13$, multiply to $-30$: \ans{$-15$ and $2$}
            \item \ans{$= (x - 15)(x + 2)$}
            \item \ans{$= \left(x - \tfrac{15}{10}\right)\!\left(x + \tfrac{2}{10}\right)$}
            \item \ans{$= \left(x - \tfrac{3}{2}\right)\!\left(x + \tfrac{1}{5}\right)$}
            \item \ans{$= (2x - 3)(5x + 1)$}
        \end{itemize}
    \end{enumerate}

    \vspace{0.1in}

    \item Simplify $\sqrt{75} - \sqrt{12}$.
    \begin{itemize}
        \item \ans{$\sqrt{75} = \sqrt{25 \cdot 3} = 5\sqrt{3}$}
        \item \ans{$\sqrt{12} = \sqrt{4 \cdot 3} = 2\sqrt{3}$}
        \item \ans{$5\sqrt{3} - 2\sqrt{3} = 3\sqrt{3}$}
    \end{itemize}

\end{enumerate}
\end{tcolorbox}
