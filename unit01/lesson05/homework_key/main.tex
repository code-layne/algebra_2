\begin{notesbox}{Practice}
Show your work. Read direction from the \emph{sign} of $r$ and strength from its \emph{size};
give every slope its units; and check each prediction against the range of the data.

\begin{enumerate}[leftmargin=1.9em, itemsep=10pt]
  \item \textbf{Match $r$ to the plot.} Assign each value $r\in\{\,0.98,\ -0.60,\ 0.05\,\}$ to a
        plot, and name each plot's direction.
    \begin{center}
    \begin{tikzpicture}[scale=0.32]
      \lgrid{-0.5}{6}{-0.5}{7}
      \dpt{1}{1}\dpt{2}{2}\dpt{3}{3}\dpt{4}{5}\dpt{5}{5}
      \node[font=\scriptsize\bfseries] at (2.5,-1.8){(P)};
    \end{tikzpicture}
    \hspace{0.4cm}
    \begin{tikzpicture}[scale=0.32]
      \lgrid{-0.5}{6}{-0.5}{7}
      \dpt{1}{5}\dpt{2}{3}\dpt{3}{4}\dpt{4}{2}\dpt{5}{3}
      \node[font=\scriptsize\bfseries] at (2.5,-1.8){(Q)};
    \end{tikzpicture}
    \hspace{0.4cm}
    \begin{tikzpicture}[scale=0.32]
      \lgrid{-0.5}{6}{-0.5}{7}
      \dpt{1}{3}\dpt{2}{5}\dpt{3}{2}\dpt{4}{4}\dpt{5}{3}
      \node[font=\scriptsize\bfseries] at (2.5,-1.8){(R)};
    \end{tikzpicture}
    \end{center}
    (P) $r=$ \ans{$0.98$}, direction \ans{positive} \quad (Q) $r=$ \ans{$-0.60$}, direction
    \ans{negative} \quad (R) $r=$ \ans{$0.05$}, direction \ans{none}

  \boxguard[16]
  \item \textbf{Sign versus size.} Two studies report $r=-0.85$ and $r=0.45$.
    \begin{center}
    \renewcommand{\arraystretch}{1.4}
    \begin{tabularx}{\linewidth}{@{}X|X@{}}
    \textbf{(a) $r=-0.85$} & \textbf{(b) $r=0.45$} \\
    \hline
    Direction \ans{negative}; \; strength \ans{strong} & Direction \ans{positive}; \; strength \ans{moderate} \\
    \end{tabularx}
    \end{center}
    (c) A classmate says ``$r=-0.85$ is the weaker one, because it is negative.'' What is wrong
    with that?
    \ansline{The sign only says which way the trend runs; strength is the \emph{size}, $|{-0.85}|=0.85$.}
    \ansline{$0.85>0.45$, so $r=-0.85$ is the stronger linear relationship --- it just slopes down.}

  \boxguard[24]
  \item \textbf{From a data table.} A biologist measures a seedling's height every five days.
    \begin{center}
    \renewcommand{\arraystretch}{1.2}
    \begin{tabular}{|l|c|c|c|c|c|c|c|}
    \hline
    day $x$ & $0$ & $5$ & $10$ & $15$ & $20$ & $25$ & $30$ \\ \hline
    height $y$ (cm) & $5$ & $10$ & $11$ & $18$ & $20$ & $26$ & $29$ \\ \hline
    \end{tabular}
    \end{center}
    Technology gives $\hat y=0.8x+5$ with $r=0.99$.
    \begin{enumerate}[label=(\alph*), leftmargin=1.9em, itemsep=4pt]
      \item The slope $0.8$ means: each additional \ans{day}, the height grows by about
            \ans{$0.8$} \ans{cm} (units!). \; The intercept $5$ is the predicted height on
            day \ans{$0$}.
      \item Predict the height on day $20$, then compare with the table.
        \begin{work}
          \hat y &= 0.8(20) + 5 \\
                 &= 16 + 5 \\
                 &= 21
        \end{work}
        The table says \ans{$20$} cm; the prediction is \ans{interpolation} (interpolation /
        extrapolation), and it is close.
      \item Predict the height on day $100$.
        \begin{work}
          \hat y &= 0.8(100) + 5 \\
                 &= 80 + 5 \\
                 &= 85
        \end{work}
        This is \ans{extrapolation}. Why should the biologist not trust it? \ans{day $100$ is far past the $0$--$30$ data; growth slows, so the line has broken}
    \end{enumerate}

  \boxguard[16]
  \item \textbf{The special values.} Say what the scatterplot looks like, or why it cannot happen.
    \begin{enumerate}[label=(\alph*), leftmargin=1.9em, itemsep=4pt]
      \item $r=1$ exactly: the points \ans{lie exactly on one rising line}
      \item $r=0$: does this prove the two variables are unrelated? \ans{no} \; A tight
            \emph{curved} pattern can have $r\approx0$, because $r$ measures only a
            \ans{linear} fit.
      \item A study reports $r=1.3$. What must be true? \ans{impossible --- $r$ is always between $-1$ and $1$}
    \end{enumerate}

  \boxguard[18]
  \item \textbf{Correlation versus causation.} Across an elementary school, \emph{shoe size} and
        \emph{reading level} have a strong positive correlation.
    \begin{enumerate}[label=(\alph*), leftmargin=1.9em, itemsep=4pt]
      \item Does a bigger shoe \emph{cause} better reading? \ans{no} \; Name the lurking
            variable that pushes both up: \ans{age}
      \item If the study looked only at $9$-year-olds, would you expect the correlation to stay
            strong? Explain.
            \ansline{No --- with age held fixed, the lurking variable can no longer push both up together.}
            \ansline{Among children of one age, shoe size tells you little about reading level.}
    \end{enumerate}

  \item \textbf{Multiple choice (SOL-style).} Which correlation coefficient indicates the
        \emph{strongest} linear relationship?
    \begin{enumerate}[label=\Alph*., leftmargin=2.2em, itemsep=1pt]
      \item $r=0.35$
      \item $r=-0.82$ \hfill \textcolor{keyred}{\textbf{$\leftarrow$ correct}}
      \item $r=0.60$
      \item $r=-0.10$
    \end{enumerate}
    Say in one sentence how you ruled out one of the others.
    \ansline{Strength is $|r|$: $0.82$ beats $0.60$, $0.35$, and $0.10$ --- the minus sign is direction, not weakness.}
    \ansline{C is the largest \emph{positive} value, but $|{-0.82}|>0.60$, so C is not the strongest.}
\end{enumerate}
\end{notesbox}

\begin{extensionbox}
\textbf{Which line fits better?} For the five data points below, two candidate lines are proposed:
$y_1=1.5x+0.5$ and $y_2=x+2$. Fill in each line's predictions and its \emph{error} (error $=$
data $-$ prediction) at every point.
\begin{center}
\renewcommand{\arraystretch}{1.25}
\begin{tabular}{|l|c|c|c|c|c|}
\hline
$x$ & $1$ & $2$ & $3$ & $4$ & $5$ \\ \hline
data $y$ & $2$ & $3$ & $5$ & $6$ & $8$ \\ \hline
$y_1=1.5x+0.5$ & \ans{$2$} & \ans{$3.5$} & \ans{$5$} & \ans{$6.5$} & \ans{$8$} \\ \hline
error & \ans{$0$} & \ans{$-0.5$} & \ans{$0$} & \ans{$-0.5$} & \ans{$0$} \\ \hline
$y_2=x+2$ & \ans{$3$} & \ans{$4$} & \ans{$5$} & \ans{$6$} & \ans{$7$} \\ \hline
error & \ans{$-1$} & \ans{$-1$} & \ans{$0$} & \ans{$0$} & \ans{$1$} \\ \hline
\end{tabular}
\end{center}
Total error \emph{size} (add the absolute values): \; $y_1$: \ans{$1.0$} \quad $y_2$:
\ans{$3.0$}. \; The better fit is \ans{$y_1$}, because its predictions are \ans{closest}
to the data overall --- which is exactly what technology minimises.
\end{extensionbox}
