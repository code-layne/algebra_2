\vspace{4pt}
\textbf{3. Slope and direction.} For $y=ax+b$, the slope is $a$ and the $y$-intercept is $b$.
\begin{itemize}[leftmargin=1.6em, itemsep=3pt]
  \item For $y=3x-5$: slope $=$ \ans{$3$}, \; $y$-intercept $=$ \ans{$-5$}.
  \item A line with a \emph{positive} slope goes \ans{up} (up / down) as you read left to right;
        a \emph{negative} slope goes \ans{down} (up / down).
\end{itemize}
