\vspace{6pt}
\textbf{3. Slope and direction.} For $y=ax+b$, the slope is $a$ and the $y$-intercept is $b$.
\begin{itemize}[leftmargin=1.6em, itemsep=4pt]
  \item For $y=3x-5$: \; slope $=$ \ans{$3$}, \; $y$-intercept $=$ \ans{$-5$}.
  \item A line with a \emph{positive} slope goes \ans{up} (up / down) as you read left to
        right; a \emph{negative} slope goes \ans{down}.
  \item In $y=3x-5$, each time $x$ goes up by $1$, $y$ goes up by \ans{$3$}.
\end{itemize}
