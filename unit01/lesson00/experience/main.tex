\begin{headlinebox}{forestbg}
\textbf{Freezing Point.} On a cold weekend the number that matters most is \emph{zero}: above it
the roads are wet, below it they are ice. With your group, work through the two temperature
stories below using what you already know --- tables, rules, and points on a graph --- and be
ready to explain \emph{how you know} each answer from the \textbf{graph}.
\end{headlinebox}

\vspace{0.04in}

\begin{minipage}[t]{0.62\linewidth}
\vspace{0pt}
\begin{itemize}[leftmargin=1.1em, itemsep=8pt]
  \item A \textbf{linear function} has a \blank{2.4cm} rate of change, so its graph is a
        \blank{2.4cm}. Form: $f(x) = mx + b$.
  \item \textbf{Slope} $m = \dfrac{\Delta y}{\Delta x} = \dfrac{\text{rise}}{\text{run}}$.
        For $f$: $m = \blank{1.0cm}$.
  \item \textbf{$y$-intercept} $(0,b)$: where the graph crosses the $y$-axis (set $x=0$).
        For $f$: $(0,\blank{0.9cm})$.
  \item \textbf{$x$-intercept / zero} $(a,0)$: where it crosses the $x$-axis; the input with
        $f(a)=0$ (solve $f(x)=0$). For $f$: $(\blank{0.9cm},0)$.
\end{itemize}
\end{minipage}

\vspace{8pt}
\begin{itemize}[leftmargin=1.1em, itemsep=8pt]
  \item \textbf{Increasing / decreasing:} read left to right. \\[4pt]
        \mbox{$m>0$: \blank{2.2cm}} \quad \mbox{$m<0$: \blank{2.2cm}} \quad
        \mbox{$m=0$: \blank{2.2cm}}
  \item \textbf{Positive / negative:} $f(x)>0$ where the graph is \blank{1.8cm} the $x$-axis;
        $f(x)<0$ where it is \blank{1.8cm}. The sign can only switch at a \blank{1.8cm}. \\[4pt]
        For $f$: positive when $x \blank{1.4cm}$, negative when $x \blank{1.4cm}$. \\[4pt]
        \emph{``Positive'' (where it sits) and ``increasing'' (which way it moves) are different
        questions} --- Sunday's line was positive \emph{and} falling.
  \item \textbf{Domain / range} of a line that is not horizontal: \blank{2.8cm} /
        \blank{2.8cm}. A horizontal line $f(x)=b$ has range \blank{1.6cm} and no zero (unless
        $b=0$). In a \emph{story}, the domain is only the inputs that make sense.
\end{itemize}
