\begin{notesbox}{Practice: Reading Linear Functions}
Eight problems. For every graph, table, or rule, read the features \emph{and} be able to say how you
know. Show your thinking where there is room.

\begin{enumerate}[leftmargin=1.9em, itemsep=10pt]
  \item \textbf{An increasing line.} Read every feature of the graph.
    \begin{center}
    \begin{minipage}[c]{0.38\linewidth}\centering
    \begin{tikzpicture}[scale=0.42]
      \lgrid{-4}{4}{-4}{6}
      \draw[very thick, forest] (-3,-4)--(2,6);
      \fill[charcoal] (0,2) circle (3pt);
      \fill[charcoal] (-1,0) circle (3pt);
    \end{tikzpicture}
    \end{minipage}\hfill
    \begin{minipage}[c]{0.58\linewidth}
    \renewcommand{\arraystretch}{1.45}
    \begin{tabular}{@{}l l@{}}
      Domain: \blank{2.4cm} & Range: \blank{2.4cm} \\
      $y$-intercept: \blank{2.0cm} & Zero: \blank{2.0cm} \\
      Slope: \blank{2.0cm} & Incr.\ or decr.? \blank{2.0cm} \\
      Positive when $x$ \blank{1.6cm} & Negative when $x$ \blank{1.6cm}
    \end{tabular}
    \end{minipage}
    \end{center}

  \item \textbf{A decreasing line.} Read every feature of the graph.
    \begin{center}
    \begin{minipage}[c]{0.38\linewidth}\centering
    \begin{tikzpicture}[scale=0.42]
      \lgrid{-3}{4}{-4}{5}
      \draw[very thick, forest] (-1.5,5)--(3,-4);
      \fill[charcoal] (0,2) circle (3pt);
      \fill[charcoal] (1,0) circle (3pt);
    \end{tikzpicture}
    \end{minipage}\hfill
    \begin{minipage}[c]{0.58\linewidth}
    \renewcommand{\arraystretch}{1.45}
    \begin{tabular}{@{}l l@{}}
      Domain: \blank{2.4cm} & Range: \blank{2.4cm} \\
      $y$-intercept: \blank{2.0cm} & Zero: \blank{2.0cm} \\
      Slope: \blank{2.0cm} & Incr.\ or decr.? \blank{2.0cm} \\
      Positive when $x$ \blank{1.6cm} & Negative when $x$ \blank{1.6cm}
    \end{tabular}
    \end{minipage}
    \end{center}

  \item \textbf{From a table.} A linear function is given by the table below. (Watch the step in $x$!)
    \begin{center}
    \renewcommand{\arraystretch}{1.25}
    \begin{tabular}{|c|c|c|c|c|}
    \hline
    $x$ & $0$ & $2$ & $4$ & $6$ \\ \hline
    $f(x)$ & $-6$ & $-3$ & $0$ & $3$ \\ \hline
    \end{tabular}
    \end{center}
    Slope $m = \dfrac{\Delta y}{\Delta x} = \blank{1.6cm}$ \quad $y$-intercept: \blank{1.8cm} \quad
    zero: \blank{1.8cm} \\[3pt]
    Increasing or decreasing? \blank{2.4cm} \quad $f(x)<0$ when $x$ \blank{1.6cm}

  \item \textbf{In context.} A pool already holds $600$ gallons when a hose is turned on; it adds
        $150$ gallons per minute, so the volume after $t$ minutes is $W(t) = 600 + 150t$.
    \begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=4pt]
      \item The slope $150$ means: \blank{7.4cm}
      \item The $y$-intercept $(0,600)$ means: \blank{6.8cm}
      \item Find the zero of $W$ by solving $W(t)=0$.
        \begin{work}
          600 + 150t &= 0 \\
                150t &= -600 \\
                   t &= -4
        \end{work}
        Does this zero mean anything in the story? Explain. \writelines{2}
      \item Which inputs $t$ make sense here, and is $W$ increasing or decreasing on them?
            \blank{6.0cm}
    \end{enumerate}

  \item \textbf{A horizontal line.} The graph of $c(x) = -2$ is a flat line through $(0,-2)$.
    \begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=4pt]
      \item What is its slope? \blank{1.6cm} \quad Is it increasing, decreasing, or constant?
            \blank{2.6cm}
      \item What is its range? \blank{2.6cm} \quad Does it have a zero? \blank{2.2cm}
      \item Is $c(x)$ positive or negative? \blank{3.0cm}
    \end{enumerate}

  \item \textbf{Same zero, opposite behavior.} Both lines below have their zero at $x=1$.
    \begin{center}
    \begin{minipage}[c]{0.38\linewidth}\centering
    \begin{tikzpicture}[scale=0.42]
      \lgrid{-3}{5}{-4}{4}
      \draw[very thick, forest] (-2.5,-3.5)--(4.5,3.5) node[above left, font=\scriptsize, forest]{$f$};
      \draw[very thick, navy] (-2.5,3.5)--(4.5,-3.5) node[below left, font=\scriptsize, navy]{$g$};
      \fill[charcoal] (1,0) circle (3pt);
    \end{tikzpicture}
    \end{minipage}\hfill
    \begin{minipage}[c]{0.58\linewidth}
    \begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=4pt]
      \item Which line is increasing? \blank{1.0cm} \; Which is decreasing? \blank{1.0cm}
      \item $f(x)>0$ when $x$ \blank{1.4cm} \qquad $g(x)>0$ when $x$ \blank{1.4cm}
      \item At $x=3$, is $f$ positive or negative? \blank{1.8cm} \; Is $g$? \blank{1.8cm}
    \end{enumerate}
    \end{minipage}
    \end{center}

  \item \textbf{Explain.} A line is not horizontal. Why does the sign of $f(x)$ (positive vs.\
        negative) change at the \emph{zero} and nowhere else? \writelines{3}

  \item \textbf{Multiple choice.} Consider the linear function $y = -3x + 6$. Which statement is
        true?
    \begin{enumerate}[label=\Alph*., leftmargin=2.2em, itemsep=1pt]
      \item The function is increasing.
      \item The function is positive for all $x > 2$.
      \item The zero of the function is at $x = 2$.
      \item The $y$-intercept is $(0,-3)$.
    \end{enumerate}
    Explain why each of the \emph{other} three statements is false. \writelines{3}
\end{enumerate}
\end{notesbox}
