% DEBRIEF 1 — intercepts and slope
\begin{frame}[t, plain]
  \forestheader{Debrief: naming what you found (1)}
  \sectionlabel[redacc]{Saturday: $T(h)=2h-6$}
  \begin{columns}[T]
    \begin{column}{0.60\textwidth}
      \small
      \begin{itemize}\setlength{\itemsep}{5pt}
        \item ``Same $+2$ every hour'' $\to$ \redink{constant rate of change}. A rule with one is a
              \redink{linear function}, $f(x)=mx+b$; the constant is the \redink{slope}
              $m=\dfrac{\Delta y}{\Delta x}=\dfrac{\text{rise}}{\text{run}}$.
        \item The crossing on the vertical axis, $(0,-6)$ $\to$ the \redink{$y$-intercept} $(0,b)$:
              the starting value. Set $x=0$.
        \item The crossing on the horizontal axis, $(3,0)$ $\to$ the \redink{$x$-intercept}, the
              \redink{zero}: the input where the output is $0$. Solve $f(x)=0$.
      \end{itemize}
    \end{column}
    \begin{column}{0.38\textwidth}
      \centering
      \begin{tikzpicture}[x=0.42cm, y=0.2cm]
        \tempgrid
        \draw[very thick, forest] (0,-6)--(7,8);
        \draw[semithick, redacc, dashed] (4,2)--(5,2)--(5,4);
        \node[redacc, font=\tiny, right] at (5.05,3){$+2$};
        \fill[redacc] (0,-6) circle (3pt) node[right, font=\tiny, redacc]{$(0,-6)$};
        \fill[redacc] (3,0) circle (3pt)  node[above left, font=\tiny, redacc]{$(3,0)$};
      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}

% DEBRIEF 2 — direction and sign
\begin{frame}[t, plain]
  \forestheader{Debrief: naming what you found (2)}
  \sectionlabel[redacc]{Saturday vs.\ Sunday --- same zero, opposite behavior}
  \begin{columns}[T]
    \begin{column}{0.60\textwidth}
      \small
      \begin{itemize}\setlength{\itemsep}{5pt}
        \item ``Going up'' / ``going down'' $\to$ \redink{increasing} / \redink{decreasing}, read
              left to right. Decided by the \redink{sign of the slope}: $+2$ vs.\ $-2$.
        \item ``Above freezing'' / ``below freezing'' $\to$ \redink{positive} ($f(x)>0$, graph above
              the axis) / \redink{negative} ($f(x)<0$, below). The sign can only switch at a
              \redink{zero}.
        \item Saturday: negative \emph{before} $h=3$, positive after. Sunday: positive before
              $h=3$, negative after.
      \end{itemize}
    \end{column}

% DEBRIEF 3 — positive is not increasing
\begin{frame}[t, plain]
  \forestheader{``Positive'' is not ``increasing''}
  \sectionlabel[goldacc]{The key distinction --- your answer to 2c}
  \begin{itemize}\setlength{\itemsep}{7pt}
    \item At 5 p.m.\ Sunday the temperature is $4^\circ$: \textbf{positive}. And it is
          \textbf{falling}. Both at once.
    \item \textbf{Positive / negative} asks where the graph \emph{sits}: above or below the
          $x$-axis. It switches at the \textbf{zero}.
    \item \textbf{Increasing / decreasing} asks which way the graph \emph{moves} as you read
          left to right --- the sign of the \textbf{slope}.
  \end{itemize}
  \vspace{2pt}
  \begin{block}{They are independent}
    Sunday's line $T(h)=6-2h$ falls everywhere, yet $T>0$ for all $h<3$. A line can be positive
    and decreasing, or negative and increasing.
  \end{block}
\end{frame}

% DEBRIEF 4 — flat line, domain and range
\begin{frame}[t, plain]
  \forestheader{Debrief: the flat line, and the story vs.\ the math}
  \sectionlabel[redacc]{One example from me, one question from you}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \small
      \begin{itemize}\setlength{\itemsep}{5pt}
        \item My example: a thermostat holds $T(h)=21$ all day. Change per hour $0$ $\to$ slope
              $0$, a \redink{constant function}. Never crosses: \redink{no zero}, positive
              everywhere. Only one output: \redink{range $\{21\}$}.
        \item Your 2e: the \emph{story} accepts only the evening hours, but the \emph{rule}
              accepts any real number. \redink{Domain} = allowed inputs, \redink{range} =
              resulting outputs.
        \item A line that is not horizontal: domain all reals, range all reals. In a story, keep only
              the inputs that make sense.
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \centering

% QUICKNOTES SUMMARY
\begin{frame}[t, plain]
  \forestheader{QuickNotes: six characteristics of $f(x)=2x-4$}
  \sectionlabel{Fill the box in your packet}
  \begin{columns}[T]
    \begin{column}{0.40\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.40]
        \lgrid{-2}{6}{-6}{6}
        \draw[very thick, forest] (-1,-6)--(5,6);
        \draw[semithick, navy, dashed] (2,0)--(3,0)--(3,2);
        \fill[charcoal] (0,-4) circle (3.5pt) node[left, font=\tiny]{$(0,-4)$};
        \fill[charcoal] (2,0) circle (3.5pt) node[above left, font=\tiny]{$(2,0)$};
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.58\textwidth}
      \footnotesize
      \begin{enumerate}\setlength{\itemsep}{3pt}
        \item \textbf{Slope:} $\tfrac{\text{rise}}{\text{run}}=\tfrac{2}{1}=2$ (constant)
        \item \textbf{$y$-intercept:} $(0,-4)$
        \item \textbf{$x$-intercept / zero:} $(2,0)$, where $f(x)=0$
        \item \textbf{Increasing} (slope $>0$)
        \item \textbf{Positive} for $x>2$, \textbf{negative} for $x<2$
        \item \textbf{Domain:} all reals \quad \textbf{Range:} all reals
      \end{enumerate}
    \end{column}
  \end{columns}
\end{frame}

% APPLICATION — worked together
\begin{frame}[t, plain]
  \forestheader{Application: the car-wash card}
  \sectionlabel[goldacc]{We do this one together $\cdot$ 7 minutes}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \small
      A gift card starts with \textbf{\$40}. Each wash costs \textbf{\$8}, so after $w$ washes the
      balance is $B(w) = 40 - 8w$.
      \begin{itemize}\setlength{\itemsep}{4pt}
        \item What does the \textbf{slope} $-8$ mean here? The \textbf{$y$-intercept} $(0,40)$?
        \item Solve $B(w) = 0$. What does the \textbf{zero} mean --- in \emph{washes}?
        \item Which inputs $w$ make sense? Is $B$ increasing or decreasing on them?
      \end{itemize}
      \vspace{2pt}
      \begin{block}{Now change one number}
        A wash costs \textbf{\$10} instead. Sooner or later? And on the graph --- does the line
        \emph{start} somewhere else, get \emph{steeper}, or both?
      \end{block}
