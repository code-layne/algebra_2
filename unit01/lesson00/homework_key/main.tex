\begin{notesbox}{Practice: Reading Linear Functions}
Eight problems. For every graph, table, or rule, read the features \emph{and} be able to say how you
know. Show your thinking where there is room.

\begin{enumerate}[leftmargin=1.9em, itemsep=10pt]
  \item \textbf{An increasing line.} Read every feature of the graph.
    \begin{center}
    \begin{minipage}[c]{0.38\linewidth}\centering
    \begin{tikzpicture}[scale=0.42]
      \lgrid{-4}{4}{-4}{6}
      \draw[very thick, forest] (-3,-4)--(2,6);
      \fill[charcoal] (0,2) circle (3pt);
      \fill[charcoal] (-1,0) circle (3pt);
    \end{tikzpicture}
    \end{minipage}\hfill
    \begin{minipage}[c]{0.58\linewidth}
    \renewcommand{\arraystretch}{1.45}
    \begin{tabular}{@{}l l@{}}
      Domain: \ans{all reals} & Range: \ans{all reals} \\
      $y$-intercept: \ans{$(0,2)$} & Zero: \ans{$(-1,0)$} \\
      Slope: \ans{$2$} & Incr.\ or decr.? \ans{increasing} \\
      Positive when $x$ \ans{$>-1$} & Negative when $x$ \ans{$<-1$}
    \end{tabular}
    \end{minipage}
    \end{center}

  \item \textbf{A decreasing line.} Read every feature of the graph.
    \begin{center}
    \begin{minipage}[c]{0.38\linewidth}\centering
    \begin{tikzpicture}[scale=0.42]
      \lgrid{-3}{4}{-4}{5}
      \draw[very thick, forest] (-1.5,5)--(3,-4);
      \fill[charcoal] (0,2) circle (3pt);
      \fill[charcoal] (1,0) circle (3pt);
    \end{tikzpicture}
    \end{minipage}\hfill
    \begin{minipage}[c]{0.58\linewidth}
    \renewcommand{\arraystretch}{1.45}
    \begin{tabular}{@{}l l@{}}
      Domain: \ans{all reals} & Range: \ans{all reals} \\
      $y$-intercept: \ans{$(0,2)$} & Zero: \ans{$(1,0)$} \\
      Slope: \ans{$-2$} & Incr.\ or decr.? \ans{decreasing} \\
      Positive when $x$ \ans{$<1$} & Negative when $x$ \ans{$>1$}
    \end{tabular}
    \end{minipage}
    \end{center}

  \item \textbf{From a table.} A linear function is given by the table below. (Watch the step in $x$!)
    \begin{center}
    \renewcommand{\arraystretch}{1.25}
    \begin{tabular}{|c|c|c|c|c|}
    \hline
    $x$ & $0$ & $2$ & $4$ & $6$ \\ \hline
    $f(x)$ & $-6$ & $-3$ & $0$ & $3$ \\ \hline
    \end{tabular}
    \end{center}
    Slope $m = \dfrac{\Delta y}{\Delta x} = {\color{keyred}\mathbf{\tfrac{3}{2}}}$ \quad $y$-intercept: \ans{$(0,-6)$} \quad
    zero: \ans{$(4,0)$} \\[3pt]
    Increasing or decreasing? \ans{increasing} \quad $f(x)<0$ when $x$ \ans{$<4$}

  \item \textbf{In context.} A pool already holds $600$ gallons when a hose is turned on; it adds
        $150$ gallons per minute, so the volume after $t$ minutes is $W(t) = 600 + 150t$.
    \begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=4pt]
      \item The slope $150$ means: \ans{the pool gains $150$ gallons every minute}
      \item The $y$-intercept $(0,600)$ means: \ans{when the hose is turned on ($t=0$) the pool already holds $600$ gal}
      \item Find the zero of $W$ by solving $W(t)=0$.
        \begin{work}
          600 + 150t &= 0 \\
                150t &= -600 \\
                   t &= -4
        \end{work}
        Does this zero mean anything in the story?
        \ansline{No. $t=-4$ is four minutes \emph{before} the hose was turned on, which is outside the story --- the pool was never empty while we were filling it.}
      \item Which inputs $t$ make sense here, and is $W$ increasing or decreasing on them?
            \ans{$t\ge 0$ (until the pool is full); increasing}
    \end{enumerate}

  \item \textbf{A horizontal line.} The graph of $c(x) = -2$ is a flat line through $(0,-2)$.
    \begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=4pt]
      \item What is its slope? \ans{$0$} \quad Is it increasing, decreasing, or constant?
            \ans{constant}
      \item What is its range? \ans{$\{-2\}$} \quad Does it have a zero? \ans{no}
      \item Is $c(x)$ positive or negative? \ans{negative everywhere}
    \end{enumerate}

  \item \textbf{Same zero, opposite behavior.} Both lines below have their zero at $x=1$.
    \begin{center}
    \begin{minipage}[c]{0.38\linewidth}\centering
    \begin{tikzpicture}[scale=0.42]
      \lgrid{-3}{5}{-4}{4}
      \draw[very thick, forest] (-2.5,-3.5)--(4.5,3.5) node[above left, font=\scriptsize, forest]{$f$};
      \draw[very thick, navy] (-2.5,3.5)--(4.5,-3.5) node[below left, font=\scriptsize, navy]{$g$};
      \fill[charcoal] (1,0) circle (3pt);
    \end{tikzpicture}
    \end{minipage}\hfill
    \begin{minipage}[c]{0.58\linewidth}
    \begin{enumerate}[label=(\alph*), leftmargin=1.8em, itemsep=4pt]
      \item Which line is increasing? \ans{$f$} \; Which is decreasing? \ans{$g$}
      \item $f(x)>0$ when $x$ \ans{$>1$} \qquad $g(x)>0$ when $x$ \ans{$<1$}
      \item At $x=3$, is $f$ positive or negative? \ans{positive} \; Is $g$? \ans{negative}
    \end{enumerate}
    \end{minipage}
    \end{center}

  \item \textbf{Explain.} A line is not horizontal. Why does the sign of $f(x)$ (positive vs.\
        negative) change at the \emph{zero} and nowhere else?
        \ansline{To go from above the $x$-axis to below it (or back), a line has to \emph{cross} the axis, and the crossing is the zero. A line that is not horizontal crosses the axis exactly once, so it changes sign exactly once --- there.}

  \item \textbf{Multiple choice.} Consider the linear function $y = -3x + 6$. Which statement is
        true?
    \begin{enumerate}[label=\Alph*., leftmargin=2.2em, itemsep=1pt]
      \item The function is increasing.
      \item The function is positive for all $x > 2$.
      \item The zero of the function is at $x = 2$. \textcolor{keyred}{\textbf{$\leftarrow$ correct}}
      \item The $y$-intercept is $(0,-3)$.
    \end{enumerate}
    Explain why each of the \emph{other} three statements is false.
    \ansline{A: slope $-3<0$, so it is decreasing. B: decreasing with zero at $2$ means positive for $x<2$, not $x>2$. D: the $y$-intercept is $(0,6)$ --- $-3$ is the slope. C is true: $-3(2)+6=0$.}
\end{enumerate}
\end{notesbox}
