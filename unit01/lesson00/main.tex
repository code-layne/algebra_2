\begin{tcolorbox}[colback=forestbg, colframe=forest, boxrule=0.9pt, arc=2mm,
    left=2mm, right=2mm, top=1.5mm, bottom=1.5mm]
    \textbf{Primary Objective:} Students can read the graph (or table, or rule) of a linear function
    and name its key characteristics --- \emph{domain} and \emph{range}, the \emph{$y$-intercept} and
    the \emph{$x$-intercept} (the \emph{zero}), the \emph{slope} as a \emph{constant rate of change},
    whether the line is \emph{increasing}, \emph{decreasing}, or \emph{constant}, and the intervals
    where the function is \emph{positive} or \emph{negative} --- justify each feature from the graph,
    and interpret it in a context. This is the ``read-a-graph'' toolkit the whole course reuses.
    \\[2pt]
    \textbf{Standards (2023 VA SOL):} \textbf{A2.F.2} --- investigate and analyze the characteristics
    of functions graphically. This Lesson~0 launches that toolkit on linear graphs:
    \textbf{A2.F.2a} (determine domain, range, zeros, and intercepts from a graph),
    \textbf{A2.F.2c} (intervals on which a graph is increasing, decreasing, or constant), and
    \textbf{A2.F.2f} (determine $f(x)$ from a graph and interpret $x$ and $f(x)$ in context). Slope
    as a constant rate of change is reactivated Algebra~1 prerequisite knowledge.
    \\[2pt]
    \textbf{Lesson model:} \emph{Experience first, formalize later.} Students work the activity in
    small groups using only prior knowledge; the teacher circulates with questions, cues, and prompts
    (not answers); the debrief attaches the formal vocabulary to what the groups already found; an
    application is worked together; and Check Your Understanding applies it to new contexts. The four
    parts are one student document, \textbf{Experience \& Formalize}. No separate direct-instruction
    block, and \textbf{no homework} --- Check Your Understanding is the lesson's whole practice set
    and is \emph{not scored}.
\end{tcolorbox}

\begin{skillbox}[Learning Targets \& Key Understandings]{goldbox}
\begin{tabularx}{\linewidth}{@{}X|X@{}}
\textbf{Learning targets (student language)} & \textbf{Key understandings (the ``why'')} \\[2pt]
\hline
\vspace{-6pt}
\begin{itemize}[leftmargin=1.1em, nosep, after=\vspace{2pt}]
  \item I can read the \textbf{$y$-intercept}, the \textbf{zero}, and the \textbf{slope} of a
        linear function from a graph, table, or rule, and say what each means in a situation.
  \item I can tell whether a line is \textbf{increasing}, \textbf{decreasing}, or \textbf{constant},
        and give the intervals where it is \textbf{positive} or \textbf{negative}.
  \item I can state the \textbf{domain} and \textbf{range} of a line --- as pure math, and as the
        inputs that make sense in a story.
\end{itemize}
&
\vspace{-6pt}
\begin{itemize}[leftmargin=1.1em, nosep, after=\vspace{2pt}]
  \item A function is \emph{linear} exactly when its rate of change is \emph{constant} --- the same
        slope everywhere; that is why the graph is straight.
  \item Intercepts are where the graph meets an axis; the zero is the input that makes the output $0$,
        and it is the only place a (non-horizontal) line can change sign.
  \item \textbf{``Positive'' is not ``increasing.''} Sign describes \emph{where the graph sits}
        (above/below the axis); increasing/decreasing describes \emph{which way it moves}. A line can
        be positive and falling at once.
  \item Every non-vertical line has domain all reals; the range is all reals unless the line is
        horizontal. A \emph{story} restricts the domain to the inputs that make sense.
\end{itemize}
\end{tabularx}
\end{skillbox}

\begin{skillbox}[{Vocabulary, Concepts, \& Theorems}]{sky}
\renewcommand{\arraystretch}{1.3}
\begin{tabularx}{\linewidth}{@{}l X@{}}
\textbf{Linear function} & A function whose graph is a straight line; it can be written $f(x)=mx+b$ and has a \emph{constant} rate of change. \\
\textbf{Slope / rate of change} & $m=\dfrac{\Delta y}{\Delta x}=\dfrac{\text{rise}}{\text{run}}$: how much the output changes per unit change in the input. \\
\textbf{$y$-intercept} & The point $(0,b)$ where the graph crosses the $y$-axis --- the output when $x=0$ (the ``starting value'' in a story). \\
\textbf{$x$-intercept (zero)} & The point $(a,0)$ where the graph crosses the $x$-axis; the input $a$ with $f(a)=0$. \\
\textbf{Domain / range} & The set of allowed inputs ($x$) / the set of resulting outputs ($y$), read from the graph. \\
\textbf{Increasing / decreasing} & The graph rises ($m>0$) / falls ($m<0$) as you read it left to right. \\
\textbf{Constant function} & A horizontal line $f(x)=b$; slope $0$, neither increasing nor decreasing; range $\{b\}$. \\
\textbf{Positive / negative interval} & The $x$-values where the graph is \emph{above} the $x$-axis ($f(x)>0$) / \emph{below} it ($f(x)<0$). \\
\end{tabularx}
\end{skillbox}

\begin{fixedskillbox}[Warm-Up --- Activate Prior Knowledge]{forestbg}
    The Warm-Up rehearses exactly the Algebra~1 skills the activity will lean on: \textbf{(1)} reading
    ordered pairs off a grid, including points \emph{on} an axis (the seed of ``intercept'');
    \textbf{(2)} evaluating $g(x)=3x-6$ and writing outputs as points, ending with ``for what input is
    $g(x)=0$?'' (the seed of ``zero''); and \textbf{(3)} continuing a table with a constant step and
    writing its rule $y=3x+5$ (the seed of ``slope'' and $mx+b$). Debrief item~3 aloud, briefly: a
    constant change in output for equal steps in input is what makes a rule \emph{linear} --- but do
    \emph{not} name slope or intercepts yet. The activity will surface them.
\end{fixedskillbox}

\boxguard
\begin{skillbox}[Experience \& Formalize --- The Activity: \emph{Freezing Point}]{forestbg}
\textbf{Launch (2 min).} ``On a cold weekend the temperature that matters most is zero --- above it
the roads are wet, below it they are ice.'' Two stories: Saturday morning warming from $-6\,^\circ$C
at $2^\circ$/hr, and Sunday evening cooling from $6\,^\circ$C at $2^\circ$/hr. Groups of 3--4, one
packet each but one shared conversation. Tell them: \emph{``Use what
you already know --- tables, rules, points. For every answer, be ready to show me where you see it on
the graph.''} Graphs are pre-drawn; students label and circle.

\begin{multicols}{2}
\textbf{What students do.} Part~1 builds the whole toolkit on one rising line: complete the table,
write $T(h)=2h-6$, circle the two axis crossings and say what each means, find the per-hour change
and locate it in the rule \emph{and} on the graph, decide ``going up,'' and write the below/above
freezing hours as inequalities. Part~2 repeats it on a falling line, forces the crux question (2c):
at 5~p.m.\ the temperature is $4^\circ$ --- positive --- and falling, and closes with the story-vs-math
distinction (2e): the story only accepts evening hours, while the rule accepts every real number.
The flat line and the same-zero comparison are saved for the debrief and Check Your Understanding.

\columnbreak
\textbf{What the teacher does.} Circulate. Listen before speaking. When a group stalls, give a
\emph{question, cue, or prompt}, not an answer:
\begin{itemize}[leftmargin=1.1em, nosep]
  \item 1c/1d: ``Put your finger where the line touches the axis. What are the two numbers there?
        Which one is a time and which is a temperature?''
  \item 1e: ``Go from $h=1$ to $h=2$ on the graph. How far up did you move? Now from $4$ to $5$.''
  \item 2c: ``Is $4$ a positive number? Is the \emph{next} hour warmer or colder? Can both be true?''
  \item 2e: ``Could I plug in $h=100$? $h=-3$? What does the \emph{rule} say --- never mind the story.''
\end{itemize}
While circulating, pick groups whose work you will put on the board for the debrief: a clean Part~1
table + labeled graph, a group with a good 2c explanation, and one good answer to 2e.
\end{multicols}
\end{skillbox}

\boxguard
\begin{skillbox}[Debrief --- Formalize (what goes on the board, in a second color)]{forestbg}
Work through the displayed student work in order, and \textbf{attach each formal name to something a
group already wrote}. Students record each line in their QuickNotes as you go.
\begin{multicols}{2}
\begin{enumerate}[leftmargin=1.3em, nosep, itemsep=3pt]
  \item \textbf{Constant change $\Rightarrow$ linear; the change is the slope.} Next to the table:
        ``same $+2$ every hour'' $\to$ \emph{constant rate of change}; a rule with constant rate is a
        \emph{linear function}, $f(x)=mx+b$; the constant is the \emph{slope} $m=\Delta y/\Delta x$.
  \item \textbf{The vertical crossing is the $y$-intercept} $(0,b)$ --- the starting value; find it by
        setting $x=0$. \textbf{The horizontal crossing is the $x$-intercept, the \emph{zero}} --- the
        input where the output is $0$; find it by solving $f(x)=0$.
  \item \textbf{Up/down is increasing/decreasing} and is decided by the \emph{sign of the slope}
        ($+2$ vs.\ $-2$). Then pose the flat line yourself: ``a thermostat holds $21^\circ$ all
        day'' --- slope $0$, \emph{constant}, never crosses, one output.
  \item \textbf{Above/below the axis is positive/negative}, and it can only switch at a zero. Put
        Saturday and Sunday side by side: same zero at $h=3$, opposite sign pattern.
  \item \textbf{Positive $\ne$ increasing.} Use the group's 2c answer verbatim: $4^\circ$ and falling.
        ``Positive'' is where the graph \emph{sits}; ``increasing'' is which way it \emph{moves}.
        Say it twice; this is the misconception the unit test will probe.
  \item \textbf{Domain and range.} Build on 2e: the math line takes all reals in and out
        (horizontal: range $\{b\}$, no zero); the story keeps only the inputs that make sense.
\end{enumerate}
\columnbreak
\textbf{QuickNotes.} Then fill the packet's QuickNotes box together on $f(x)=2x-4$: slope $2$,
$y$-intercept $(0,-4)$, zero $(2,0)$, increasing, positive for $x>2$ / negative for $x<2$, domain and
range all reals. Keep it to the six lines --- the box is a summary of what the groups discovered, not a
new lecture.

\textbf{Why this order.} Each formal term lands on a picture the students already drew or circled, so
the vocabulary is a \emph{label} rather than a \emph{definition to memorize}. The flat line is the
one example the teacher poses cold (it is no longer in the activity); it goes late because it is the
exception to three rules at once (slope $0$, no zero, one-value range).
\end{multicols}
\end{skillbox}

\boxguard
\begin{skillbox}[Application --- \emph{The Car-Wash Card} (worked together, 7 min)]{forestbg}
Straight out of QuickNotes and into a model, with the class holding the pen: a \$40 card, \$8 a wash,
$B(w)=40-8w$. Item~1 asks what the slope and the $y$-intercept \emph{mean} here (not what they are);
item~2 solves $B(w)=0$ in the packet's \texttt{work} block and interprets $w=5$; item~3 is the
meaningful-domain question (whole washes, $0$ to $5$).

\textbf{Item~4 is the point of the whole box.} ``What if a wash cost \$10?'' The card still starts at
\$40, so the $y$-intercept does not move --- only the \emph{steepness} changes, and a steeper drop
reaches zero \emph{sooner} ($w=4$). Ask it in that order: \emph{``Where does the line start now?
Same. So what did change? Then why does it run out sooner?''} A student who answers ``the line
starts lower'' is reading the \$10 as a starting value rather than a rate --- send them back to the
graph, not to the algebra.
\end{skillbox}

\boxguard
\begin{skillbox}[Check Your Understanding --- the lesson's practice, \emph{not scored}]{redbox}
Six items in the packet, new contexts. \textbf{There is no homework:} this is the lesson's entire
practice set, done in class, and the student keeps the packet. \textbf{These carry no point value ---
do not collect them for a grade} (the cover's score column reads \textbf{NA}); spot-check instead.
\begin{itemize}[leftmargin=1.2em, nosep]
  \item[\textbf{1.}] the core procedure, read off a \emph{rule}: $f(x)=2x+2$ (the graph and the
        table are items~2 and~3, so the three representations are all covered);
  \item[\textbf{2.}] the \emph{contrast pair}: two lines with the \emph{same} zero at $x=1$, one
        rising and one falling, closing with (d) ``why can the sign change only at the zero?'';
  \item[\textbf{3.}] a table whose $x$-step is $2$ --- the slope is $\tfrac32$, not $3$;
  \item[\textbf{4.}] the constant function the activity dropped, plus its boundary: $c(x)=b$ has a
        zero only when $b=0$, and then \emph{every} input is one;
  \item[\textbf{5.}] a model in a fresh context, $W(t)=600+150t$, whose zero $t=-4$ is a real feature
        of the line and a \emph{meaningless} moment in the story --- the activity's 2e point again;
  \item[\textbf{6.}] an SOL-style multiple-choice item on $y=-3x+6$.
\end{itemize}
Pairs for 1--3, then solo for 4--6. \textbf{Item~6 is the formative check:} sort responses into
``chose C (got it) / chose B (fused \emph{positive} with \emph{right of the zero}) / chose A (fused
\emph{positive} with \emph{increasing}) / chose D (read the slope as the intercept)'' to decide
whether Lesson~1.1 opens with a one-minute re-look at the Sunday line. Ten minutes will not cover six
items for everyone: \textbf{priority is 1, 2, 5, 6} (procedure, contrast, model, check); \textbf{3 and
4 are the early-finisher bank} and can be picked up at the start of 1.1.
\end{skillbox}

\boxguard
\begin{skillbox}[Watch For (while circulating)]{redbox}
\begin{itemize}[leftmargin=1.2em, nosep]
  \item swapping the intercepts --- reading the $y$-intercept off the $h$-axis or vice versa (Part~1c/d);
  \item calling Sunday's rate ``$2$'' instead of ``$-2$'' --- the sign \emph{is} the direction (2a/2b);
  \item fusing ``positive'' with ``increasing'' (2c, CYU~2 and~6) --- the lesson's target misconception;
  \item writing the flat line's range as ``all reals'' or giving it a zero (CYU~4);
  \item reading CYU~3's slope as $3$ --- the $x$-step is $2$, so $m=\tfrac{\Delta y}{\Delta x}=\tfrac32$;
  \item answering 2e's second half with the story's domain --- push ``forget the story; what does the \emph{rule} allow?''
\end{itemize}
Cold-call prompts: ``Where is the zero, and what does $T$ equal there?'' ``Rising or falling --- and
what is the sign of the rate?'' ``Name an hour where $T<0$.''
\end{skillbox}

\boxguard
\begin{skillbox}[Close \& Preview]{goldbox}
\textbf{Nothing is assigned.} Say it out loud --- there is no homework in this course; the practice
was Check Your Understanding and the students keep the packet. \textbf{What changed today:} they
walked in able to read points off a graph and walked out able to name what a line's features
\emph{mean} --- where it starts, where it crosses zero, how fast it changes, which way it heads, and
where it sits above or below the axis. \textbf{Preview:} Lesson~1.1, \emph{Linear Functions: slope,
rate of change, and forms of a line} --- today they \emph{read} those features off a line; next they
run it backwards and use slope and a point to \emph{write} the equation, then graph by transforming
the parent line $y=x$.
\end{skillbox}

% ── Teacher notes — one per component, in packet order ──
\begin{teachernote}[Warm-Up]
Items 1 and 2 plant ``a point on an axis'' and ``the input that makes the output $0$'' without
naming them; item~3 plants ``constant step $\Rightarrow$ rule $mx+b$.'' Debrief only item~3, and only
the step: resist saying ``slope'' --- let the word arrive in the debrief attached to the groups' own
``$+2$ every hour.'' Five minutes; if item~1 runs long, skip the ``on the axis'' follow-up.
\end{teachernote}

\begin{teachernote}[Experience \& Formalize]
\textbf{Activity.} Part~1 gets about 11 of the 20 minutes; Part~2 goes faster because the moves
repeat on a falling line. If a group finishes early, ask them to compare the two graphs: ``both cross
at $h=3$ --- write one sentence using \emph{before} and \emph{after}.'' \textbf{Debrief.} Do not
re-teach from scratch --- every term should be written \emph{on top of} a displayed student answer.
The two must-land moments are 2c (positive and falling) and the Saturday/Sunday same-zero
comparison; if time is short, cut the QuickNotes walkthrough to slope / intercepts / sign and let
students finish the box from the slides --- and keep QuickNotes to half a page, it is a summary and
not a second lesson. \textbf{Application.} Seven minutes, worked together: ask what the slope and the
intercept \emph{mean} before anyone computes, then let a student drive the \texttt{work} block for
$40-8w=0$. Save two of the seven minutes for item~4 (\$10 a wash) --- it is the concept check, not a
bonus. \textbf{Check Your Understanding.} Two \texttt{work} blocks (Application~2 and CYU~5c) ---
students solve in the reserved space and the key prints the steps in the same space, so the two files
paginate alike. Priority items are 1, 2, 5, 6; 3 and 4 are the early-finisher bank. Item~6 is the
formative check. \textbf{Not scored} --- spot-check item~6 rather than collecting the page.
\end{teachernote}
