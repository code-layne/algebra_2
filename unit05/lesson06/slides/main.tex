% THE NOTATION
\begin{frame}[t, plain]
  \forestheader{Wired in series: $(f\circ g)(x)=f(g(x))$}
  \sectionlabel[goldacc]{The notation}
  \begin{center}
  \begin{tikzpicture}[every node/.style={font=\small}]
    \node (x) at (0,0) {$x$};
    \node[draw=forest, very thick, rounded corners, fill=forest!12, minimum width=1.5cm,
          minimum height=0.9cm] (g) at (2.6,0) {$g$};
    \node[draw=navy, very thick, rounded corners, fill=navy!12, minimum width=1.5cm,
          minimum height=0.9cm] (f) at (7.0,0) {$f$};
    \node (out) at (10.2,0) {$f\big(g(x)\big)$};
    \draw[->, very thick, charcoal] (x)--(g);
    \draw[->, very thick, charcoal] (g)--node[above, font=\scriptsize]{$g(x)$}(f);
    \draw[->, very thick, charcoal] (f)--(out);
    \node[font=\scriptsize, forest] at (2.6,-0.95) {runs \textbf{first}};
    \node[font=\scriptsize, navy] at (7.0,-0.95) {runs \textbf{second}};
  \end{tikzpicture}
  \end{center}
  \vspace{2pt}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \footnotesize
      \begin{itemize}\setlength{\itemsep}{4pt}
        \item Read it ``\emph{$f$ of $g$ of $x$}.''
        \item The function \textbf{next to the $x$} goes first.
        \item $g$ is the \textbf{inner}, $f$ is the \textbf{outer}.
        \item \textbf{Work inside out} --- and write the middle number down.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \footnotesize
      \begin{block}{$\circ$ is not $\cdot$}
        $(f\circ g)(x)\ne f(x)\cdot g(x)$. \\[3pt]
        The circle means \emph{feed}, not \emph{multiply}.
      \end{block}
      \vspace{3pt}
      You already did one: in the Warm-Up you found $g(-1)=2$ and then $f(2)=1$. \\[3pt]
      That was $(f\circ g)(-1)=1$.
    \end{column}
  \end{columns}
\end{frame}

% NUMERICALLY
\begin{frame}[t, plain]
  \forestheader{Two steps, always --- never skip the middle number}
  \sectionlabel[goldacc]{Numerically}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \footnotesize
      $f(x)=2x-3$, \quad $g(x)=x^{2}+1$
      \vspace{4pt}
      \begin{center}
      \renewcommand{\arraystretch}{1.35}
      \begin{tabular}{|l|c|c|c|}
      \hline
      & \textbf{inside} & \textbf{outside} & \textbf{answer} \\ \hline
      $(f\circ g)(3)$ & $g(3)=10$ & $f(10)$ & $17$ \\ \hline
      $(g\circ f)(3)$ & $f(3)=3$ & $g(3)$ & $10$ \\ \hline
      $(f\circ f)(4)$ & $f(4)=5$ & $f(5)$ & $7$ \\ \hline
      \end{tabular}
      \end{center}
      \vspace{4pt}
      Same functions, same input, \textbf{different answers}.
    \end{column}
    \begin{column}{0.48\textwidth}
      \footnotesize
      From a table --- no formulas needed:
      \vspace{3pt}
      \begin{center}
      \renewcommand{\arraystretch}{1.3}
      \begin{tabular}{|c|c|c|c|c|c|}
      \hline
      $x$ & $-2$ & $-1$ & $0$ & $1$ & $2$ \\ \hline
      $s(x)$ & $1$ & $2$ & $-1$ & $0$ & $-2$ \\ \hline
      $t(x)$ & $0$ & $-2$ & $1$ & $-1$ & $2$ \\ \hline
      \end{tabular}
      \end{center}
      \vspace{4pt}
      \begin{block}{One input, three answers}
        $(s\circ t)(0)=s(1)=0$ \\[2pt]
        $(t\circ s)(0)=t(-1)=-2$ \\[2pt]
        $(s\circ s)(0)=s(-1)=2$
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% ALGEBRAICALLY
\begin{frame}[t, plain]
  \forestheader{Substitute the whole function, in parentheses}
  \sectionlabel[goldacc]{Algebraically}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \footnotesize
      $f(x)=2x-3$, \quad $g(x)=x^{2}+1$
      \vspace{4pt}
      \[ (f\circ g)(x)=f\big(x^{2}+1\big) \]
      \[ =2\big(x^{2}+1\big)-3=2x^{2}-1 \]
      \vspace{-6pt}
      \[ (g\circ f)(x)=g\big(2x-3\big) \]
      \[ =\big(2x-3\big)^{2}+1=4x^{2}-12x+10 \]
    \end{column}
    \begin{column}{0.48\textwidth}
      \footnotesize
      \begin{block}{Check against the numbers}
        At $x=3$: $2(9)-1=17$ \checkmark \\[3pt]
        and $4(9)-36+10=10$ \checkmark \\[3pt]
        --- the same two answers as the table.
      \end{block}
      \vspace{4pt}
      \begin{center}
      \textbf{Write the parentheses before you write what goes in them.} \\[3pt]
      Drop them and $\big(2x-3\big)^{2}$ becomes $2x-3^{2}$.
      \end{center}
    \end{column}
  \end{columns}
\end{frame}

% TRANSFORMATIONS ARE COMPOSITIONS
\begin{frame}[t, plain]
  \forestheader{Every transformation in 5.4 was a composition}
  \sectionlabel[goldacc]{The payoff}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \footnotesize
      \begin{itemize}\setlength{\itemsep}{4pt}
        \item A change \textbf{inside} --- $f(x-h)$, $f(kx)$ --- is an inner machine getting at the
              input \emph{before} $f$ sees it.
        \item A change \textbf{outside} --- $a\,f(x)+k$ --- is an outer machine acting on $f$'s
              answer.
      \end{itemize}
      \vspace{5pt}
      \begin{block}{$2\sqrt{x-5}+1$ is three machines}
        subtract $5$ \; $\rightarrow$ \; take the root \; $\rightarrow$ \; double and add $1$
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \footnotesize
      \begin{block}{Why inside changes feel backwards}
        $\sqrt{x-4}$ shifts \emph{right}, not left. \\[4pt]
        The inner machine hands the root a number that is $4$ \textbf{smaller} --- so $x$ has to be
        $4$ \textbf{bigger} to make up for it.
      \end{block}
      \vspace{4pt}
      \begin{center}
      That question has been open since Lesson 5.4. \\ This is the answer.
      \end{center}
    \end{column}
  \end{columns}
\end{frame}

% DOMAIN
\begin{frame}[t, plain]
  \forestheader{An input has to survive \emph{both} stages}
  \sectionlabel{Domain}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \footnotesize
      \begin{block}{Two gates}
        \textbf{Gate 1:} $x$ must be in the domain of the inner function $g$. \\[3pt]
        \textbf{Gate 2:} $g(x)$ must be in the domain of the outer function $f$.
      \end{block}
      \vspace{4pt}
      $f(x)=\sqrt{x}$, $g(x)=2x-10$:
      \[ (f\circ g)(x)=\sqrt{2x-10}, \quad x\ge5 \]
      \vspace{-8pt}
      \begin{center}
      Gate 1 lets everything through. Gate 2 does the work.
      \end{center}
    \end{column}
    \begin{column}{0.50\textwidth}
      \footnotesize
      \textbf{Now the surprise.} $f(x)=x^{2}$, $g(x)=\sqrt{x}$:
      \vspace{3pt}
      \begin{center}
      \renewcommand{\arraystretch}{1.4}
      \begin{tabular}{|l|l|}
      \hline
      $\left(\sqrt{x}\,\right)^{2}=x$ & but domain $x\ge0$ \\ \hline
      $\sqrt{x^{2}}=|x|$ & domain: all reals \\ \hline
      \end{tabular}
      \end{center}
      \vspace{4pt}
      \begin{block}{Read the machines, not the answer}
        The formula simplifies to $x$ --- and still refuses $x=-9$, because Gate 1 turned it away
        before the squaring ever ran. \\[3pt]
        Simplifying erases the \emph{evidence}, not the restriction.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% DECOMPOSITION
\begin{frame}[t, plain]
  \forestheader{Running it backwards: what two machines built this?}
  \sectionlabel[goldacc]{Decomposition}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \footnotesize
      \begin{center}
      \renewcommand{\arraystretch}{1.4}
      \begin{tabular}{|l|l|l|}
      \hline
      $h(x)$ & \textbf{inner $g$} & \textbf{outer $f$} \\ \hline
      $\sqrt{3x+1}$ & $3x+1$ & $\sqrt{x}$ \\ \hline
      $(x-5)^{3}$ & $x-5$ & $x^{3}$ \\ \hline
      $\sqrt[3]{x}+7$ & $\sqrt[3]{x}$ & $x+7$ \\ \hline
      \end{tabular}
      \end{center}
    \end{column}
    \begin{column}{0.44\textwidth}
      \footnotesize
      \begin{block}{How to find the inner function}
        Ask what your \emph{hands} would do first if you were evaluating $h(4)$ on a calculator.
        That is $g$.
      \end{block}
      \vspace{4pt}
      More than one decomposition is correct --- but this one is always useful.
    \end{column}
  \end{columns}
\end{frame}

% SUMMARY
\begin{frame}[t, plain]
  \forestheader{Composition, in one table}
  \sectionlabel{Summary}
  \begin{center}
  \footnotesize
  \renewcommand{\arraystretch}{1.35}
  \begin{tabular}{lll}
  \hline
   & \textbf{$(f\circ g)(x)=f(g(x))$} & \textbf{$(g\circ f)(x)=g(f(x))$} \\
  \hline
  Which runs first? & $g$ --- the one next to the $x$ & $f$ \\
  Read it & inside out & inside out \\
  Are they equal? & \multicolumn{2}{l}{usually \textbf{not} --- order is part of the answer} \\
  Domain & \multicolumn{2}{l}{the inputs that clear \emph{both} gates} \\
  How to find it & \multicolumn{2}{l}{read the machines, \emph{not} the simplified rule} \\
  Not to be confused with & \multicolumn{2}{l}{$f(x)\cdot g(x)$ --- the circle is not a
  multiplication dot} \\
  \hline
  \end{tabular}
  \end{center}
  \vspace{8pt}
  \begin{center}
    $f(x)=\sqrt{x}$ and $g(x)=x-4$ give $\sqrt{x-4}$ one way and $\sqrt{x}-4$ the other. \\[3pt]
    \textbf{Two functions. Two orders. Two different graphs.}
  \end{center}
\end{frame}

\begin{frame}[t, plain]
  \forestheader{Where Unit 5 goes next}
  \sectionlabel{Connections}
  \textbf{Today:} the output of one function becomes the input of another, $(f\circ g)(x)=f(g(x))$
  --- worked inside out, with the order and the two-stage domain both doing real work.\\[8pt]
  \begin{itemize}\setlength{\itemsep}{3pt}
    \item \textbf{5.1} $\sqrt{x^{2}}=|x|$ --- the rule that made the second half of today's domain
          slide come out the way it did.
    \item \textbf{5.4} Transformations were compositions all along, which is finally \emph{why}
          inside changes go the wrong way.
    \item \textbf{5.5} Raising both sides to the $n$th power undid a radical --- one operation
          undoing another, now with a notation.
    \item \textbf{5.7} \textbf{Inverse functions.} Some pairs give back $x$ in \emph{both} orders.
          That is the test, $f(g(x))=g(f(x))=x$ --- and it explains 5.0's reflection over $y=x$, and
          why $y=x^{2}$ needs a restricted domain first.
  \end{itemize}
\end{frame}
