\begin{notesbox}{Getting Ready: Skills We'll Use Today}
\small Three skills. Today's whole lesson is these three, run in order, on one function at a time.

\vspace{4pt}
\textbf{1. Factor, restrict, cancel, sort} (Lessons 5.0 and 5.1). \; $h(x)=\dfrac{x^2-9}{x^2-2x-3}$
\begin{enumerate}[label=(\alph*), leftmargin=2.1em, itemsep=3pt]
  \item Factor both: \; $h(x)=\dfrac{(\rule{1.5cm}{0.4pt})(\rule{1.5cm}{0.4pt})}%
        {(\rule{1.5cm}{0.4pt})(\rule{1.5cm}{0.4pt})}$
  \item Restrictions, read off the \emph{original} denominator: $x\neq\blank{1.0cm}$ and
        $x\neq\blank{1.0cm}$.
  \item Cancel the common factor. Simplified: \blank{2.4cm}
  \item One restriction cancelled and one did not. The one that cancelled gives a
        \blank{1.6cm} at $x=\blank{0.8cm}$; the one that survived gives a \blank{3.4cm} at
        $x=\blank{0.8cm}$.
\end{enumerate}

\vspace{4pt}
\textbf{2. The degree comparison} (Lesson 5.0). Give the horizontal asymptote --- or say there is none.
\begin{enumerate}[label=(\alph*), leftmargin=2.1em, itemsep=3pt]
  \item $y=\dfrac{2x}{x^2+5}$: \; top degree \blank{0.7cm}, bottom degree \blank{0.7cm}, so
        HA \blank{1.6cm}
  \item $y=\dfrac{3x^2-1}{x^2+4}$: \; top degree \blank{0.7cm}, bottom degree \blank{0.7cm}, so
        HA \blank{1.6cm}
  \item $y=\dfrac{x^3}{x^2-1}$: \; top degree \blank{0.7cm}, bottom degree \blank{0.7cm}, so
        HA \blank{1.6cm}
\end{enumerate}

\vspace{4pt}
