% THE BUILD
\begin{frame}[t, plain]
  \forestheader{The build --- five steps, always in this order}
  \sectionlabel[goldacc]{Post this}
  \begin{enumerate}\setlength{\itemsep}{5pt}
    \item \textbf{Factor} the top and the bottom completely.
    \item \textbf{Restrictions} off the \emph{original} denominator. Cancel test: a factor that cancels
          $\Rightarrow$ \textbf{hole} (give its coordinates); a factor that survives $\Rightarrow$
          \textbf{vertical asymptote}.
    \item \textbf{Degree comparison} $\Rightarrow$ horizontal asymptote, or ``none.''
    \item \textbf{Intercepts, computed.} $x$-int: zeros of the \emph{simplified} numerator.
          $y$-int: $f(0)$.
    \item \textbf{Sign chart.} Boundary points $=$ $x$-intercepts $+$ vertical asymptotes. One test
          value per interval.
  \end{enumerate}
  \vspace{4pt}
  {\footnotesize Then assemble: domain as a union, end behavior, above/below by interval.}
\end{frame}

% HOLE OR WALL
\begin{frame}[t, plain]
  \forestheader{Step 2: hole or wall?}
  \sectionlabel{The cancel test}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Cancels $\Rightarrow$ hole}
        \[ \frac{x-4}{(x-4)(x+1)}=\frac{1}{x+1} \]
        $x=4$ is a \textbf{hole} --- one missing point. Height: put $4$ into the \emph{simplified}
        form, $\frac15$.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Survives $\Rightarrow$ wall}
        \[ \frac{x+5}{(x-2)(x+3)} \]
        Nothing cancels, so $x=2$ and $x=-3$ are both \textbf{vertical asymptotes}.
      \end{block}
    \end{column}
  \end{columns}
  \vspace{8pt}
  {\footnotesize \textbf{Order matters.} Restrictions come off the \emph{original} denominator.
  Simplify first and the hole disappears from your answer --- the same error that becomes an extraneous
  solution in Lesson 5.6.}
\end{frame}

% DEGREE COMPARISON
\begin{frame}[t, plain]
  \forestheader{Step 3: the degree comparison}
  \sectionlabel{Recall from 5.0}
  \centering
  \renewcommand{\arraystretch}{1.5}
  \begin{tabular}{|l|c|l|}
  \hline
  \textbf{Case} & \textbf{Horizontal asymptote} & \textbf{Example} \\ \hline
  top $<$ bottom & $y=0$ & $\dfrac{x-1}{x^2-x-6}$ \\ \hline
  top $=$ bottom & ratio of leading coefficients & $\dfrac{2x^2-8}{x^2-x-6} \Rightarrow y=2$ \\ \hline
  top $>$ bottom & \textbf{none} & $\dfrac{x^3}{x^2-1}$ \\ \hline
  \end{tabular}

  \vspace{8pt}
  \begin{block}{}
    Compare the \emph{whole} top with the \emph{whole} bottom --- never one factor with another.
    And ``none'' is a complete answer: the graph still has end behavior, the arms just do not flatten.
  \end{block}
\end{frame}

      \[ f(x)=\frac{x-1}{(x+2)(x-3)} \]
      \begin{itemize}\setlength{\itemsep}{3pt}
        \item Restrictions $-2$ and $3$; nothing cancels.

        \item $x$-int: numerator zero at $x=1$ $\Rightarrow$ $(1,0)$.
        \item $y$-int: $f(0)=\dfrac{-1}{-6}=\dfrac16$.
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{8pt}
  \begin{block}{Everything is known --- and the graph is still not determined}
    \centering That is the Hook. One more step.
  \end{block}

% CROSSING THE HA
\begin{frame}[t, plain]
  \forestheader{Does it cross its horizontal asymptote?}
  \sectionlabel[goldacc]{Three claims, one reason}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \[ f(x)=0 \quad\Longleftrightarrow\quad x-1=0 \]
      \begin{itemize}\setlength{\itemsep}{3pt}
        \item A fraction is zero only when its \textbf{numerator} is.
        \item Here the numerator has a zero: $x=1$.
        \item \textbf{Yes} --- and the crossing is at the $x$-intercept $(1,0)$.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Putting the three together}
        \textbf{5.0:} a rational graph \emph{may} cross its HA. \\[3pt]
        \textbf{5.4:} a transformed parent \emph{never} does --- nonzero constant numerator. \\[3pt]
        \textbf{5.5:} same fact, both times. A numerator with a zero makes a crossing
        \emph{certain}.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

      \begin{itemize}\setlength{\itemsep}{2pt}
        \item $x+2$ cancels $\Rightarrow$ \textbf{hole} at $\left(-2,\frac45\right)$.
        \item $x-3$ survives $\Rightarrow$ \textbf{wall} $x=3$.
        \item Equal degrees $\Rightarrow$ \textbf{HA} $y=1$.
        \item $x$-int $(2,0)$; \; $y$-int $\left(0,\frac23\right)$.
      \end{itemize}
      \vspace{2pt}
      {\footnotesize It is yesterday's anchor family, right $3$ and up $1$ --- with one point punched
      out. \textbf{5.5 did not replace 5.4. It surrounded it.}}
    \end{column}
  \end{columns}
\end{frame}

% THE TRAP
\begin{frame}[t, plain]
  \forestheader{Two traps in Example B}
  \sectionlabel[goldacc]{Hold the line}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{$(-2,0)$ is not an $x$-intercept}
        The \emph{original} numerator $x^2-4$ is zero at $-2$. \\[3pt]
        But $-2$ is not in the domain --- it is the hole. \\[3pt]
        Only zeros of the \textbf{simplified} numerator count.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{The hole is not a boundary point}
        Sign chart boundaries here: $2$ and $3$ only. \\[3pt]
        The sign cannot change at a hole --- the function is the same expression on both sides of it.
      \end{block}
    \end{column}
  \end{columns}
  \vspace{8pt}
  {\footnotesize And the domain still excludes $-2$: $(-\infty,-2)\cup(-2,3)\cup(3,\infty)$.
  A hole costs you an input just as surely as a wall does.}
\end{frame}
