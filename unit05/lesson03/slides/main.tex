% HOOK
\begin{frame}[t, plain]
  \forestheader{Both students are right. So why does one form win?}
  \sectionlabel{Hook}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \[ \frac{1}{\sqrt2} \qquad\text{vs.}\qquad \frac{\sqrt2}{2} \]
      \vspace{-8pt}
      \footnotesize
      \begin{itemize}\setlength{\itemsep}{5pt}
        \item $1 \div 1.414214 = 0.7071\dots$
        \item $1.414214 \div 2 = 0.7071\dots$
      \end{itemize}
      \vspace{5pt}
      {\footnotesize Same number. Every textbook still insists on the second.}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{It is 1850 and you have no calculator}
        Which long division would you rather do? \\[4pt]
        $1 \div 1.414214$ \quad or \quad $1.414214 \div 2$
      \end{block}
      \vspace{4pt}
      {\footnotesize A radical on \textbf{top} is harmless. A radical on the \textbf{bottom} is a
      nuisance. \\[4pt]
      \textcolor{goldacc}{\textbf{How did they get from one to the other?}} Multiply top and bottom
      by $\sqrt2$ --- and $\sqrt2\cdot\sqrt2=2$.}
    \end{column}
  \end{columns}
\end{frame}

% MULTIPLYING
\begin{frame}[t, plain]
  \forestheader{The product property, run \emph{forwards}}
  \sectionlabel[goldacc]{Multiplying}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \[ \sqrt[n]{a}\cdot\sqrt[n]{b}=\sqrt[n]{ab} \]
      \vspace{-8pt}
      \begin{block}{The move}
        \centering Outsides times outsides. \\ Insides times insides. \\
        \textbf{Then} simplify.
      \end{block}
      \vspace{4pt}
      \footnotesize
      $2\sqrt5\cdot3\sqrt{10}=6\sqrt{50}=30\sqrt2$ \\[3pt]
      $\sqrt[3]4\cdot\sqrt[3]6=\sqrt[3]{24}=2\sqrt[3]3$
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Never skip the last step}
        $\sqrt6\cdot\sqrt{10}=\sqrt{60}=2\sqrt{15}$ \\[4pt]
        {\footnotesize Neither factor simplified on its own. The \emph{product} created a perfect
        square out of nothing.}
      \end{block}
      \vspace{5pt}
      {\footnotesize \textcolor{goldacc}{\textbf{$\sqrt a\cdot\sqrt a=a$}} --- a square root times
      itself erases the radical. \\[4pt]
      $\left(\sqrt7\right)^{2}=7$, \; $\left(3\sqrt5\right)^{2}=45$. \\[4pt]
      A cube root needs \textbf{three} copies. The index still sets the group size.}
    \end{column}
  \end{columns}
\end{frame}

% DISTRIBUTING AND FOIL
\begin{frame}[t, plain]
  \forestheader{Radicals distribute and FOIL exactly like binomials}
  \sectionlabel{Expanding}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \footnotesize
      \begin{itemize}\setlength{\itemsep}{6pt}
        \item $\sqrt5\left(\sqrt3+\sqrt{10}\right)=\sqrt{15}+\sqrt{50}=\sqrt{15}+5\sqrt2$
        \item $\sqrt2\left(4-\sqrt2\right)=4\sqrt2-2$
        \item $\left(2+\sqrt3\right)\left(5-\sqrt3\right)$ \\[3pt]
              \qquad $=10-2\sqrt3+5\sqrt3-3=7+3\sqrt3$
      \end{itemize}
      \vspace{6pt}
      {\footnotesize Simplify each product \emph{separately} after distributing.}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{The error of the day}
        $\left(\sqrt7+\sqrt2\right)^{2} \ne 9$ \\[4pt]
        $=7+\sqrt{14}+\sqrt{14}+2 = 9+2\sqrt{14}$
      \end{block}
      \vspace{4pt}
      {\footnotesize There is a \textbf{middle term}, for the same reason
      \textcolor{keyred}{$(x+4)^{2}\ne x^{2}+16$} --- which you rejected in the Warm-Up. \\[5pt]
      Your turn: $\left(3-\sqrt5\right)^{2}=14-6\sqrt5$.}
    \end{column}
  \end{columns}
\end{frame}

% CONJUGATES
\begin{frame}[t, plain]
  \forestheader{One special pair comes out \emph{rational}}
  \sectionlabel[goldacc]{Conjugates}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \[ \left(4+\sqrt3\right)\left(4-\sqrt3\right) \]
      \vspace{-14pt}
      \[ = 16 \underbrace{- 4\sqrt3 + 4\sqrt3}_{\text{cancel}} - 3 = 13 \]
      \vspace{-4pt}
      \footnotesize
      Two things happen at once: the middle terms \textbf{cancel}, and each square \textbf{erases
      its radical}. What is left has no radical anywhere.
      \vspace{6pt}

      {\footnotesize Contrast: $\left(4+\sqrt3\right)^{2}=19+8\sqrt3$. \; Same numbers --- and the
      radical survives.}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Conjugates}
        \centering
        Same terms, \textbf{opposite} middle sign. \\[5pt]
        $\left(a+\sqrt b\right)\left(a-\sqrt b\right)=a^{2}-b$ \\[4pt]
        $\left(\sqrt a+\sqrt b\right)\left(\sqrt a-\sqrt b\right)=a-b$
      \end{block}
      \vspace{4pt}
      {\footnotesize \textbf{You have met this before.} Lesson 2.4:
      $(a+bi)(a-bi)=a^{2}+b^{2}$ --- the same move made an \emph{imaginary} number real. \\[4pt]
      Why one plus and one minus? \; $i^{2}=-1$, but $\left(\sqrt b\right)^{2}=+b$.}
    \end{column}
  \end{columns}
\end{frame}

% DIVIDING
\begin{frame}[t, plain]
  \forestheader{Dividing --- and the condition we skipped yesterday}
  \sectionlabel{Quotients}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \footnotesize
      \begin{itemize}\setlength{\itemsep}{7pt}
        \item $\dfrac{\sqrt{48}}{\sqrt3}=\sqrt{16}=4$
        \item $\dfrac{6\sqrt{20}}{2\sqrt5}=3\sqrt4=6$
        \item $\dfrac{\sqrt{40x^{3}}}{\sqrt{5x}}=\sqrt{8x^{2}}=2x\sqrt2$
      \end{itemize}
      \vspace{6pt}
      {\footnotesize Two radicals with the same index pull together under one radical --- and the
      quotient is usually friendlier than either piece.}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{But this one does not come out}
        \centering $\dfrac{\sqrt3}{\sqrt5}=\sqrt{\dfrac{3}{5}}$
      \end{block}
      \vspace{4pt}
      {\footnotesize A radical is stuck in the \textbf{denominator} --- simplest-form condition
      \textbf{3}, the one Lesson 5.2 deliberately left open. \\[5pt]
      \textcolor{goldacc}{\textbf{Time to finish it.}}}
    \end{column}
  \end{columns}
\end{frame}

% RATIONALIZING: MONOMIAL
\begin{frame}[t, plain]
  \forestheader{Rationalizing: multiply by a form of $1$}
  \sectionlabel[goldacc]{One radical}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \[ \frac{1}{\sqrt2}\cdot\frac{\sqrt2}{\sqrt2}=\frac{\sqrt2}{2}
         \qquad
         \frac{3}{\sqrt5}\cdot\frac{\sqrt5}{\sqrt5}=\frac{3\sqrt5}{5} \]
      \vspace{-6pt}
      \begin{block}{Why it is legal}
        \centering $\dfrac{\sqrt b}{\sqrt b}=1$, and multiplying by $1$ changes the
        \textbf{form}, never the \textbf{value}. \\[3pt]
        {\footnotesize Exactly like $\frac34=\frac{15}{20}$.}
      \end{block}
    \end{column}
    \begin{column}{0.46\textwidth}
      \footnotesize
      $\dfrac{5}{\sqrt3}=\dfrac{5\sqrt3}{3}$ \qquad
      $\sqrt{\dfrac{3}{5}}=\dfrac{\sqrt{15}}{5}$ \\[5pt]
      $\dfrac{\sqrt7}{\sqrt{12}}=\dfrac{\sqrt7}{2\sqrt3}=\dfrac{\sqrt{21}}{6}$
      \; {\scriptsize (simplify the bottom \emph{first})}
      \vspace{8pt}

      \begin{block}{A higher index changes the multiplier}
        $\dfrac{1}{\sqrt[3]2}\cdot\dfrac{\sqrt[3]4}{\sqrt[3]4}=\dfrac{\sqrt[3]4}{2}$ \\[4pt]
        {\footnotesize $\sqrt[3]2$ alone is not enough --- you need \textbf{three} copies.
        Multiply by whatever \emph{completes the group}.}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% RATIONALIZING: BINOMIAL
\begin{frame}[t, plain]
  \forestheader{A binomial on the bottom? Use its conjugate}
  \sectionlabel{Two terms}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \[ \frac{6}{3-\sqrt5}\cdot\frac{3+\sqrt5}{3+\sqrt5}
         = \frac{6\left(3+\sqrt5\right)}{9-5}
         = \frac{9+3\sqrt5}{2} \]
      \vspace{-6pt}
      \footnotesize
      \begin{itemize}\setlength{\itemsep}{4pt}
        \item Multiplying by $\sqrt5$ would leave a radical down there. The conjugate is
              \emph{guaranteed} to clear it.
        \item Check for a common factor at the end --- this one reduced.
      \end{itemize}
      \vspace{5pt}
      {\footnotesize $\dfrac{4}{\sqrt5-1}=\sqrt5+1$ \qquad
      $\dfrac{\sqrt2}{\sqrt7+\sqrt3}=\dfrac{\sqrt{14}-\sqrt6}{4}$}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{block}{Three ways this goes wrong}
        \footnotesize
        1. Multiplying only the \emph{denominator} --- that is not multiplying by $1$. \\[4pt]
        2. Using the \emph{same} binomial: $\left(3-\sqrt5\right)^{2}=14-6\sqrt5$, still
        irrational. \\[4pt]
        3. Forgetting to FOIL a binomial \emph{numerator} --- four products, not two.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% ERROR SLIDE
\begin{frame}[t, plain]
  \forestheader{The four ways today goes wrong}
  \sectionlabel[goldacc]{Watch out}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \footnotesize
      \begin{block}{1. Not finishing}
        $\sqrt6\cdot\sqrt{10}=\sqrt{60}$ --- true, unfinished. \; $60=4\cdot15$, so $2\sqrt{15}$.
      \end{block}
      \vspace{4pt}
      \begin{block}{2. Dropping the middle term}
        $\left(\sqrt5+\sqrt2\right)^{2}\ne7$. \; It is $7+2\sqrt{10}$.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \footnotesize
      \begin{block}{3. Wrong multiplier}
        $\dfrac{5}{2+\sqrt3}\cdot\dfrac{2+\sqrt3}{2+\sqrt3}$ accomplishes nothing. Flip the sign:
        $2-\sqrt3$.
      \end{block}
      \vspace{4pt}
      \begin{block}{4. Only the bottom}
        Multiplying just the denominator by the conjugate \textbf{changes the number}. Top and
        bottom, always.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% GOLDEN RATIO
\begin{frame}[t, plain]
  \forestheader{Rationalizing makes an equality \emph{visible}}
  \sectionlabel{Extension}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \[ \varphi=\frac{1+\sqrt5}{2}\approx 1.6180 \]
      \vspace{-10pt}
      \footnotesize
      \begin{tabular}{lll}
      \hline
      $\varphi$ & $1/\varphi$ & $\varphi^{2}$ \\
      $1.6180$ & $0.6180$ & $2.6180$ \\
      \hline
      \end{tabular}
      \vspace{6pt}

      {\footnotesize The same decimal three times. The reciprocal is one \emph{less}; the square is
      one \emph{more}.}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Why? Rationalize.}
        $\dfrac{1}{\varphi}=\dfrac{2}{1+\sqrt5}\cdot\dfrac{1-\sqrt5}{1-\sqrt5}
        =\dfrac{\sqrt5-1}{2}=\varphi-1$
      \end{block}
      \vspace{4pt}
      {\footnotesize Cut a $1\times1$ square off a $\varphi\times1$ rectangle. The leftover has
      ratio $\dfrac{1}{\varphi-1}=\dfrac{2}{\sqrt5-1}=\varphi$ --- \textbf{another golden
      rectangle}, forever. \\[5pt]
      \textcolor{goldacc}{\textbf{You cannot see that until the radical leaves the denominator.}}}
    \end{column}
  \end{columns}
\end{frame}
