\begin{notesbox}{Practice}
Every answer needs \emph{two} parts: the simplest form \textbf{and} the restrictions. Factor every
denominator before you build an LCD, and put the second numerator in \textbf{parentheses} every single
time you subtract.

\begin{enumerate}[leftmargin=1.9em, itemsep=11pt]
  \item \textbf{Like denominators.} Combine, simplify, restrict.
    \begin{enumerate}[label=(\alph*), leftmargin=2.1em, itemsep=5pt]
      \item $\dfrac{3x}{x+7}+\dfrac{21}{x+7}=\;$\blank{2.2cm} \quad $x\neq$ \blank{1.4cm}
      \item $\dfrac{7x-1}{x^2-1}-\dfrac{3x+3}{x^2-1}=\;$\blank{2.2cm} \quad $x\neq$ \blank{1.8cm}
      \item $\dfrac{x^2}{x+5}-\dfrac{25}{x+5}=\;$\blank{2.2cm} \quad $x\neq$ \blank{1.4cm}
    \end{enumerate}
    \vspace{2pt}
    Your answer to (a) is a number and your answer to (c) is a polynomial. Neither has a denominator
    left. So what does each restriction do to the graph? \blank{5.0cm}

  \item \textbf{Monomial LCDs.}
    \begin{enumerate}[label=(\alph*), leftmargin=2.1em, itemsep=5pt]
      \item $\dfrac{3}{4x}+\dfrac{5}{6x}=\;$\blank{2.2cm} \quad LCD used: \blank{1.6cm}
      \item $\dfrac{7}{x^2}-\dfrac{2}{5x}=\;$\blank{2.2cm} \quad LCD used: \blank{1.6cm}
      \item $\dfrac{4}{3a}+\dfrac{1}{a^2b}=\;$\blank{2.2cm} \quad restrictions: \blank{2.2cm}
    \end{enumerate}

  \item \textbf{Binomial and quadratic LCDs.} Factor first. One of these has \emph{opposite}
        binomials for denominators, and one of them cancels at the end.
    \begin{enumerate}[label=(\alph*), leftmargin=2.1em, itemsep=7pt]
      \item $\dfrac{2}{x+1}+\dfrac{3}{x-2}=\;$\blank{2.6cm} \quad $x\neq$ \blank{1.8cm}
      \item $\dfrac{x}{x-3}-\dfrac{2}{x+3}=\;$\blank{2.6cm} \quad $x\neq$ \blank{1.8cm}
      \item $\dfrac{4}{x-6}-\dfrac{48}{x^2-36}=\;$\blank{2.6cm} \quad $x\neq$ \blank{1.8cm}
      \item $\dfrac{6}{x-1}+\dfrac{3}{1-x}=\;$\blank{2.6cm} \quad $x\neq$ \blank{1.4cm}
      \item $\dfrac{5}{x^2+4x+3}+\dfrac{2}{x^2-9}=\;$\blank{2.6cm} \quad $x\neq$ \blank{2.2cm}
    \end{enumerate}

  \item \textbf{Complex fractions.} Simplify each as a quotient of two simple fractions, then state
        \emph{all} restrictions --- the inner denominators \emph{and} the main fraction bar.
    \begin{enumerate}[label=(\alph*), leftmargin=2.1em, itemsep=7pt]
      \item \TallMath{\frac{\;\dfrac{1}{x}+\dfrac{1}{6}\;}{\;\dfrac{1}{x}-\dfrac{1}{6}\;}}
            $=\;$\blank{2.2cm} \quad $x\neq$ \blank{1.8cm}
      \item \TallMath{\frac{\;\dfrac{2}{x}-\dfrac{2}{5}\;}{\;x-5\;}}
            $=\;$\blank{2.2cm} \quad $x\neq$ \blank{1.8cm}
      \item \TallMath{\frac{\;\dfrac{x}{x+2}\;}{\;\dfrac{x^2}{x^2-4}\;}}
            $=\;$\blank{2.2cm} \quad $x\neq$ \blank{2.2cm}
    \end{enumerate}

  \item \textbf{A round trip.} A cyclist rides $30$ miles out to a lake at $x$ mi/h and rides the same
        $30$ miles home at $x+5$ mi/h.
    \begin{enumerate}[label=(\alph*), leftmargin=2.1em, itemsep=5pt]
      \item Total time $T(x)=$ \blank{4.4cm} hours (one fraction, simplified).
      \item Check your formula: if $x=10$, part (a) gives \blank{1.2cm} hours, and adding the two legs
            by hand gives \blank{1.2cm} hours.
      \item The algebra excludes two values of $x$. Which two? \blank{1.8cm} \; Which values of $x$
            make sense for a bike ride? \blank{1.8cm}
      \item \textbf{Average speed} for the whole trip is total distance divided by total time --- a
            complex fraction. Simplify $\dfrac{60}{T(x)}=$ \blank{3.0cm} mi/h. \\[3pt]
            At $x=10$ that gives \blank{1.4cm} mi/h. Is that the average of $10$ and $15$?
            \blank{1.0cm} \; Explain in one sentence. \blank{4.4cm}
    \end{enumerate}

  \item \textbf{Justify.} Are $\dfrac{1}{x-2}+\dfrac{1}{x+2}$ and $\dfrac{2x}{x^2-4}$ the same
        expression? Then answer the same question for $\dfrac{3}{x-3}-\dfrac{18}{x^2-9}$ and
        $\dfrac{3}{x+3}$. Answer both in terms of \emph{inputs}: which inputs does each form accept,
        and where do they agree?
        \writelines{3}
\end{enumerate}
\end{notesbox}
